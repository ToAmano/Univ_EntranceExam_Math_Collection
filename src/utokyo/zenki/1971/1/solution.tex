\begin{oframed}
変数$t$が$0$から$\pi$まで動く時，
     \begin{align}
     x=2\cos\left(t-\dfrac{\pi}{6}\right) , y=\cos\left(t+\dfrac{\pi}{3}\right)
     \end{align}
によってあらわされる点$(x,y)$と原点$(0,0)$との間の距離の最大値，最小値およびそれをとる$t$
の値を求めよ．     
\end{oframed}

\setlength{\columnseprule}{0.4pt}
\begin{multicols}{2}
{\bf[解]}\2 題意の距離の$2$乗を$L(t)$とおけば
     \begin{align}
     L(t)&=x^2+y^2 \\
     &=4\cos^2\left(t-\dfrac{\pi}{6}\right)+\cos^2\left(t+\dfrac{\pi}{3}\right)  \\
     &=\left(\sqrt{3}c+s\right)^2+\left(\dfrac{1}{2}c-\dfrac{\sqrt{3}}{2}s\right)^2 \\
     &=\dfrac{13}{4}c^2+\dfrac{7}{4}s^2+\dfrac{3\sqrt{3}}{2}sc  \\
     &=\dfrac{7}{4}+\dfrac{3}{2}c^2+\dfrac{3\sqrt{3}}{4}\sin2t \\
     &=\dfrac{5}{2}+\dfrac{3}{4}(\cos2t+\sqrt{3}\sin2t) \\
     &=\dfrac{5}{2}+\dfrac{3}{2}\sin\left(2t+\dfrac{\pi}{6}\right)
     \end{align}
となる．$0\le t\le \pi$に注意して，求める距離$L$の最大小は，
     \begin{align}
     \left\{
          \begin{array}{ll}
          \min L=1 &\left(t=\dfrac{2\pi}{3}\right)     \\
          \max L=2 &\left(t=\dfrac{\pi}{6}\right)
          \end{array}
     \right.\tag{答}
     \end{align}     
となる．
