\paragraph{【新課程】}
$\theta$を$0\le\theta<2\pi$なる実数とする．
簡単のため$\cos\theta = c$，$s = \sin\theta$とおくと，
円柱側面の点$P$は$P(c, s, z)$とおける．
$P$が$D$の領域にある時，
\begin{align}
  0\le z \le 1-x+\sqrt{3}y \label{eq:1}
\end{align}
を満たすから代入して
\begin{align}
  0 \leqq z \leqq 1 - c + \sqrt{3}s \label{eq:2}
\end{align}
である．

$z$の存在条件から
\begin{align}
  &0 \leqq 1 - c + \sqrt{3}s \nonumber\\
  &\Longleftrightarrow \cos\left(\theta+\dfrac{\pi}{3}\right) \leqq \dfrac{1}{2} \nonumber\\
  &\Longleftrightarrow 0 \leqq \theta \leqq \dfrac{4\pi}{3}
\end{align}
となる．ただし$2$行目で三角関数の合成を用いた．

したがって，求める面積$T$は
\begin{align}
  T
  &= \int_0^{\frac{4\pi}{3}} (1 - c + \sqrt{3}s) \, d\theta \nonumber\\
  &= \Bigl[ \theta - s - \sqrt{3}c \Bigr]_0^{\frac{4\pi}{3}} \nonumber\\
  &= 2\sqrt{3} + \dfrac{4\pi}{3}
\end{align}
となる．$\cdots\text{（答）}$

時刻$t$における点$P$の座標を$(x(t), 0)$とする．
条件より$x(0) = 0$であり，$P$は$x$軸上を正の向きに動くから，$t > 0$において$x(t) > 0$である．
$x \geqq 0$において$\dfrac{x^2+1}{x+1} > 0$であるから，時刻$t$において囲まれる図形の面積$S$は
\begin{align}
  S = \int_0^x \dfrac{u^2+1}{u+1} \, du \label{eq:3}
\end{align}
と表される．両辺を$x$で微分すると
\begin{align}
  \dfrac{dS}{dx} = \dfrac{x^2+1}{x+1}
\end{align}
である．合成関数の微分法により
\begin{align}
  \dfrac{dS}{dt} = \dfrac{dS}{dx} \cdot \dfrac{dx}{dt} = \dfrac{x^2+1}{x+1} \dfrac{dx}{dt} \label{eq:4}
\end{align}
となる．題意より$P$が点$(x, 0)$を通過するときの変化率は$\dfrac{dS}{dt} = x^2+1$であるから，\cref{eq:4}の左辺に代入して
\begin{align}
  \dfrac{x^2+1}{x+1} \dfrac{dx}{dt} = x^2+1
\end{align}
が成り立つ．$x^2+1 > 0$であるから両辺を$x^2+1$で割ると
\begin{align}
  \dfrac{1}{x+1} \dfrac{dx}{dt} = 1 \Longleftrightarrow \dfrac{1}{x+1}dx = dt
\end{align}
を得る．変数分離形として両辺を$t$で積分すると
\begin{align}
  \int \dfrac{1}{x+1} \, dx = \int \, dt
\end{align}
より，
\begin{align}
  \log_e(x+1) = t + C
\end{align}
である．ただし$C$は積分定数である．
$t = 0$のとき$x = 0$であるから，$\log_e 1 = 0 + C$より$C = 0$である．
したがって
\begin{align}
  \log(x+1) = t \Longleftrightarrow x+1 = e^t \Longleftrightarrow x = e^t - 1 \label{eq:5}
\end{align}
である．これを\cref{eq:3}に代入して面積$S$は
\begin{align}
  S
  &= \int_0^x \dfrac{u^2+1}{u+1} \, du \\
  &= \int_0^x \left[u - 1 + \dfrac{2}{u+1}\right]\, du \\
  &= \left[ \dfrac{u^2}{2} - u + 2\log(u+1) \right]_0^x \\
  &= \dfrac{x^2}{2} - x + 2\log(x+1)
\end{align}
である．\cref{eq:5}を代入して
\begin{align}
  S &= \dfrac{(e^t - 1)^2}{2} - (e^t - 1) + 2t \nonumber\\
  &= \dfrac{e^{2t} - 2e^t + 1}{2} - e^t + 1 + 2t \nonumber\\
  &= \dfrac{1}{2}e^{2t} - 2e^t + 2t + \dfrac{3}{2} \tag{答}
\end{align}
である．
