与えられた条件
\begin{align}
  0 \le \alpha < \pi/2 \label{eq:1}
\end{align}
の元で考える．

題意の条件$z>0$から，$P$は$xy$平面上にはなく，$A, B, C$ とは異なることに注意する．

\begin{align}
  \vec{AP} =
  \begin{pmatrix} x-1 \\ y-1 \\ z
  \end{pmatrix}, \quad
  \vec{BP} =
  \begin{pmatrix} x-1 \\ y+1 \\ z
  \end{pmatrix}, \quad
  \vec{CP} =
  \begin{pmatrix} x \\ y \\ z
  \end{pmatrix}
\end{align}

(ロ)～(ニ)より，$z$軸の方向ベクトル$\vec{n} =
\begin{pmatrix} 0 \\ 0 \\ 1
\end{pmatrix}$ として，
\begin{align}
  \begin{cases}
    \vec{n} \cdot \vec{AP} = |\vec{n}| |\vec{AP}| \cos\pi/4 \\
    \vec{n} \cdot \vec{BP} = |\vec{n}| |\vec{BP}| \cos\pi/4 \\
    \vec{n} \cdot \vec{CP} = |\vec{n}| |\vec{CP}| \cos\alpha
  \end{cases}
  \iff
  \begin{cases}
    z = \sqrt{(x-1)^2 + (y-1)^2 + Z^2} \cdot \dfrac{1}{\sqrt{2}} \\
    z = \sqrt{(x-1)^2 + (y+1)^2 + Z^2} \cdot \dfrac{1}{\sqrt{2}} \\
    z = \sqrt{x^2 + y^2 + Z^2} \cos\alpha
  \end{cases}
\end{align}
\cref{eq:1}および$z>0$から両辺二乗しても同値だから，
\begin{align}
  z^2 &= (x-1)^2 + (y+1)^2 = (x-1)^2 + (y-1)^2  \label{eq:2} \\
  z^2 &= \cdot \tan^2\alpha = x^2 + y^2 \label{eq:3}
\end{align}
である．\cref{eq:2}の右側の不等式から，
\begin{align}
  (y+1)^2 = (y-1)^2 \iff y=0 \label{eq:4}
\end{align}
だから，\cref{eq:2,eq:3}に代入し直して
\begin{align}
  & z^2 = (x-1)^2 + 1 = x^2 - 2x + 2  \label{eq:5} \\
  & z^2 \cdot \tan^2\alpha = x^2  \label{eq:6}
\end{align}
である．従って\cref{eq:1,eq:4,eq:5,eq:6}および$z>0$を満たす点$P(x,y,z)$の数を求めれば良い以下$\alpha$が$0$か否かで場合分けする．

\subparagraph{$1^\circ$ $\alpha \neq 0$ の時}

\cref{eq:6}および$z>0$から
\begin{align}
  z^2 = \left(\dfrac{x}{\tan\alpha}\right)^2 \quad \therefore z = \left|\dfrac{x}{\tan\alpha}\right| \label{eq:7}
\end{align}
だから，\cref{eq:5}に代入して
\begin{align}
  & \left(\dfrac{x}{\tan\alpha}\right)^2 = x^2-2x+2 \\
  \therefore
  & (t^2-1) x^2 - 2t^2 x + 2t^2 = 0 \label{eq:8}
\end{align}
である．ただし$t=\tan\alpha$とおいた．

\cref{eq:8}を満たす$x$に対して\cref{eq:7}から$z$が一意に定まる．$y=0$で固定だから，結局\cref{eq:8}を満たす$x$の数が点$P$の数に等しい．

\cref{eq:8}は$t^2\neq 1$の時には$x$についての二次関数で，判別式$D$として
\begin{align}
  D/4 = t^4 - (t^2-1) \cdot 2t^2 = t^2 (2 - t^2)
\end{align}
だから，解の数は
\begin{align}
  \begin{cases}
    D/4 > 0 \iff 2 > t^2 \text{ の時 } 2 \text{ 個} \\
    D/4 = 0 \iff 2 = t^2 \text{ の時 } 1 \text{ 個} \\
    D/4 < 0 \iff 2 < t^2 \text{ の時 } 0 \text{ 個}
  \end{cases}
\end{align}
である．解が存在する時は
\begin{align}
  x = \dfrac{t^2 \pm \sqrt{t^2(2-t^2)}}{t^2 - 1}
\end{align}
より\cref{eq:7}に代入して
\begin{align}
  z = \Bigl| \dfrac{x}{t} \Bigr| = \Bigl| \dfrac{t \pm \sqrt{2-t^2}}{t^2 - 1} \Bigr| \label{eq:9}
\end{align}
である．

一方で$t^2 = 1 \iff \alpha = \pi/4$ ($\because$ \cref{eq:1}) の時，
\cref{eq:8}は$x=1$となり，さらに\cref{eq:7}から$z=1$となって，$P$の数は$1$つである．

\subparagraph{$2^\circ$ $\alpha = 0$ の時}

この時は\cref{eq:6}から$x=0$だから，\cref{eq:5}から$z=\sqrt{2}$となる．
よって$P$は一つである．
\casesend

以上をまとめて，$P$の数と対応する$z$の座標は
\begin{align}
  \begin{cases}
    \alpha = 0 \text{ の時}，Z = \sqrt{2} \text{ の } 1 \text{ つ} \\
    0 < \tan\alpha < \sqrt{2} \land \alpha \neq \pi/4 \text{ の時}，Z = \Bigl| \dfrac{t \pm \sqrt{2-t^2}}{t^2 - 1} \Bigr| \text{ の } 2 \text{ つ} \\
    \tan\alpha = \sqrt{2} \text{ の時}，Z = \sqrt{2} \text{ の } 1 \text{ つ} \\
    \sqrt{2} < \tan\alpha \text{ の時}，0 \text{ 個} \\
    \alpha = \pi/4 \text{ の時}，Z = 1 \text{ の } 1 \text{ つ}
  \end{cases}
\end{align}
