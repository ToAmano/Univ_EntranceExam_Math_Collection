与えられた数列
\begin{align}
  \begin{cases}
    a_{n+1}=\dfrac{a_n}{(1+a_n)^2} \\
    a_1 = \dfrac{1}{2}
  \end{cases} \label{eq:1}
\end{align}
について考える．

\paragraph{(1)}

$b_n=\dfrac{1}{a_n}$を\cref{eq:1}に代入すると
\begin{align}
  \begin{cases}
    b_1 = 2 \\
    b_{n+1} = b_n + \dfrac{1}{b_n} + 2
  \end{cases} \label{eq:2}
\end{align}
である．

$b_n > 2n \quad (n \ge 2)$ を数学的帰納法によって示す．
$n=2$ の時，\cref{eq:2}から
\begin{align}
  b_2 = 4 + \dfrac{1}{2}
\end{align}
なので題意は成立する．
そこで$n=k \in \mathbb{N}_{\ge 2}$ での成立を仮定すると$n=k+1$について
\begin{align}
  b_{k+1} = b_k + \dfrac{1}{b_k} + 2 > 2(k + 1)
\end{align}
となり，$n=k+1$ でも題意が成立する．
以上より数学的帰納法によって題意は示された．

(1) から $n \ge 2$ に対して $b_n=\dfrac{1}{a_n} > 2n \ \therefore \ a_n < \dfrac{1}{2n}$ である．$n \to \infty$ をかんがえるので $n$ は十分大きいとして良く，上図で面積比較して，
\begin{align}
  \dfrac{1}{n} \sum_{k=1}^n a_k
  = \dfrac{a_1}{n} + \dfrac{1}{n} \sum_{k=2}^n a_k
  &< \dfrac{1}{n} \left( \dfrac{1}{2} + \sum_{k=2}^n \dfrac{1}{2k} \right) \\
  &< \dfrac{1}{2n} \left( 1 + \int_1^n \dfrac{1}{x} dx \right)
  = \dfrac{1+\log n}{2n} \\
  \therefore \ \dfrac{1}{n} \sum_{k=1}^n a_k &< \dfrac{1 + \log n}{2n} \to 0 \quad (n \to \infty)
\end{align}
を得る．

さらに，\cref{eq:1}より帰納的に $0 < a_n$が成り立つから，挟み撃ちの定理から
\begin{align}
  \lim_{n\to\infty} \dfrac{1}{n} \sum_{k=1}^n a_k = 0
\end{align}
である．

\paragraph{(3)}

\cref{eq:2}に$a_n=\dfrac{1}{b_n}$を代入して
\begin{align}
  b_{n+1} - b_n = 2 + a_n
\end{align}
だから，$n=1, 2, \cdots, N$ として足して，
\begin{align}
  b_{N+1} - b_1 = 2N + \sum_{k=1}^N a_k
\end{align}
を得る．
両辺を $N+1$ でわって，さらに$b_{N+1}=\dfrac{1}{a_{N+1}}$，$b_1=2$を代入して
\begin{align}
  \dfrac{1}{(N+1)a_{N+1}} &= \dfrac{2}{N+1} + \dfrac{2N}{N+1} + \dfrac{1}{N+1} \sum_{k=1}^N a_k
\end{align}
だから，$N\to\infty$の極限をとると最終項は(2)の結果により$0$に収束するから
\begin{align}
  \lim_{N \to \infty}\dfrac{1}{(N+1)a_{N+1}} = 2
\end{align}
を得る．$0 < x$ で $\dfrac{1}{x}$ が連続なことから，求める極限は
\begin{align}
  \lim_{n\to\infty} na_n = \dfrac{1}{2}
\end{align}
である．

{\bf [別解]}
\paragraph{(3)}

まず$b_n \le 2n + \log n$ を帰納法で示す．
$n=1$ の時は
\begin{align}
  b_1 \le 2
\end{align}
より成立するので，$n = k \in \mathbb{N}$ での成立を仮定する．
仮定および\cref{eq:2}より
\begin{align}
  b_{k+1}
  = b_k + \dfrac{1}{b_k} + 2
  \le 2(k+1) + \log k + \dfrac{1}{2k} \quad (\because 2k \le b_k) \label{eq:3}
\end{align}
である．

ここで新しく
\begin{align}
  f(x) = \log(x+1) - \log x - \dfrac{1}{2x}
\end{align}
とおくと，一階微分は
\begin{align}
  f'(x)
  = \dfrac{1}{x+1} - \dfrac{1}{x} + \dfrac{1}{2x^2}
  = \dfrac{1-x}{2x^2(x+1)}
\end{align}
だから，増減表は下表となる．

よって $1 \le x$ では $f(x) > 0$ つまり $\log(k+1) > \log k + \dfrac{1}{2k}$ だから，\cref{eq:3}に代入して
\begin{align}
  b_{k+1} \le 2(k+1) + \log(k+1)
\end{align}
となる．よって $n=k+1$ でも成立する．

以上から数学的帰納法により
\begin{align}
  2n < \dfrac{1}{a_n} < 2n + \log n
  \quad \therefore
  \dfrac{1}{2n+\log n} < a_n < \dfrac{1}{2n}
\end{align}
である．両辺に$n$をかけて
\begin{align}
  \dfrac{1}{2+\dfrac{\log n}{n}} < na_n < \dfrac{1}{2}
\end{align}
である．$n\to\infty$とすると両辺は$\dfrac{1}{2}$に収束するから，
挟み撃ちの定理より求める極限値は
\begin{align}
  \lim_{n\to\infty} n a_n = \dfrac{1}{2}
\end{align}
である．
