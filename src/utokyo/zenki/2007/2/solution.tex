{\bf [解]}

題意より$n \in \mathbb{Z}_{\ge 2}, 2 \le k \le n$について考える．
簡単のため
\begin{align}
  \begin{cases}
    \angle OP_0 P_1 = \angle OP_{k-1}P_{k} = \theta \\
    |OP_{k-1}| = b_k \\
  \end{cases} \label{eq:1}
\end{align}
とおく．題意より
\begin{align}
  \begin{cases}
    \angle P_{k-1}OP_k = \dfrac{\pi}{n} \\
    |P_{k-1}P_{k} | = a_{k} \\
    b_1 = 1 \\
    b_2 = 1 + \dfrac{1}{n}
  \end{cases} \label{eq:2}
\end{align}
である．

さて$\triangle O P_{k-1} P_k$ に正弦定理を用いると
\begin{align}
  \dfrac{a_k}{\sin \dfrac{\pi}{n}}
  = \dfrac{b_{k+1}}{\sin \theta}
  = \dfrac{b_k}{\sin\left(\theta + \dfrac{\pi}{n}\right)} \label{eq:3}
\end{align}
だから，$b_{k+1},b_{k}$について
\begin{align}
  b_{k+1} = \dfrac{\sin \theta}{\sin\left(\theta + \dfrac{\pi}{n}\right)} b_k \label{eq:4}
\end{align}
である．初期条件\cref{eq:2}より\cref{eq:4}で$k=1$として
\begin{align}
  1 + \dfrac{1}{n} = \dfrac{\sin \theta}{\sin\left(\theta + \dfrac{\pi}{n}\right)} \label{eq:5}
\end{align}
だから，\cref{eq:4}に代入して
\begin{align}
  b_{k+1} = \left(1 + \dfrac{1}{n}\right) b_k
\end{align}
だから，\cref{eq:2}の初期条件$b_1=1$の元で漸化式を繰り返し用いて
\begin{align}
  b_k = \left(1 + \dfrac{1}{n}\right)^{k-1}b_1 = \left(1 + \dfrac{1}{n}\right)^{k-1} \label{eq:6}
\end{align}
である．簡単のため$\alpha=1+\dfrac{1}{n}$とおくと$b_k=\alpha^{k-1}$である．

次に，$a_k$を求めるために$\triangle O P_k P_{k-1}$ に余弦定理を用いて
\begin{align}
  |P_{k}P_{k+1}|^2
  & = |OP_{k}|^2+|OP_{k-1}|^2-2|OP_{k}||OP_{k-1}|\cos\angle P_{k}OP_{k-1} \\
  a_k^2
  & = b_{k}^2 + b_{k+1}^2 - 2b_{k}b_{k+1}\cos\dfrac{\pi}{n}  \\
  \therefore
  a_{k}
  & = \alpha^{k-1}\sqrt{1+\alpha^2-2\alpha\cos\dfrac{\pi}{n}}
\end{align}
だから，$k=1,2,\cdots,n$ として足し合わせて
\begin{align}
  S_n
  & = \dfrac{1 - \alpha^n}{1 - \alpha} \sqrt{1 + \alpha^2 - 2\alpha \cos \dfrac{\pi}{n}} \\
  & = n(\alpha^n-1)sqrt{1+\left(1+\dfrac{1}{n}\right)^2-2\left(1+\dfrac{1}{n}\right)\cos \dfrac{\pi}{n}} \\
  & = (\alpha^n-1)\sqrt{2n^2\left(1-\cos\dfrac{\pi}{n}\right)+2n(1-\cos\dfrac{\pi}{n}) + 1} \label{eq:7}
\end{align}
である．ここで$n\to\infty$の時$t=\dfrac{1}{n}\to 0$であって，
\begin{align}
  & \lim_{n\to\infty} n^2\left(1-\cos\dfrac{\pi}{n}\right)
  = \lim_{t\to 0} \pi^2\dfrac{1-\cos t\pi}{\pi^2t^2} = \dfrac{\pi^2}{2} \\
  & \lim_{n\to\infty} n(1-\cos\dfrac{\pi}{n})
  = \lim_{t\to 0} \dfrac{1-\cos t\pi}{t} = 0 \\
  & \lim_{n\to\infty} \alpha^n
  = e
\end{align}
だから\cref{eq:7}に代入して
\begin{align}
  \lim_{n\to\infty} S_n (e - 1) \sqrt{\pi^2 + 1}
\end{align}
である．
