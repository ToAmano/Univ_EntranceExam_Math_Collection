\paragraph{(1)}

わかりやすさのため$B=P_0, C=Q_0$ と置く．
$\triangle P_n Q_n D$， $\triangle AD Q_n$ に注目し，$|DQ_{n+1}|$, $|Q_{n+1}Q_{n}|$の長さの相似をふた通りで表すと
\begin{align}
  \begin{cases}
    |DQ_{n+1}| : |Q_{n+1}Q_{n}| = a - x_{n+1} : x_{n+1} & (\triangle AD Q_n)\\
    |DQ_{n+1}| : |Q_{n+1}Q_{n}| = x_{n+1} : x_{n} - x_{n+1} & (\triangle P_n Q_n D)
  \end{cases}
\end{align}
だからこれらが等しいことにより
\begin{align}
  & x_{n+1} : x_n - x_{n+1} = a - x_{n+1} : x_{n+1} \\
  \therefore
  &x_{n+1} = \dfrac{a x_n}{a + x_n} \label{eq:1}
\end{align}
である．

$x_n \neq 0$ だから，$y_n = \dfrac{1}{x_n}$ として\cref{eq:1}の両辺逆数をとると
\begin{align}
  y_{n+1} = y_n + \dfrac{1}{a} \label{eq:2}
\end{align}
である．初項は$y_0 = \dfrac{1}{x_0} = \dfrac{1}{b}$ だから漸化式\cref{eq:2}をくり返し用いて，
\begin{align}
  y_n = \dfrac{n}{a} + \dfrac{1}{b}
\end{align}
である．よって$x_n$は
\begin{align}
  x_n
  = \dfrac{1}{y_n}
  = \dfrac{1}{\dfrac{n}{a} + \dfrac{1}{b}} = \dfrac{ab}{a + bn} \label{eq:3}
\end{align}
である．

\paragraph{(2)}

台形$ABCD$の高さ$H$とする．
辺 $P_{n+1} Q_{n+1}$ を底辺とみたときの $\triangle D P_{n+1} Q_n$ （あるいは$\triangle D P_n Q_n$）の高さ $h_n$ は$\triangle DCB$と$\triangle D P_nQ_n$の相似に着目して
\begin{align}
  & x_n: b = h_n: H \\
  & h_n = \dfrac{H}{b} x_n \quad (n \ge 1) \label{eq:4}
\end{align}
である．

従って三角形$\triangle D P_{n+1} Q_n$の面積$F_n$:
\begin{align}
  F_n &= \dfrac{1}{2} h_n x_{n+1}
\end{align}
に\cref{eq:3,eq:4}を代入して
\begin{align}
  F_n
  &= \dfrac{H}{2b} x_n x_{n+1}\\
  &= \dfrac{H}{2b} \cdot \dfrac{ab}{a+bn} \cdot \dfrac{ab}{a+b(n+1)} \\
  &= \Bigl( \dfrac{1}{a+bn} - \dfrac{1}{a+b(n+1)} \Bigr) \dfrac{H a^2}{2}
\end{align}
となり，和および極限は
\begin{align}
  \sum_{k=1}^n F_k
  &= \dfrac{Ha^2}{2} \sum_{k=1}^{n} \Bigl( \dfrac{1}{a+bk} - \dfrac{1}{a+b(k+1)} \Bigr) \\
  &= \dfrac{Ha^2}{2} \Bigl( \dfrac{1}{a+b} - \dfrac{1}{a+b(n+1)} \Bigr) \\
  &\xrightarrow{n \to \infty} \dfrac{H a^2}{2(a+b)}
\end{align}
である．
