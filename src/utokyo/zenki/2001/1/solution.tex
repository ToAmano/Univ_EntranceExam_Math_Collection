\noindent
したがって，
\begin{align}
|AP| = |BQ| &= \dfrac{2}{3}\sqrt{3} \quad \cdots \textcircled{1}
\end{align}
となる．$O$ から $AB$ に下ろした垂足を $R$ とする．
\begin{align}
|ER| &= \dfrac{3}{2} \quad \cdots \textcircled{2}
\end{align}
\begin{align}
|OR| &= \sqrt{r^2 - \left(\dfrac{3}{2}\right)^2} \\
&= \sqrt{r^2 - \dfrac{9}{4}} \quad \cdots \textcircled{3} \quad (\because \textcircled{2})
\end{align}
\begin{align}
|OE| &= \sqrt{|OP|^2 + \left(\dfrac{1}{3}\sqrt{3}\right)^2} \\
&= \sqrt{\left\{r^2 - \left(\dfrac{2}{3}\sqrt{3}\right)^2\right\} + \left(\dfrac{1}{3}\sqrt{3}\right)^2} \\
&= \sqrt{r^2 - 1} \quad \cdots \textcircled{4} \quad (\because \textcircled{1})
\end{align}
\textcircled{2}～\textcircled{4}から，$|ER|$ を2通りで表して ($E-O-R$ は一直線上)，
\begin{align}
\dfrac{3}{2} &= \sqrt{r^2 - \dfrac{9}{4}} + \sqrt{r^2 - 1}
\end{align}
$t = r^2$ として，両辺正から2乗して
\begin{align}
\dfrac{9}{4} &= 2t - \dfrac{13}{4} + 2\sqrt{\left(t - \dfrac{9}{4}\right)(t - 1)}
\end{align}
\begin{align}
2 - t &= \sqrt{\left(t - \dfrac{9}{4}\right)(t - 1)}
\end{align}
$t < 2$ のもとで2乗して
\begin{align}
-4t + 4 &= -\dfrac{13}{4}t + \dfrac{9}{4}
\end{align}
\begin{align}
t &= \dfrac{13}{9} \quad (t < 2 \text{ をみたす})
\end{align}
$r > 0$ から
\begin{align}
r &= \sqrt{t} = \dfrac{\sqrt{13}}{3} \hfill \text{\slash\slash}
\end{align}
