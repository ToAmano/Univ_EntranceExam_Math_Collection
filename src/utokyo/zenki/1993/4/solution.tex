{\bf [解]} $p,q \in \mathbb{R}$ とする．
\begin{align}
(f(x))^2 = x^{2n} + p^2 x^2 + q^2 + 2(p x^{n+1} + q x^n + pq x)
\end{align}
だから，
\begin{align}
I = \int_0^1 (x^{2n} + p^2 x^2 + q^2) \, dx + \int_{-1}^1 (p x^{n+1} + q x^n) \, dx \quad \dots \text{①}
\end{align}
である．

\bigskip
\noindent
$1^\circ$ $n \in \text{odd}$ の時\\
①から
\begin{align}
I &= \int_0^1 (x^{2n} + p^2 x^2 + q^2) \, dx + 2 \int_0^1 p x^{n+1} \, dx \\
&= p^2 \int_0^1 x^2 \, dx + 2p \int_0^1 x^{n+1} \, dx + q^2 \int_0^1 1 \, dx + \int_0^1 x^{2n} \, dx \\
&= \dfrac{1}{3} p^2 + \dfrac{2}{n+2} p + q^2 + \dfrac{1}{2n+1} \\
&= \dfrac{1}{3} \Bigl( p + \dfrac{3}{n+2} \Bigr)^2 + q^2 - \dfrac{3}{(n+2)^2} + \dfrac{1}{2n+1}
\end{align}
だから，$n \in \mathbb{N}$ より $(p,q) = \Bigl(-\dfrac{3}{n+2}, 0\Bigr)$ で $\min I = \dfrac{1}{2n+1} - \dfrac{3}{(n+2)^2}$

\bigskip
\noindent
$2^\circ$ $n \in \text{even}$ の時\\
①から
\begin{align}
I &= \int_0^1 (x^{2n} + p^2 x^2 + q^2 + 2q x^n) \, dx \\
&= q^2 + \dfrac{2q}{n+1} + \dfrac{1}{3} p^2 + \dfrac{1}{2n+1} \\
&= \Bigl( q + \dfrac{1}{n+1} \Bigr)^2 + \dfrac{1}{3} p^2 + \dfrac{1}{2n+1} - \dfrac{1}{(n+1)^2}
\end{align}
から $(p,q) = \Bigl(0, -\dfrac{1}{n+1}\Bigr)$ で $\min I = \dfrac{1}{2n+1} - \dfrac{1}{(n+1)^2}$

\bigskip
\noindent
以上まとめて，
\begin{align}
\begin{cases}
n \in \text{odd} \dots (p,q) = \Bigl(-\dfrac{3}{n+2}, 0\Bigr) \text{の時} \quad \min I = \dfrac{(n-1)^2}{(2n+1)(n+2)^2} \\[1.5ex]
n \in \text{even} \dots (p,q) = \Bigl(0, -\dfrac{1}{n+1}\Bigr) \text{の時} \quad \min I = \dfrac{n^2}{(2n+1)(n+1)^2}
\end{cases}
\quad \text{\#\#}
\end{align}
