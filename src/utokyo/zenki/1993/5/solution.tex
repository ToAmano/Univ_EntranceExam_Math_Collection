従って$t_k(n)$に関する漸化式は最初の数に応じて以下のようになる．

\subparagraph{$1^\circ$ 最初の数が$1$の時，すなわち$k=1,3,4$ の時}

$S$ では 1 だったから，$n+1$ 回目について$n$ 回目で場合分けして
\begin{align}
  t_k(n+1)
  &= (1-q) t_k(n) + p(1 - t_k(n)) \\
  &= (1-p-q) t_k(n) + p \\
  \therefore
  t_k(n+1) - \dfrac{p}{p+q}
  &= (1-p-q) \Bigl( t_k(n) - \dfrac{p}{p+q} \Bigr)
\end{align}
を得る．$t_k(0) = 1$ からくり返し用いて
\begin{align}
  t_k(n)
  &= (1-p-q)^n \Bigl( 1 - \dfrac{p}{p+q} \Bigr) + \dfrac{p}{p+q} \\
  &= \dfrac{1}{p+q} \{ p + q(1-p-q)^n \} \label{eq:1}
\end{align}

\subparagraph{$2^\circ$ 最初の数が$0$の時，すなわち$k=2,5$ の時}

$1^\circ$ と同様に考えて
\begin{align}
  t_k(n+1)
  &= (1-p) t_k(n) + q(1 - t_k(n)) \\
  &= (1-p-q) t_k(n) + q \\
  \therefore
  t_k(n+1) - \dfrac{q}{p+q}
  &= (1-p-q) \Bigl( t_k(n) - \dfrac{q}{p+q} \Bigr)
\end{align}
を得る．$t_k(0) = 1$ からくり返し用いて
\begin{align}
  t_k(n) = \dfrac{1}{p+q} \{ q + p(1-p-q)^n \} \label{eq:2}
\end{align}
である．
\casesend

\cref{eq:1,eq:2}から，第$n$回目の書写しで$S$と一致する確率$C(n)$は，
$t_k(n)\, (k=1,2,3,4,5)$の積で
\begin{align}
  C(n)
  &= \prod_{k=1}^5 t_k(n) \\
  &= \left(\dfrac{p + q(1-p-q)^n}{p+q} \right)^3\left(\dfrac{q + p(1-p-q)^n}{p+q}\right)^2
\end{align}
である．$|1-p-q|<1$だから，$n\to\infty$での極限値は
\begin{align}
  \lim_{n\to\infty} C(n) =  \dfrac{p^3 q^2}{(p+q)^5}
\end{align}
である．
