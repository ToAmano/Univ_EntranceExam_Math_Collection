与えられた領域
\begin{align}
  \begin{cases}
    C_1: y \ge \dfrac{1}{2} x^2 \\
    C_2: \dfrac{x^2}{1/2} + \dfrac{y^2}{1/32} \le 1
  \end{cases} \label{eq:1}
\end{align}
を図示すると上図の斜線部のようになる．これは$y$軸に関して線対称である．
簡単のため
\begin{align}
  \begin{cases}
    f(x) = \dfrac{1}{2} x^2 \\
    g(x) = \sqrt{\dfrac{1}{32} - \dfrac{x^2}{16}}
  \end{cases} \label{eq:2}
\end{align}
および，逆関数を
\begin{align}
  \begin{cases}
    f^{-1}(y) = \sqrt{2y} \\
    g^{-1}(y) = \sqrt{\dfrac{1}{2}-16y^2}
  \end{cases} \label{eq:3}
\end{align}

とおく．

\cref{eq:1}を$x$軸の周りに回転した立体の体積$V_1$は，対称性から$0\le x$の部分を$x$軸周りに回転させた領域の$2$倍だから
\begin{align}
  \dfrac{1}{2} V_1
  &= \pi \int_0^{\frac{1}{2}} \left[(g(x))^2-(f(x))^2\right] dx \\
  &= \pi \int_0^{\frac{1}{2}} \left[ \left( \dfrac{1}{32} - \dfrac{x^2}{16} \right) - \dfrac{1}{4} x^4 \right] dx \\
  &= \left[ -\dfrac{1}{20} x^5 - \dfrac{1}{48} x^3 + \dfrac{1}{32} x \right]_0^{\frac{1}{2}} \\
  &= \dfrac{11\pi}{15} \left(\dfrac{1}{2}\right)^6 \\
  \therefore
  V_1 &= \dfrac{11\pi}{15} \left(\dfrac{1}{2}\right)^5 \label{eq:4}
\end{align}
である．

ついで\cref{eq:1}を$y$軸の周りに回転した立体の体積$V_2$は，対称性から$0\le x$の部分を$y$軸周りに回転させてれば良いから
\begin{align}
  \dfrac{1}{\pi} V_2
  &= \int_0^{\frac{1}{8}} \left(f^{-1}(y)\right)^2 dy
  + \int_{\frac{1}{8}}^{\frac{\sqrt{2}}{8}} \left(g^{-1}(y)\right)^2 dy \\
  &= \int_0^{\frac{1}{8}} (2y) dy + \int_{\frac{1}{8}}^{\frac{\sqrt{2}}{8}} \left( \dfrac{1}{2} - 16y^2 \right) dy \\
  &= \left[ y^2 \right]_0^{\frac{1}{8}} + \left[ \dfrac{1}{2} y - \dfrac{16}{3} y^3 \right]_{\frac{1}{8}}^{\frac{\sqrt{2}}{8}} \\
  &= \dfrac{1}{64} + \dfrac{1}{1} \dfrac{\sqrt{2}-1}{8} - \dfrac{16}{3} \dfrac{2\sqrt{2}-1}{8^3} \\
  &= \dfrac{1}{64} + \dfrac{\sqrt{2}-1}{16} - \dfrac{1}{3} \dfrac{2\sqrt{2}-1}{32} \\
  &= \dfrac{1}{3} \dfrac{1}{64} \left[ -7 + 8\sqrt{2} \right] \\
  \therefore
  V_2 &= \dfrac{\pi}{192} (-7 + 8\sqrt{2}) \label{eq:5}
\end{align}
である．

\cref{eq:4,eq:5}をまとめて
\begin{align}
  V_1 = \dfrac{11}{15} \left(\dfrac{1}{2}\right)^5 \pi, \quad
  V_2 = \dfrac{\pi}{192} (-7 + 8\sqrt{2}) \label{eq:6}
\end{align}
である．

\paragraph{(2)}

\cref{eq:6} から，
\begin{align}
  \dfrac{V_2}{V_1}
  &= \dfrac{\dfrac{1}{3} \left(\dfrac{1}{2}\right)^6 (-7 + 8\sqrt{2})}{\dfrac{11}{15} \left(\dfrac{1}{2}\right)^5} \\
  &= \dfrac{\dfrac{1}{6}(-7 + 8\sqrt{2})}{\dfrac{11}{15}} \\
  &= \dfrac{5}{22} (-7 + 8\sqrt{2})
\end{align}
だから
\begin{align}
  \dfrac{5}{22} (-7 + 8\sqrt{2}) < 1
  &\iff
  -7 + 8\sqrt{2} < \dfrac{22}{5}
  \iff
  8\sqrt{2} < \dfrac{57}{5} \label{eq:7}
\end{align}
である．ここで
\begin{align}
  64 \cdot 2 \cdot 25 = 3200 < 3249 = (57)^2
\end{align}
の平方根を取れば
\begin{align}
  8\sqrt{2} < \dfrac{57}{5}
\end{align}
だから\cref{eq:7}は成立する．よって
\begin{align}
  \dfrac{V_2}{V_1} < 1
\end{align}
である．
