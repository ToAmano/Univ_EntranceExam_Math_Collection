与えられた漸化式
\begin{align}
  \begin{cases}
    \overrightarrow{OQ_n}=\overrightarrow{OP_n}-(\vec{a}\cdot\overrightarrow{OP_n})\vec{a} \\
    \overrightarrow{OP_{n+1}}=4\{ \overrightarrow{OQ_n}-(\vec{b}\cdot\overrightarrow{OQ_n})\vec{b} \} \\
    x_1 = 1, y_1 = 0
  \end{cases} \label{eq:1}
\end{align}
について考える．

簡単のため$c = \cos\theta, s = \sin\theta$ とおく．
また，
\begin{align}
  \begin{cases}
    A_n = \vec{a} \cdot \overrightarrow{OP_n} \\
    B_n = \vec{b} \cdot \overrightarrow{OP_n} \\
    T = \vec{a} \cdot \vec{b}
  \end{cases}\label{eq:2}
\end{align}
とおくと，$P_n(x_n,y_n)$より
\begin{align}
  \begin{cases}
    A_n = cx_n + sy_n \\
    B_n = \dfrac{1}{2}\left(\sqrt{3}x_n+y_n\right) \\
    T   = \dfrac{1}{2}\left(\sqrt{3}c+s\right)
  \end{cases}
\end{align} \label{eq:3}
である．

\cref{eq:1}の第一式に\cref{eq:2}を代入すると
\begin{align}
  \overrightarrow{OQ_n}=\overrightarrow{OP_n}-A_n\vec{a} \label{eq:4}
\end{align}
だから，$\vec{b}$との内積をとって\cref{eq:2}に注意して
\begin{align}
  \vec{b}\cdot\overrightarrow{OQ_n}=B_n - TA_n \label{eq:5}
\end{align}
である．よって\cref{eq:1}で$P_n$の漸化式を立てると
\begin{align}
  \overrightarrow{OP_{n+1}}
  &= 4\{ \overrightarrow{OQ_n}-(B_n - TA_n)\vec{b} \} \quad (\because\text{\cref{eq:5}})\\
  &= 4\{ \overrightarrow{OP_n}-A_n\vec{a} - (B_n - TA_n)\vec{b}\} \quad (\because\text{\cref{eq:4}})
\end{align}
だから，\cref{eq:3}を代入して成分表示すると
\begin{align}
  \overrightarrow{OP_{n+1}} =
  \begin{pmatrix}
    x_{n+1} \\ y_{n+1}
  \end{pmatrix}
  &= 4
  \begin{pmatrix}
    x_{n} \\ y_{n}
  \end{pmatrix}
  -4A_n
  \begin{pmatrix}
    c \\ s
  \end{pmatrix}
  -2(B_n-TA_n)
  \begin{pmatrix}
    \sqrt{3} \\ 1
  \end{pmatrix} \\
  \therefore
  &
  \begin{cases}
    x_{n+1} = 4x_{n} - 4A_nc - 2\sqrt{3}(B_n-TA_n) \\
    y_{n+1} = 4y_{n} - 4A_ns - 2(B_n-TA_n)
  \end{cases}
\end{align}
である．さらに\cref{eq:3}を代入して
\begin{align}
  x_{n+1}
  &= 4x_{n} - 4(cx_n+sy_n)c - \sqrt{3}\left[\sqrt{3}x_n+y_n-(\sqrt{3}c+s)(cx_n+sy_n)\right] \\
  &= (4 - 4c^2 - 3 + \sqrt{3}c(\sqrt{3}c+s)) x_n
  + (-4sc - \sqrt{3} + \sqrt{3}s(\sqrt{3}c+s)) y_n \\
  &= (1 - c^2 + \sqrt{3}sc) x_n + (-sc - \sqrt{3}c^2) y_n \\
  &= s(s + \sqrt{3}s) x_n - c(s + \sqrt{3}c) y_n \label{eq:6}
\end{align}
および
\begin{align}
  y_{n+1}
  &= 4y_{n} - 4(cx_n+sy_n)s - \left[\sqrt{3}x_n+y_n-(\sqrt{3}c+s)(cx_n+sy_n)\right] \\
  &= (-4cs-\sqrt{3}+c(\sqrt{3}c+s))x_n
  + (4-4s^2-1 +s(\sqrt{3}c+s))y_n \\
  &= (-3sc-\sqrt{3}s^2)x_n + (3c^2+\sqrt{3}cs)y_n \\
  &= -\sqrt{3}s(s+\sqrt{3}c)x_n + \sqrt{3}c(\sqrt{3}c+s)y_n \label{eq:7}
\end{align}
である．
ここで
\begin{align}
  \begin{cases}
    \alpha = s(s + \sqrt{3}c) \\
    \beta = c(s + \sqrt{3}c)
  \end{cases} \label{eq:8}
\end{align}
とすると，
\cref{eq:6,eq:7}は
\begin{align}
  \begin{cases}
    x_{n+1} = \alpha x_n - \beta y_n \\
    y_{n+1} = -\sqrt{3} (\alpha x_n - \beta y_n) = -\sqrt{3}x_{n+1}
  \end{cases} \label{eq:9}
\end{align}
を得る．第二式を第一式に代入して
\begin{align}
  x_{n+1} &= (\alpha + \sqrt{3}\beta) x_n \quad (n \ge 2)
\end{align}
だからくり返し用いて，
\begin{align}
  \begin{cases}
    x_n = (\alpha + \sqrt{3}\beta)^{n-2} x_2 & (n \ge 2) \\
    y_n = -\sqrt{3} (\alpha + \sqrt{3}\beta)^{n-2} x_2 & (n \ge 2)
  \end{cases}
\end{align}
を得る．ここで\cref{eq:1,eq:9}より$x_2$は
\begin{align}
  x_2 = \alpha x_1 = \alpha
\end{align}
だから
\begin{align}
  \begin{cases}
    x_n = \alpha (\alpha + \sqrt{3}\beta)^{n-2} & (n \ge 2)\\
    y_n = -\sqrt{3}\alpha (\alpha + \sqrt{3}\beta)^{n-2} & (n \ge 2)
  \end{cases} \label{eq:10}
\end{align}
である．

したがって数列 $\{x_n\}, \{y_n\}$ が収束する条件は，
\begin{align}
  -1 < \alpha + \sqrt{3}\beta \le 1 \quad \text{or} \quad \alpha = 0
\end{align}
である． $\alpha, \beta$ に\cref{eq:8}を代入して
\begin{align}
  \begin{cases}
    -1 < (s + \sqrt{3}c)^2 \le 1 \\
    s = 0 \\
    s + \sqrt{3}c = 0
  \end{cases}
\end{align}
ここで第三式は第一式の場合に含まれるから
\begin{align}
  \begin{cases}
    -1 \le s + \sqrt{3}c \le 1 \\
    s = 0
  \end{cases}
  \begin{cases}
    -1 \le 2\sin\left(\theta+\dfrac{\pi}{3}\right) \le 1 \\
    s = 0
  \end{cases}
  \begin{cases}
    \dfrac{5}{6}\pi \le \theta + \dfrac{\pi}{3} \le \dfrac{7}{6}\pi \\
    \dfrac{11}{6}\pi \le \theta + \dfrac{\pi}{3} \le \dfrac{13}{6}\pi \\
    \theta = 0, \pi
  \end{cases}
  \begin{cases}
    \dfrac{1}{2}\pi \le \theta \le \dfrac{5}{6}\pi \\
    \dfrac{3}{2}\pi \le \theta \le \dfrac{11}{6}\pi \\
    \theta = 0, \pi
  \end{cases}
\end{align}
である．

収束値は\cref{eq:10}において$\alpha+\sqrt{3}\beta=1$かどうかで異なるので以下場合分けする．

\subparagraph{$\alpha+\sqrt{3}\beta\neq 1$即ち $\dfrac{1}{2}\pi < \theta < \dfrac{5}{6}\pi, \ \dfrac{3}{2}\pi < \theta < \dfrac{11}{6}\pi, \ \theta = 0, \pi$ の時}

この時は$x_n, y_n$ 共に 0 に収束する．

\subparagraph{$\alpha+\sqrt{3}\beta=1$即ち $\theta = \dfrac{1}{2}\pi, \dfrac{5}{6}\pi, \dfrac{3}{2}\pi, \dfrac{11}{6}\pi$ の時}

\cref{eq:10}から$x_n \to \alpha, \ y_n \to -\sqrt{3}\alpha$ となるから，$\alpha = s(s + \sqrt{3}c)$ において$\alpha$の値で場合分けして
\begin{align}
  \begin{cases}
    \theta = \dfrac{1}{2}\pi, \dfrac{3}{2}\pi \text{の時，} x_n \to 1, \ y_n \to -\sqrt{3} \\
    \theta = \dfrac{5}{6}\pi, \dfrac{11}{6}\pi \text{の時，} x_n \to -\dfrac{1}{2}, \ y_n \to \dfrac{\sqrt{3}}{2}
  \end{cases}
\end{align}
である．
\casesend

以上をまとめて求める極限値は
\begin{align}
  \begin{cases}
    \theta = 0, \ \dfrac{1}{2}\pi < \theta < \dfrac{5}{6}\pi, \ \theta = \pi, \ \dfrac{3}{2}\pi < \theta < \dfrac{11}{6}\pi \text{ の時}, & (x_n, y_n) \longrightarrow (0, 0) \\
    \theta = \dfrac{1}{2}\pi, \ \dfrac{3}{2}\pi & (x_n, y_n) \longrightarrow (1, -\sqrt{3}) \\
    \theta = \dfrac{5}{6}\pi, \ \dfrac{11}{6}\pi & (x_n, y_n) \longrightarrow \left(-\dfrac{1}{2}, \dfrac{\sqrt{3}}{2}\right)
  \end{cases}
\end{align}
である．
