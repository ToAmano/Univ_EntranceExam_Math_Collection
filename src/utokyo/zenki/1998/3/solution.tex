円 $C_n$ の中心 $O_n$ とする． $O_{n+2}$ から $x = x_n, x_{n+1}$ への垂足を各 $H_n, H_{n+1}$ とおく．$O_{n+1}$から$x=x_n$への垂足を$I_{n}$とおく．
$\triangle O_n H_n O_{n+2}, \triangle O_{n+1} H_{n+1} O_{n+2}, \triangle O_nO_{n+1}I_{n}$ に三平方を用いて，
\begin{align}
  &
  \begin{cases}
    (r_{n+2} + r_n)^2     = (x_n - x_{n+2})^2 + (r_{n+2} - r_n)^2 \\
    (r_{n+2} + r_{n+1})^2 = (x_{n+1} - x_{n+2})^2 + (r_{n+2} - r_{n+1})^2 \\
    (r_{n+1} + r_{n})^2   = (x_{n}-x_{n+1})^2 + (r_{n+1}-r_{n})^2
  \end{cases} \\
  &
  \begin{cases}
    2\sqrt{r_n r_{n+2}}     = |x_n - x_{n+2}| \\
    2\sqrt{r_{n+1} r_{n+2}} = |x_{n+2} - x_{n+1}| \\
    2\sqrt{r_n r_{n+1}}     = |x_{n}-x_{n+1}|
  \end{cases}
  \quad (\because r_k > 0)
\end{align}
である．
図から $n$: even の時 $x_n < x_{n+2} < x_{n+1}$， $n$: odd の時 $x_{n+1} < x_{n+2} < x_n$ であることより，
\begin{align}
  \begin{cases}
    2\sqrt{r_n r_{n+2}}     = (-1)^n (x_{n+2} - x_n) \\
    2\sqrt{r_{n+1} r_{n+2}} = (-1)^n (x_{n+1} - x_{n+2}) \\
    2\sqrt{r_n r_{n+1}}     = (-1)^n (x_{n+1} - x_n)
  \end{cases} \label{eq:1}
\end{align}
である．

\paragraph{(1)}

$q_n \in \mathbb{Z}$ を数学的帰納法で示す．
題意より$n=0, 1$ の時は
\begin{align}
  r_0 = r_1 = \dfrac{1}{2} \\
  \therefore
  q_0 = q_1 = 1 \label{eq:2}
\end{align}
より整数だから，以下 $n=k, k+1 \ (k \in \mathbb{Z}_{\ge 0})$ での成立を仮定する．
\cref{eq:1}で第一式と第二式を足して第三式を引くと
\begin{align}
  2\sqrt{r_k r_{k+2}} + 2\sqrt{r_{k+1} r_{k+2}} &= 2\sqrt{r_k r_{k+1}}
\end{align}
である．$q_n = \dfrac{1}{2\sqrt{r_n}}$ を代入して
\begin{align}
  \dfrac{1}{q_k q_{k+2}} + \dfrac{1}{q_{k+1} q_{k+2}} &= \dfrac{1}{q_k q_{k+1}}
\end{align}
である．
両辺に $q_k q_{k+1} q_{k+2}$ をかけて，
\begin{align}
  q_{k+2} &= q_k + q_{k+1} \label{eq:3}
\end{align}
で，仮定より右辺が整数だから$n=k+2$でも$q_n$は整数である．

以上から数学的帰納法により題意は示された． \hfill $\blacksquare$

\paragraph{(2)}

まず$p_n$の満たす漸化式を求める．
\cref{eq:1}の第一式，第二式の両辺を左辺で割って
\begin{align}
  \begin{cases}
    1  = (-1)^n \left(\dfrac{x_{n+2}}{2\sqrt{r_n r_{n+2}}} - \dfrac{x_n}{2\sqrt{r_n r_{n+2}}}\right) \\
    1 =  (-1)^n\left(\dfrac{x_{n+1}}{2\sqrt{r_{n+1} r_{n+2}}}-\dfrac{x_{n+2}}{2\sqrt{r_{n+1} r_{n+2}}}\right)
  \end{cases}
\end{align}
である．ここに$q_n=\dfrac{1}{\sqrt{2r_n}},p_n=\dfrac{x_n}{\sqrt{2r_n}}$を代入して
\begin{align}
  \begin{cases}
    (-1)^n = (q_n p_{n+2} - q_{n+2} p_n)  \\
    (-1)^n = (q_{n+2} p_{n+1} - q_{n+1} p_{n+2})
  \end{cases}
\end{align}
だから，辺々引き算して
\begin{align}
  q_{n+2} p_{n+1} - q_{n+1} p_{n+2} &= q_n p_{n+2} - q_{n+2} p_n \\
  \therefore
  q_{n+2} (p_{n+1}+p_{n}) &= p_{n+2}(q_{n+1}+q_{n})
\end{align}
である．\cref{eq:3}を代入して$q_n+q_{n+1}>0$で両辺を割って
\begin{align}
  p_{n+2} = p_{n+1}+p_{n} \label{eq:4}
\end{align}
である．
初期条件は$n=0,1$の時$p_0=0,p_1=1$であり，\cref{eq:4}から$p_2=1$である．
\cref{eq:3}より$p_n,q_n$は同じ漸化式に従い，初期条件は$q_0=q_1=1,p_1=p_2=1$だから
\begin{align}
  p_{n+1} = q_n \label{eq:5}
\end{align}
である．よって(1)および$p_0=0$と合わせて，$p_n\in \mathbb{Z}$である．

以下$p_n$ と $q_n$ が互いに素であることを巣学的帰納法で示す．
$n=0, 1$ の時 $p_0 = 0, p_1 = 1$ となって成立するので， 以下 $n=k, k+1 \ (k \in \mathbb{Z}_{\ge 0})$ での成立を仮定する．

\cref{eq:3,eq:5}より
\begin{align}
  \text{$p_{k+2}$ と $q_{k+2}$ が互いに素」}\\
  \iff
  \text{「$q_{k+1}$ と $q_{k+2}$ が互いに素」} \quad (\because\text{\cref{eq:5}})\\
  \iff
  \text{「$q_{k+1}$ と $q_{k+1} + q_k$ が互いに素」} \quad (\because\text{\cref{eq:3}})\\
  \iff
  \text{「$q_{k+1}$ と $q_k$ が互いに素」} \\
  \iff
  \text{「$q_{k+1}$ と $p_{k+1}$ が互いに素」} \quad (\because\text{\cref{eq:5}})
\end{align}
となり，最終行は仮定より成立するから $q_{k+2}$と$p_{k+2}$も互いに素である．

以上から数学的帰納法により題意は示された．\hfill $\blacksquare$

\paragraph{(3)}

\cref{eq:5}および$(p_0,q_0)=(0,1)$より$x_n=\dfrac{p_n}{q_n}$は
\begin{align}
  x_n &=
  \begin{cases}
    0 & (n = 0) \\
    \dfrac{q_{n-1}}{q_n} & (n \ge 1)
  \end{cases} \label{eq:6}
\end{align}
となる．

従って\cref{eq:3}の両辺を $q_{k+1}$ でわって，
\begin{align}
  \dfrac{q_{k+2}}{q_{k+1}} &= \dfrac{q_k}{q_{k+1}}+1 \\
  \therefore
  \dfrac{1}{x_{n+2}} &= x_{n+1} + 1 \quad (n \ge 0) \\
  \therefore
  x_{n+1} &= \dfrac{1}{1 + x_n} \quad (n \ge 1)
\end{align}
である． $\alpha = \dfrac{1}{1 + \alpha}$ と辺々引き算して
\begin{align}
  x_{n+1} - \alpha &= \dfrac{1}{1 + x_n} - \dfrac{1}{1 + \alpha} \\
  &= \dfrac{-1}{(1 + \alpha)(1 + x_n)} (x_n - \alpha)
\end{align}
である．ここで\cref{eq:3,eq:6}より$0\le x_n \le 1$だから，両辺絶対値をとって
\begin{align}
  |x_{n+1} - \alpha|
  &= \left| \dfrac{1}{(1 + \alpha)(1 + x_n)} \right| |x_n - \alpha| \\
  &\le  \left| \dfrac{1}{(1 + \alpha)} \right| |x_n - \alpha| \\
  &=   \left| \alpha \right| |x_n - \alpha| \\
\end{align}
なので，$n=1, 2,\cdots,$とくり返し用いて，
\begin{align}
  |x_n - \alpha| &< \left(\alpha\right)^{n-1} |x_1 - \alpha|
\end{align}
である．$0<\alpha<1$より右辺は$n\to\infty$で$0$に収束するから，
はさみうちの定理より
\begin{align}
  \lim_{n\to\infty} = \alpha = \dfrac{-1 + \sqrt{5}}{2}
\end{align}
である．
