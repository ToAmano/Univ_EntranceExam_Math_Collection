{\bf [解]} $S_1 - \int_1^a \log x \, dx = a(\log a - 1) + 1 \quad \dots \text{① である．}$

\bigskip

\paragraph{(1)}

$\square ABCD$ の面積を $S'$ とすると，
\begin{align}
S' &= \triangle ABD D' + \triangle D' B C \\
&= \dfrac{1}{2}(\log a + \log b)(a - b) + \dfrac{1}{2} \log b \cdot (b - 1)
\end{align}
図より明らかに $S' < S_1$ だから，$S'$ の最大値こそ $S_2$ である．

\begin{align}
\dfrac{d}{db} S' = \dfrac{1}{2b}(a - 1) - \dfrac{1}{2} \log a
\end{align}
より，極大を与える $\Bigl(\because \dfrac{a-1}{\log a} < a\Bigr)$

従って，
\begin{align}
S_2 &= S'\Big|_{b = \frac{a-1}{\log a}} \\
&= \dfrac{1}{2}(a \log a - a + 1) + \dfrac{1}{2} \dfrac{a-1}{\log a} \log\Bigl(\dfrac{a-1}{\log a}\Bigr)
\end{align}

\paragraph{(2)}
\begin{align}
\dfrac{S_2}{S_1} &= \dfrac{\dfrac{1}{2}(a \log a - a + 1) + \dfrac{1}{2}(a-1)\{\log(a-1) - \log\log a\}}{a \log a - a + 1} \\
&= \dfrac{1}{2} + \dfrac{1}{2} \underbrace{\dfrac{(a-1)\{\log(a-1) - \log\log a\}}{a \log a - a + 1}}_{T}
\end{align}

\begin{align}
T = \dfrac{\Bigl(1 - \dfrac{1}{a}\Bigr) \Bigl\{ 1 + \dfrac{\log(1 - \dfrac{1}{a})}{\log a} - \dfrac{\log\log a}{\log a} \Bigr\}}{1 - \dfrac{1}{\log a} + \dfrac{1}{a \log a}} \longrightarrow 1 \quad (a \to \infty)
\end{align}
\begin{align}
\Bigl[ \log a = A \to \infty \text{ より } \dfrac{\log(\log a)}{\log a} = \dfrac{\log A}{A} \to 0 \Bigr]
\end{align}
より，
\begin{align}
\dfrac{S_2}{S_1} \longrightarrow 1 \quad (a \to \infty) \text{ である．}
\end{align}
