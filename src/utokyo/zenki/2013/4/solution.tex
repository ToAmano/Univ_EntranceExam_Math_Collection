$\angle APB = \alpha, \angle APC = \beta$ とおく．
上図のように $P$ を原点とし $A$ を $x$ 軸正の方向とする座標をとる．
$P$ が $\triangle ABC$ の内部にあるから，$C$ が $y \ge 0$ にくるようにする．
この時 $\overrightarrow{PA}, \overrightarrow{PB}, \overrightarrow{PC}$ に平行な単位ベクトルを各々 $\vec{a}, \vec{b}, \vec{c}$ として，パラメータ$\alpha,\beta$を用いて
\begin{align}
  \vec{a} =
  \begin{pmatrix} 1 \\ 0
  \end{pmatrix}, \\
  \quad \vec{b} =
  \begin{pmatrix} \cos(-\alpha) \\ \sin(-\alpha)
  \end{pmatrix} =
  \begin{pmatrix} \cos(\alpha) \\ -\sin(\alpha)
  \end{pmatrix}, \\
  \quad \vec{c} =
  \begin{pmatrix} \cos\beta \\ \sin\beta
  \end{pmatrix}
\end{align}
と書ける．題意から $\vec{a} + \vec{b} + \vec{c} = \vec{0}$ だから，
\begin{align}
  \begin{cases}
    1 + \cos\alpha + \cos\beta = 0 \\
    \sin\alpha = \sin\beta
  \end{cases} \label{eq:1}
\end{align}
である．$\cos^2\alpha + \sin^2\alpha = 1$ に代入して
\begin{align}
  2 + 2\cos\beta = 1 \quad \therefore \cos\beta = -\dfrac{1}{2} \label{eq:2}
\end{align}
である．
$0 \le \beta \le \pi$ から $\beta = \dfrac{2}{3}\pi$ で，\cref{eq:1}から $\alpha = \dfrac{2}{3}\pi$ が従う．よって，
\begin{align}
  \angle APB = \dfrac{2}{3}\pi, \quad \angle APC = \dfrac{2}{3}\pi
\end{align}
である．

\paragraph{(2)}

$|\overrightarrow{PA}| = a, |\overrightarrow{PB}| = b, |\overrightarrow{PC}| = c$ とすると，
\begin{align}
  \overrightarrow{PA} = a
  \begin{pmatrix} 1 \\ 0
  \end{pmatrix}, \\
  \quad \overrightarrow{PB} = -\dfrac{b}{2}
  \begin{pmatrix} 1 \\ \sqrt{3}
  \end{pmatrix}, \\
  \quad \overrightarrow{PC} = \dfrac{c}{2}
  \begin{pmatrix} -1 \\ \sqrt{3}
  \end{pmatrix}
\end{align}
である．したがって，
\begin{align}
  \overrightarrow{AB} =
  \begin{pmatrix} -\dfrac{1}{2}b - a \\ -\dfrac{\sqrt{3}}{2}b
  \end{pmatrix},
  \quad \overrightarrow{AC} =
  \begin{pmatrix} -\dfrac{1}{2}c - a \\ \dfrac{\sqrt{3}}{2}c
  \end{pmatrix} \label{eq:3}
\end{align}
である．
$\angle CAB = \dfrac{\pi}{2}$より$\overrightarrow{AC}\cdot \overrightarrow{AB} = 0$であり，また $|AB| = 1, |AC| = \sqrt{3}$ だから\cref{eq:3}を代入して
\begin{align}
  \begin{cases}
    (-\dfrac{1}{2}b - a)(-\dfrac{1}{2}c - a) - \dfrac{3}{4}bc = 0 \\
    (\dfrac{1}{2}b + a)^2 + \dfrac{3}{4}b^2 = 1 \\
    (\dfrac{1}{2}c + a)^2 + \dfrac{3}{4}c^2 = 3
  \end{cases}
  \iff
  \begin{cases}
    a^2 + \dfrac{1}{2}(ac + ab - bc) = 0 \\
    a^2 + b^2 + ab = 1 \\
    a^2 + c^2 + ac = 3
  \end{cases} \label{eq:4}
\end{align}
である．第一式から$c\neq a$の時
\begin{align}
  b = \dfrac{a(2a+c)}{c-a} \label{eq:5}
\end{align}
だから第二式に代入して
\begin{align}
  a^2 + \dfrac{a^2(2a+c)^2}{(c-a)^2} + \dfrac{a^2(2a+c)}{c-a} = 1 \\
  a^2(c-a)^2 + a^2(2a+c)^2 + a^2(2a+c)(c-a) = (c-a)^2 \\
  a^2\left[ 3a^2 + 3ac + 3c^2 \right] = (c-a)^2 \label{eq:6}
\end{align}
である．\cref{eq:4}の第三式を代入して
\begin{align}
  9a^2 = (c-a)^2 \quad
  \therefore
  3a = \pm(c-a) \quad
  \therefore
  c= 4a, -2a
\end{align}
を得る．$a,c>0$より
\begin{align}
  c=4a \label{eq:7}
\end{align}
であるから\cref{eq:5}に代入して
\begin{align}
  b = \dfrac{6a^2}{3a} = 2a \label{eq:8}
\end{align}
である．これを\cref{eq:4}の第二式に代入して
\begin{align}
  7a^2 = 1 \quad \therefore a = \dfrac{\sqrt{7}}{7}
\end{align}
だから\cref{eq:7,eq:8}より
\begin{align}
  b = \dfrac{2\sqrt{7}}{7}, c = \dfrac{4\sqrt{7}}{7}
\end{align}
である．まとめて
\begin{align}
  a = \dfrac{\sqrt{7}}{7}, \quad
  b = \dfrac{2\sqrt{7}}{7}, \quad
  c = \dfrac{4\sqrt{7}}{7}
\end{align}
である．
