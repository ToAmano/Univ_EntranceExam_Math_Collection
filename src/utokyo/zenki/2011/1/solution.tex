Pから $l$ に下ろした垂足を H とすると，$\triangle PQR$の面積$S(a)$は
\begin{align}
  S(a)
  = \dfrac{1}{2} |PH||QR|
  = \dfrac{1}{2} |PH|\cdot 2|HQ|
  = |PH||HQ| \label{eq:2}
\end{align}
である．

点と直線の距離公式から
\begin{align}
  |PH| = \dfrac{|a-1|}{\sqrt{a^2+1}} \label{eq:3}
\end{align}
だから，$\triangle PQH$に三平方の定理を用いて
\begin{align}
  |HQ| = \sqrt{|PQ|^2-|PH|^2} = \sqrt{1 - \dfrac{(a-1)^2}{a^2+1}} = \sqrt{\dfrac{2a}{a^2+1}} \label{eq:4}
\end{align}
である．\cref{eq:3,eq:4}を\cref{eq:2}に代入して
\begin{align}
  S(a) = \dfrac{\sqrt{2a}|a-1|}{a^2+1} \label{eq:5}
\end{align}
である．

\paragraph{(2)}

$0 < a < 1$ の時\cref{eq:5}で絶対値を外して
\begin{align}
  S(a) = \dfrac{\sqrt{2a}(1-a)}{a^2+1} = \sqrt{\dfrac{2a(1-a)^2}{(a^2+1)^2}}
\end{align}
である．根号の中身を$f(a)$とおく：
\begin{align}
  f(a) = \dfrac{2a(1-a)^2}{(a^2+1)^2}
\end{align}
と，一階微分は
\begin{align}
  f'(a)
  &= 2 \dfrac{ (3a^2-4a+1)(a^2+1)^2 - a(1-a)^2 \cdot 2(a^2+1) \cdot 2a }{(a^2+1)^4} \\
  &= \dfrac{2}{(a^2+1)^3} \left[ (3a-1)(a-1)(a^2+1) - 4a^2(1-a)^2 \right] \\
  &= \dfrac{2(1-a)}{(a^2+1)^2} (a+1) (a^2-4a+1)
\end{align}
だから，$f$および$S$の増減表は以下のようになる．
