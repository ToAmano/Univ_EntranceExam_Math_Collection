\paragraph{【新課程】}

\subparagraph{(1)}

直線$l$の方向ベクトルを$\vec{l}$，直線$m$の方向ベクトルを$\vec{m}$とすると，
題意より
\begin{align}
  \vec{l} &=
  \begin{pmatrix} 2-1 \\ 1-1 \\ 1-0
  \end{pmatrix}
  =
  \begin{pmatrix} 1 \\ 0 \\ 1
  \end{pmatrix} \\
  \vec{m} &=
  \begin{pmatrix} 1-1 \\ 3-1 \\ 2-1
  \end{pmatrix}
  =
  \begin{pmatrix} 0 \\ 2 \\ 1
  \end{pmatrix}
\end{align}
である．
これらに垂直なベクトルとして
\begin{align}
  \vec{p} =
  \begin{pmatrix} -2 \\ -1 \\ 2
  \end{pmatrix}
\end{align}
が取れる．

求める平面$\pi$は直線$l$を含むので点$(1,1,0)$を通り，かつ$\vec{p}$を法線ベクトルとするから，平面$\pi$の方程式は
\begin{align}
  -2(x-1) - (y-1) + 2(z-0) = 0 \iff -2x - y + 2z + 3 = 0 \quad \cdots\text{（答）}
\end{align}
である．

\subparagraph{(2)}

$n$上の点$X$をパラメータ$t$および定数$a,b,c$を用いて
\begin{align}
  \overrightarrow{OX} =
  \begin{pmatrix} 2 \\ 0 \\ 1
  \end{pmatrix}
  + t
  \begin{pmatrix} a \\ b \\ c
  \end{pmatrix} \label{eq:1}
\end{align}
とかく．これが$\ell,m$と交わるので，(1)から
\begin{align}
  \overrightarrow{OX} =
  \begin{pmatrix} 1 \\ 1 \\ 0
  \end{pmatrix}
  + s
  \begin{pmatrix} 1 \\ 0 \\ 1
  \end{pmatrix} \label{eq:2} \\
  \overrightarrow{OX} =
  \begin{pmatrix} 1 \\ 1 \\ 1
  \end{pmatrix}
  + u
  \begin{pmatrix} 0 \\ 2 \\ 1
  \end{pmatrix} \label{eq:3}
\end{align}
を満たす$s,u\in\mathbb{R}$がある．
\cref{eq:1}においてそれぞれ対応する$t$を$t_1,t_2$として
\begin{align}
  \begin{cases}
    2+at_1 = 1 + s \\
    bt_1 = 1 \\
    1+ct_1 = s \label{eq:4}
  \end{cases}
\end{align}
および
\begin{align}
  \begin{cases}
    2+at_2 = 1  \\
    bt_2   = 1 + 2u \\
    1+ct_2 = 1 + u
  \end{cases} \label{eq:5}
\end{align}
を満たす．このような$s,u,t_1,t_2$が存在するような$a,b,c$について考える．
まず\cref{eq:4}において第一式と第三式から$s$を消去して
\begin{align}
  2+at_1 = 1 + (1+ct_1) \quad \therefore (a-c)t_1 = 0
\end{align}
より$a=c$または$t_1=0$である．$t_1=0$は第二式を満たさず不適だから
\begin{align}
  a=c \label{eq:6}
\end{align}
である．次に\cref{eq:5}において第一式と第三式に\cref{eq:6}を代入して
\begin{align}
  at_2 = -1 = u
\end{align}
だから，これを第二式に代入して$u$を消去すると
\begin{align}
  bt_2 = 1 + 2(-1) = -1
\end{align}
だから，
\begin{align}
  at_2 = bt_2 = -1
\end{align}
となり，
\begin{align}
  a = b \label{eq:7}
\end{align}
である．\cref{eq:6,eq:7}より$a=b=c$が必要であり，この時$s=2,u=-1$として\cref{eq:4,eq:5}は満たされる．よって求める交点の座標は
\begin{align}
  l \text{ と } n \text{ の交点: } (3, 1, 2), \qquad
  m \text{ と } n \text{ の交点: } (1, -1, 0) \quad \cdots\text{（答）}
\end{align}
である．

\paragraph{【旧課程】}
\paragraph{(1)}
与えられた関数
\begin{align}
  f(x) = -\cos x - \dfrac{a}{4}\cos 2x \quad (0\le x \le 2\pi) \label{eq:8}
\end{align}
について考える．
倍角公式$\cos 2x = 2\cos^2 x - 1$ より$t=\cos x$とおくと$f(x)$は$t$の二次関数：
\begin{align}
  f(x)
  & =  -\cos x - \dfrac{a}{4}\left(2\cos^2 x - 1\right) \\
  & =  -\dfrac{a}{2}t^2 - t + \dfrac{a}{4} \label{eq:9}
\end{align}
になる．$0 \leqq x \leqq 2\pi$ より $-1 \leqq t \leqq 1$ だから，この範囲での$f(x)$の最大最小を考えれば良い．平方完成して
\begin{align}
  f(x)
  &= -\dfrac{a}{2}\left( t + \dfrac{1}{a} \right)^2 + \dfrac{1}{2a} + \dfrac{a}{4}
\end{align}
だから，この放物線は上に凸で，軸は直線 $t = -\dfrac{1}{a} < 0$ である．
$a>0$の大きさによって場合分けする．

\paragraph{(2)}
$f(x)$ を$x$で微分して倍角公式を用いると
\begin{align}
  f'(x) &= \sin x + \dfrac{a}{2}\sin 2x = \sin x (1 + a\cos x) \\
  f''(x) &= \cos x + a\cos 2x = 2a\cos^2 x + \cos x - a
\end{align}
である．$t = \cos x$（$-1 \leqq t \leqq 1$）とおき，
\begin{align}
  h(t) = 2at^2 + t - a
\end{align}
とする．$2$次方程式 $h(t) = 0$ の判別式は $D = 1 + 8a^2 > 0$ であるから，相異なる$2$つの実数解
\begin{align}
  \alpha = \dfrac{-1 - \sqrt{1+8a^2}}{4a}, \qquad \beta = \dfrac{-1 + \sqrt{1+8a^2}}{4a} \quad (\alpha < \beta)
\end{align}
をもつ．
ここで
\begin{align}
  h(0) = -a < 0, \qquad h(1) = a + 1 > 0
\end{align}
であるから，常に $0 < \beta < 1$ である．
一方，$\alpha$ については
\begin{align}
  h(-1) = 2a(-1)^2 + (-1) - a = a - 1
\end{align}
であるから，$a$ の値によって次のように場合分けされる．

$h(-1) = a - 1 < 0$ より $\alpha < -1$ である．
したがって，$-1 \leqq t \leqq 1$ において $h(t) = 0$ となるのは $t = \beta$ のみである．
