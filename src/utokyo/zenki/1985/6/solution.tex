\begin{align}
  AB &= 8 \\
  BC &= \sqrt{(8-6)^2+(2\sqrt{3})^2} = \sqrt{4+12} = 4 \\
  CA &= \sqrt{6^2+(2\sqrt{3})^2} = \sqrt{36+12} = 4\sqrt{3}
\end{align}
である．$BC^2+CA^2=16+48=64=AB^2$だから，三角形$ABC$は$\angle C=90°$の直角三角形であり，
$\sin A=BC/AB=1/2$，$\sin B=CA/AB=\sqrt{3}/2$から$\angle A=30°$，$\angle B=60°$である．この様子を下図に示す．

下図のように，点$P$が辺上を動くときにできる半径$1$の円の通過部分の面積$S$は，三角形$ABC$から各辺を内側に距離$1$だけ移した小さい相似な三角形を除いた部分($\triangle P$）と，半径$1$の円（各頂点における扇形の中心角の和は外角の和$2\pi$に等しいから，合わせてちょうど1つの円になる，$\bigcirc Q$）と，3辺に沿った3つの長方形($\square X, \square Y, \square Z$)に分けられる．すなわち

  最後に$\triangle P$について考える．上図で内側の小さい三角形$A'B'C'$について考えると
  \begin{align}
    \begin{cases}
      |A'C'| = |AC| - 1 - \dfrac{1}{\sin \pi/6} - \dfrac{1}{\tan \pi/6} = 4\sqrt{3}-(3+\sqrt{3}) = 3\sqrt{3} - 3\\
      |B'C'| = |BC| - 1 - \dfrac{1}{\sin \pi/3} - \dfrac{1}{\tan \pi/3} = 4 - \left(1 +\dfrac{2}{\sqrt{3}} + \dfrac{1}{\sqrt{3}}\right) = 4- \left(1+\sqrt{3}\right)= 3 - \sqrt{3}
    \end{cases}
  \end{align}
  だから，$\triangle P$は$\triangle ABC$から$\triangle A'B'C'$の面積を引いたもので
  \begin{align}
    \triangle P
    &= \triangle ABC - \triangle A'B'C' \\
    &= \dfrac{|AC||CB|}{2} - \dfrac{|A'C'||C'B'|}{2} \\
    &= \dfrac{16\sqrt{3}}{2} - \dfrac{(3\sqrt{3}-3)(3-\sqrt{3})}{2} \\
    &= 8\sqrt{3} - (-9+6\sqrt{3}) \\
    &= 9+2\sqrt{3} \label{eq:4}
  \end{align}
  である．

  \cref{eq:2,eq:3,eq:4}を\cref{eq:1}に代入して求める面積は
  \begin{align}
    S &= (9+2\sqrt{3}) + \pi + (12+4\sqrt{3}) \\
    &= 21+6\sqrt{3}+\pi \quad \cdots\text{(答)}
  \end{align}
  である．

  (1)と同様に図形をラベルすると，それぞれの面積は
  \begin{align}
    \bigcirc Q = s^2 \pi
  \end{align}
  \begin{align}
    \square X+\square Y+\square Z = s\left(12+4\sqrt{3}\right)
  \end{align}
  \begin{align}
    \triangle P
    &= 8\sqrt{3} - \dfrac{1}{2}\left[4\sqrt{3}-\left(3+\sqrt{3}\right)s\right]\left[4 - \left(1+\sqrt{3}\right)s\right] \\
    &= 2\sqrt{3}\left(1+\sqrt{3}\right)s + 2\left(3+\sqrt{3}\right)s - \dfrac{1}{2}\left(3+\sqrt{3}\right)\left(1+\sqrt{3}\right)s^2 \\
    &= 4\sqrt{3}\left(1+\sqrt{3}\right)s - \left(3+2\sqrt{3}\right)s^2
  \end{align}
  だから，
  \begin{align}
    S(z)
    &= - \left(3+2\sqrt{3}\right)s^2 + 4\sqrt{3}\left(1+\sqrt{3}\right)s + s^2 \pi + s\left(12+4\sqrt{3}\right) \\
    &= - \left(3+2\sqrt{3}\right)s^2 + 8\sqrt{3}\left(1+\sqrt{3}\right)s + s^2 \pi \label{eq:5}
  \end{align}
  である．

  もとめる体積$V$は対称性（$S(z)$は$z$の偶関数）から
  \begin{align}
    \dfrac{1}{2}V = \int_0^1 S(z)\,dz \label{eq:6}
  \end{align}
  である．

  $z=c=\cos\theta$と置換すると，$dz=-s\,d\theta$であり，$z:0\to1$は$\theta:\dfrac{\pi}{2}\to0$に対応するから，\cref{eq:5,eq:6}より
  \begin{align}
    \dfrac{1}{2}V = \int_0^{\frac{\pi}{2}} \left[\left\{-(3+2\sqrt{3})+\pi\right\}s^2+2(12+4\sqrt{3})s \right]s\,d\theta \label{eq:7}
  \end{align}
  となる．各項を個別に計算すると
  \begin{align}
    (\text{第一項})
    &= \int_0^{\frac{\pi}{2}} s^3\,d\theta \\
    &= \int_0^{\frac{\pi}{2}} \dfrac{1}{4}(3s-\sin3\theta)\,d\theta \\
    &= \left[-\dfrac{3}{4}c+\dfrac{1}{12}\cos3\theta\right]_0^{\frac{\pi}{2}} \\
    &= \dfrac{3}{4}-\dfrac{1}{12} \\
    &= \dfrac{2}{3} \\
    (\text{第二項})
    &= \int_0^{\frac{\pi}{2}} s^2\,d\theta \\
    &= \dfrac{1}{2}\left[\theta-\dfrac{1}{2}\sin2\theta\right]_0^{\frac{\pi}{2}} \\
    &= \dfrac{\pi}{4}
  \end{align}
  だから，\cref{eq:7}に代入して
  \begin{align}
    \dfrac{1}{2}V
    = (\pi-3-2\sqrt{3})\cdot\dfrac{2}{3} + 2(12+4\sqrt{3})\cdot\dfrac{\pi}{4}
  \end{align}
  となり，
  \begin{align}
    V
    &= \dfrac{4}{3}(\pi-3-2\sqrt{3}) + (12+4\sqrt{3})\pi \\
    &= 4\left(\dfrac{10}{3}+\sqrt{3}\right)\pi - 4\left(1+\dfrac{2}{3}\sqrt{3}\right) \quad \cdots\text{(答)}
  \end{align}
  である．
