{\bf [解]} $P(\alpha,\alpha^2)$，$Q(\beta,\beta^2)$（$\alpha<\beta\ \cdots①$）とおく．
\begin{align}
PQ:y=(\alpha+\beta)x-\alpha\beta
\end{align}
だから，
\begin{align}
m&=\alpha+\beta\\
L&=\sqrt{(\beta-\alpha)^2+(\beta^2-\alpha^2)^2}=(\beta-\alpha)\sqrt{1+(\alpha+\beta)^2}\quad(\because①)
\end{align}
$\cdots②$である．$PQ$の中点$M\left(\dfrac{\alpha+\beta}{2},\dfrac{\alpha^2+\beta^2}{2}\right)$から
\begin{align}
h=\dfrac{\alpha^2+\beta^2}{2}\quad\cdots③
\end{align}
である．②から，$\beta-\alpha=\dfrac{L}{\sqrt{1+m^2}}$だから，
\begin{align}
\alpha^2+\beta^2=\dfrac{1}{2}\left[(\alpha+\beta)^2+(\beta-\alpha)^2\right]=\dfrac{1}{2}\left(m^2+\dfrac{L^2}{1+m^2}\right)
\end{align}
なので，③に代入して
\begin{align}
h=\dfrac{1}{4}\left(m^2+\dfrac{L^2}{1+m^2}\right)
\end{align}

\paragraph{(2)}

$\alpha\beta=\dfrac{1}{4}\left[(\alpha+\beta)^2-(\beta-\alpha)^2\right]=\dfrac{1}{4}\left(m^2-\dfrac{L^2}{1+m^2}\right)$だから，$\alpha,\beta$は$x$の2次式
\begin{align}
x^2-mx+\dfrac{1}{4}\left(m^2-\dfrac{L^2}{1+m^2}\right)=0
\end{align}
の2実解で，判別式を$D$として
\begin{align}
D=m^2-\left(m^2-\dfrac{L^2}{1+m^2}\right)=\dfrac{L^2}{1+m^2}>0
\end{align}
だから，$m$は任意の実数値をとる．$t=m^2$とおくと，したがって$t\ge0$である．
\begin{align}
4h=t+\dfrac{L^2}{1+t}=:f(t)
\end{align}
とすると
\begin{align}
f'(t)=1-\dfrac{L^2}{(1+t)^2}=\dfrac{t^2+2t+1-L^2}{(1+t)^2}
\end{align}
だから，$L$によって以下のようになる．

$1^\circ$ $0<L\le1$の時，$t\ge0$の時，$f'(t)\ge0$で，$f$は単調増加だから，$t=0$で最小，
\begin{align}
h_{\min}=\dfrac{1}{4}L^2
\end{align}

$2^\circ$ $1\le L$の時，$p=-1+L$として考える．$f'(t)$の符号は$t^2+2t+1-L^2=(t-p)(t+1+L)$の符号と一致し，$t\ge0>-1-L$だから，$f'$は$t=p$で符号が負から正に変わる．
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & $\cdots$ & $p$ & $\cdots$ \\
\hline
$f'$ & & $-$ & $0$ & $+$ \\
$f$ & & $\searrow$ & & $\nearrow$ \\
\end{tabular}
\end{center}
したがって，$t=p$で最小，$1+p=L$だから
\begin{align}
f(p)=p+\dfrac{L^2}{1+p}=(L-1)+\dfrac{L^2}{L}=(L-1)+L=2L-1
\end{align}
\begin{align}
h_{\min}=\dfrac{1}{4}(2L-1)
\end{align}

以上まとめて
\begin{align}
h_{\min}=\begin{cases}\dfrac{1}{4}L^2&(0<L\le1)\\[2mm]\dfrac{1}{4}(2L-1)&(1\le L)\end{cases}
\end{align}
