図のように球 $S$ に内接する球の列 $S_n$，$n=1,2,3,\cdots$ がある．
$S$ の中心 $O$ と $S_n$ の中心 $O_n$ はすべて同一平面上にあり，
$O_{n+1}$ は $S_n$ の表面上にあって，
この平面上において $O_{n+2}$ と $O_n$ は直線 $OO_{n+1}$ に関して互いに反対側にある．
また $S$ の半径は $a$，$S_n$ の半径は $\displaystyle\dfrac{a}{2^n}$ である．
このとき，

\begin{figure}[H]
  \centering
  \begin{tikzpicture}[scale=1.5, >=stealth, every node/.style={font=\small}]
    \coordinate (O) at (0,0);
    \draw[thick, gray!80!black] (O) circle (2.0);
    \node[above=2pt] at (O) {$O$};
    \node at (1.5, 1.7) {$S$};

    % 球 S_1: 半径 1.0, 中心 (0, -1.0)
    \coordinate (O1) at (0, -1.0);
    \draw[very thick, blue!85!black] (O1) circle (1.0);
    \node[circle, fill=blue!85!black, inner sep=1.2pt] at (O1) {}; \node[left=2pt] at (O1) {$O_1$};
    \node at (-0.4, -1.3) {$S_1$};

    % 球 S_2: 半径 0.5, 中心 (1.5*sin(41.41°), -1.5*cos(41.41°)) ≈ (0.9922, -1.125)
    % |O O2| = 1.5, |O1 O2| = 1.0
    \coordinate (O2) at (0.9922, -1.125);
    \draw[very thick, teal!85!black] (O2) circle (0.5);
    \node[circle, fill=teal!85!black, inner sep=1.2pt] at (O2) {}; \node[below right=1pt] at (O2) {$O_2$};
    \node at (1.1, -0.6) {$S_2$};

    % 球 S_3: 半径 0.25, 中心 (1.75*cos(-33.24°), 1.75*sin(-33.24°)) ≈ (1.464, -0.959)
    % |O O3| = 1.75, |O2 O3| = 0.5
    \coordinate (O3) at (1.464, -0.959);
    \draw[very thick, orange!90!black] (O3) circle (0.25);
    \node[circle, fill=orange!90!black, inner sep=1.0pt] at (O3) {}; \node[above right=1pt] at (O3) {$O_3$};
    \node at (1.55, -0.65) {$S_3$};

    \draw[dashed, gray!70!black] (O) -- (O1);
    \draw[dashed, gray!70!black] (O) -- (O2);
    \draw[dashed, gray!70!black] (O) -- (O3);
  \end{tikzpicture}

\begin{enumerate}
  \item $S_n$ と $S_{n+1}$ の共通部分の体積 $v_n$ を求めよ．
  \item $m=1,2,3,\cdots$ に対して，$\displaystyle V_m=\sum_{n=1}^m v_n$ とおくとき，
    $\displaystyle V=\lim_{m\to\infty}V_m$ を求めよ．
\end{enumerate}
