平面上に点列 $P_0,P_1,P_2,\cdots,P_n,\cdots$ があり，
$P_0$，$P_1$ の座標はそれぞれ $(0,0)$，$(1,0)$ である．
また，任意の自然数 $n$ に対し，
線分 $P_nP_{n+1}$ の長さは，線分 $P_{n-1}P_n$ の長さの2倍で，
半直線 $P_nP_{n+1}$ が半直線 $P_{n-1}P_n$ となす角は $120^\circ$ である．
$P_{3n}$ の座標を求めよ．

\begin{figure}[htbp]
  \centering
  \begin{tikzpicture}[scale=1.3, >=stealth, every node/.style={font=\small}]
    \draw[->, thick, gray!80!black] (-2.8, 0) -- (2.0, 0) node[right] {$x$};
    \draw[->, thick, gray!80!black] (0, -2.5) -- (0, 2.5) node[above] {$y$};
    \node[below left=2pt] at (0, 0) {$O$};

    \coordinate (P0) at (0, 0);
    \coordinate (P1) at (1, 0);
    \coordinate (P2) at (0, {sqrt(3)});
    \coordinate (P3) at (-2, {-sqrt(3)});

    % 折れ線
    \draw[line width=1.2pt, blue!85!black] (P0) -- (P1) -- (P2) -- (P3);

    % 方向の延長点線
    \draw[dashed, gray!70!black] (P1) -- ++(0.7, 0);
    \draw[dashed, gray!70!black] (P2) -- ++({0.7*cos(120)}, {0.7*sin(120)});

    % 角度マーク
    \draw[->, thick, orange!85!black] ($(P1)+(0.4, 0)$) arc (0:120:0.4);
    \node[orange!85!black, font=\footnotesize] at ($(P1)+(60:0.65)$) {$120^\circ$};

    \draw[->, thick, orange!85!black] ($(P2)+({0.4*cos(120)}, {0.4*sin(120)})$) arc (120:240:0.4);
    \node[orange!85!black, font=\footnotesize] at ($(P2)+(180:0.65)$) {$120^\circ$};

    % 点
    \node[circle, fill=blue!85!black, inner sep=1.3pt] at (P0) {};
    \node[circle, fill=blue!85!black, inner sep=1.3pt] at (P1) {};
    \node[below=2pt] at (P1) {$P_1$};
    \node[circle, fill=blue!85!black, inner sep=1.3pt] at (P2) {};
    \node[above right=1pt] at (P2) {$P_2$};
    \node[circle, fill=blue!85!black, inner sep=1.3pt] at (P3) {};
    \node[below left=1pt] at (P3) {$P_3$};
  \end{tikzpicture}
  \caption{点列 $P_0, P_1, P_2, P_3$ の配置}
  \label{fig:point_sequence}
\end{figure}
