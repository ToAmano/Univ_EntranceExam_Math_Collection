{\bf [解]}

与えられた放物線
\begin{align}
  \begin{cases}
    C_1:y=-x^2+1 \\
    C_2:y={(x-u)}^2+u
  \end{cases} \label{eq:1}
\end{align}
について考える．

\paragraph{(1)}

$C_1$と$C_2$が共有点を持つ時，$y$を消去した
\begin{align}
  & -x^2+1=(x-u)^2+u\\
  \therefore
  & 2x^2-2ux+u^2+u-1=0 \label{eq:2}
\end{align}
が実解を持つ．判別式を$D$として $D/4\ge0$ だから，
\begin{align}
  D/4=u^2-2(u^2+u-1)\ge0\iff-u^2-2u+2\ge0\iff-1-\sqrt{3}\le u\le-1+\sqrt{3}
\end{align}
となり，求める値は
\begin{align}
  (a,b)=(-1-\sqrt{3},-1+\sqrt{3})
\end{align}
である．

\paragraph{(2)}

与えられた式を$f(u)$として
\begin{align}
  f(u) = 2|x_1y_2-x_2y_1| \label{eq:3}
\end{align}
を求める．

$x_1,x_2$は\cref{eq:2}の$2$実数解である．
対称性から$x_1\le x_2$としてよい．

$(x_k,y_k)$は$C_1$上の点でもあるから
\begin{align}
  y_k=-x_k^2+1\quad(k=1,2) \label{eq:4}
\end{align}
なので\cref{eq:3}に代入して
\begin{align}
  \dfrac{1}{2}f(u)
  &=\left|x_1(1-x_2^2)-x_2(1-x_1^2)\right|\\
  &=\left|x_1-x_1x_2^2-x_2+x_1^2x_2\right|\\
  &=\left|(x_1-x_2)+(x_1-x_2)x_1x_2\right|\\
  &=|x_2-x_1||1+x_1x_2| \\
  \therefore
  f(u) &= 2|x_2-x_1||1+x_1x_2| \label{eq:5}
\end{align}
である．

ここで，$x_1\le x_2$だから，\cref{eq:2}の解と係数の関係および解の公式より
\begin{align}
  \begin{cases}
    x_2-x_1=\sqrt{-u^2-2u+2} \\
    x_1x_2=\dfrac{1}{2}(u^2+u-1)
  \end{cases}
\end{align}
である．\cref{eq:5}に代入して
\begin{align}
  f(u)
  &=2\sqrt{-u^2-2u+2}\left(1+\dfrac{u^2+u-1}{2}\right)\\
  &=\sqrt{-u^2-2u+2}\,(u^2+u+1) \label{eq:6}
\end{align}
である．

\paragraph{(3)}

$t=u+1$とおいて\cref{eq:6}に代入して
\begin{align}
  f(u)
  &=\sqrt{-(t-1)^2+3}\,\left((t-1)^2+(t-1)+1\right)\\
  &=\sqrt{-t^2+3}\,(t^2-t+1)
\end{align}
であるから，(1)の$(a,b)$の値を用いて求める定積分は
\begin{align}
  I
  &=\int_{a}^{b}f(u)\, du \\
  &=\int_{a+1}^{b+1}f(t)\, dt  \quad (t=u+1) \\
  &=\int_{-\sqrt{3}}^{\sqrt{3}}\sqrt{-t^2+3}\,(t^2-t+1)\,dt \\
  &=2\int_0^{\sqrt{3}}\sqrt{-t^2+3}\,(t^2+1)\,dt \label{eq:7}
\end{align}
である．ただし奇関数の積分が$0$になることを用いた．
以下各項を計算する．

まず第一項は$t=\sqrt{3}\sin\theta$とおくと$\dfrac{dt}{d\theta}=\sqrt{3}\cos\theta$だから，
\begin{align}
  (\text{第一項})
  &= \int_0^{\sqrt{3}}t^2\sqrt{-t^2+3}\,dt \\
  &=\int_0^{\pi/2}3\sin^2\theta\cdot\sqrt{3}\cos\theta\cdot\sqrt{3}\cos\theta\,d\theta\\
  &=9\int_0^{\pi/2}\dfrac{1}{4}\sin^22\theta\,d\theta\\
  &=\dfrac{9}{4}\cdot\dfrac{1}{2}\left[\theta-\dfrac{1}{4}\sin4\theta\right]_0^{\pi/2} \\
  &=\dfrac{9}{16}\pi
\end{align}
である．

第二項は，半径$\sqrt{3}$の四分円の面積に等しく，
\begin{align}
  (\text{第二項})
  &=\int_0^{\sqrt{3}}\sqrt{-t^2+3}\,dt=\dfrac{1}{4}\pi(\sqrt{3})^2=\dfrac{3}{4}\pi
\end{align}
である．

以上を\cref{eq:7}に代入して
\begin{align}
  I=2\left(\dfrac{9}{16}\pi+\dfrac{3}{4}\pi\right)=\dfrac{21}{8}\pi
\end{align}
である．
