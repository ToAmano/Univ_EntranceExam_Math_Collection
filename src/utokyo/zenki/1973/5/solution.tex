{\bf [解答]} $y$の存在条件から
\begin{align}
  \dfrac{t}{(1+t^2)x}\le\dfrac{1}{1+x^2}　
\end{align}
である．$0<x,1<t$に注意して
\begin{align}
  & t(1+x^2) \le (1+t^2)x \\
  \Longleftrightarrow
  & tx^2-(1+t^2)x+t \le 0 \\
  \Longleftrightarrow
  &(x-t)\left(x-\dfrac{1}{t}\right)\ge0 \\
  \Longleftrightarrow
  &\dfrac{1}{t}\le x\le t
\end{align}
である．

従って，
\begin{align}
  f(t)&=\int_{1/t}^t\left(\dfrac{1}{1+x^2}-\dfrac{t}{(1+t^2)x}\right)dx \nonumber\\
  &=\int_{1/t}^t\dfrac{1}{1+x^2}dx-\dfrac{t}{1+t^2}\int_{1/t}^t\dfrac{1}{x}dx\label{eq:1}
\end{align}
両辺$t$で微分する．各項計算して，
\begin{align}
  \cdot&\left[\int_{1/t}^t\left(\dfrac{1}{1+x^2}dx\right)\right]'=\dfrac{2}{1+t^2} \\
  \cdot&\left[\dfrac{t}{1+t^2}\int_{1/t}^t\dfrac{1}{x}dx\right]'=\dfrac{2}{1+t^2}+\dfrac{1-t^2}{(1+t^2)^2}2\log t
\end{align}
であるから，\cref{eq:1}に代入して
\begin{align}
  f'(t)=\dfrac{-2(1-t^2)}{(1+t^2)^2}\log t
\end{align}
である．$\cdots$(答)
