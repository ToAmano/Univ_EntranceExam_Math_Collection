題意の操作をした時の遷移図は上のように$0 \le z \le 15$かどうかで場合分けされる．
ここで題意より
\begin{align}
  K(z) =
  \begin{cases}
    z    & (0 \le z \le 15) \\
    30-z & (16 \le z \le 30)
  \end{cases}
\end{align}
であるから，一回$L=0$または$L=30$になった場合はそれ以降にボールの移動は起こらない．
従って任意の自然数$n$に対して
\begin{align}
  P_n(0) = 0, \quad P_n(30) = 1 \label{eq:1}
\end{align}
である．

\paragraph{(1)}

$1$回目の操作で場合分けする．$m \ge 2$とする．

\subparagraph{$1^\circ \ 0 \le x \le 15$の時}

$1$回目の操作後，確率$\dfrac{1}{2}$で$L=2x$，残り確率$\dfrac{1}{2}$で$L=0$になるから
\begin{align}
  P_m(x) = \dfrac{1}{2} P_{m-1}(2x) + \dfrac{1}{2} P_{m-1}(0)
\end{align}
である．\cref{eq:1}より第二項は$0$だから
\begin{align}
  P_m(x) = \dfrac{1}{2} P_{m-1}(2x) \label{eq:2}
\end{align}
である．

\subparagraph{$2^\circ \ 16 \le x \le 30$の時}

$1$回目の操作後，確率$\dfrac{1}{2}$で$L=2x-30$，残り確率$\dfrac{1}{2}$で$L=30$になるから
\begin{align}
  P_m(x) = \dfrac{1}{2} P_{m-1}(2x-30) + \dfrac{1}{2} P_{m-1}(30)
\end{align}
である．\cref{eq:1}より第二項は$1$だから
\begin{align}
  P_m(x) = \dfrac{1}{2} P_{m-1}(2x-30) + \dfrac{1}{2} \label{eq:3}
\end{align}
である．
\casesend

\cref{eq:2,eq:3}をまとめて，
\begin{align}
  P_m(x) =
  \begin{cases}
    \dfrac{1}{2} P_{m-1}(2x) & (0 \le x \le 15) \\
    \dfrac{1}{2} P_{m-1}(2x-30) + \dfrac{1}{2} & (16 \le x \le 30)
  \end{cases} \label{eq:4}
\end{align}
である．

\paragraph{(2)}

$x=10$からはじめて\cref{eq:4}を繰り返し用いて
\begin{align}
  P_{2n+2}(10)
  &= \dfrac{1}{2} P_{2n+1}(20) \\
  &= \dfrac{1}{2}\left(\dfrac{1}{2} P_{2n}(10) + \dfrac{1}{2}\right) \\
  &= \dfrac{1}{4} P_{2n}(10) + \dfrac{1}{4}
\end{align}
であり，変形して
\begin{align}
  P_{2n+2}(10) - \dfrac{1}{3} = \dfrac{1}{4} \left(P_{2n}(10) - \dfrac{1}{3}\right) \label{eq:5}
\end{align}
となる．

初期条件$P_2(10)$について，\cref{eq:2}より
\begin{align}
  P_2(10) = \dfrac{1}{2}P_1(20) = \dfrac{1}{2}\cdot\dfrac{1}{2}=\dfrac{1}{4}
\end{align}
である．\cref{eq:5}を繰り返し用いて
\begin{align}
  P_{2n}(10)
  &= \left(\dfrac{1}{4}\right)^{n-1} \left(\dfrac{1}{4} - \dfrac{1}{3}\right) + \dfrac{1}{3} \\
  &= \dfrac{1}{3} \left\{ 1 - \left(\dfrac{1}{4}\right)^n \right\}
\end{align}
である．

\paragraph{(3)}

$x=6$からはじめて\cref{eq:4}を繰り返し用いて
\begin{align}
  P_{4n+4}(6)
  &= \dfrac{1}{2} P_{4n+3}(12) = \dfrac{1}{4} P_{4n+2}(24) \\
  &= \dfrac{1}{4}\left(\dfrac{1}{2} P_{4n+1}(18) + \dfrac{1}{2}\right) \\
  &= \dfrac{1}{8} \left(\dfrac{1}{2} P_{4n}(6) + \dfrac{1}{2}\right) + \dfrac{1}{8} \\
  &= \dfrac{1}{16} P_{4n}(6) + \dfrac{3}{16} \\
  \therefore
  P_{4n+4}(6) - \dfrac{1}{5}
  &= \dfrac{1}{16} \left(P_{4n}(6) - \dfrac{1}{5}\right) \label{eq:6}
\end{align}
となる．

初期条件$P_4(6)$について，\cref{eq:2}より
\begin{align}
  P_4(6)
  = \dfrac{1}{2}P_3(12)
  = \dfrac{1}{4}P_2(24)
  = \dfrac{1}{8}\left(P_1(18)+1\right)
  = \dfrac{1}{8}\left(\dfrac{1}{2}+1\right)
  = \dfrac{3}{16}
\end{align}
だから，\cref{eq:6}をくり返し用いて，
\begin{align}
  P_{4n}(6)
  &= \left(\dfrac{1}{16}\right)^{n-1} \left(\dfrac{3}{16} - \dfrac{1}{5}\right) + \dfrac{1}{5} \\
  &= \dfrac{1}{5} \left\{ 1 - \left(\dfrac{1}{16}\right)^n \right\}
\end{align}
である．
