\begin{enumerate}
  \item すべての自然数$k$に対して，次の不等式を示せ．
    \begin{align*}
      \dfrac{1}{2(k+1)}<\int_0^1\dfrac{1-x}{k+x}dx<\dfrac{1}{2k}
    \end{align*}
  \item $m>n$であるようなすべての自然数$m$と$n$に対して，次の不等式を示せ．
    \begin{align*}
      \dfrac{m-n}{2(m+1)(n+1)}<\log\dfrac{m}{n}-\sum_{k=n+1}^m\dfrac{1}{k}<\dfrac{m-n}{2mn}
    \end{align*}
\end{enumerate}
