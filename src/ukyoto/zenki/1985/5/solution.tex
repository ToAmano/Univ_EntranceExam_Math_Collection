2枚の硬貨A, Bとし，表の出る確率が各々 $a, b$ だとする．
確率変数 $X_A, X_B$ を以下のように定める．
\begin{align}
  X_A =
  \begin{cases} 0 & (A\text{が裏}) \\ 1 & (A\text{が表})
  \end{cases} \quad
  X_B =
  \begin{cases} 0 & (B\text{が裏}) \\ 1 & (B\text{が表})
  \end{cases}
\end{align}
すると，$X = X_A + X_B$ で，$X_A, X_B$ は独立だから，
\begin{align}
  \begin{cases}
    P(X=0) = P(X_A=0) \cdot P(X_B=0) \\
    P(X=1) = P(X_A=0) \cdot P(X_B=1) + P(X_A=1) \cdot P(X_B=0) \\
    P(X=2) = P(X_A=1) \cdot P(X_B=1)
  \end{cases} \label{eq:1}
\end{align}
である．$P(X_A=0) = 1-a, P(X_A=1) = a$ などを代入して
\begin{align}
  \begin{cases}
    P(X=0) = (1-a)(1-b) \\
    P(X=1) = (1-a)b + (1-b)a \\
    P(X=2) = ab
  \end{cases} \label{eq:2}
\end{align}
である．

\paragraph{(1)}

$P(X=k) = {}_2\mathrm{C}_k \cdot p^k (1-p)^{2-k}$ の時，\cref{eq:2}に代入して，
\begin{align}
  \begin{cases}
    (1-p)^2 = (1-a)(1-b) = 1 + ab - (a+b) \\
    2p(1-p) = a + b - 2ab \\
    p^2 = ab
  \end{cases}
\end{align}
$\alpha = a+b, \beta = ab$ として，
\begin{align}
  \begin{cases}
    \beta - \alpha = (1-p)^2 - 1 = p(p-2) \\
    -2\beta + \alpha = 2p(1-p) \\
    \beta = p^2
  \end{cases}
\end{align}
第3式を1, 2式に代入して
\begin{align}
  \begin{cases}
    \alpha = p^2 - p(p-2) = 2p \\
    \alpha = 2p(1-p) + 2p^2 = 2p
  \end{cases}
\end{align}
だから$a, b$ は $x$ の2次式 $x^2 - 2px + p^2 = 0$ の2解で，$(a, b) = (p, p)$とすれば良い．

\paragraph{(2)}

$P(X=k) = \dfrac{1}{3}$ の時，\cref{eq:2}に代入して同様に$\alpha=a+b,\beta=ab$として
\begin{align}
  \dfrac{1}{3} = 1 + \beta - \alpha = \alpha - 2\beta = \beta
\end{align}
である．これを解くと$(\alpha, \beta) = \left(1, \dfrac{1}{3}\right)$ だから，$a, b$ は $t$ の2次式 $t^2 - t + \dfrac{1}{3} = 0$ の2解だが，この判別式を $D$ として
\begin{align}
  D = 1 - \dfrac{4}{3} = -\dfrac{1}{3} < 0
\end{align}
から，$a, b \in \mathbb{R}$ に反し矛盾する．よって $(a, b)$ は存在しない．
