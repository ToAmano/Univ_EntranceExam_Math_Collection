\paragraph{【甲・乙共通】}

{\bf [解答]}

与えられた数列
\begin{align}
  \begin{cases}
    0<x,y,\qquad y\ne x \\
    a_{n+1} = xa_{n} + y^{n+1}
  \end{cases} \label{eq:1}
\end{align}
について考える．
$y \neq 0$より$b_n=\dfrac{a_n}{y^n}$によって$\{b_n\}$を定めると，\cref{eq:1}より
\begin{align}
  \begin{cases}
    b_{n+1}=\dfrac{x}{y}b_n+1 \\
    b_1 = 0
  \end{cases} \label{eq:2}
\end{align}
である．変形して
\begin{align}
  b_{n+1}-\dfrac{y}{y-x}=\dfrac{x}{y}\left(b_n-\dfrac{y}{y-x}\right)
\end{align}
だから漸化式をくり返し用いて
\begin{align}
  b_n=\left(\dfrac{x}{y}\right)^{n-1}\left(-\dfrac{y}{y-x}\right)+\dfrac{y}{y-x}
\end{align}
であり，$a_n=y^nb_n$より
\begin{align}
  a_n
  &=\dfrac{y}{y-x}\left(y^n-x^{n-1}y\right) \\
  &=\dfrac{y^2}{y-x}\left(y^{n-1}-x^{n-1}\right)
\end{align}
である．これが収束する条件は \cref{eq:1}から $y^{n-1}-x^{n-1}$ が収束する条件と等しい．

\subparagraph{(i) $y>x>0$のとき}

この時は
\begin{align}
  y^{n-1}-x^{n-1}=y^{n-1}\left\{1-\left(\dfrac{x}{y}\right)^{n-1}\right\}
\end{align}
が収束するには，$0<y\leqq1$が条件である．

\subparagraph{(ii) $x>y>0$のとき}

(i)と同じく$0<x\leqq1$が条件である．
\casesend

以上まとめて求める条件は
\begin{align}
  0<x<y\leqq1\quad\text{又は}\quad0<y<x\leqq1
\end{align}
だから図示して図の斜線部である．
境界は線分 $x = 1 (0 < y < 1)$ および $y = 1 (0 < x < 1)$ を含み，
端点および直線 $y = x$ は含まない．
