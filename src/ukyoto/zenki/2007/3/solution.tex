\paragraph{【甲・乙共通】}

{\bf [解答]}

与えられた関係式
\begin{align}
  a+d=-(b+c),\label{eq:1}\\
  ad-bc=-p,\label{eq:2}\\
  d\leqq c\leqq b\leqq a \label{eq:3}
\end{align}
について考える．

$a+d=\alpha, ad=\beta$とおくと$a,d$が整数だから$\alpha,\beta$も整数で，\cref{eq:1,eq:2}および二次方程式の解と係数の関係から
\begin{itemize}
  \item $a,d$は$X^2-\alpha X+\beta=0$の2解 \\
  \item $b,c$は$X^2+\alpha X+\beta+p=0$の2解
\end{itemize}
となる．だからそれぞれ$X=a,b,c,d$を代入して
\begin{align}
  \begin{cases}
    a^2-\alpha a+\beta=0 \\
    d^2-\alpha d+\beta=0 \\
    c^2+\alpha c+\beta+p=0 \\
    b^2+\alpha b+\beta+p=0
  \end{cases}
\end{align}
である．
$a$と$b$の式を辺々引いて，
\begin{align}
  & a^2-b^2-\alpha(a+b)=p \\
  & (a+b)(a-b-\alpha)=p \\
  \therefore
  -(a+b)(b+d)=p
\end{align}
である．\cref{eq:3}および$a,b,c,d$は整数，$p$は素数だから
\begin{align}
  (b+d,a+b)=(-1,p),(-p,1) \label{eq:4}
\end{align}
である．同じことを$a,c$の式に対して行えば
\begin{align}
  & -(a+c)(c+d)=p \\
  \therefore
  & (c+d,a+c)=(-1,p),(-p,1) \label{eq:5}
\end{align}
である．

\cref{eq:4,eq:5}から$c+d\leqq b+d\leqq a+c\leqq a+b$だから
\begin{align}
  (c+d,b+d,a+c,a+b)=(-1,-1,p,p),\quad(-p,-1,1,p),\quad(-p,-p,1,1)
\end{align}
である．\cref{eq:1}から$a+d=-(b+c)$だからこれを満たすのは
\begin{align}
  (c+d,b+d,a+c,a+b) = (-p,-1,1,p) \label{eq:6}
\end{align}
の時である．この時$a,b,c$を$d$のみで表すと
\begin{align}
  a = p+1+d,
  \quad
  b = -d-1,
  \quad
  c = -p-d
\end{align}
だから\cref{eq:1,eq:2}は自動的に満たされる．\cref{eq:3}に代入すると
\begin{align}
  & d \le -p-d \le -d-1 \le p+1+d \\
  \therefore
  & \dfrac{-p-2}{2} \le d \le \dfrac{-p}{2}
\end{align}
である．$p$が$3$以上の素数，$d$が整数だから
\begin{align}
  d = -\dfrac{p+1}{2}
\end{align}
である．
この時全ての条件が満たされて十分で，よって
\begin{align}
  a=\dfrac{p+1}{2},\qquad b=\dfrac{p-1}{2},\qquad c=-\dfrac{p-1}{2}
\end{align}
である．

以上から
\begin{align}
  (a,b,c,d)=\left(\dfrac{p+1}{2},\dfrac{p-1}{2},-\dfrac{p-1}{2},-\dfrac{p+1}{2}\right)
\end{align}
である．

{\bf [別解]}

$d$を消去して解くこともできる．
条件$a+b+c+d=0$より$d=-(a+b+c)$であり，これを$ad-bc=-p$に代入すると
\begin{align}
  -a(a+b+c)-bc = -p
  \iff a^2+ab+ac+bc = p
  \iff (a+b)(a+c) = p \label{eq:7}
\end{align}
を得る．条件$a \geqq b \geqq c \geqq d$より
\begin{align}
  a+b \geqq a+c
\end{align}
である．また，$a+b+c+d=0$かつ$a \geqq b \geqq c \geqq d$において，$a \leqq 0$と仮定すると$a=b=c=d=0$となり$ad-bc=-p$を満たさないから$a>0$である．
さらに$a+b \leqq 0$と仮定すると$b \leqq -a < 0$となり，$d \leqq c \leqq b < 0$から$a+b+c+d < a - 3a = -2a < 0$となって矛盾する．
したがって$a+b > 0$であり，\cref{eq:7}より$a+c > 0$も従う．

$p$は素数であり$a+b \geqq a+c > 0$だから，\cref{eq:7}より
\begin{align}
  \begin{cases}
    a+b = p \\
    a+c = 1
  \end{cases}
  \iff
  \begin{cases}
    b = p-a \\
    c = 1-a
  \end{cases}
\end{align}
に限られる．これより$d = -(a+b+c) = a-p-1$となる．
条件$a \geqq b \geqq c \geqq d$に代入すると
\begin{align}
  a \geqq p-a \geqq 1-a \geqq a-p-1
\end{align}
となる．$p-a \geqq 1-a$は$p \geqq 3$より常に成り立つ．
残りの不等式を整理すると
\begin{align}
  \begin{cases}
    a \geqq p-a \iff a \geqq \dfrac{p}{2} \\
    1-a \geqq a-p-1 \iff a \leqq \dfrac{p+2}{2}
  \end{cases}
  \iff \dfrac{p}{2} \leqq a \leqq \dfrac{p+2}{2}
\end{align}
である．$p$は$3$以上の素数（奇数）であり，$a$は整数だから
\begin{align}
  a = \dfrac{p+1}{2}
\end{align}
に定まる．このとき
\begin{align}
  b = \dfrac{p-1}{2}, \quad c = -\dfrac{p-1}{2}, \quad d = -\dfrac{p+1}{2}
\end{align}
となり，与えられたすべての条件を満たす．
以上より
\begin{align}
  (a,b,c,d)=\left(\dfrac{p+1}{2},\,\dfrac{p-1}{2},\,-\dfrac{p-1}{2},\,-\dfrac{p+1}{2}\right)
\end{align}
である．
