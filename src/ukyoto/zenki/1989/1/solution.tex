$\angle A_n O B_n = \theta_n$ とおくと，題意から
帰納的に
\begin{align}
  \theta_n = \left(\dfrac{1}{2}\right)^{n-1} \theta \label{eq:1}
\end{align}
である．右上図で
\begin{align}
  OA_n = a_n, \quad
  OB_{n+1} = a_{n+1}, \quad
  a_1 = 1 \label{eq:2}
\end{align}
だから$\triangle OA_nB_{n+1}$に着目すると
\begin{align}
  & |OB_{n+1}| = |OA_n|\sin\angle OA_nB_{n+1} \\
  \therefore
  & a_{n+1} = a_n \cos\left(\left(\dfrac{1}{2}\right)^n \theta\right)
\end{align}
となり，これを解くと
\begin{align}
  a_n
  &= \cos\left\{ \left(\dfrac{1}{2}\right)^{n-1} \theta \right\} \cdots \cos\dfrac{\theta}{8} \cos\dfrac{\theta}{4} \cos\dfrac{\theta}{2} \\
  &= \prod_{i=1}^{n-1}\cos\left\{ \left(\dfrac{1}{2}\right)^{i}\theta\right\} \label{eq:3}
\end{align}
である．

\paragraph{(1)}

\cref{eq:3}より
\begin{align}
  a_3 \cdot \sin\dfrac{\theta}{4}
  = \sin\dfrac{\theta}{4} \cdot \cos\dfrac{\theta}{4} \cdot \cos\dfrac{\theta}{2}
  = \dfrac{1}{2} \sin\dfrac{\theta}{2} \cdot \cos\dfrac{\theta}{2}
  = \dfrac{1}{4} \sin\theta
\end{align}
である．

\paragraph{(2)}

新しく
\begin{align}
  A_n = a_n \sin\dfrac{\theta}{2^{n-1}}
\end{align}
とおくと，(1)と同じように計算して
\begin{align}
  A_n = \left(\dfrac{1}{2}\right)^{n-1} \sin\theta
\end{align}
だから
\begin{align}
  a_n
  = \dfrac{ \left(\dfrac{1}{2}\right)^{n-1} \theta }{ \sin\left\{ \left(\dfrac{1}{2}\right)^{n-1} \theta \right\} } \cdot \dfrac{\sin\theta}{\theta}
  \xrightarrow{n \to \infty} \dfrac{\sin\theta}{\theta}
\end{align}
である．
