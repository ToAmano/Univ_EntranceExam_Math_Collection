{\bf [解答]}

与えられた方程式
\begin{align}
  f(x) = x^4+ax^3+bx^2+cx+1 = 0 \label{eq:1}
\end{align}
について考える．係数が整数だから2つの虚数解はお互いに複素共役であり，また$x=0$は解ではない．
そこで2つの整数解を$\alpha,\beta \ (\neq 0)$，2つの虚数解を$z,\bar z \ (\neq 0)$とおく．解と係数の関係から
\begin{align}
  a&=-(\alpha+\beta+z+\bar z) \label{eq:2}\\
  1&=\alpha\beta z\bar z=\alpha\beta|z|^2 \label{eq:3}\\
  c&=-\beta z\bar z - \alpha z\bar z - \alpha\beta \bar z - \alpha\beta z \label{eq:4}\\
  b&=\alpha\beta+(\alpha+\beta)(z+\bar z)+|z|^2 \label{eq:5}
\end{align}
である．\cref{eq:4}はさらに変形して\cref{eq:3}より
\begin{align}
  c&=-\alpha\beta z\bar z\left(\dfrac{1}{\alpha}+\dfrac{1}{\beta}+\dfrac{1}{z}+\dfrac{1}{\bar z}\right)\\
  &=-\left(\dfrac{1}{\alpha}+\dfrac{1}{\beta}+\dfrac{z+\bar z}{|z|^2}\right) \label{eq:6}
\end{align}

題意より$\alpha,\beta, a$が整数だから\cref{eq:2}より
\begin{align}
  z+\bar z \in \mathbb{Z} \label{eq:7}
\end{align}
である．よって\cref{eq:5}より
\begin{align}
  |z|^2 \in \mathbb{Z}
\end{align}
が従う．しかるに\cref{eq:3}において $\alpha,\beta,|z|^2$ が全て整数だから
\begin{align}
  |z|^2=1,\ \alpha\beta=1
  \quad\therefore\
  |z|=1,\ (\alpha,\beta)=(1,1),(-1,-1) \label{eq:8}
\end{align}
である．

そこで$z=e(\theta)=\cos\theta+i\sin\theta \ (0 \le \theta < 2\pi)$とおくと，\cref{eq:7}より
\begin{align}
  & 2\cos\theta \in \mathbb{Z}
  \quad\therefore
  \cos\theta = 0, \pm 1, \pm \dfrac{1}{2}
\end{align}
だが，このうち$\cos\theta=\pm 1$つまり$\theta=0,\pi$の時は$z$が実数になって不適なので
\begin{align}
  \cos\theta = 0, \pm \dfrac{1}{2} \label{eq:9}
\end{align}
である．

以下，$(\alpha,\beta,z)$が\cref{eq:8,eq:9}の時に解と係数の関係\cref{eq:2,eq:5,eq:6}を満たすかどうかを調べる．

\subparagraph{$1^\circ$ $(\alpha,\beta)=(1,1)$の時}

\cref{eq:2,eq:5,eq:6}に代入して
\begin{align}
  a = - (2+2\cos\theta),
  \quad
  b = 2(1+2\cos\theta),
  \quad
  c = -(2+2\cos\theta)
\end{align}
となり，\cref{eq:1}に代入して元の方程式は
\begin{align}
  f(x)
  &=x^4-(2+2\cos\theta)x^3+2(1+2\cos\theta)x^2-(2+2\cos\theta)x+1 \\
  &=(x-1)^2(x^2-2\cos\theta x+1)
\end{align}
であり，$\cos\theta = 0, \pm\dfrac{1}{2}$いずれも$x^2-2\cos\theta x+1=0$の判別式$\cos^2\theta-1<0$を満たし十分である．よって
\begin{align}
  (a,b,c)=(-2,2,-2),\ (-3,4,-3),\ (-1,0,-1) \label{eq:10}
\end{align}
である．

\subparagraph{$2^\circ$ $(\alpha,\beta)=(-1,-1)$の時}

\cref{eq:2,eq:5,eq:6}に代入して
\begin{align}
  a=2-2\cos\theta,\quad b=2(1-2\cos\theta),\quad c=2-2\cos\theta
\end{align}
となり，\cref{eq:1}に代入して元の方程式は
\begin{align}
  f(x)
  &=x^4+(2-2\cos\theta)x^3+2(1-2\cos\theta)x^2+(2-2\cos\theta)x+1 \\
  &=(x+1)^2(x^2-2\cos\theta x+1)
\end{align}
であり，$\cos\theta = 0, \pm\dfrac{1}{2}$いずれも$x^2-2\cos\theta x+1=0$の判別式$\cos^2\theta-1<0$を満たし十分である．よって
\begin{align}
  (a,b,c)=(2,2,2),\ (1,0,1),\ (3,4,3)
\end{align}
である．

\begin{align}
  (a,b,c)=(\pm2,2,\pm2),\ (\pm1,0,\pm1),\ (\pm3,4,\pm3)
\end{align}
である．
