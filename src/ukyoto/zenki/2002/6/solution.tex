(ii)から，
\begin{align}
  e(\theta)=\cos\dfrac{\pi\theta}{180}+i\sin\dfrac{\pi\theta}{180} \quad (0<\theta< 90^{\circ}) \label{eq:1}
\end{align}
とおくと
\begin{align}
  z_{n+1}-z_n=e(\theta)(z_n-z_{n-1}) \label{eq:2}
\end{align}
である．$z_0=0$，$z_1=a$とあわせてくり返し用いると
\begin{align}
  z_{n+1}-z_n= a\{e(\theta)\}^n \label{eq:3}
\end{align}
を得る．$\alpha_n=\dfrac{z_n}{\{e(\theta)\}^n}$とおくと
\begin{align}
  & \alpha_{n+1}=\dfrac{1}{e(\theta)}\alpha_n+\dfrac{a}{e(\theta)}\\
  & \alpha_{n+1}-\dfrac{a}{e(\theta)-1}=\dfrac{1}{e(\theta)}\left\{\alpha_n-\dfrac{a}{e(\theta)-1}\right\}\quad(\because\text{\cref{eq:1}より}e(\theta)\neq 1)
\end{align}
となる．$\alpha_0=0$に注意して漸化式をくり返し用いると
\begin{align}
  \alpha_n
  &=\left(\dfrac{1}{e(\theta)}\right)^n\left(0-\dfrac{a}{e(\theta)-1}\right)+\dfrac{a}{e(\theta)-1}\\
  &=\dfrac{a}{e(\theta)-1}\left[-\left(\dfrac{1}{e(\theta)}\right)^n+1\right]\\
  \therefore
  z_n&=\dfrac{a}{e(\theta)-1}\left\{e(n\theta)-1\right\}
\end{align}
である．ただし$\{e(\theta)\}^n=e(n\theta)$を用いた．

これが$z_0=0$に一致する$n$（$n\ge1$）が存在する条件は，$a\ne0$，$e(\theta)-1\ne 0$（$\because\text{\cref{eq:1}}$）に注意して
\begin{align}
  &e(n\theta)=1\text{なる}n\ge1\text{が存在する}\\
  \iff&n\theta=360m\text{なる}n\in\mathbb N,\ m\in\mathbb Z\text{が存在する}\\
  \iff&\theta=\dfrac{m}{n}\cdot 360^{\circ}\text{なる}n\in\mathbb N,\ m\in\mathbb Z\text{が存在する}\\
  \iff&\theta\text{が有理数}
\end{align}
である．よって示された．\qed
