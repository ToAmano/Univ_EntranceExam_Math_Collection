\section*{第1問}

{\bf [解]}
$\sin\theta=s$，$\cos\theta=c$と書く．
\begin{align*}
f'(\theta)&=-4\sin4\theta-8s\cdot c\\
&=-8\sin2\theta\cos2\theta-8c\cdot s\\
&=-8c\cdot s(2\cos2\theta+1)\\
&=-8c\cdot s(4c^2-1)
\end{align*}

から，下表を得る．

\begin{center}
\begin{tabular}{c|ccccccccc}
$\theta$ & $0$ & & $\pi/3$ & & $\pi/2$ & & $2\pi/3$ & & $3\pi/4$ \\
\hline
$f'$ & & $-$ & $0$ & $+$ & $0$ & $-$ & $0$ & $+$ & \\
\hline
$f$ & $1$ & $\searrow$ & $-\dfrac72$ & $\nearrow$ & $-3$ & $\searrow$ & $-\dfrac72$ & $\nearrow$ & $-3$ \\
\end{tabular}
\end{center}

したがって，$\max=1$，$\min=-\dfrac72$．

\section*{第3問}

{\bf [解]}
$f(x)=x^2-x+\dfrac{n-1}{n^2}$とおくと，$f(x)=\left(x-\dfrac1n\right)\left(x-\dfrac{n-1}n\right)$である．したがって，$x=\dfrac1n,\dfrac{n-1}n$として，
\begin{align*}
\left(\frac1n\right)^{2n}&=a_n\left(\frac1n\right)+b_n\\
\left(\frac{n-1}n\right)^{2n}&=a_n\left(\frac{n-1}n\right)+b_n
\end{align*}
である．$n\to\infty$の時を考えるので，$n\ne2$として良い．辺々引いて，
\begin{align*}
\frac{2-n}na_n=\left(\frac1n\right)^{2n}-\left(\frac{n-1}n\right)^{2n}\quad\cdots①
\end{align*}

①から，$n\to\infty$の時
\begin{align*}
a_n=\frac1{\frac1n-1}\left\{\left(\frac1n\right)^{2n}-\left(1-\frac1n\right)^n\left(\frac{n-1}n\right)^n\right\}\longrightarrow e^{-2}
\end{align*}
\begin{align*}
b_n=\left(\frac1n\right)^{2n}-\frac1n\cdot a_n\longrightarrow0\quad(\because a_n\to e^{-2})．
\end{align*}

\section*{第5問}

{\bf [解]}
$C$が$\pm1$を通るので，$C$の中心は$\beta$とおける（純虚数）．
\begin{align*}
|1-\beta|^2=|-1-\beta|^2=|\alpha-\beta|^2=A\quad\cdots①
\end{align*}

さらに，$\alpha=k\zeta$（$k\in\mathbb R$）とおける．ただし$|\zeta|=1$．①から
\begin{align*}
|\beta|^2-\beta-\bar\beta+1=|\beta|^2-\bar\alpha\beta-\alpha\bar\beta+|\alpha|^2
\end{align*}

$\beta$は純虚数で，$\beta+\bar\beta=0$より
\begin{align*}
1=k^2-k\bar\zeta\beta-k\zeta\bar\beta\quad\cdots②
\end{align*}

又，
\begin{align*}
\left|-\frac1{\bar\alpha}-\beta\right|^2=\left|\frac\zeta k+\beta\right|^2=|\beta|^2+\frac1k\zeta\bar\beta+\frac1k\bar\zeta\beta+\frac1{k^2}
\end{align*}

一方，②の両辺を$k^2$で割って
\begin{align*}
\frac1{k^2}+\frac1k\bar\zeta\beta+\frac1k\zeta\bar\beta=1
\end{align*}

だから①に代入して
\begin{align*}
\left|-\frac1{\bar\alpha}-\beta\right|^2=|\beta|^2+1=A．
\end{align*}

したがって，$C$は$-\dfrac1{\bar\alpha}$も通る．$\blacksquare$

\section*{第6問}

{\bf [解]}
全ての操作の後，$N+1$に赤が入っているのは，$N$回目の操作後に赤の入っている箱を$T$として，$N+1$回目で$T$と交換する時である（$T\ne N+1$）．$T\ne N+1$となる確率は，排反を考えて，
\begin{align*}
1-\left(\frac{N-1}N\right)^N
\end{align*}

だからもとめるのは
\begin{align*}
\frac1N\left\{1-\left(\frac{N-1}N\right)^N\right\}．
\end{align*}
