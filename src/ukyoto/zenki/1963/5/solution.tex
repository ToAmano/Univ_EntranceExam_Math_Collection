題意より$\triangle O P_n P_{n+1}$ と $\triangle O P_{n+1} P_{n+2}$ の相似比は，
\begin{align}
  |O P_0|: |O P_{1}| \quad \therefore a: |O P_{1}| \label{eq:1}
\end{align}
である．
ここで $\triangle O P_0 P_1$ に正弦定理を用いて，
\begin{align}
  & \dfrac{|O P_1|}{\sin \alpha} = \dfrac{|OP_0|}{\sin(\alpha+\theta)} \\
  & \dfrac{|O P_1|}{\sin \alpha} = \dfrac{a}{\sin(\alpha+\theta)} \\
  \therefore
  & |O P_1| = \dfrac{\sin \alpha}{\sin(\alpha+\theta)} \cdot \alpha \label{eq:2}
\end{align}
だから\cref{eq:1}に代入して，$a \neq 0$より
\begin{align}
  |O P_0|: |O P_1| = 1: \dfrac{\sin \alpha}{\sin(\alpha+\theta)} \label{eq:3}
\end{align}
である．従って繰り返し用いれば
\begin{align}
  |OP_n| = \left(\dfrac{\sin \alpha}{\sin(\alpha+\theta)}\right)^n a \label{eq:4}
\end{align}
である．$\lim_{n\to\infty} P_n = O$となる必要十分条件は$\lim_{n\to\infty}|OP_n|=0$となることだから，\cref{eq:4}よりこの条件は
\begin{align}
  0 < \dfrac{\sin \alpha}{\sin(\alpha+\theta)} < 1 \label{eq:5}
\end{align}
である．

\subparagraph{(ロ)}

$\triangle O P_n P_{n+1}$ の面積 $S_n$ とする．
まず$n=0$の時
\begin{align}
  S_0
  = \dfrac{1}{2} |O P_0| |O P_1| \sin\theta
  = \dfrac{1}{2} \dfrac{\sin\alpha}{\sin(\alpha+\theta)}a^2\sin \theta
  = \dfrac{1}{2}pa^2\sin \theta \label{eq:6}
\end{align}
である．ただし簡単のため
\begin{align}
  p = \dfrac{\sin \alpha}{\sin(\alpha+\theta)} \label{eq:7}
\end{align}
とおいた．
\cref{eq:1}から$S_{n+1}$と$S_{n}$の関係は
\begin{align}
  S_{n+1} = p^2 S_n
\end{align}
である．繰り返し用いて\cref{eq:6}から
\begin{align}
  S_n = (p^2)^{n} \cdot \dfrac{1}{2} p a^2 \sin\theta = \dfrac{1}{2} p^{2n+1} a^2\sin\theta
\end{align}
である．したがって(イ)の条件 $0 < p < 1$ とあわせて，求める無限和は
\begin{align}
  S = \lim_{n \to \infty} \sum_{k=0}^n S_k = \dfrac{1}{2} p a^2\sin\theta \dfrac{1}{1-p^2}
\end{align}
だから，\cref{eq:6}より $S$ は $\triangle O P_0 P_1$ の
\begin{align}
  \dfrac{1}{1-p^2} = \dfrac{1}{1 - \left( \dfrac{\sin \theta}{\sin(\alpha+\theta)} \right)^2} \text{ (倍)}
\end{align}
である．

\subparagraph{ハ}

$a_n = |P_n P_{n+1}|$ とおくと，\cref{eq:1}より
\begin{align}
  a_{n+1} = p a_n \label{eq:8}
\end{align}
であり，初項$a_0$について，$\triangle O P_0 P_1$ に正弦定理を用いて
\begin{align}
  \dfrac{|P_0 P_1|}{\sin \theta}
  = \dfrac{a}{\sin(\alpha+\theta)}
  \quad \therefore
  a_0
  = \dfrac{\sin \theta}{\sin(\alpha+\theta)} a
\end{align}
だから，\cref{eq:8}をくり返し用いて，
\begin{align}
  a_n = \dfrac{\sin \theta}{\sin(\alpha+\theta)} \alpha \cdot p^{n-1}
\end{align}
となるので，(イ)の条件$0<p<1$より
\begin{align}
  L
  &= \lim_{n \to \infty} \sum_{k=0}^n a_k \\
  &= \dfrac{\sin \theta \cdot a}{\sin(\alpha+\theta)} \cdot \dfrac{1}{1-p} \quad (\because |p| < 1) \\
  &= \dfrac{\sin \theta \cdot a}{\sin(\alpha+\theta)} \cdot \dfrac{\sin(\alpha+\theta)}{\sin(\alpha+\theta) - \sin \alpha} \\
  &= \dfrac{\sin \theta}{\sin(\alpha+\theta) - \sin \alpha} \cdot a
\end{align}
である．$\theta \to +0$ として，
\begin{align}
  L
  = \dfrac{\theta}{\sin(\alpha+\theta) - \sin \theta} \cdot \dfrac{\sin \theta}{\theta} \cdot \alpha \longrightarrow{\theta \to +0} \dfrac{\alpha}{\cos \alpha}
\end{align}
が求める極限値である．
