{\bf [解答]}

\paragraph{(イ)}

与えられた方程式
\begin{align}
  f(x) = x^2-(a+c)x+(ac-b^2)=0 \label{eq:1}
\end{align}
について考える．

判別式 $D$ として
\begin{align}
  D
  &= (a+c)^2 - 4(ac-b^2) \\
  &= 4b^2 + (a-c)^2 \ge 0
\end{align}
より\cref{eq:1}は実根を持つ．

\paragraph{(ロ)}

\cref{eq:1}の二つの解を$\alpha,\beta \ (\alpha\le\beta)$とする．
\cref{eq:1}を変形すると
\begin{align}
  f(x) = (x-a)(x-c) - b^2 \label{eq:2}
\end{align}
だから，$f(a)=f(c)=-b^2\le 0$となり，$\alpha,\beta$との位置関係は
\begin{align}
  \alpha \le a,c \le \beta \label{eq:3}
\end{align}
となる．

与えられた数
\begin{align}
  \gamma
  & = \dfrac{a+c}{2} - \dfrac{(a-c)(p^2-q^2)+4bpq}{2(p^2+q^2)} \\
  & = \dfrac{2(cp^2+aq^2)-4bpq}{2(p^2+q^2)} \\
  & = \dfrac{cp^2+aq^2-2bpq}{p^2+q^2} \label{eq:4}
\end{align}
と$\alpha,\beta$の大小を比べよう．

\subparagraph{$1^{\circ} \alpha,\gamma$について}

まず$\alpha,\gamma$について\cref{eq:4}から
\begin{align}
  \gamma - \alpha
  & = \dfrac{cp^2+aq^2-2bpq}{p^2+q^2} - \alpha \\
  & = \dfrac{(c-\alpha)p^2+(a-\alpha)q^2-2bpq}{p^2+q^2}
\end{align}
だが，この正負は分子
\begin{align}
  g(p) = (c-\alpha)p^2-2bpq + (a-\alpha)q^2
\end{align}
の正負と等しい．
$c=\alpha$の時は\cref{eq:2}から$b=0$だから
\begin{align}
  g(p) = (a-\alpha)q^2 \ge 0 (\because\text{\cref{eq:3}})
\end{align}
である．$c\neq \alpha$なら\cref{eq:3}から$g(p)$は$p$の二次係数が正の二次式で，$g(p)=0$の判別式$D_g$として
\begin{align}
  D_g/4 = [b^2 - (c-\alpha)(a-\alpha)]q^2 = 0 (\because f(\alpha)=0)
\end{align}
より$g(p) \ge 0$である．よっていずれの場合も$g(p)\ge0$つまり
\begin{align}
  \gamma \ge \alpha \label{eq:5}
\end{align}
が成り立つ．

\subparagraph{$2^{\circ} \beta,\gamma$について}

この時\cref{eq:4}より
\begin{align}
  \beta - \gamma =
  & = \beta - \dfrac{cp^2+aq^2-2bpq}{p^2+q^2} \\
  & = \dfrac{(\beta - c)p^2+(\beta-a)q^2+2bpq}{p^2+q^2}
\end{align}
だが，この正負は分子
\begin{align}
  h(p) = (\beta - c)p^2+(\beta-a)q^2+2bpq
\end{align}
の正負と等しい．
$c=\beta$の時は\cref{eq:2}から$b=0$だから
\begin{align}
  h(p) = (\beta-a)q^2 \ge 0 (\because\text{\cref{eq:3}})
\end{align}
である．$c\neq \alpha$なら\cref{eq:3}から$h(p)$は$p$の二次係数が正の二次式で，
$h(p)=0$の判別式$D_h$として
\begin{align}
  D_h/4 = [b^2 - (\beta - c)(\beta -a)]q^2 = 0 (\because f(\beta)=0)
\end{align}
より$h(p) \ge 0$である．よっていずれの場合も$h(p)\ge0$つまり
\begin{align}
  \gamma \le \beta \label{eq:6}
\end{align}
が成り立つ．
\casesend

以上\cref{eq:5,eq:6}から常に$\alpha \le \gamma \le \beta$が成立する．
