$\triangle ABC$ の重心を $G$ としてこれを表す複素数を $g$ とすると\cref{eq:1}から，
\begin{align}
  \begin{cases}
    g = \dfrac{\alpha + \beta + \gamma}{3} = 1 \\
    |GA|=|GB|=|GC|=1
  \end{cases} \label{eq:2}
\end{align}
だから，$\triangle ABC$は$1$を中心とする半径$1$の円周上にある．
$B,C$の位置関係に応じて，$\beta - 1, \gamma - 1$ は，$\alpha - 1$ を $\dfrac{2}{3}\pi, \dfrac{4}{3}\pi$ 回転させた複素数で，$e(\theta) = \cos \theta + i \sin \theta$ とおくと，$z = \alpha - 1$ から
\begin{align}
  \beta-1, \gamma-1 =
  \begin{cases}
    e\left(\dfrac{2}{3}\pi\right) (\alpha - 1) = \dfrac{1}{2}(-1 + \sqrt{3}i)z \\
    e\left(\dfrac{4}{3}\pi\right) (\alpha - 1) = \dfrac{1}{2}(-1 - \sqrt{3}i)z
  \end{cases}
\end{align}
より
\begin{align}
  (\beta, \gamma) = \left( \dfrac{1}{2}(-1 \pm \sqrt{3}i)z + 1 , \; \dfrac{1}{2}(-1 \mp \sqrt{3}i)z + 1 \right) \quad (\text{複号同順}) \label{eq:3}
\end{align}
である．

\paragraph{(2)}

\cref{eq:2}に$z=\alpha-1$を代入すると
\begin{align}
  |z|=1 \label{eq:4}
\end{align}
だから，$z=e(\theta)$とおける．

\cref{eq:3}を$\alpha\beta\gamma$に代入して
\begin{align}
  \alpha \beta \gamma
  = \left(z+1 \right)\left[ e\left(\dfrac{2}{3}\pi\right)z + 1 \right] \left[ e\left(\dfrac{4}{3}\pi\right)z + 1 \right] \label{eq:5}
\end{align}
だから，
\begin{align}
  \begin{cases}
    e(\theta) + 1 = 2 \cos \dfrac{\theta}{2} e\left(\dfrac{\theta}{2}\right) \\
    e(\theta_1)e(\theta_2) = e(\theta_1+\theta_2)
  \end{cases}
\end{align}
に注意して$z=e(\theta)$を\cref{eq:5}に代入すると
\begin{align}
  \alpha\beta\gamma
  &=  \left[ z+1 \right]\left[ e\left(\dfrac{2}{3}\pi+\theta\right) + 1 \right] \left[ e\left(\dfrac{4}{3}\pi+\theta\right) + 1 \right] \\
  &=  \left[ 2 \cos \dfrac{\theta}{2} e\left(\dfrac{\theta}{2}\right) \right]
  \left[ 2 \cos \dfrac{\theta+2\pi/3}{2} e\left(\dfrac{\theta+2\pi/3}{2}\right)\right]
  \left[ 2 \cos \dfrac{\theta+4\pi/3}{2} e\left(\dfrac{\theta+4\pi/3}{2}\right)\right] \\
  &=  8\cos \dfrac{\theta}{2} \cos \left(\dfrac{\theta}{2} + \dfrac{\pi}{3}\right) \cos \left(\dfrac{\theta}{2} + \dfrac{2\pi}{3}\right) e\left(\dfrac{3}{2}\theta + \pi\right) \label{eq:6}
\end{align}
を得る．係数部分をさらに変形すると積和の公式から
\begin{align}
  \cos \dfrac{\theta}{2} \cos \left(\dfrac{\theta}{2} + \dfrac{\pi}{3}\right) \cos \left(\dfrac{\theta}{2} + \dfrac{2\pi}{3}\right)
  &= \dfrac{1}{2} \cos \left(\dfrac{\theta}{2} + \dfrac{\pi}{3}\right) \left[ \cos \left(\theta + \dfrac{2}{3}\pi\right) + \cos \dfrac{2}{3}\pi \right] \\
  &= \dfrac{1}{2}\cos \left(\dfrac{\theta}{2} + \dfrac{\pi}{3}\right) \left[ \cos 2\left(\dfrac{\theta}{2} + \dfrac{1}{3}\pi\right) - \dfrac{1}{2} \right]
\end{align}
であり，$t = \cos(\dfrac{\theta}{2} + \dfrac{\pi}{3})$ として倍角公式を使うと
\begin{align}
  (\text{係数})
  &= \dfrac{1}{2}t(2t^2 - 1 - \dfrac{1}{2}) \\
  &= \dfrac{4t^3 - 3t}{4}
\end{align}
となる．さらに三倍角の公式から
\begin{align}
  (\text{係数}) &= \dfrac{1}{4}\cos \left(\dfrac{3}{2}\theta + \pi\right)
\end{align}
となる．これを\cref{eq:6}に代入して
\begin{align}
  \alpha\beta\gamma
  &= 2\cos \left(\dfrac{3}{2}\theta + \pi\right)e\left(\dfrac{3}{2}\theta + \pi\right) \label{eq:7}
\end{align}
を得る．これを\cref{eq:1}に代入して
\begin{align}
  1 = |\alpha\beta\gamma| = \left|2\cos \left(\dfrac{3}{2}\theta + \pi\right)\right|
\end{align}
だから，絶対値を外して
\begin{align}
  \cos \left(\dfrac{3}{2}\theta + \pi\right) = \pm \dfrac{1}{2} \label{eq:8}
\end{align}
となる．題意の条件
\begin{align}
  0 \le \arg \alpha \le \arg \beta \le \arg \gamma < 2\pi \label{eq:9}
\end{align}
と(1)から
\begin{align}
  & 0 \le \arg \alpha = \theta \le \dfrac{2\pi}{3} \\
  \therefore
  & \pi \le \dfrac{3}{2}\theta + \pi \le 2\pi \label{eq:10}
\end{align}
として良いから，\cref{eq:8}の解は
\begin{align}
  \dfrac{3}{2}\theta + \pi = \dfrac{4\pi}{3}, \dfrac{5\pi}{3} \label{eq:11}
\end{align}
である．この時\cref{eq:7}に代入すると
\begin{align}
  \alpha\beta\gamma = -e\left(\dfrac{4\pi}{3}\right), e\left(\dfrac{5\pi}{3}\right)
\end{align}
だから，このうち\cref{eq:1}の第4式の条件を満たすのは前者，つまり
\begin{align}
  \dfrac{3}{2}\theta + \pi = \dfrac{4\pi}{3} \quad \therefore \theta = \dfrac{2\pi}{9} \label{eq:12}
\end{align}
の時である．

この時\cref{eq:9}および(1)から，
\begin{align}
  \begin{cases}
    \alpha = z + 1 = e\left(\dfrac{2}{9}\pi\right) + 1 = 2 \cos \dfrac{\pi}{9} e\left(\dfrac{\pi}{9}\right) \\
    \beta = e\left(\dfrac{2}{3}\pi\right) z + 1 = e\left(\dfrac{8}{9}\pi\right) + 1 = 2 \cos \dfrac{4}{9}\pi e\left(\dfrac{4}{9}\pi\right) \\
    \gamma = e\left(\dfrac{4}{3}\pi\right) z + 1 = e\left(\dfrac{14}{9}\pi\right) + 1 = 2 \cos \dfrac{7}{9}\pi e\left(\dfrac{7}{9}\pi\right) = 2 \cos \dfrac{2}{9}\pi e\left(\dfrac{16}{9}\pi\right)
  \end{cases}
\end{align}
となり，
\begin{align}
  (\arg \alpha, \arg \beta, \arg \gamma) = \left( \dfrac{\pi}{9}, \dfrac{4}{9}\pi, \dfrac{16}{9}\pi \right)
\end{align}
である．

$z$のまま変形していく手段もある．$|\alpha \beta \gamma| = 1$ に (1)の\cref{eq:3}を代入し計算すると
\begin{align}
  \alpha\beta\gamma
  &= (z+1)\left\{ \dfrac{1}{2}(-1 - \sqrt{3}i)z + 1 \right\} \left\{ \dfrac{1}{2}(-1 + \sqrt{3}i)z + 1 \right\} \\
  &= (z+1)\left\{ (1 - \dfrac{1}{2}z)^2 + \dfrac{3}{4}z^2 \right\} \\
  &= (z+1)(z^2 - z + 1) \\
  &= z^3 + 1
\end{align}
だから，両辺絶対値をとって
\begin{align}
  |\alpha \beta \gamma| = 1 \iff |z^3 + 1| = 1
\end{align}
となる．$z = e(\theta)$ として(2)で使った関係式より
\begin{align}
  1 &= |e(3\theta) + 1| = 2 \left| \cos \dfrac{3}{2}\theta \right| \left| e\left(\dfrac{3}{2}\theta\right) \right|
\end{align}
となる．以下は[解答]と全く同様に解ける．
