\noindent
題意から扇形 $A_{k+2} A_{k+1} B_k$ ($k \ge 0, B_0 = A_1$) の半径を $r_k$ とすると
\[
  \begin{cases}
    r_0 = 1 \cdot l_n \\
    r_{k+1} = r_k + l_n
  \end{cases}
\]
から, 等差数列の公式より, $r_k = (k+1) l_n$ であり, 中心角は $\dfrac{2\pi}{n}$ だから, 扇形 $A_{k+2} A_{k+1} B_k$ の面積 $T_k$ は
\[
  T_k = \dfrac{1}{2} \cdot \dfrac{2\pi}{n} \cdot r_k^2 = \dfrac{\pi}{n} l_n^2 (k+1)^2
\]

\noindent
よって, 扇形の面積の総和 $T$ は
\[
  T = \sum_{k=0}^{n-1} \dfrac{\pi}{n} l_n^2 (k+1)^2
\]
\[
  = \sum_{k=1}^n \dfrac{\pi}{n} l_n^2 k^2
\]
\[
  = \dfrac{1}{6} \pi (n+1)(2n+1) l_n^2 \quad \dots \text{②}
\]

\noindent
一方, 正 $n$ 角形の面積 $S$ は, 下図の三角形の面積の $n$ 倍で,

\[
  \dfrac{S}{n} = \dfrac{1}{2} \cdot r^2 \cdot \sin \dfrac{2\pi}{n}
\]
\[
  S = \dfrac{1}{2} n r^2 \sin \dfrac{2\pi}{n} \quad \dots \text{③}
\]

\vspace{1em}
\noindent
①, ②, ③から
\[
  S_n = S + T = \dfrac{1}{2} n r^2 \sin \dfrac{2\pi}{n} + \dfrac{\pi}{6} (n+1)(2n+1) r^2 \cdot 4 \sin^2 \dfrac{\pi}{n}
\]
\[
  = \dfrac{1}{2} r^2 \cdot \dfrac{\sin \dfrac{2\pi}{n}}{\dfrac{2\pi}{n}} \cdot 2\pi + \dfrac{2\pi}{3} r^2 \left( \dfrac{\sin \dfrac{\pi}{n}}{\dfrac{\pi}{n}} \right)^2 \pi^2 \left( 1 + \dfrac{1}{n} \right) \left( 2 + \dfrac{1}{n} \right)
\]
\[
  \to \pi r^2 + \dfrac{4}{3} \pi^3 r^2 \quad \left( n \to \infty, \text{この時 } \dfrac{2\pi}{n}, \dfrac{\pi}{n} \to 0 \right)
\]
