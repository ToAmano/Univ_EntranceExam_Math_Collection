\[
  Q\left(a \cos\left(\dfrac{\pi}{2}-\theta\right), a \sin\left(\dfrac{\pi}{2}-\theta\right)\right) = Q(a \sin\theta, a \cos\theta)
\]

\begin{enumerate}[(1)]
  \item $\tan \dfrac{\alpha}{2} = t$ とおくと $t = \dfrac{r}{a}$ だから
    \[
      \tan \alpha = \dfrac{2t}{1-t^2} = \dfrac{2(r/a)}{1-(r/a)^2} = \dfrac{2ra}{a^2-r^2}
    \]

  \item $\vec{QA} =
    \begin{pmatrix} -r - a \sin\theta \\ -a \cos\theta
    \end{pmatrix}$, $\vec{QB} =
    \begin{pmatrix} r - a \sin\theta \\ -a \cos\theta
    \end{pmatrix}$ から
    \begin{align}
      \tan \beta &= \dfrac{|\vec{QA} \cdot \vec{QB} \cdot \sin\beta|}{\vec{QA} \cdot \vec{QB}} \\
      &= \dfrac{|(-r - a s)(-a c) - (r - a s)(-a c)|}{(r - a s)(-r - a s) + a^2 c^2} \quad (c = \cos\theta, s = \sin\theta) \\
      &= \dfrac{|2 r a c|}{a^2 - r^2} = \dfrac{2 r a c}{a^2 - r^2} \quad (\because 2 r a c > 0)
    \end{align}
    だから
    \[
      \tan \alpha \cdot c = \dfrac{2 r a c}{a^2 - r^2} = \tan \beta
    \]
    である．
\end{enumerate}
