{\bf [解答]}

(イ) の両辺絶対値を取って
\begin{align}
  |z_k|^p = 1 \ \therefore |z_k| = 1, \ z_k \neq 1 \label{eq:1}
\end{align}
である．よって
\begin{align}
  z_k = e(\theta_k) = \cos\theta_k + i\sin\theta_k \ (0 < \theta < 2\pi) \label{eq:2}
\end{align}
とおける．これを(イ)，(ロ)に代入して
\begin{align}
  \begin{cases}
    e(p\theta_k) = 1 \\
    e(\sum_{k=1}^{n}\theta_k) = 1
  \end{cases} \label{eq:3}
\end{align}
である．これを満たす$\{\theta_k\}$ の個数が $a_n$ である．

第1式から自然数 $m_k$ を用いて
\begin{align}
  \theta_k = \dfrac{2\pi}{p}m_k\ (m_k = 1, 2, \cdots, p-1) \label{eq:4}
\end{align}
とおける．したがって $\{\theta_k\}$ の個数は $\{m_k\}$ の数に他ならない．
\cref{eq:3}の第二式に代入して
\begin{align}
  & \sum_{k=1}^{n} \dfrac{2\pi}{p}m_k = 2t\pi \quad (t\in\mathbb{Z}) \\
  \therefore
  & \sum_{k=1}^{n} m_k \equiv 0 \pmod p \label{eq:5}
\end{align}
である．

\paragraph{(1)}

$n=2$の時は\cref{eq:5}を満たす$(m_1,m_2)$の組み合わせは
\begin{align}
  (m_1,m_2) = (1,p-1), (2,p-2), \cdots, (p-1,1)
\end{align}
の$p-1$通りだから
\begin{align}
  a_2 = p-1 \label{eq:6}
\end{align}
である．

同様に考えて$n=3$の時は\cref{eq:5}を満たす$(m_1,m_2,m_3)$の組み合わせは$1 \le m_k \le p-1$より
\begin{align}
  m_1+m_2+m_3 = p, 2p
\end{align}
だから，これを満たす組を数える．まず
\begin{align}
  m_1+m_2+m_3 = p
\end{align}
の時は$p$個の玉の間に$2$本の棒を入れる場合の数に等しく
\begin{align}
  {}_{p-1}C_{2} = \dfrac{(p-1)(p-2)}{2}
\end{align}
である．次に
\begin{align}
  m_1+m_2+m_3 = 2p
  \quad\therefore
  (p-m_1)+(p-m_2)+(p-m_3) = p
\end{align}
において，$1 \le p-m_k \le p-1$だからこの場合の場合の数も同じく
\begin{align}
  {}_{p-1}C_{2} = \dfrac{(p-1)(p-2)}{2}
\end{align}
だけある．よって
\begin{align}
  a_3 = 2\dfrac{(p-1)(p-2)}{2} = (p-1)(p-2) \label{eq:7}
\end{align}
である．

\paragraph{(2)}

$S_n = \sum_{k=1}^n m_k$ とおく．$m_k = 1, 2, \cdots, p-1$ から\cref{eq:5}について
\begin{itemize}
  \item $S_n \equiv 0 \pmod p$の時: $S_{n+1} \not\equiv 0$ \\
  \item $S_n \not\equiv 0 \pmod p$の時: 対応する$m_{n+1}$がただ1つあって$S_{n+1}\equiv 0$となる
\end{itemize}
$S_n$のとり方は個別の$m_k$が$p-1$通りの値をとるから合計で$(p-1)^n$通りある．
よって$S_{n+1} \equiv 0$ となる場合の数は
\begin{align}
  a_{n+1} = (p-1)^n - a_n \label{eq:8}
\end{align}
である．

\paragraph{(3)}

$b_n = \dfrac{a_n}{(p-1)^n}$ とおく．\cref{eq:8}に代入して
\begin{align}
  & b_{n+1} = -\dfrac{1}{p-1} b_n + \dfrac{1}{p-1} \\
  & b_{n+1} - \dfrac{1}{p} = -\dfrac{1}{p-1} \left(b_n - \dfrac{1}{p}\right)
\end{align}
である．\cref{eq:6}より$b_2=\dfrac{1}{p-1}$だから漸化式を繰り返し用いて
\begin{align}
  b_n &= \left(-\dfrac{1}{p-1}\right)^{n-1} \left(-\dfrac{1}{p}\right) + \dfrac{1}{p} \\
  \therefore
  a_n &= \dfrac{1}{p} \left[ (p-1)^n + (-1)^n (p-1) \right]
\end{align}
である．

{\bf [別解]}
\paragraph{(2)}
漸化式は他の表現方法もあるのでそれを調べよう．$a_{n+2}$ についてかんがえる．

\subparagraph{$1^\circ\ m_{n+1} + m_{n+2} \equiv 0$ の時}

このような $(m_{n+1}, m_{n+2})$ のくみあわせは $1 \le m_i \le p-1$ から，$m_{n+1}$ を定めればただ $1$ 通りに定まり$p-1$ 通りである．また$S_{n+2} \equiv 0 \iff S_n \equiv 0$ だから，$m_1 \sim m_n$ の定め方は $a_n$ 通りである．あわせて
\begin{align}
  (p-1) a_n \text{ 通り}
\end{align}

\subparagraph{$2^\circ\ m_{n+1} + m_{n+2} \equiv r\ (r=1, 2, \cdots, p-1)$ の時}

$m_{n+1}$ のえらび方は $1, 2, \cdots, p-1$ から $r$ をのぞいた $p-2$ 通りある．
$m_{n+2}$ は，
\begin{align}
  \begin{cases}
    m_{n+1} < r \text{ ならば } m_{n+2} = r - m_{n+1} \\
    m_{n+1} > r \text{ ならば } m_{n+2} = p + r - m_{n+1}
  \end{cases}
\end{align}
と各々 $1$ 通りに定まる．また$S_{n+2} \equiv 0 \iff S_n + r \equiv 0$ だから，$m_1 \sim m_n$ の定め方は $a_{n+1}$ 通りある．あわせて
\begin{align}
  (p-2) a_{n+1}
\end{align}
通りである．
\casesend

以上で全ての場合がつくされ，$1^\circ, 2^\circ$ は排反だから
\begin{align}
  a_{n+2} = (p-2)a_{n+1} + (p-1)a_n
\end{align}
である．
