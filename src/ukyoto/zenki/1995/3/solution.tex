{\bf [解答]} 3交点の $x$ 座標を，小さい順に $\alpha, \beta, \gamma$ とおく．

又，$f(x) = ax^2 + bx + c$ とする．
\begin{align}
  x^3 - f(x) = (x-\alpha)(x-\beta)(x-\gamma) \cdots \text{\textcircled{1}}
\end{align}
各点での接線を $l_1, l_2, l_3$ として，これらは
\begin{align}
  y = 3k^2 x - 2k^3 \quad (k=\alpha, \beta, \gamma) \cdots \text{\textcircled{2}}
\end{align}
で表される．\textcircled{2} が $P$ を通ることから
\begin{align}
  2k^3 - 3pk^2 + q = 0 \cdots \text{\textcircled{3}}
\end{align}
したがって，$\alpha, \beta, \gamma$ は $k$ の3次式 \textcircled{3} の3実解で，\textcircled{1} から
\begin{align}
  x^3 - f(x) = x^3 - \dfrac{3}{2}p x^2 + \dfrac{q}{2}
\end{align}
係数比較して
\begin{align}
  a = \dfrac{3}{2}p, \quad b = 0, \quad c = -\dfrac{q}{2} \cdots (1)
\end{align}

\paragraph{(2)}

$p, q$ の条件は $x$ の3次方程式 $x^3 - \dfrac{3}{2}px^2 + \dfrac{q}{2} = 0 \cdots \text{\textcircled{4}}$ が3異実解を持つこと．\textcircled{4} の左辺 $g(x)$ とおく．$g'(x) = 3x^2 - 3px$ から，$p$ によって下のように分かれる

$1^\circ \ p = 0$ の時

$g'(x) \ge 0$ となり，$g(x)$ は単調増加し，\textcircled{4} が3異実解を持つことはない．

$2^\circ \ p > 0$ の時

\begin{center}
  \begin{table}[htb]
    \centering
    \caption{$g(x)$ の増減表}
    \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      $x$ & $\cdots$ & $0$ & $\cdots$ & $p$ & $\cdots$ \\
      \hline
      $g'$ & $+$ & $0$ & $-$ & $0$ & $+$ \\
      \hline
      $g$ & $\nearrow$ & & $\searrow$ & & $\nearrow$ \\
      \hline
    \end{tabular}
  \end{table}
\end{center}
上図から，条件は
\begin{align}
  & g(0) > 0 \land g(p) < 0 \\
  \iff & \dfrac{q}{2} > 0 \land - \dfrac{p^3}{2} + \dfrac{q}{2} < 0 \\
  \iff & q > 0 \land -p^3 + q < 0
\end{align}

$3^\circ \ p < 0$ の時

$2^\circ$ と同様に，条件は
\begin{align}
  g(p) > 0 \land g(0) < 0 \iff q < 0 \land -p^3 + q > 0
\end{align}

以上まとめて
\begin{align}
  \begin{cases}
    p > 0, \quad 0 < q < p^3 \\
    \text{又は} \\
    p < 0, \quad p^3 < q < 0
  \end{cases}
\end{align}
