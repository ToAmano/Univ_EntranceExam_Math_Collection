\paragraph{(1)}

与えられた曲線
\begin{align}
  C_1:y=f(x)=x^3,
  \quad
  C_2:y=g(x)=ax^2+bx+c \label{eq:1}
\end{align}
について考える．
$C_1,C_2$の3交点の $x$ 座標を小さい順に $\alpha, \beta, \gamma$ とおくと，
因数定理から
\begin{align}
  f(x)-g(x) = (x-\alpha)(x-\beta)(x-\gamma) \label{eq:2}
\end{align}
である．また各点での$C_1$の接線を $l_1, l_2, l_3$ とすると
\begin{align}
  y = 3k^2 x - 2k^3 \quad (k=\alpha, \beta, \gamma) \label{eq:3}
\end{align}
で表される．\cref{eq:3} が $P(p,q)$ を通ることから
\begin{align}
  h(k) = 2k^3 - 3pk^2 + q = 0 \label{eq:4}
\end{align}
である．したがって $\alpha, \beta, \gamma$ は $k$ の3次式 \cref{eq:4} の3実解である．
一方で\cref{eq:1}から$f(x)-g(x)=0$の3実解でもあるから，\cref{eq:1,eq:4}が比例係数をのぞいて等しく，
\begin{align}
  f(x)-g(x) = x^3 - \dfrac{3}{2}p x^2 + \dfrac{q}{2}
\end{align}
となる．係数比較して
\begin{align}
  a = \dfrac{3}{2}p, \quad b = 0, \quad c = -\dfrac{q}{2} \label{eq:5}
\end{align}
である．

\paragraph{(2)}

$p, q$ の条件は $k$ の3次方程式\cref{eq:4}が3異実解を持つことである．
一階微分は
\begin{align}
  h'(x) = 3x^2 - 3px = 3x(x-p) \label{eq:6}
\end{align}
だから$p$が$0$かいなかに応じて$h(x)$の挙動は以下のように変化する．

\subparagraph{$1^\circ \ p = 0$ の時}

\begin{table}[htb]
  \centering
  \caption{$h(x)$ の増減表（$p=0$ のとき）}
  \label{tab:variation_p0}
  \begin{tabular}{|c|c|c|c|}
    \hline
    $x$ & $\cdots$ & $0$ & $\cdots$ \\
    \hline
    $h'$ & $+$ & $0$ & $+$ \\
    \hline
    $h$ & $\nearrow$ & $q$ & $\nearrow$ \\
    \hline
  \end{tabular}
\end{table}

この時\cref{eq:6}より$h'(x) \ge 0$ となるから $h(x)$ は単調増加する．
よって\cref{eq:4}が3異実解を持つことはない．

\subparagraph{$2^\circ \ p > 0$ の時}

\begin{table}[htb]
  \centering
  \caption{$h(x)$ の増減表}
  \begin{tabular}{|c|c|c|c|c|c|}
    \hline
    $x$ & $\cdots$ & $0$ & $\cdots$ & $p$ & $\cdots$ \\
    \hline
    $g'$ & $+$ & $0$ & $-$ & $0$ & $+$ \\
    \hline
    $g$ & $\nearrow$ & & $\searrow$ & & $\nearrow$ \\
    \hline
  \end{tabular}
\end{table}

この時\cref{eq:6}より$h(x)$の増減表は上のようになる．
よって$h(x)=0$が3異実解を持つ条件は
\begin{align}
  & h(0) > 0 \land h(p) < 0 \\
  \iff & \dfrac{q}{2} > 0 \land - \dfrac{p^3}{2} + \dfrac{q}{2} < 0 \\
  \iff & q > 0 \land -p^3 + q < 0 \label{eq:7}
\end{align}
である．

\subparagraph{$3^\circ \ p < 0$ の時}

\begin{table}[htb]
  \centering
  \caption{$h(x)$ の増減表（$p<0$ のとき）}
  \label{tab:variation_p_neg}
  \begin{tabular}{|c|c|c|c|c|c|}
    \hline
    $x$ & $\cdots$ & $p$ & $\cdots$ & $0$ & $\cdots$ \\
    \hline
    $h'$ & $+$ & $0$ & $-$ & $0$ & $+$ \\
    \hline
    $h$ & $\nearrow$ & & $\searrow$ & & $\nearrow$ \\
    \hline
  \end{tabular}
\end{table}

この時\cref{eq:6}より$h(x)$の増減表は上のようになる．
よって$h(x)=0$が3異実解を持つ条件は
\begin{align}
  h(p) > 0 \land h(0) < 0 \iff q < 0 \land -p^3 + q > 0 \label{eq:8}
\end{align}
である．
\casesend

以上\cref{eq:7,eq:8}をまとめて
\begin{align}
  \begin{cases}
    p > 0, \quad 0 < q < p^3 \\
    \text{又は} \\
    p < 0, \quad p^3 < q < 0
  \end{cases}
\end{align}
である．
