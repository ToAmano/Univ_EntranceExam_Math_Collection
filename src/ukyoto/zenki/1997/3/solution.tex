与えられた二つの放物線を
\begin{align}
  C_1: y=x^2+1, C_2: y=kx^2 \label{eq:1}
\end{align}
とする．$C_1$上の点$P(a,a^2+1)$における接線は
\begin{align}
  L: y = f(x) = 2ax - a^2 + 1 \label{eq:2}
\end{align}
である．$L$と$C_2$の交点の$x$座標を$\alpha, \beta \ (\alpha<\beta)$とする．
$C_1$と$C_2$の交点の$x$座標は$p = \sqrt{\dfrac{1}{k-1}}$とおいて$\pm p$である．

$C_1$と$C_2$で囲まれた部分の面積$S_1$，$L$と$C_2$で囲まれた面積$S_2$とすると，題意から
\begin{align}
  S_1 = 2S_2 \cdots \label{eq:3}
\end{align}
である．$S_1,S_2$はそれぞれ
\begin{align}
  S_1
  &= \int_{-p}^p \{ x^2+1 - kx^2 \} dx \\
  &= \dfrac{k-1}{6} (2p)^3 \\
  &= \dfrac{4}{3} (k-1) \left( \dfrac{1}{k-1} \right)^{\frac{3}{2}} \\
  &= \dfrac{4}{3} \dfrac{1}{\sqrt{k-1}} \label{eq:4}
\end{align}
および
\begin{align}
  S_2
  & = \int_\alpha^\beta \{ f(x) - kx^2 \} dx \\
  &= \dfrac{k}{6}(\beta-\alpha)^3 \label{eq:5}
\end{align}
となる．$\alpha \le \beta$およびこれらが$f(x)-kx^2=0$の解であることから
\begin{align}
  \beta - \alpha = \dfrac{2}{k}\sqrt{(1-k)a^2+k}
\end{align}
だから\cref{eq:5}に代入して，
\begin{align}
  S_2
  & = \dfrac{k}{6} \left( \dfrac{2}{k} \right)^3 \left\{ (1-k)a^2+k \right\}^{\frac{3}{2}}\\
  & = \dfrac{4}{3}\dfrac{1}{k^2} \left\{ (1-k)a^2+k \right\}^{\frac{3}{2}} \label{eq:6}
\end{align}
である．

\cref{eq:4,eq:6}を\cref{eq:3}に代入して
\begin{align}
  S_2 & = \dfrac{4}{3}\dfrac{1}{k^2} \left\{ (1-k)a^2+k \right\}^{\frac{3}{2}} \\
  & = \dfrac{2}{3} \dfrac{1}{\sqrt{k-1}} \\
  \dfrac{1}{\sqrt{k-1}} & = \dfrac{2}{k^2}\left\{ (1-k)a^2+k \right\}^{\frac{3}{2}}
\end{align}
である．両辺0以上から$\dfrac{2}{3}$乗して，
\begin{align}
  & \left(\dfrac{1}{k-1}\right)^{\frac{1}{3}} = \left(\dfrac{2}{k^2}\right)^{\frac{2}{3}} \{ (1-k)a^2+k \} \\
  \therefore
  & a^2 = - \dfrac{1}{k-1}\left[ \left(\dfrac{k^4}{4(k-1)}\right)^{\frac{1}{3}} - k \right] \label{eq:7}
\end{align}
である．

\paragraph{(2)}

\cref{eq:7}を変形して求める極限値は
\begin{align}
  a^2
  & = - \left\{ \dfrac{k^4}{4(k-1)^4} \right\}^{\frac{1}{3}} + \dfrac{k}{k-1} \\
  & = - \left( \dfrac{1}{4} \right)^{\frac{1}{3}} \left( \dfrac{1}{1-1/k} \right)^{\frac{4}{3}} + \dfrac{1}{1-1/k} \\
  & \xrightarrow{k\to\infty} 1 - \left( \dfrac{1}{4} \right)^{\frac{1}{3}}
\end{align}
である．
