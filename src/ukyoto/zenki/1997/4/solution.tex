\paragraph{(1)}

題意の不等式
\begin{align}
  \int_\alpha^\beta \sin xdx + \int_{\pi-\beta}^{\pi-\alpha} \sin xdx >
  (\beta-\alpha)(\sin\alpha+\sin(\pi-\beta))
  \quad  \left(0\leqq\alpha<\beta\leqq\dfrac{\pi}{2} \right) \label{eq:1}
\end{align}
について考える．左辺第二項で$t = \pi - x$と置換すると$\dfrac{dt}{dx} = -1$で
\begin{align}
  \int_{\pi-\beta}^{\pi-\alpha} \sin xdx
  & =  \int_\beta^\alpha \sin(\pi-t) dt (-1) \\
  & = \int_\alpha^\beta \sin t dt
\end{align}
だから，\cref{eq:1}は
\begin{align}
  \int_0^\beta \sin t dt > \dfrac{1}{2}(\beta-\alpha)(\sin\alpha + \sin\beta) \label{eq:2}
\end{align}
を示せば良い．\cref{eq:2}の左辺，右辺は下図の斜線部の面積である．

\end{align}
で，これが\cref{eq:2}に他ならない．よって題意は示された．

\paragraph{(2)}

\cref{eq:2}で $(\alpha, \beta) = \left( \dfrac{k\pi}{8}, \dfrac{k+1}{8}\pi \right) \ (k=0,1,2,3)$とすると，これは$0 \le \alpha < \beta \le \dfrac{\pi}{2}$をみたすので，
\begin{align}
  \int_{\frac{k\pi}{8}}^{\frac{k+1}{8}\pi} \sin t dt > \dfrac{1}{2} \cdot \dfrac{\pi}{8} \left( \sin\dfrac{k\pi}{8} + \sin\dfrac{k+1}{8}\pi \right)
\end{align}
だから，$k=0,1,2,3$について足し合わせると
\begin{align}
  \int_0^{\frac{\pi}{2}} \sin t dt
  &> \dfrac{\pi}{16} \sum_{k=0}^{3}\left(\sin\dfrac{k\pi}{8} + \sin\dfrac{k+1}{8}\pi\right) \\
  &= \dfrac{\pi}{16} \sum_{k=0}^{3}\left(\sin\dfrac{k\pi}{8} + \sin\dfrac{8-(k+1)}{8}\pi\right) \quad (\because \sin(\pi-x)=\sin x) \\
  &= \dfrac{\pi}{16} \sum_{k=0}^{7} \sin\dfrac{k\pi}{8}
\end{align}
である．左辺の積分は$1$であり，右辺は$k=0$の項は$0$だから
\begin{align}
  1 > \dfrac{\pi}{16} \sum_{k=1}^{7} \sin\dfrac{k\pi}{8}
  \therefore
  \dfrac{16}{\pi} > \sum_{k=1}^7 \sin\dfrac{k\pi}{8}
\end{align}
である．
