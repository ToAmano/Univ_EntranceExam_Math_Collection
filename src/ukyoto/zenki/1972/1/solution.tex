{\bf [解答]}

与えられた方程式の左辺と右辺を
\begin{align}
  \vec{S}_n = \sum_{k=1}^n \vec{A}_k, \quad
  \vec{T}_n = \sum_{k=1}^n \vec{B}_k \label{eq:1}
\end{align}
とおく．

ベクトルの2つの公理
\begin{align}
  & \vec{A} + \vec{B} = \vec{B} + \vec{A} \label{eq:2} \\
  & (\vec{A} + \vec{B}) + \vec{C} = \vec{A} + (\vec{B} + \vec{C}) \label{eq:3}
\end{align}
を用いて
\begin{align}
  \vec{S}_n = \vec{T}_n \label{eq:4}
\end{align}
を数学的帰納法により示す．

証明の前に以下の補題Iを示す．

\begin{lemma}[補題I]
  任意のベクトル $\vec{a}_1, \dots, \vec{a}_n$ に対し ($n \ge 2$)
  \begin{align}
    \vec{a}_1 + \cdots + \vec{a}_n = \vec{a}_2 + \vec{a}_3 + \cdots + \vec{a}_n + \vec{a}_1 \quad \cdots \label{eq:5}
  \end{align}
  が成り立つ．
\end{lemma}

\begin{proof}
  証明は数学的帰納法による. $n=2$ の時は\cref{eq:5}は\cref{eq:2}に他ならない.
  そこで$n=k$ での成立を仮定して$n=k+1$ の時を考えると，
  \begin{align}
    \vec{a}_1 + \cdots + \vec{a}_{k+1}
    &= \vec{a}_1 + \cdots + \vec{a}_{k-1} + (\vec{a}_k + \vec{a}_{k+1}) \quad (\because \text{\cref{eq:3}}) \\
    &= \vec{a}_2 + \cdots + (\vec{a}_k + \vec{a}_{k+1}) + \vec{a}_1 \quad (\because \text{仮定}) \\
    &= \vec{a}_2 + \cdots + \vec{a}_k + \vec{a}_{k+1} + \vec{a}_1 \quad (\because \text{\cref{eq:3}})
  \end{align}
  である．ゆえに $n=k+1$ でも\cref{eq:5}は成立する．
  以上から帰納法により\cref{eq:5}は示された.
\end{proof}

以下数学的帰納法によって\cref{eq:4}を示す．

まず$n=1$ の時は\cref{eq:4}は
\begin{align}
  \vec{A}_1 = \vec{B}_1
\end{align}
で，これは定義から成立する.

そこで $i \in \mathbb{N}$ に対し, $n \le i$ での\cref{eq:4}の成立を仮定する.
$\vec{A},\vec{B}$の定義から,
\begin{align}
  \vec{A}_{i+1} = \vec{B}_{k+1} \quad (k=0,1,\cdots,i)
\end{align}
なる $k$ があることに注意する. この$k$の値で場合分けする．

\subparagraph{$1^\circ$ $k=i$ の時}

この時は
\begin{align}
  \vec{T}_{i+1}
  &= \vec{T}_i + \vec{B}_{i+1} \\
  &= \vec{S}_i + \vec{A}_{i+1} \quad (\because \text{仮定}, \vec{A}_{i+1} = \vec{B}_{i+1}) \\
  &= \vec{S}_{i+1}
\end{align}
となり, $n=i+1$ でも\cref{eq:4}は成立する．

\subparagraph{$2^\circ$ $k \ne i$ の時}

この時は
\begin{align}
  \vec{T}_{i+1}
  &= \vec{B}_1 + \dots + \vec{B}_k + \vec{B}_{k+1} + \vec{B}_{k+2} + \dots + \vec{B}_{i+1} \\
  &= \vec{B}_{k+1} + \vec{B}_1 + \dots + \vec{B}_k + \vec{B}_{k+2} + \dots + \vec{B}_{i+1} \quad (\because \text{仮定}) \\
  &= \vec{B}_1 + \dots + \vec{B}_k + \vec{B}_{k+2} + \dots + \vec{B}_{i+1} + \vec{B}_{k+1} \quad (\because \text{補題1}) \\
  &= \vec{S}_{i+1} \quad (\because 1^\circ \text{に帰着})
\end{align}
となり，$n=i+1$ でも\cref{eq:4}は成立する．
\casesend

以上 $1^\circ, 2^\circ$ から, $n=i+1$ でも\cref{eq:4}は成立する.
ゆえに数学的帰納法により\cref{eq:4}は示された.
