[\textbf{解}]
\begin{align}
F(x) &= \int_2^x \dfrac{1}{2}\left( \dfrac{1}{t+1} - \dfrac{3}{t+3} \right) dt \\
&= \dfrac{1}{2} \left[ \log(t+1) - 3\log(t+3) \right]_2^x \\
&= \dfrac{1}{2} \left( \log(x+1) - 3\log(x+3) + 3\log 3 \right)
\end{align}
だから
\begin{align}
F(x) - \log x &= -\dfrac{3}{2}\log 3 + \dfrac{3}{2}\log(x+3) - \dfrac{1}{2}\log(x+1) - \log x \\
&= -\dfrac{3}{2}\log 3 + \dfrac{1}{2}\log \dfrac{(x+3)^3}{(x+1)x^2} \\
&= -\dfrac{3}{2}\log 3 + \dfrac{1}{2}\log \dfrac{(1 + \dfrac{3}{x})^3}{1 + \dfrac{1}{x}} \longrightarrow -\dfrac{3}{2}\log 3 \quad (x \to \infty)
\end{align}
$(\because \log x \text{ は連続})$
