[解] $n$回目の試行前袋に赤玉が $i$ コ入っている確率 $P_{i,n}$ とおく．
\begin{align}
  \paragraph{(1)}

  \
  \begin{cases}
    n+1\text{回目の試行前に赤玉が1コ} \cdots \quad Q_{1,n} + P_{2,n} \\
    \qquad \qquad \text{〃} \qquad \qquad 2\text{コ} \cdots \quad Q_{2,n} + P_{3,n} \\
    \qquad \qquad \text{〃} \qquad \qquad 3\text{コ} \cdots \quad Q_{3,n}
  \end{cases}
\end{align}
又
\begin{align}
  Q_{1,n} = (N+2)P_{1,n} \quad Q_{2,n} = \dfrac{N+1}{2} P_{2,n} \quad Q_{3,n} = \dfrac{N}{3} P_{3,n}
\end{align}
だから
\begin{align}
  \begin{cases}
    P_{1,n+1} = \dfrac{1}{N+3} ( (N+2) P_{1,n} + P_{2,n} ) \\
    P_{2,n+1} = \dfrac{2}{N+3} ( \dfrac{N+1}{2} P_{2,n} + P_{3,n} ) \\
    P_{3,n+1} = \dfrac{3}{N+3} \cdot \dfrac{N}{3} P_{3,n} = \dfrac{N}{N+3} P_{3,n}
  \end{cases} \cdots \#
\end{align}

\paragraph{(2)}

$p_n = P_{1,n} + P_{2,n} + P_{3,n}$ だから(1)より
\begin{align}
  p_{n+1} = \dfrac{N+2}{N+3} p_n \quad \cdots \text{①}
\end{align}
又，$p_1 = \dfrac{3}{N+3}$ だから，①と等比数列の公式から
\begin{align}
  p_n = \left(\dfrac{N+2}{N+3}\right)^{n-1} \dfrac{3}{N+3}
\end{align}
