\noindent
[\textbf{解}]
\begin{enumerate}
  \item[(1)] $n$ 個の頂点の中から3つ選べば三角形が1つ対応するので，
    \[
      _n\mathrm{C}_3 = \dfrac{1}{6}n(n-1)(n-2)
    \]

  \item[(2)] (1)から，鈍角三角形，直角三角形のものをのぞけばよい．

    \begin{enumerate}
      \item[$1^\circ$] $n = 2k\ (k \in \mathbb{N}_{\ge 2})$ の時

        1つの頂点を $A_1$ とする．$A_1 A_{k+1}$ が外接円の直径だから，鈍角をつくるには右図の分領域から2点を考えればよく，
        \[
          \begin{cases}
            k \ge 3 \text{のとき} & _{k-1}\mathrm{C}_2 \text{通り} \\
            k \le 2 \text{の時} & 0 \text{通り}
          \end{cases}
        \]
        1つの頂点が他にある場合と重複を考えて，
        \[
          \begin{cases}
            k \ge 3 \text{の時} & 2k \cdot _{k-1}\mathrm{C}_2 / 2 = k(k-1)(k-2) \text{通り} \\
            k \le 2 \text{の時} & 0 \text{通り}
          \end{cases}
        \]

        \begin{figure}[htb]
          \centering
          \begin{tikzpicture}[scale=1.55,
            every node/.style={font=\small}]
            % n=8 illustrates n=2k: three vertices on each open semicircle.
            \draw[thin, gray!55] (0,0) circle[radius=1.2];
            \draw[line width=1.3pt, teal!75!black]
            (90:1.2) arc[start angle=90, end angle=-90, radius=1.2];
            \draw[line width=1.3pt, orange!85!black]
            (-90:1.2) arc[start angle=-90, end angle=-270, radius=1.2];
            \draw[dashed, gray!70] (90:1.2)--(-90:1.2);
            \foreach \ang in {45,0,-45,-135,180,135}
            \fill[blue!85!black] (\ang:1.2) circle(1.4pt);
            \fill[red!80!black] (90:1.2) circle(1.6pt)
            node[above=2pt] {$A_1$};
            \fill[red!80!black] (-90:1.2) circle(1.6pt)
            node[below=2pt] {$A_{k+1}$};
            \node[teal!75!black, font=\footnotesize] at (1.65,0)
            {$k-1$ 個};
            \node[orange!85!black, font=\footnotesize] at (-1.65,0)
            {$k-1$ 個};
          \end{tikzpicture}
          \caption{偶数個の頂点：直径 $A_1A_{k+1}$ の両側にある頂点}
          \label{fig:1966_6_even_vertices}
        \end{figure}

        一方，直角三角形は1つの直径となる辺に対し，のこり1つの辺のえらび方が $2k-2$ 通りあるから，
        \[
          k(2k-2) \text{通り}
        \]

      \item[$2^\circ$] $n = 2k-1$ の時

        1°と同様に考える．1つの頂点を $A_1$ に固定した時，鈍角三角形は
        \[
          \begin{cases}
            k \ge 3 \text{のとき} & _{k-1}\mathrm{C}_2 \text{通り} \\
            k \le 2 \text{の時} & 0 \text{通り}
          \end{cases}
        \]
        だから，あわせて
        \[
          \begin{cases}
            k \ge 3 \text{の時} & (2k-1) \cdot _{k-1}\mathrm{C}_2 \text{通り} \\
            k = 2 \text{の時} & 0 \text{通り}
          \end{cases}
        \]
        直角三角形は存在しない．

        \begin{figure}[htb]
          \centering
          \begin{tikzpicture}[scale=1.55,
            every node/.style={font=\small}]
            % n=7 illustrates n=2k-1; the opposite point is between A_k and A_{k+1}.
            \draw[line width=1.2pt, blue!85!black] (0,0)
            circle[radius=1.2];
            \draw[dashed, gray!70] (90:1.2)--(-90:1.2);
            \foreach \ang in {38.571,-12.857,-167.143,141.429}
            \fill[blue!85!black] (\ang:1.2) circle(1.4pt);
            \fill[red!80!black] (90:1.2) circle(1.6pt)
            node[above=2pt] {$A_1$};
            \fill[teal!75!black] (-64.286:1.2) circle(1.6pt)
            node[below right=2pt] {$A_k$};
            \fill[orange!85!black] (-115.714:1.2) circle(1.6pt)
            node[below left=2pt] {$A_{k+1}$};
            \fill[gray!75] (-90:1.2) circle(1.1pt);
          \end{tikzpicture}
          \caption{奇数個の頂点：$A_1$ の反対側は $A_k$ と $A_{k+1}$ の間}
          \label{fig:1966_6_odd_vertices}
        \end{figure}
    \end{enumerate}

    以上から，求める数は $n=3$ の時 1，$n=4$ の時 0 で，$n \ge 5$ の時
    \[
      n \in \text{even の時}: \quad \dfrac{1}{6}n(n-1)(n-2) - \dfrac{n}{2}\left(\dfrac{n}{2}-1\right)\left(\dfrac{n}{2}-2\right) - \dfrac{n}{2}(n-2) = \dfrac{1}{24}n(n-2)(n-4)
    \]
    \[
      n \in \text{odd の時}: \quad \dfrac{1}{6}n(n-1)(n-2) - \dfrac{1}{2}n\left(\dfrac{n-1}{2}\right)\left(\dfrac{n-3}{2}\right) = \dfrac{1}{24}n(n+1)(n-1)
    \]
    これらは $n=3, 4$ でも成立するから
    \[
      \begin{cases}
        n \in \text{odd} & \dfrac{1}{24}n(n+1)(n-1) \\
        n \in \text{even} & \dfrac{1}{24}n(n-2)(n-4)
      \end{cases}
    \]
    である．
\end{enumerate}
