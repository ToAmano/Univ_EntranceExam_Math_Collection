[\textbf{解}] $P$ が $ABCD$ を1周する時，
\[
  \begin{cases}
    AB \text{上} \dots y = 1, -1 \le x \le 1 & \dots \text{①} \\
    BC \text{上} \dots x = -1, -1 \le y \le 1 & \dots \text{②} \\
    CD \text{上} \dots y = -1, -1 \le x \le 1 & \dots \text{③} \\
    DA \text{上} \dots x = 1, -1 \le y \le 1 & \dots \text{④}
  \end{cases}
\]
である．従って，$Q(u, v)$ のキセキは以下．($a > 1$ に注意)

\begin{enumerate}
  \item[$1^\circ$] ①の時
    \[
      u = a \cdot a^x, \quad v = \dfrac{1}{a} \cdot a^{2x} \quad (-1 \le x \le 1)
    \]
    から，
    \[
      v = \dfrac{1}{a^3} u^2 \quad (1 \le u \le a^2)
    \]

  \item[$2^\circ$] ②の時
    \[
      u = \dfrac{1}{a} \cdot a^y, \quad v = \dfrac{1}{a^2} \cdot a^{-y} \quad (-1 \le y \le 1)
    \]
    から
    \[
      v = \dfrac{1}{a^3} \dfrac{1}{u} \quad \left(\dfrac{1}{a^2} \le u \le 1\right)
    \]

  \item[$3^\circ$] ③の時
    \[
      u = \dfrac{1}{a} \cdot a^x, \quad v = a \cdot a^{2x} \quad (-1 \le x \le 1)
    \]
    から
    \[
      v = a^3 u^2 \quad \left(\dfrac{1}{a^2} \le u \le 1\right)
    \]

  \item[$4^\circ$] ④の時
    \[
      u = a \cdot a^y, \quad v = a^2 \cdot a^{-y} \quad (-1 \le y \le 1)
    \]
    から
    \[
      v = a^3 \dfrac{1}{u} \quad (1 \le u \le a^2)
    \]
\end{enumerate}

図示して右図のようになる．求める面積 $S$ のうち $\dfrac{1}{a^2} \le u \le 1$ のものを $S_1$, $1 \le u \le a^2$ のものを $S_2$ として，
\[
  S = S_1 + S_2 \quad \dots \text{⑤}
\]

\begin{align}
  S_1 &= \int_{\frac{1}{a^2}}^1 \left( a^3 u^2 - \dfrac{1}{a^3 u} \right) du \\
  &= \left[ \dfrac{a^3}{3} u^3 - \dfrac{1}{a^3} \log u \right]_{\frac{1}{a^2}}^1 \\
  &= \dfrac{a^3}{3} - \left( \dfrac{1}{3a^3} + \dfrac{2}{a^3} \log a \right) \\
  &= \dfrac{1}{3}a^3 - \dfrac{1}{3a^3} - \dfrac{2}{a^3}\log a \\[10pt]
  S_2 &= \int_1^{a^2} \left( a^3 \dfrac{1}{u} - \dfrac{1}{a^3} u^2 \right) du \\
  &= \left[ a^3 \log u - \dfrac{1}{3a^3} u^3 \right]_1^{a^2} \\
  &= 2a^3 \log a - \dfrac{1}{3}a^3 + \dfrac{1}{3a^3}
\end{align}
⑤へ代入して，
\[
  S = 2\left(a^3 - \dfrac{1}{a^3}\right)\log a
\]
