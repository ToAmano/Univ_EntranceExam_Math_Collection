与えられた変換
\begin{align}
  u = a^{x+y}, \quad
  v = a^{2x-y} \label{eq:1}
\end{align}
について考える．

$P(x,y)$ が $ABCD$ を1周する時，各辺上での座標は
\begin{align}
  \begin{cases}
    AB \text{上} \dots y = 1, -1 \le x \le 1  \\
    BC \text{上} \dots x = -1, -1 \le y \le 1 \\
    CD \text{上} \dots y = -1, -1 \le x \le 1 \\
    DA \text{上} \dots x = 1, -1 \le y \le 1
  \end{cases} \label{eq:2}
\end{align}
である．従って，$a>0$にも注意して$Q(u, v)$ の軌跡は以下のようになる．

\subparagraph{$1^\circ$ $P$が辺$AB$上を動く時}

この時\cref{eq:1,eq:2}から$Q$の座標は
\begin{align}
  u = a \cdot a^x, \quad v = \dfrac{1}{a} \cdot a^{2x} \quad (-1 \le x \le 1)
\end{align}
と書けるから$x$を消去して
\begin{align}
  v = \dfrac{1}{a^3} u^2 \quad (1 \le u \le a^2)
\end{align}
である．

\subparagraph{$2^\circ$ $P$が辺$BC$上を動く時}

この時\cref{eq:1,eq:2}から$Q$の座標は
\begin{align}
  u = \dfrac{1}{a} \cdot a^y, \quad v = \dfrac{1}{a^2} \cdot a^{-y} \quad (-1 \le y \le 1)
\end{align}
と書けるから$y$を消去して
\begin{align}
  v = \dfrac{1}{a^3} \dfrac{1}{u} \quad \left(\dfrac{1}{a^2} \le u \le 1\right)
\end{align}
である．

\subparagraph{$3^\circ$ $P$が辺$CD$上を動く時}

この時\cref{eq:1,eq:2}から$Q$の座標は
\begin{align}
  u = \dfrac{1}{a} \cdot a^x, \quad v = a \cdot a^{2x} \quad (-1 \le x \le 1)
\end{align}
と書けるから$x$を消去して
\begin{align}
  v = a^3 u^2 \quad \left(\dfrac{1}{a^2} \le u \le 1\right)
\end{align}
である．

\subparagraph{$4^\circ$ $P$が辺$DA$上を動く時}

この時\cref{eq:1,eq:2}から$Q$の座標は
\begin{align}
  u = a \cdot a^y, \quad v = a^2 \cdot a^{-y} \quad (-1 \le y \le 1)
\end{align}
と書けるから$y$を消去して
\begin{align}
  v = a^3 \dfrac{1}{u} \quad (1 \le u \le a^2)
\end{align}
である．
\casesend

これらをまとめると
\begin{align}
  \begin{cases}
    v = \dfrac{1}{a^3} u^2 &  (1 \le u \le a^2) \\
    v = \dfrac{1}{a^3} \dfrac{1}{u} & \left(\dfrac{1}{a^2} \le u \le 1\right) \\
    v = a^3 u^2 & \left(\dfrac{1}{a^2} \le u \le 1\right) \\
    v = a^3 \dfrac{1}{u} & (1 \le u \le a^2)
  \end{cases} \label{eq:3}
\end{align}
であり，図示すると下図である．

求める面積 $S$ のうち $\dfrac{1}{a^2} \le u \le 1$ のものを $S_1$, $1 \le u \le a^2$ のものを $S_2$ として，
\begin{align}
  S = S_1 + S_2 \label{eq:4}
\end{align}
であり，各項計算すると\cref{eq:3}より
\begin{align}
  S_1
  &= \int_{\frac{1}{a^2}}^1 \left( a^3 u^2 - \dfrac{1}{a^3 u} \right) du \\
  &= \left[ \dfrac{a^3}{3} u^3 - \dfrac{1}{a^3} \log u \right]_{\frac{1}{a^2}}^1 \\
  &= \dfrac{a^3}{3} - \left( \dfrac{1}{3a^3} + \dfrac{2}{a^3} \log a \right) \\
  &= \dfrac{1}{3}a^3 - \dfrac{1}{3a^3} - \dfrac{2}{a^3}\log a
\end{align}
および
\begin{align}
  S_2
  &= \int_1^{a^2} \left( a^3 \dfrac{1}{u} - \dfrac{1}{a^3} u^2 \right) du \\
  &= \left[ a^3 \log u - \dfrac{1}{3a^3} u^3 \right]_1^{a^2} \\
  &= 2a^3 \log a - \dfrac{1}{3}a^3 + \dfrac{1}{3a^3}
\end{align}
だから\cref{eq:4}に代入して求める面積は
\begin{align}
  S = 2\left(a^3 - \dfrac{1}{a^3}\right)\log a
\end{align}
