\noindent
[\textbf{解}] $P$ での接線は $\dfrac{x_1}{a^2}x - \dfrac{y_1}{b^2}y = 1 \dots \text{①}$ であるから，$Q\left(\dfrac{a^2}{x_1}, 0\right)$ である．

\begin{enumerate}
  \item[(1)] まず $\dfrac{y_1^2}{b^2} = \dfrac{x_1^2}{a^2} - 1$ 及び $y_1, b > 0$ から
    \[
      y_1 = b \sqrt{\dfrac{x_1^2}{a^2} - 1}
    \]
    だから
    \[
      \triangle OPQ = \dfrac{1}{2} \dfrac{a^2}{x_1} b \sqrt{\dfrac{x_1^2}{a^2} - 1}
    \]

  \item[(2)]
    \[
      \begin{aligned}
        \triangle OPQ &= \dfrac{ab}{2} \sqrt{\dfrac{x_1^2 - a^2}{x_1^2}} \\
        &= \dfrac{ab}{2} \sqrt{1 - a^2 / x_1^2} \xrightarrow{x_1 \to \infty} \dfrac{ab}{2}
      \end{aligned}
    \]
\end{enumerate}
