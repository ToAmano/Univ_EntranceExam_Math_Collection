\begin{enumerate}
  \item[(1)] $P$ が $AB$ 上にある時, $t$ にたいして
    \begin{align}
      \overrightarrow{OP} = \overrightarrow{OA} + t \overrightarrow{AB}
    \end{align}
    とかける．$t$ にたいして
    \begin{align}
      \overrightarrow{OP} = (1-t)\overrightarrow{OA} + t\overrightarrow{OB}
    \end{align}
    だから $a = 1-t, b = t$ とおけば, 十分性が示される $\dots \text{①}$

    次に, $\overrightarrow{OP} = a\overrightarrow{OA} + b\overrightarrow{OB}, a+b=1$ とおける時, $b=1-a$ から
    \begin{align}
      \overrightarrow{OP} &= a\overrightarrow{OA} + (1-a)\overrightarrow{OB} \\
      &= \overrightarrow{OB} + a\overrightarrow{BA}
    \end{align}
    となり, たしかに $P$ は $AB$ 上にある．よって必要性が示された $\dots \text{②}$

    ①, ② から示された．(終)

  \item[(2)] $p+q+r=0$ から, $r=-(p+q)$ だから,
    \begin{align}
      p(\overrightarrow{OA} - \overrightarrow{OC}) + q(\overrightarrow{OB} - \overrightarrow{OC}) = 0
    \end{align}
    \begin{align}
      p\overrightarrow{CA} + q\overrightarrow{CB} = 0 \quad \dots \text{③}
    \end{align}
    ここで, $A, B, C$ は一直線上にないから③をみたす $p, q$ は $p=q=0$ のみ．よって
    \begin{align}
      r = 0
    \end{align}
    となるから $(p, q, r) = (0, 0, 0)$ である．(終)

  \item[(3)] $\overrightarrow{OQ} = \overrightarrow{OA}' + \overrightarrow{AQ}' = \overrightarrow{OA} + x\overrightarrow{AB} + y\overrightarrow{AC}$ ($\because$題意)から,
    \begin{align}
      \overrightarrow{OQ} &= (1-x-y)\overrightarrow{OA} + x\overrightarrow{OB} + y\overrightarrow{OC} \\
      &= l\overrightarrow{OA} + m\overrightarrow{OB} + n\overrightarrow{OC}
    \end{align}
    (2)から, $\overrightarrow{OA}, \overrightarrow{OB}, \overrightarrow{OC}$ の係数比較ができて
    \begin{align}
      l = 1-(x+y), \quad m = x, \quad n = y \quad \dots \text{④}
    \end{align}
    これは $l+m+n = 1$ をみたし十分．したがって④のように一意に表せ, 又必ず存在する．(終)
\end{enumerate}
