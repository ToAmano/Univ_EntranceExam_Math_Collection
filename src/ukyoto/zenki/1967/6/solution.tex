{\bf [解答]} $y_1 = f(x) = \dfrac{\pi}{4}\dfrac{1}{x} - \dfrac{\pi}{4} + 1$, $y_2 = g(x) = \sqrt{2}\cos \dfrac{\pi}{4}x$ とおく．

\begin{enumerate}
  \item[(1)] $y_1 - y_3 = \dfrac{\pi}{4}\dfrac{1}{x} - \dfrac{\pi}{4} + 1 - \dfrac{\pi}{4} + \dfrac{\pi}{4}x - 1 = \dfrac{\pi}{4}\left(x + \dfrac{1}{x} - 2\right)$ であり, $0 < x \le 1$ では
    AM-GMから
    \[
      y_1 - y_3 \ge \dfrac{\pi}{4}(2-2) = 0
    \]
    等号成立は $x=1$ $\dots \text{①}$

    $y_3 - y_2 = \dfrac{\pi}{4}(1-x) + 1 - \sqrt{2}\cos\dfrac{\pi}{4}x$
    \[
      = \dfrac{\pi}{4}(1-x) + \sqrt{2}\left(\cos\dfrac{\pi}{4} - \cos\dfrac{\pi}{4}x\right)
    \]
    これは $x=1$ で等号成立． $0 < x < 1$ の時, 平均値の定理から
    \[
      \cos\dfrac{\pi}{4} - \cos\dfrac{\pi}{4}x = (1-x) \cdot \left(-\dfrac{\pi}{4}\right) \sin\dfrac{\pi}{4}c \quad (0 < c < 1)
    \]
    をみたす $c$ があるので
    \[
      y_3 - y_2 = \dfrac{\pi}{4}(1-x)\left[ 1 - \sqrt{2}\sin\dfrac{\pi}{4}c \right] > 0 \quad (\because \sqrt{2}\sin\dfrac{\pi}{4}c < 1)
    \]
    だから, ②とあわせて
    \[
      y_3 \ge y_2 \quad (\text{等号成立は } x=1) \quad \dots \text{②}
    \]
    ①, ②から, $0 < x < 1$ では $y_1 > y_2$ となり, 交点を持たず, $x=1$ での $(1,1)$ のみである．(終)

  \item[(2)] 右図から, もとめる面積 $S$ として
    \[
      \begin{aligned}
        S &= \int_0^{1/2} \left(1 + \dfrac{\pi}{4} - \sqrt{2}\cos\dfrac{\pi}{4}x\right) dx + \int_{1/2}^1 (f(x) - g(x)) dx \\
        &= \int_0^{1/2} \left(1 + \dfrac{\pi}{4}\right) dx + \int_{1/2}^1 f(x) dx - \int_0^1 g(x) dx \quad \dots \text{③}
      \end{aligned}
    \]
    ここで,
    \begin{itemize}
      \item $\int_0^{1/2} \left(1 + \dfrac{\pi}{4}\right) dx = \dfrac{1}{2}\left(1 + \dfrac{\pi}{4}\right)$
      \item $\int_{1/2}^1 f(x) dx = \left[ \dfrac{\pi}{4}\log x + \left(1-\dfrac{\pi}{4}\right)x \right]_{1/2}^1 = \dfrac{\pi}{4}\log 2 + \dfrac{1}{2}\left(1-\dfrac{\pi}{4}\right)$
      \item $\int_0^1 g(x) dx = \sqrt{2} \dfrac{4}{\pi} \left[ \sin\dfrac{\pi}{4}x \right]_0^1 = \dfrac{4}{\pi}$
    \end{itemize}
    を③に代入して
    \[
      \begin{aligned}
        S &= \dfrac{1}{2}\left(1+\dfrac{\pi}{4}\right) + \dfrac{\pi}{8}\log 2 + \dfrac{1}{2}\left(1-\dfrac{\pi}{4}\right) + \dfrac{4}{\pi} \\
        &= 1 + \dfrac{4}{\pi} + \dfrac{\pi}{8}\log 2
      \end{aligned}
    \]
\end{enumerate}
