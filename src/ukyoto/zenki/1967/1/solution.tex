\noindent
[\textbf{解}]
\begin{enumerate}
  \item[(1)] $x = \pm 1$ は明らかに解でないから，
    \[
      k = \dfrac{-x(x-3)(x+3)}{3(x+1)(x-1)} = \dfrac{1}{3}\left[ -x + \dfrac{8x}{x^2-1} \right] \equiv f(x) \cdot \dfrac{1}{3}
    \]
    \[
      f'(x) = -1 + \dfrac{8(x^2-1) - 8x \cdot 2x}{(x^2-1)^2} = -1 + \dfrac{-8x^2-8}{(x^2-1)^2} = \dfrac{-(x^2+3)^2}{(x^2-1)^2} < 0
    \]
    から $f(x)$ は $x < -1, -1 < x < 1, 1 < x$ の各区間で単調減少．これと，
    \[
      \begin{aligned}
        f(x) &\longrightarrow \mp \infty \quad (x \to \pm \infty) \\
        f(x) &\longrightarrow \mp \infty \quad (x \to -1 \pm 0) \qquad (\text{複号同順}) \\
        f(x) &\longrightarrow \mp \infty \quad (x \to 1 \pm 0)
      \end{aligned}
    \]
    から $y=f(x)$ のグラフは下図．よって示すべきことは明らか．

    \begin{figure}[htb]
      \centering
      \begin{tikzpicture}[x=1.1cm, y=0.85cm,

  \item[(2)] (1)より，
    \[
      f\left(1+\dfrac{2}{k}\right) = \dfrac{-\left(1+\dfrac{2}{k}\right)\left(-2+\dfrac{2}{k}\right)\left(4+\dfrac{2}{k}\right)}{3\left(2+\dfrac{2}{k}\right)\dfrac{2}{k}} = \dfrac{-(2+k)(2-2k)(4+2/k)}{6(2k+2)}
    \]
    \[
      = \dfrac{-(k-1)(k+2)(2k+1)}{3k(k+1)}
    \]
    から
    \[
      k - \dfrac{(k-1)(k+2)(2k+1)}{3k(k+1)} = \dfrac{k^3+3k+2}{3k(k+1)} > 0 \quad (\because k > 0)
    \]
    より，$k > f\left(1+\dfrac{2}{k}\right)$ となり，$1 < 1+\dfrac{2}{k}$ とあわせて，正の解はただ1つあって
    \[
      1 < x < 1 + \dfrac{2}{k}
    \]
    をみたす．
\end{enumerate}
