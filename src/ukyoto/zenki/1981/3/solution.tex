{\bf [解答]}

\begin{align}
  f(x) = k(x^{m+1}-1) - (x^m-1) \label{eq:1}
\end{align}
とおいて，$f(x) \ge 0$ が $0 < x$ の任意の$x$で成り立つ$k$の条件をしらべる．
\begin{align}
  f'(x)
  &= k(m+1)x^m - mx^{m-1} \\
  &= \{k(m+1)x - m\} x^{m-1}
\end{align}
から$f(x)$の増減表は以下のようになる．

\begin{table}[htb]
  \centering
  \caption{$f(x)$ の増減表}
  \begin{tabular}{|c|c|c|c|c|}
    \hline
    $x$ & $0$ & $\cdots$ & $\dfrac{m}{k(m+1)}$ & $\cdots$ \\ \hline
    $f'$ & & $-$ & $0$ & $+$ \\ \hline
    $f$ & & $\searrow$ & & $\nearrow$ \\ \hline
  \end{tabular}
\end{table}

よって，
\begin{align}
  f(x)
  &\ge f\left(\dfrac{m}{k(m+1)}\right) \\
  &= k \left[ \left\{\dfrac{m}{k(m+1)}\right\}^{m+1} - 1 \right] - \left[ \left\{\dfrac{m}{k(m+1)}\right\}^m - 1 \right] \\
  &= \left\{\dfrac{m}{k(m+1)}\right\}^m \left( \dfrac{m}{m+1} - 1 \right) + (1-k) \label{eq:2}
\end{align}
である．
$x \to \infty$ とすれば$f(x)>0$には$k > 0$ が必要なので，$t = \dfrac{1}{k}$ とおいて\cref{eq:2}の右辺を$g(t)$とすると
\begin{align}
  \begin{cases}
    g(t) = - \dfrac{m^m}{(m+1)^{m+1}} t^m + 1 - \dfrac{1}{t} \\
    g'(t) = - \dfrac{m^{m+1}}{(m+1)^{m+1}} t^{m-1} + \dfrac{1}{t^2}
  \end{cases}
\end{align}
である．よって$g(t)$の増減表は以下のようになる．

\begin{table}[htb]
  \centering
  \caption{$g(t)$ の増減表}
  \begin{tabular}{|c|c|c|c|c|}
    \hline
    $t$ & $0$ & $\cdots$ & $\dfrac{m+1}{m}$ & $\cdots$ \\ \hline
    $g'$ & & $+$ & $0$ & $-$ \\ \hline
    $g$ & & $\nearrow$ & $0$ & $\searrow$ \\ \hline
  \end{tabular}
\end{table}

    \node[below left=2pt] at (O) {$O$};

    % 曲線 y = g(t) （m=1 の形状：t=(m+1)/m=2 で軸に接する）
    \draw[line width=1.4pt, teal!75!black, domain=0.42:3.0, samples=70]
    plot (\x, {-0.25*\x + 1 - 1/\x});
    \node[teal!85!black, font=\footnotesize, above right] at (2.4, -0.6) {$y = g(t)$};

    % 最大値の点
    \node[below=3pt, font=\footnotesize] at (2, 0) {$\frac{m+1}{m}$};
    \node[circle, fill=red!85!black, inner sep=1.4pt] at (2, 0) {};
    \node[above=3pt, red!85!black, font=\footnotesize] at (2, 0) {最大値 $0$};
  \end{tikzpicture}
  \caption{関数 $y=g(t)$ の概形（$t=\dfrac{m+1}{m}$ で最大値 $0$ をとる）}
  \label{fig:gt_graph}
\end{figure}

これから
\begin{align}
  g(t) \le g\left(\dfrac{m+1}{m}\right) = 0
\end{align}
だから，常に $f(x) \ge 0$ がみたされるには，$t = \dfrac{m+1}{m} \Leftrightarrow k = \dfrac{m}{m+1}$ が条件である．
