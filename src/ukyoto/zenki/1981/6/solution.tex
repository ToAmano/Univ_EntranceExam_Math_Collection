[解1] (1) $0 < a \le x$ とする。部分積分から、
\begin{align}
\int_a^x e^{-\dfrac{t^2}{2}} dt &= \int_a^x \dfrac{1}{t} \cdot t e^{-\dfrac{t^2}{2}} dt \\
&= \left[ \dfrac{1}{t} (-e^{-\dfrac{t^2}{2}}) \right]_a^x - \int_a^x \dfrac{1}{t^2} e^{-\dfrac{t^2}{2}} dt \\
&\le \dfrac{1}{a} e^{-\dfrac{a^2}{2}} - \dfrac{1}{x} e^{-\dfrac{x^2}{2}} \quad \text{(答)} \quad \left(\because \dfrac{1}{t^2} e^{-\dfrac{t^2}{2}} > 0, \text{等号成立は } x=a\right)
\end{align}

\int_3^b e^{-\dfrac{t^2}{2} + 2t} dt &= \int_3^b \dfrac{1}{-t+2} (-t+2) e^{-\dfrac{t^2}{2} + 2t} dt \\
&= \left[ \dfrac{1}{-t+2} e^{-\dfrac{t^2}{2} + 2t} \right]_3^b - \int_3^b \dfrac{1}{(-t+2)^2} e^{-\dfrac{t^2}{2} + 2t} dt \\
&< e^{\dfrac{3}{2}} + \dfrac{1}{2-b} e^{-\dfrac{b^2}{2} + 2b} \quad \left(\because \dfrac{e^{-\dfrac{t^2}{2} + 2t}}{(-t+2)^2} > 0\right) \\
&< e^{\dfrac{3}{2}} \quad \text{(答)} \quad \left(\because \dfrac{1}{2-b} e^{-\dfrac{b^2}{2} + 2b} < 0\right)
\end{align}
