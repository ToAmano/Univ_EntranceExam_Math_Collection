{\bf [解答]}

\paragraph{(1)}

与えられた不等式
\begin{align}
  \int_a^xe^{-\frac{t^2}{2}}dt\leqq\dfrac{1}{a}e^{-\frac{a^2}{2}}-\dfrac{1}{x}e^{-\frac{x^2}{2}} \label{eq:1}
\end{align}
について考える．

$0 < a \le x$ とする．\cref{eq:1}の左辺を部分積分して
\begin{align}
  \int_a^x e^{-\frac{t^2}{2}} dt
  &= \int_a^x \dfrac{1}{t} \cdot t e^{-\frac{t^2}{2}} dt \\
  &= \left[ \dfrac{1}{t} (-e^{-\frac{t^2}{2}}) \right]_a^x - \int_a^x \dfrac{1}{t^2} e^{-\frac{t^2}{2}} dt \\
  &= \dfrac{1}{a} e^{-\frac{a^2}{2}} - \dfrac{1}{x} e^{-\frac{x^2}{2}} - \int_a^x \dfrac{1}{t^2} e^{-\frac{t^2}{2}} dt \\
  &\le \dfrac{1}{a} e^{-\frac{a^2}{2}} - \dfrac{1}{x} e^{-\frac{x^2}{2}} \quad \text{(答)} \quad \left(\because \dfrac{1}{t^2} e^{-\frac{t^2}{2}} > 0, \text{等号成立は } x=a\right)
\end{align}
である．よって\cref{eq:1}は示された．

\paragraph{(2)}

与えられた不等式
\begin{align}
  \int_3^be^{-\frac{t^2}{2}+2t}<e^{\frac{3}{2}} \quad (b>3) \label{eq:2}
\end{align}
について考える．左辺の被積分関数に$\dfrac{-t+2}{-t+2}$をかけて
\begin{align}
  \int_3^b e^{-\frac{t^2}{2} + 2t} dt
  &= \int_3^b \dfrac{1}{-t+2} (-t+2) e^{-\frac{t^2}{2} + 2t} dt \\
  &= \left[ \dfrac{1}{-t+2} e^{-\frac{t^2}{2} + 2t} \right]_3^b - \int_3^b \dfrac{1}{(-t+2)^2} e^{-\frac{t^2}{2} + 2t} dt \\
  &< e^{\frac{3}{2}} + \dfrac{1}{2-b} e^{-\frac{b^2}{2} + 2b} \quad \left(\because \dfrac{e^{-\frac{t^2}{2} + 2t}}{(-t+2)^2} > 0\right) \\
  &< e^{\frac{3}{2}} \quad \text{(答)} \quad \left(\because \dfrac{1}{2-b} e^{-\frac{b^2}{2} + 2b} < 0\right)
\end{align}
である．よって\cref{eq:2}は示された．

{\bf [別解]}

\paragraph{(1)}

\begin{align}
  f(x) = \dfrac{1}{a} e^{-\frac{a^2}{2}} - \dfrac{1}{x} e^{-\frac{x^2}{2}} - \int_a^x e^{-\frac{t^2}{2}} dt
\end{align}
とおくと，
\begin{align}
  f'(x) = \dfrac{1}{x^2} e^{-\frac{x^2}{2}} > 0
\end{align}
及び $f(a) = 0$ から，$a \le x$ に対して $f(x) \ge f(a)$ である．よって示された．
