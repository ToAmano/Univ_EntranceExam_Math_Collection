[解] $c = \cos\theta \ (-1 \le c \le 1)$ とおくと
\begin{equation*}
  (\text{与式}) \iff a(2c^2-1) + bc - 1 < 0 \quad \cdots \text{\textcircled{1}}
\end{equation*}
\textcircled{1}の左辺$f(c)$とおき，$f(c)<0$ となる条件をしらべる．

$f'(c) = 2ac+b$ である．

$1^\circ \ a=0$の時，$f(c)$は高々一次関数だから
\begin{align}
  &f(1) < 0 \land f(-1) < 0 \\
  \iff & -1 < b < 1
\end{align}

$2^\circ \ a>0$の時

$f(c)$の軸 $c = -\dfrac{b}{4a}$ によらず，$f(c)$は $c = \pm 1$で最大で
\begin{align}
  &f(1) < 0 \land f(-1) < 0 \\
  \iff & a+b-1 < 0 \land a-b-1 < 0
\end{align}

$3^\circ \ a<0$の時
\begin{align}
  &\text{\textcircled{ア}} \ -1 \le -\dfrac{b}{4a} \le 1 \text{の時} \quad f(-\dfrac{b}{4a}) < 0 \iff \dfrac{(a+\dfrac{1}{2})^2}{1/4} + \dfrac{b^2}{2} < 1 \quad \cdots \text{A} \\
  &\text{\textcircled{イ}} \ -\dfrac{b}{4a} \le -1 \text{の時} \quad f(-1) < 0 \\
  &\text{\textcircled{ウ}} \ -\dfrac{b}{4a} \ge 1 \text{の時} \quad f(1) < 0
\end{align}

以上まとめて
\begin{itemize}
  \item $a \ge 0 \land a-1 < b < 1-a$
  \item $4a \le b \le -4a \land a < 0 \land \dfrac{(a+\dfrac{1}{2})^2}{1/4} + \dfrac{b^2}{2} < 1$
  \item $4a \ge b \land a < 0 \land a-1 < b$
  \item $-4a \le b \land a < 0 \land b < -a+1$
\end{itemize}
