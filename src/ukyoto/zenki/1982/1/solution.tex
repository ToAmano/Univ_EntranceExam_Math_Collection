\end{figure}

与えられた曲線を
\begin{align}
  C: y = f(x) = x^3 + ax \label{eq:1}
\end{align}
とおく．$P$の$x$座標を$t (\neq 0)$とすると，$f'(t) = 3t^2 + a$から$P$での$C$の接線$l$は
\begin{align}
  l: y = g(x) = (3t^2 + a)x - 2t^3
\end{align}
だから
\begin{align}
  f(x) - g(x) = x^3 - 3t^2x + 2t^3 = (x-t)^2(x+2t) \label{eq:2}
\end{align}
となる．従って $x = -2t$ ($x \neq t$, $\because t \neq 0$) を$x$座標とする点$Q$で交わる．

\end{figure}

$x \le 0$の部分の面積$S_1$, $x \ge 0$の部分の面積$S_2$とする．
$y = f(x)$は原点対称性だから$t>0$の時 $S_1 : S_2 = A : B$ なら$t<0$の時$S_1 : S_2 = B : A$である．よって$t>0$の時のみ考える．

具体的に計算すると
\begin{align}
  S_1
  &= \int_{-2t}^0 \{ f(x) - g(x) \} dx \\
  &= \int_{-2t}^0 \{ (x+2t) - 3t \}^2 (x+2t) dx \\
  &= \left[ \dfrac{1}{4} (x+2t)^4 - 2t(x+2t)^3 + \dfrac{9t^2}{2} (x+2t)^2 \right]_{-2t}^0 \\
  &= \left( \dfrac{16}{4} - 16 + 18 \right) t^4 = 6t^4 \label{eq:3}
\end{align}
および
\begin{align}
  S_1+S_2
  &= \int_{-2t}^{t} \{ f(x) - g(x) \} dx \\
  &= \int_{-2t}^{t}  (x - t)^2 (x+2t) dx \\
  &= \dfrac{(3t)^42!1!}{4!} = \dfrac{27t^4}{4} \label{eq:4}
\end{align}
より\cref{eq:3}を代入して
\begin{align}
  S_2 = \dfrac{27t^4}{4}-6t^4 = \dfrac{3}{4}t^4
\end{align}
である．以上から
\begin{align}
  \begin{cases}
    t>0 \text{の時}: S_1 : S_2 = 8 : 1 \\
    t<0 \text{の時}: S_1 : S_2 = 1 : 8
  \end{cases}
\end{align}
である．

{\bf [別解]}

(2)の$S_2$は直接積分すると以下のように検算できる．
\begin{align}
  S_2
  &= \int_0^t \{ f(x) - g(x) \} dx \\
  &= \int_0^t (x-t)^2 (x+2t) dx \\
  &= \left[ \dfrac{1}{4} (x-t)^4 + t (x-t)^3 \right]_0^t \\
  &= -\dfrac{1}{4} t^4 + t^4 = \dfrac{3}{4} t^4
\end{align}
