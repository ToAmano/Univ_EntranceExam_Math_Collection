与えられた関数を
\begin{align}
  \begin{cases}
    y=\sqrt{x} \\
    y=\dfrac{2}{a}x+1-\dfrac{1}{a}
  \end{cases} \quad (0 < a \le 2) \label{eq:1}
\end{align}
について考える．

これらが共有点を持つ時，連立した
\begin{align}
  \sqrt{x} = \dfrac{2x}{a} + 1 - \dfrac{1}{a} \label{eq:2}
\end{align}
が $x$ について実数解を持つ．$ x = \dfrac{1}{2}$は代入するとわかるように解ではないから以下 $x \neq \dfrac{1}{2}$ とする．\cref{eq:2}を$a$について整理すると
\begin{align}
  f(x) = \dfrac{\sqrt{x}-1}{2x-1} = \dfrac{1}{a} \label{eq:3}
\end{align}
を得る．$f(x)$を微分すると
\begin{align}
  f'(x)
  & = \dfrac{\dfrac{1}{2\sqrt{x}}(2x-1) - 2(\sqrt{x}-1)}{(2x-1)^2} \\
  & = \dfrac{-x + 2\sqrt{x} - 1/2}{\sqrt{x}(2x-1)^2}
\end{align}
だから $p = \dfrac{2-\sqrt{2}}{2}, \ q = \dfrac{2+\sqrt{2}}{2}$ とおくと，
\begin{align}
  f'(x) = \dfrac{-(\sqrt{x}-p)(\sqrt{x}-q)}{\sqrt{x}(2x-1)^2}
\end{align}
となるので$f(x)$の増減表は以下のようになる．

\end{table}

無限遠および$x=\dfrac{1}{2}$での極限は
\begin{align}
  \lim_{x \to 1/2 \mp 0} f(x) = \pm\infty \quad (\text{複号同順}),
  \quad
  \lim_{x\to\infty} = 0
\end{align}
だから，$y=f(x)$のグラフの概形は下図のようになる．

\paragraph{(2)}

与式を展開して整理すると
\begin{align}
  & 4x^2 + \{4(a-1)-a^2\}x + (a-1)^2 = 0 \\
  \therefore
  & g(x) = 4x^2 - (a-2)^2 x + (a-1)^2 = 0 \label{eq:4}
\end{align}
である．この判別式を $D$ とすると
\begin{align}
  D
  & = (a-2)^4 - 16(a-1)^2 \\
  & = \{(a-2)^2 + 4(a-1)\}\{(a-2)^2 - 4(a-1)\} \\
  & = a^2(a^2-8a+8)
\end{align}
だから，$t = 4-2\sqrt{2}$ として$0 < a \le 2$と合わせて
\begin{itemize}
  \item $0 < a \le t$ の時，$D \ge 0$ から，\cref{eq:4}は実数解を持つ．\\
  \item $t < a \le 2$ の時，$ D<0$ から，\cref{eq:4}は実数解を持たない．
\end{itemize}
となる．以下この二つを個別に考える．

\subparagraph{$1^\circ$ $0 < a \le t$ の時}

\cref{eq:4}の二つの実数解は，端点について$g(0) \ge 0$および軸について$x = \dfrac{(a-2)^2}{8} \ge 0$からいずれも非負である．従って$|\alpha| \le |\beta| \therefore \alpha \le \beta$ で，
\begin{align}
  |\beta|
  & = \beta \\
  & = \dfrac{1}{8}\{(a-2)^2 + \sqrt{a^2(a^2-8a+8)}\} \\
  & = \dfrac{1}{8}\{(a-2)^2 + a\sqrt{a^2-8a+8}\}
\end{align}
である．この最小値を求めれば良い．両辺$a$で微分して
\begin{align}
  8\dfrac{d|\beta|}{da}
  &= 2(a-2) + \sqrt{a^2-8a+8} + \dfrac{a(a-4)}{\sqrt{a^2-8a+8}} \\
  &= 2(a-2) + \dfrac{a^2-8a+8 + a(a-4)}{\sqrt{a^2-8a+8}} \\
  &= 2(a-2) + \dfrac{2(a^2-6a+4)}{\sqrt{a^2-8a+8}}
\end{align}
だから，この符号は
\begin{align}
  \dfrac{d|\beta|}{da} \ge 0
  \iff
  \dfrac{a^2-6a+4}{\sqrt{a^2-8a+8}} \ge 2-a
\end{align}
で決まる．$0<a\le 3-\sqrt{5}$の時は両辺が$0$以上，$3-\sqrt{5} < a \le t$の時は左辺が負で右辺が正となり不適だから$0<a \le 3-\sqrt{5}$の元で二乗して，
\begin{align}
  & (a^2-6a+4)^2 \ge (a-2)^2(a^2-8a+8) \\
  \iff
  & a^4-12a^3+44a^2-48a+16 \ge a^4-12a^2+44a^2-64a+32 \\
  \iff
  & a \ge 1
\end{align}
となる．しかし$0<a\le 3-\sqrt{5} < 1$よりこれを満たす$a$はない．
従って区間内で$\dfrac{d|\beta|}{da} < 0$つまり$|\beta|$は単調減少だから，
最小値は$a = t$の時の
\begin{align}
  \min |\beta| = \beta(a=t) = \dfrac{(t-2)^2}{8} = \dfrac{(2-2\sqrt{2})^2}{8} = \dfrac{3-2\sqrt{2}}{2} \label{eq:5}
\end{align}
である．ただし$a=t$の時は$D=0$であることを用いた．

\subparagraph{$2^\circ$ $t < a \le 2$ の時}

この時は$D<0$となり\cref{eq:4}は虚数解を持つ．\cref{eq:4}は実数係数だから$\beta=\bar{\alpha}$となり，$|\alpha|=|\beta|$だから，解と係数の関係から
\begin{align}
  |\beta|
  & = \sqrt{|\alpha||\beta|}
  = \sqrt{|\alpha\beta|} \\
  & = \sqrt{\left|\dfrac{(a-1)^2}{4}\right|} \\
  & = \dfrac{|a-1|}{2}
\end{align}
である．$1<t<a\le2$の区間での最小値は$a\to t$の時であり
\begin{align}
  \lim_{a\to t}|\beta| = \dfrac{|t-1|}{2} = \dfrac{3-2\sqrt{2}}{2}
\end{align}
である．これは\cref{eq:5}と同じだから，$\min |\beta|$は$a=t$で連続である．
\casesend

以上 $1^\circ, 2^\circ$ から，求める$|\beta|$の最小値は$a=4-2\sqrt{2}$の時の
\begin{align}
  \min |\beta| = \dfrac{1}{2}(3-2\sqrt{2})
\end{align}
である．
