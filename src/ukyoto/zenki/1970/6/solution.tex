\paragraph{(イ)}

与えられた不等式
\begin{align}
  1<x_n<a \quad (n=2,3,\cdots\cdots) \label{eq:1}
\end{align}
を数学的帰納法で示す.

$n=2$の時\cref{eq:1}より
\begin{align}
  x_2 = \dfrac{a^2+2}{3}
\end{align}
だから
\begin{align}
  x_2 -1 = \dfrac{a^2+2}{3} - 1 = \dfrac{a^2-1}{3} > 0 \quad (\because a>1)
\end{align}
および
\begin{align}
  a - x_2 = a-\dfrac{a^2+2}{3} = \dfrac{-(a-2)(a-1)}{3} > 0 \quad (\because 1<a<2)
\end{align}
から
\begin{align}
  1 < x_2 < 2
\end{align}
であり，$n=2$では\cref{eq:1}は成立する．

そこで $n=k$ での成立を仮定して$n=k+1$の時を考えると
\begin{align}
  x_{k+1} &= \dfrac{x_k^2 + 2}{3} > \dfrac{1+2}{3} = 1 \\
  x_{k+1} &= \dfrac{x_k^2 + 2}{3} < \dfrac{a^2 + 2}{3} < a \quad (\because a^2 - 3a + 2 = (a-2)(a-1) < 0)
\end{align}
より $n=k+1$ でも\cref{eq:1}は成立する.
以上から数学的帰納法によって示された.

\paragraph{(ロ)}

$x_{n+1}-1$に\cref{eq:1}を用いて
\begin{align}
  x_{n+1} - 1
  &= \dfrac{1}{3}(x_n^2 + 2) - 1
  = \dfrac{1}{3}(x_n + 1)(x_n - 1)
  < \dfrac{1}{3}(a + 1)(x_n - 1) \quad (\because (\text{イ}))
\end{align}
より
\begin{align}
  x_{n+1} -1 \le \dfrac{a+1}{3}(x_n-1) \label{eq:2}
\end{align}
である．

\paragraph{(ハ)}

\cref{eq:2}を繰り返し用いて，(イ)から$x_n-1>0$であることと合わせて
\begin{align}
  0 < x_n - 1 \le \left(\dfrac{a+1}{3}\right)^{n-1}(a - 1)
\end{align}
である．右辺は$n\to\infty$で$0$に収束するから，挟み撃ちの定理より
\begin{align}
  \lim_{n\to\infty} x_n = 1
\end{align}
である．
