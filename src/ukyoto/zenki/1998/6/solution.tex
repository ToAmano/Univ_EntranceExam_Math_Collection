与えられた関数を
\begin{align}
  f(x) = \log(x-a) \quad (0<a<1) \label{eq:1}
\end{align}
とする．

\paragraph{(1)}

題意の体積は
\begin{align}
  V(a)
  &= \int_1^3 \pi \{ f(x) \}^2 dx \\
  &= \pi \int_{1-a}^{3-a} (\log x)^2 dx \label{eq:2}\\
  &= \pi \left[ x(\log x)^2 - 2x(\log x - 1) \right]_{1-a}^{3-a} \\
  &= \pi \left\{ (3-a) \left\{ (\log(3-a))^2 - 2\log(3-a) + 2 \right\} - (1-a) \left\{ (\log(1-a))^2 - 2\log(1-a) + 2 \right\} \right\} \label{eq:3}
\end{align}
である．

\paragraph{(2)}

\cref{eq:2}の両辺$a$で微分して
\begin{align}
  \dfrac{1}{\pi} V'(a)
  &= - (\log(3-a))^2 + (\log(1-a))^2 \\
  &= \{ \log(1-a) + \log(3-a) \} \{ \log(1-a) - \log(3-a) \} \\
  &= \log\left[(1-a)(3-a)\right] \cdot \log \dfrac{1-a}{3-a}
\end{align}
である．ここで$0<a<1$より$\log \dfrac{1-a}{3-a} < 0$に注意して$V(a)$の増減表は以下のようになる．

\begin{table}[htb]
  \centering
  \caption{$V(a)$ の増減表}
  \begin{tabular}{|c|c|c|c|c|c|}
    \hline
    $a$ & $0$ & $\cdots$ & $2-\sqrt{2}$ & $\cdots$ & $1$ \\ \hline
    $V'$ & & $-$ & $0$ & $+$ & \\ \hline
    $V$ & & $\searrow$ & & $\nearrow$ & \\ \hline
  \end{tabular}
\end{table}

従って$a=2-\sqrt{2}$の時に$V(a)$は最小となる．この時
\begin{align}
  \begin{cases}
    1-a = -1+\sqrt{2} \\
    3-a = 1+\sqrt{2}
  \end{cases}
\end{align}
だから $A_{\pm}=\log(\pm 1+\sqrt{2})$とおくと$A_{-}=-A_{+}$で，\cref{eq:3}より求める体積は
\begin{align}
  \dfrac{1}{\pi} V(a)
  &= (1+\sqrt{2}) \{ A_{+}^2 - 2A_{+} + 2 \} - (-1+\sqrt{2}) \{ A_{-}^2 - 2A_{-} + 2 \} \\
  &= (1+\sqrt{2}) \{ A_{+}^2 - 2A_{+} + 2 \} + (1-\sqrt{2}) \{ A_{-}^2 + 2A_{-} + 2 \} \\
  &= 2A_{+}^2 - 4\sqrt{2}A_{+} + 4 \\
  \therefore
  V(a) & = 2\pi \left\{ (\log(1+\sqrt{2}))^2 - 2\sqrt{2}\log(1+\sqrt{2}) + 2 \right\}
\end{align}
である．
