  {\bf [解答]} $\angle \text{BAC} = \angle \text{R}$ となるよう，3頂点 $\text{A}, \text{B}, \text{C}$ を定める．\\
  $\text{AB}=c, \text{AC}=b, \text{BC}=a$ となる\\
  内接円の中心 $\text{O}$ から各辺に\\
  下ろした垂線を $\text{H}_1 \sim \text{H}_3$ と\\
  する．題意から\\
  $2r + a + b + c = 2 \cdots \text{①}$

  \paragraph{(1)}

  $\square \text{AH}_1\text{OH}_3$ が正方形だから，
  \begin{align}
    \begin{cases}
      \text{BH}_2 = \text{BH}_1 = c-r \\
      \text{CH}_2 = \text{CH}_3 = b-r
    \end{cases}
    \quad \therefore a = b+c-2r
  \end{align}
  ①に代入して
  \begin{align}
    b+c = 1 \qquad \cdots \text{②}
  \end{align}
  だから $\quad a = 1-2r$

  \paragraph{(2)}

  $\triangle \text{ABC}$ の面積 $S(r)$ とする\\
  $S(r) = \dfrac{1}{2}bc$\\
  以下この $\max$ をもとめる．$a, b, c, r > 0$ であり，$\triangle \text{ABC}$ の存在条件から
  \begin{align}
    \begin{cases}
      b+c > b+c-2r \\
      2b+c-2r > c \\
      2c+b-2r > b
    \end{cases}
    \quad \therefore
    \begin{cases}
      2r > 0 \\
      b-r > 0 \\
      c-r > 0
    \end{cases}
  \end{align}
  $A = bc$ とおくと $b, c$ は $t$ の2次方程式 $t^2 - t + A = 0 \ (\because \text{②})$ の\\
  $r$ より大なる2実解．判別式 $D$，左辺 $f(t)$ として
  \begin{align}
    \begin{cases}
      D \ge 0 \\
      f(r) > 0 \\
      r < \dfrac{1}{2}
    \end{cases}
    \quad \therefore
    \begin{cases}
      1 - 4A \ge 0 \\
      r^2 - r + A > 0 \\
      0 < r < \dfrac{1}{2}
    \end{cases}
  \end{align}
  これを図示すると右図斜線部\\
  (境界は $A = \dfrac{1}{4}$ のみ含む)\\
  したがって
  \begin{align}
    \max S(r) = \dfrac{1}{2} \max A = \dfrac{1}{8}
  \end{align}

  [別] $\cdots a, b$ を消去．完全に\\
  内接円と三角形のカンケイから
  \begin{align}
    \dfrac{1}{2}ab = \dfrac{1}{2} \cdot r \cdot (2-2r)
  \end{align}
  \begin{align}
    ab = 2r(1-r)
  \end{align}
  (本解)から $a+b=1$ でもあるから $a,b$ は
  \begin{align}
    t^2 - t + 2r(1-r) = 0
  \end{align}
  の2実解．これが $t > r$ に2実解をもつので
  \begin{align}
    \begin{cases}
      r < \dfrac{1}{2} \\
      1 - 8r(1-r) \ge 0 \\
      r^2 - r + 2r(1-r) > 0
    \end{cases}
  \end{align}
  $\Rightarrow$ とくと $0 < r \le \dfrac{2-\sqrt{2}}{4}$ となり，代入してオワリ．
