四面体$OPQR, OABC$は底面を各々$\triangle PRQ, \triangle ABC$とみると，高さがどちらも同じだから
\begin{align}
  V_{OPQR} : V_{OABC} = \triangle PQR : \triangle ABC \label{eq:1}
\end{align}
である．$\triangle OAC, QAO$の相似から
\begin{align}
  |AQ| : |AO| = a : \sqrt{a^2+c^2}
  \therefore
  |AQ| = \dfrac{a|AO|}{\sqrt{a^2+c^2}} = \dfrac{a^2}{\sqrt{a^2+c^2}} \label{eq:2}
\end{align}
であり，同様にして
\begin{align}
  |AR| = \dfrac{a^2}{\sqrt{a^2+b^2}}, \quad |BP| = \dfrac{b^2}{\sqrt{b^2+c^2}} \label{eq:3}
\end{align}
を得る．

従って $\triangle ABC$の面積を1とすると
\begin{align}
  \begin{cases}
    \triangle ARQ = \dfrac{|AR|}{|AB|}\dfrac{|AQ|}{|AC|}\triangle ABC
    = \dfrac{a^2}{a^2+b^2} \cdot \dfrac{a^2}{a^2+c^2} \\
    \triangle BPR =  \dfrac{|BP|}{|BC|}\dfrac{|BR|}{|BA|}\triangle ABC
    = \dfrac{b^2}{a^2+b^2} \cdot \dfrac{b^2}{b^2+c^2} \\
    \triangle PCQ = \dfrac{|CP|}{|CB|}\dfrac{|CQ|}{|AC|}\triangle ABC
    = \dfrac{c^2}{b^2+c^2} \cdot \dfrac{c^2}{a^2+c^2}
  \end{cases}
\end{align}
だから
\begin{align}
  \triangle PQR
  &= \triangle ABC - \triangle ARQ - \triangle BPR - \triangle PCQ \\
  &= 1 - \dfrac{a^4(b^2+c^2) + b^4(c^2+a^2) + c^4(a^2+b^2)}{(a^2+b^2)(b^2+c^2)(c^2+a^2)} \\
  &= \dfrac{2a^2b^2c^2}{(a^2+b^2)(b^2+c^2)(c^2+a^2)}
\end{align}
となり，\cref{eq:1}に代入して求める体積比は
\begin{align}
  V_{OPQR} : V_{OABC} = 2a^2b^2c^2 : (b^2+c^2)(c^2+a^2)(a^2+b^2) \label{eq:4}
\end{align}
である．

\paragraph{(2)}

\cref{eq:4}より
\begin{align}
  & \dfrac{2a^2b^2c^2}{(b^2+c^2)(c^2+a^2)(a^2+b^2)} \le \dfrac{1}{4} \\
  \therefore
  & 8ABC \le (B+C)(C+A)(A+B) \label{eq:5}
\end{align}
を示せばよい．ただし$A=a^2, B=b^2, C=c^2$ とおいた．

$A,B,C >0$からAM-GM不等式
\begin{align}
  \begin{cases}
    B+C \ge 2\sqrt{BC} > 0 \\
    C+A \ge 2\sqrt{CA} > 0 \\
    A+B \ge 2\sqrt{AB} > 0
  \end{cases}
\end{align}
を辺々掛け合わせると\cref{eq:5}になる．よって題意は示された．
