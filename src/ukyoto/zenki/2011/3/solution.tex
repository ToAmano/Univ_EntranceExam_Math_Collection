与えられた曲線
\begin{align}
  y = x, \quad
  y = \left|\dfrac{3}{4}x^2-3\right|-2 \label{eq:1}
\end{align}
について考える．これらのグラフは上図である．

絶対値を外した方程式の解を
\begin{align}
  \begin{cases}
    \dfrac{3}{4}x^2-5=x \text{の2解}\  \alpha=-2,\beta=\dfrac{10}{3} \ (\alpha<\beta)\\
    -\dfrac{3}{4}x^2+1=x \text{の2解}\ \gamma=-2,\delta=\dfrac{2}{3} \ (\gamma<\delta)
  \end{cases} \label{eq:2}
\end{align}
とおく．

もとめる面積を$S$とすると，これは以下のように$3$つの放物線と直線の囲む面積から計算できる．
\begin{align}
  S &=
  \left( \text{
      \begin{tikzpicture}[baseline={(0,-0.1)}, scale=0.35]
        \fill[blue!15] (-2,-1.2) -- plot[domain=-2:3.33, samples=30] (\x,{0.45*\x*\x - 3.0}) -- (3.33,2.0) -- cycle;
        \fill[pattern=north east lines, pattern color=blue!70!black]
        (-2,-1.2) -- plot[domain=-2:3.33, samples=30] (\x,{0.45*\x*\x - 3.0}) -- (3.33,2.0) -- cycle;
        \draw[teal!85!black, thick] (-2.5,-1.5) -- (3.8,2.28);
        \draw[blue!85!black, thick] plot[domain=-2.5:3.6, samples=30] (\x,{0.45*\x*\x - 3.0});
        \node[below left=-1pt, font=\tiny] at (-2,-1.2) {$\alpha$};
        \node[above right=-1pt, font=\tiny] at (3.33,2.0) {$\beta$};
        \node[font=\scriptsize] at (0,2) {$S_1$};
  \end{tikzpicture}} \right)
  + 2
  \left( \text{
      \begin{tikzpicture}[baseline={(0,-0.1)}, scale=0.35]
        \fill[teal!15] (-2,-1.2) -- plot[domain=-2:0.67, samples=30] (\x,{1.5 - 0.675*\x*\x}) -- (0.67,0.4) -- cycle;
        \fill[pattern=north east lines, pattern color=teal!70!black]
        (-2,-1.2) -- plot[domain=-2:0.67, samples=30] (\x,{1.5 - 0.675*\x*\x}) -- (0.67,0.4) -- cycle;
        \draw[teal!85!black, thick] (-2.4,-1.44) -- (1.5,0.9);
        \draw[blue!85!black, thick] plot[domain=-2.4:1.2, samples=30] (\x,{1.5 - 0.675*\x*\x});
        \node[below left=-1pt, font=\tiny] at (-2,-1.2) {$\gamma$};
        \node[above right=-1pt, font=\tiny] at (0.67,0.4) {$\delta$};
        \node[font=\scriptsize] at (0,2) {$S_2$};
  \end{tikzpicture}} \right)
  - 2
  \left( \text{
      \begin{tikzpicture}[baseline={(0,-0.1)}, scale=0.35]
        \fill[blue!15] (-2,-1) -- plot[domain=-2:2, samples=30] (\x,{0.45*\x*\x - 2.8}) -- (2,-1) -- cycle;
        \fill[pattern=north east lines, pattern color=blue!70!black]
        (-2,-1) -- plot[domain=-2:2, samples=30] (\x,{0.45*\x*\x - 2.8}) -- (2,-1) -- cycle;
        \draw[teal!85!black, thick] (-2.5,-1) -- (2.5,-1);
        \draw[blue!85!black, thick] plot[domain=-2.3:2.3, samples=30] (\x,{0.45*\x*\x - 2.8});
        \node[above, font=\tiny] at (-2,-1) {$-2$};
        \node[above, font=\tiny] at (2,-1) {$2$};
        \node[font=\scriptsize] at (0,2) {$S_3$};
  \end{tikzpicture}} \right) \label{eq:3}
\end{align}

各項を個別に計算すると
\begin{align}
  S_1 &= \dfrac{3}{4}\cdot\dfrac{1}{6}(\beta-\alpha)^3=\dfrac{1}{8}(\beta-\alpha)^3，\\
  S_2 &= \dfrac{3}{4}\cdot\dfrac{1}{6}(\delta-\gamma)^3=\dfrac{1}{8}(\delta-\gamma)^3，\\
  S_3 &= \dfrac{3}{4}\cdot\dfrac{1}{6}(2+2)^3=\dfrac{1}{8}(4)^3
\end{align}
だから，これを\cref{eq:2}に代入して
\begin{align}
  S=\dfrac{1}{8}(\beta-\alpha)^3+\dfrac{1}{4}(\delta-\gamma)^3-16 \label{eq:4}
\end{align}
である．

ここで\cref{eq:2}から
\begin{align}
  \beta-\alpha=\dfrac{10}{3}+2=\dfrac{16}{3}，\qquad
  \delta-\gamma=\dfrac{2}{3}+2=\dfrac{8}{3}
\end{align}
だから，\cref{eq:4}に代入して求める面積は
\begin{align}
  S&=\dfrac{1}{8}\left(\dfrac{16}{3}\right)^3+\dfrac{1}{4}\left(\dfrac{8}{3}\right)^3-16\\
  &=\dfrac{8^3}{3^3}+\dfrac{2\cdot8^2}{3^3}-\dfrac{16\cdot3^3}{3^3}\\
  &=\dfrac{16}{3^3}(32+8-27)\\
  &=\dfrac{208}{27}
\end{align}
となる．
