円$C$の中心$P(0,a)$，半径$r$とすると，題意の$3$つの曲線は
\begin{align}
  C_1:y=f_1(x)=\sqrt{3}\log(x+1) \\
  C_2:y=f_2(x)=\sqrt{3}\log(1-x) \\
  C:  x^2 + (y-a)^2 = r^2
\end{align}
で与えられる．対称性から，$A$が$C,C_1$の接点，$B$が$C,C_2$の接点とする．
まずは$\triangle PAB$が正三角形になる条件から$a,r$を求める．
$AB$と$y$軸の交点を$H$とおくと，
\begin{align}
  |AH|=|BH| = \dfrac{|AB|}{2} = \dfrac{|AP|}{2} = \dfrac{r}{2}
\end{align}
だから，$A(r/2,f_1(r/2)),B(-r/2,f_2(-r/2))$である．
\begin{align}
  f_1'(x) = \dfrac{\sqrt{3}}{x+1}
\end{align}
と合わせて$A$での$C_1$の接線$\ell_1$は
\begin{align}
  \ell_1:y=\ell_1(x)=\dfrac{\sqrt{3}}{1+\dfrac{1}{2}r}\left(x-\dfrac{1}{2}r\right)+\sqrt{3}\log\left(1+\dfrac{1}{2}r\right) \label{eq:1}
\end{align}
であり，半径$PA\perp\ell_1$かつ$PA$と$y$軸のなす角が$\dfrac{\pi}{6}$であることから，$\ell_1$の傾きが$\tan\dfrac{\pi}{6}=\dfrac{\sqrt{3}}{3}$だから，
\begin{align}
  \dfrac{\sqrt{3}}{1+\dfrac{1}{2}r}=\dfrac{\sqrt{3}}{3}\quad\therefore\ r=4 \label{eq:2}
\end{align}
である．この時$|OP|=a$をふた通りで表すと
\begin{align}
  |OP| = |OH| + |HP| = |OH| + |AP|\cos\dfrac{\pi}{6}
\end{align}
だから
\begin{align}
  a
  & = f_1(r/2) + \dfrac{\sqrt{3}}{2}r \\
  & = \sqrt{3}\log\left(1+\dfrac{1}{2}r\right)+\dfrac{\sqrt{3}}{2}r \\
  &= \sqrt{3}\log3+2\sqrt{3} \label{eq:3}
\end{align}
を得る．

求める面積を$S$とする．対称性から$0 \le x$の部分の面積の$2$倍を求めればよく，

\begin{align}
  \dfrac{1}{2}S
  &=
  \left( \text{
      \begin{tikzpicture}[baseline={(0,0.5)}, scale=0.45, >=stealth]
        \fill[blue!15] (0,0) -- (2,0) -- (2,1.2) -- (0,2.5) -- cycle;
        \fill[pattern=north east lines, pattern color=blue!60] (0,0) -- (2,0) -- (2,1.2) -- (0,2.5) -- cycle;
        \draw[blue!85!black, thick] (0,0) -- (2,0) -- (2,1.2) -- (0,2.5) -- cycle;
        \node[below left=-1pt, font=\tiny] at (0,0) {$O$};
        \node[below right=-1pt, font=\tiny] at (2,0) {$A'$};
        \node[above right=-1pt, font=\tiny] at (2,1.2) {$A$};
        \node[above left=-1pt, font=\tiny] at (0,2.5) {$P$};
      \end{tikzpicture}
  } \right)
  -
  \left( \text{
      \begin{tikzpicture}[baseline={(0,0.5)}, scale=0.45, >=stealth]
        \fill[teal!15] (0,0) -- plot[domain=0:2, samples=30] (\x,{1.2*ln(1+\x)/ln(3)}) -- (2,0) -- cycle;
        \fill[pattern=north east lines, pattern color=teal!60] (0,0) -- plot[domain=0:2, samples=30] (\x,{1.2*ln(1+\x)/ln(3)}) -- (2,0) -- cycle;
        \draw[teal!85!black, thick] plot[domain=0:2, samples=30] (\x,{1.2*ln(1+\x)/ln(3)});
        \draw[gray!70, dashed] (2,0) -- (2,1.2);
        \draw[gray!80!black, thick] (0,0) -- (2,0);
        \node[below left=-1pt, font=\tiny] at (0,0) {$O$};
        \node[below right=-1pt, font=\tiny] at (2,0) {$A'$};
        \node[above right=-1pt, font=\tiny] at (2,1.2) {$A$};
        \node[above left=-1pt, font=\tiny, teal!85!black] at (1.5,0.9) {$C_1$};
      \end{tikzpicture}
  } \right)
  -
  \left( \text{
      \begin{tikzpicture}[baseline={(0,0.5)}, scale=0.45, >=stealth]
        \coordinate (Pc) at (0,2.5);
        \fill[orange!20] (Pc) -- (0,0.5) arc[start angle=-90, end angle=-60, radius=2] -- cycle;
        \fill[pattern=north east lines, pattern color=orange!70] (Pc) -- (0,0.5) arc[start angle=-90, end angle=-60, radius=2] -- cycle;
        \draw[orange!90!black, thick] (Pc) -- (0,0.5) arc[start angle=-90, end angle=-60, radius=2] -- cycle;
        \node[above left=-1pt, font=\tiny] at (Pc) {$P$};
        \node[above right=-1pt, font=\tiny] at (1.0,0.768) {$A$};
        \node[font=\tiny] at (0.35,1.7) {$\frac{\pi}{6}$};
      \end{tikzpicture}
  } \right) \label{eq:4}
\end{align}
と表せる．ここで点$A$から$x$軸に下ろした垂線の足を$A'(2,0)$とすると，各図形の面積は
\begin{align}
  \text{台形}OA'AP
  &= \dfrac{1}{2}(|OP| + |A'A|) \cdot |OA'| = \dfrac{1}{2}(a + \sqrt{3}\log 3) \cdot 2 = \sqrt{3}\log 3 + a \\
  \text{図形}OA'A
  &= \int_0^2\sqrt{3}\log(x+1)\,dx = \sqrt{3}\Bigl[(x+1)\log(x+1)-(x+1)\Bigr]_0^2 = \sqrt{3}(3\log 3 - 2) \\
  \text{扇形}
  &= \dfrac{1}{2}\cdot\dfrac{\pi}{6}\cdot r^2 = \dfrac{4}{3}\pi
\end{align}
となる．これらを\cref{eq:4}に代入して
\begin{align}
  \dfrac{1}{2}S
  &=2\sqrt{3}\log3+2\sqrt{3}-\left(3\sqrt{3}\log3-2\sqrt{3}+\dfrac{4}{3}\pi\right)\\
  &=4\sqrt{3}-\sqrt{3}\log3-\dfrac{4}{3}\pi \\
  \therefore
  S & =2\left(4\sqrt{3}-\sqrt{3}\log3-\dfrac{4}{3}\pi\right)
\end{align}
である．
