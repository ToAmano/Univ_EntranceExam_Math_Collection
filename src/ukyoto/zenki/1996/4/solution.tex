\begin{align}
  a_{k+1} = \left[ \dfrac{a_k + k}{3} \right] \le \left[ \dfrac{a_{k+1} + k}{3} \right] = a_{k+2} \cdots \text{①}

\begin{align}

から，
\begin{align}
