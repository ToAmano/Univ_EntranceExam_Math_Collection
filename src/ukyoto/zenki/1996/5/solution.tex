一回目に選んだ頂点 $X_1$ とすると，題意より
\begin{align}
  \overrightarrow{OP_1} = \dfrac{1}{2}(\overrightarrow{OP_0} + \overrightarrow{OX_1}) \label{eq:1}
\end{align}
であり，両辺二乗して
\begin{align}
  |\overrightarrow{OP_1}|^2
  & = \dfrac{1}{4}\{|\overrightarrow{OP_0}|^2 + |\overrightarrow{OX_1}|^2 + 2\overrightarrow{OP_0} \cdot \overrightarrow{OX_1}\} \\
  & = \dfrac{1}{4}\{|\overrightarrow{OP_0}|^2 + 1 + 2\overrightarrow{OP_0} \cdot \overrightarrow{OX_1}\} \quad (\because |\overrightarrow{OX_1}| = 1) \label{eq:2}
\end{align}
だから，$X_1=A,B,C$として期待値を計算すれば
\begin{align}
  E_1
  &= \dfrac{1}{3}\sum_{X_1=A,B,C} |\overrightarrow{OP_1}|^2 \\
  &= \dfrac{1+|\overrightarrow{OP_0}|^2}{4} + \dfrac{2}{3} \cdot \dfrac{1}{4} \overrightarrow{OP_0} \cdot (\overrightarrow{OA} + \overrightarrow{OB} + \overrightarrow{OC}) \\
  &= \dfrac{1+|\overrightarrow{OP_0}|^2}{4} \quad (\because \overrightarrow{OA} + \overrightarrow{OB} + \overrightarrow{OC} = \vec{0})
\end{align}
である．

\paragraph{(2)}

点$X$に対して$O$を始点とする位置ベクトルを小文字で$\overrightarrow{OX} = \vec{x}$ と表す．
題意から，
\begin{align}
  \vec{p}_{n+1} = \dfrac{1}{2}(\vec{p}_n + \vec{x}_{n+1})
\end{align}
だからくり返し用いて，
\begin{align}
  \vec{P}_n
  & = \dfrac{1}{2}\vec{x}_n + \dfrac{1}{4}\vec{x}_{n-1} + \cdots + \dfrac{1}{2^n}\vec{x}_1 + \dfrac{1}{2^n}\overrightarrow{OP_0} \\
  & = \dfrac{1}{2^n}\overrightarrow{OP_0} + \sum_{i=1}^n \dfrac{1}{2^{n+1-i}}\overrightarrow{OX_i} \label{eq:3}
\end{align}
である．

\paragraph{(3)}

$P_0 = O$ の時\cref{eq:3}より
\begin{align}
  \overrightarrow{OP_n} = \sum_{i=1}^n \dfrac{1}{2^{n+1-i}} \overrightarrow{OX_i}
\end{align}
だから，$Y_i$ を，$\vec{y}_i = \dfrac{1}{2^{n+1-i}} \vec{x}_i$ となるよう定めると，
\begin{align}
  |\vec{P}_n|^2
  & = |\vec{y}_1 + \vec{y}_2 + \cdots + \vec{y}_n|^2 \\
  & = \sum_{i=1}^n |\vec{y}_i|^2 + 2 \sum_{i < j} \vec{y}_i \cdot \vec{y}_j \\
  & = \sum_{i=1}^n \dfrac{1}{4^{n+1-i}} + 2 \sum_{i < j} \vec{y}_i \cdot \vec{y}_j \quad (\because |\vec{x}_i| = 1) \\
  & = \dfrac{1}{3}\{1 - (1/4)^n\} + 2 \sum_{i < j} \vec{y}_i \cdot \vec{y}_j \label{eq:4}
\end{align}
である．ここで $i \ne j$ に対して，$\triangle ABC$が一辺$\sqrt{3}$の正三角形だから
\begin{align}
  \vec{x}_i \cdot \vec{x}_j =
  \begin{cases}
    1 & (X_i = X_j) \\
    -1/2 & (X_i \ne X_j)
  \end{cases}
\end{align}
だから $\vec{x}_i \cdot \vec{x}_j$ の期待値 $P_{i,j}$ は
\begin{align}
  P_{i,j} = \dfrac{1}{3} \cdot 1 + \dfrac{2}{3} \left(-\dfrac{1}{2}\right) = 0
\end{align}
である．

従って\cref{eq:4}$の第二項\sum_{i < j} \vec{y}_i \cdot \vec{y}_j$ の期待値 $Q$ は，
\begin{align}
  Q = \sum_{i < j} \dfrac{1}{2^{n+1-i} \cdot 2^{n+1-j}} \cdot P_{i,j} = 0
\end{align}
だから\cref{eq:4}より求める期待値は
\begin{align}
  E_n = \dfrac{1}{3}(1 - (1/4)^n) + Q = \dfrac{1}{3} \{ 1 - \left(\dfrac{1}{4}\right)^n \}
\end{align}
である．

{\bf [別解]}

\paragraph{(3)}

(1)より$E_1 = \dfrac{1}{4}$ である．
ここで $n \ge 2$ の時，
\begin{align}
  \begin{cases}
    \vec{p}_n = \dfrac{1}{2} \{ \vec{p}_{n-1} + \vec{x}_n \} \\
    |\vec{p}_n|^2 = \dfrac{1}{4}\{ |\vec{p}_{n-1}|^2 + |\vec{x}_n|^2 \} + \dfrac{1}{2}\vec{p}_{n-1} \cdot \vec{x}_n
  \end{cases}\label{eq:5}
\end{align}
であり，題意より
\begin{align}
  \begin{cases}
    E(|\vec{P}_{n-1}|^2) = E_{n-1} \\
    E(|\vec{x}_n|^2) = 1 \\
    E(\vec{P}_{n-1} \cdot \vec{x}_n) = \dfrac{1}{3}\vec{P}_{n-1} \cdot (\vec{a} + \vec{b} + \vec{c}) = 0
  \end{cases}
\end{align}
を\cref{eq:5}に代入して
\begin{align}
  E_n = \dfrac{1}{4} (E_{n-1} + 1)
\end{align}
だから，初項$E_1=\dfrac{1}{4}$と合わせて
\begin{align}
  E_n = \dfrac{1}{3}\{1 - \left(\dfrac{1}{4}\right)^n\}
\end{align}
である．
