\paragraph{(3)}

$E_1 = \dfrac{1}{4} \ (\because (1)) \cdots \text{③}$ である．ここで，$n \ge 2$ の時，
\begin{align}
  \vec{P}_n = \dfrac{1}{2} \{ \vec{P}_{n-1} + \vec{x}_n \}
\end{align}
\begin{align}
  |\vec{P}_n|^2 = \dfrac{1}{4}\{ |\vec{P}_{n-1}|^2 + |\vec{x}_n|^2 \} + \dfrac{1}{2}\vec{P}_{n-1} \cdot \vec{x}_n \cdots \text{④}
\end{align}
であり，
\begin{align}
  E(|\vec{P}_{n-1}|) = E_{n-1}
\end{align}
\begin{align}
  E(|\vec{x}_n|^2) = 1
\end{align}
\begin{align}
  E(\vec{P}_{n-1} \cdot \vec{x}_n) = \dfrac{1}{3}\vec{P}_{n-1} \cdot (\vec{a} + \vec{b} + \vec{c}) = 0
\end{align}
を④に代入して
\begin{align}
  E_n = \dfrac{1}{4} (E_{n-1} + 1)
\end{align}
③とあわせて
\begin{align}
  E_n = \dfrac{1}{3}\{1 - \left(\dfrac{1}{4}\right)^n\}
\end{align} \hfill(終)

\paragraph{(3)}

$P_0 = O$ の時，
\begin{align}
  \overrightarrow{OP_n} = \sum_{i=1}^n \dfrac{1}{2^{n+1-i}} \overrightarrow{OX_i}
\end{align}
だから，$Y_i$ を，$\vec{y}_i = \dfrac{1}{2^{n+1-i}} \vec{x}_i$ となるよう定めると，
\begin{align}
  |\vec{P}_n|^2 = |\vec{y}_1 + \vec{y}_2 + \cdots + \vec{y}_n|^2
\end{align}
\begin{align}
  = \sum_{i=1}^n |\vec{y}_i|^2 + 2 \sum_{i < j} \vec{y}_i \cdot \vec{y}_j
\end{align}
\begin{align}
  = \sum_{i=1}^n \dfrac{1}{4^{n+1-i}} + 2 \sum_{i < j} \vec{y}_i \cdot \vec{y}_j \quad (\because |\vec{x}_i| = 1)
\end{align}
\begin{align}
  = \dfrac{1}{3}\{1 - (1/4)^n\} + 2 \sum_{i < j} \vec{y}_i \cdot \vec{y}_j \cdots \text{①}
\end{align}
である．ここで，$i \ne j$ に対して，
\begin{align}
  \vec{x}_i \cdot \vec{x}_j =
  \begin{cases} 1 & (X_i = X_j) \\ -1/2 & (X_i \ne X_j)
  \end{cases} \cdots \text{②}
\end{align}
だから $\vec{x}_i \cdot \vec{x}_j$ の期待値 $P_{i,j}$ は
\begin{align}
  P = \dfrac{1}{3} \cdot 1 + \dfrac{2}{3} \left(-\dfrac{1}{2}\right) = 0
\end{align}
なので，$\sum_{i < j} \vec{y}_i \cdot \vec{y}_j$ の期待値 $Q$ は，
\begin{align}
  Q = \sum_{i < j} \dfrac{1}{2^{n+1-i} \cdot 2^{n+1-j}} \cdot P_{i,j} = 0
\end{align}
だから ①より
\begin{align}
  E_n = \dfrac{1}{3}(1 - (1/4)^n) + Q = \dfrac{1}{3} \{ 1 - \left(\dfrac{1}{4}\right)^n \}
\end{align} \hfill(終)
