{\bf [解答]}

\subparagraph{(1)}

与えられた数列の式に積和の公式を用いて
\begin{align}
  f_n(x)
  & = \cos\left(x + \dfrac{a_{n+1}+a_n}{2}\right) \sin\dfrac{a_{n+1}-a_n}{2} \\
  &= \dfrac{1}{2} \left[ \sin(x+a_{n+1}) - \sin(x+a_n) \right] \label{eq:1}
\end{align}
だから，$h_n(x) = \sum_{k=1}^n f_k(x)$ とおくと
\begin{align}
  h_n(x)
  &= \dfrac{1}{2} \left[ \sin(x+a_{n+1}) - \sin(x+a_1) \right]  \\
  &= \dfrac{1}{2} \left[ \sin x \cos a_{n+1} + \sin a_{n+1} \cos x - \sin(x+a_1) \right] \label{eq:2}
\end{align}
だから，$h_n(x)$ が $n \to \infty$ で収束する条件は
\begin{align}
  \sin a_n, \ \cos a_n \text{ が収束すること．}
\end{align}
である．

\subparagraph{(ロ)}

(イ)がみたされる時
\begin{align}
  \lim_{n\to\infty} \sin a_n = \alpha, \quad
  \lim_{n\to\infty} \cos a_n = \beta \label{eq:3}
\end{align}
とおくと，\cref{eq:2}に代入して$n\to\infty$の極限をとると
\begin{align}
  F(x) = \dfrac{1}{2} \left( \beta \sin x + \alpha \cos x - \sin(x+a_1) \right) \label{eq:4}
\end{align}
となる．

従ってこれを$[0,\pi/2]$で積分して
\begin{align}
  A
  & = \int_0^{\pi/2} F(x) dx \\
  &= \dfrac{1}{2} \left[ \left[ -\beta \cos x + \alpha \sin x \right]_0^{\pi/2} - \left[ -\cos(x+a_1) \right]_0^{\pi/2} \right] \label{eq:5}
\end{align}
となる．

一方で
\begin{align}
  B
  &= \sum_{k=1}^n \left( \int_0^{\pi/2} f_k(x) dx \right) \\
  &= \dfrac{1}{2} \sum_{k=1}^n \left( \left[ -\cos(x+a_{k+1}) \right]_0^{\pi/2} + \left[ \cos(x+a_k) \right]_0^{\pi/2} \right) \\
  &= \dfrac{1}{2} \left[ \left[ \cos(x+a_1) \right]_0^{\pi/2} - \left[ \cos(x+a_{n+1}) \right]_0^{\pi/2} \right] \\
  &\xrightarrow{n\to\infty} \dfrac{1}{2} \left[ \left[ \cos(x+a_1) \right]_0^{\pi/2} - \left[ \cos x \beta - \sin x \alpha \right]_0^{\pi/2} \right] \label{eq:6}
\end{align}
である．よって\cref{eq:5,eq:6}より $A = B$ であり，2つの値は等しい．
