{\bf [解]}
$f(x)=x^3+ax^2+(b-a-1)x$

\textbf{(1)} $f'(x)=3x^2+2ax+(b-a-1)$だから，これが$x\ge0$で0以上であるような$(a,b)$を求めれば良い（$a,b\in\mathbb{R}$）．軸$x=-\dfrac{1}{3}a$で場合分け．

\textbf{$1^\circ$ $-\dfrac{1}{3}a\le0\ \therefore\ a\ge0$の時}

条件は$f'(0)\ge0\iff b\ge a+1$

\textbf{$2^\circ$ $0\le-\dfrac{1}{3}a\ \therefore\ a\le0$の時}

条件は$f'(-\tfrac13a)\ge0\iff b\ge\dfrac{1}{3}a^2+a+1$

以上をまとめ，図示して下図斜線部
\begin{align*}
\begin{cases}
a\le0\text{の時}\ b\ge\dfrac{1}{3}a^2+a+1 \\
a\ge0\text{の時}\ b\ge a+1
\end{cases}
\end{align*}

\textbf{(2)} $(a,b)$が$G$をうごく時，$0\le x$で$f(x)$は単調増加かつ$f(1)=b$から，グラフの概形は右図で，斜線部が求める面積$K$に等しい．したがって，与式$S$として
\begin{align*}
S=b-\int_0^1f(x)dx
\end{align*}
\begin{align*}
=b-\left(\dfrac{1}{4}+\dfrac{1}{3}a+\dfrac{1}{2}(b-a-1)\right)
\end{align*}
\begin{align*}
=\dfrac{1}{2}b+\dfrac{1}{6}a+\dfrac{1}{4} \quad\cdots\text{①}
\end{align*}
である．$a$を固定すると，①から，

\textbf{$a\le0$の時}
\begin{align*}
\min S=\dfrac{1}{6}a+\dfrac{1}{2}\left(\dfrac{1}{3}a^2+a+1\right)+\dfrac{1}{4}
\end{align*}
\begin{align*}
=\dfrac{1}{6}a^2+\dfrac{2}{3}a+\dfrac{3}{4}
\end{align*}
\begin{align*}
=\dfrac{1}{6}(a+2)^2+\dfrac{1}{12}\ge\dfrac{1}{12} \quad(\text{等号成立は}a=-2)
\end{align*}

\textbf{$a\ge0$の時}
\begin{align*}
\min S=\dfrac{1}{6}a+\dfrac{1}{2}(a+1)+\dfrac{1}{4}
\end{align*}
\begin{align*}
=\dfrac{2}{3}a+\dfrac{3}{4}\ge\dfrac{3}{4} \quad(\because a=0)
\end{align*}

だから，$\min S=\dfrac{1}{12}$
