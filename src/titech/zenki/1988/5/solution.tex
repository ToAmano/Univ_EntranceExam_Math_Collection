{\bf [解]}
\begin{align}
\dfrac{{}_{3n}C_n}{{}_{2n}C_n}=\dfrac{(3n)!}{n!(2n)!}\cdot\dfrac{n!n!}{(2n)!}=\dfrac{(3n)!\,n!}{(2n)!(2n)!}\equiv f(n)
\end{align}
とおく．$f(n)>0$より，$g(n)=\log f(n)^{\frac{1}{n}}$とおくと
\begin{align}
g(n)=\dfrac{1}{n}\left(\sum_{k=2n+1}^{3n}\log\dfrac{k}{n}-\sum_{k=n+1}^{2n}\log\dfrac{k}{n}\right) \quad\cdots\text{①}
\end{align}
である．ここで
\begin{align}
\begin{split}
\dfrac{1}{n}\sum_{k=n+1}^{2n}\log\dfrac{k}{n}\longrightarrow\int_1^2\log x\,dx&=2(\log2+1)-1 \\
\dfrac{1}{n}\sum_{k=2n+1}^{3n}\log\dfrac{k}{n}\longrightarrow\int_2^3\log x\,dx&=3(\log3+1)-2(\log2+1)
\end{split}
\end{align}
だから，①より
\begin{align}
g(n)\longrightarrow3\log3+4-4\log2-4=3\log3-4\log2=\log\dfrac{27}{16}
\end{align}
以上から，
\begin{align}
\lim_{n\to\infty}\left(\dfrac{{}_{3n}C_n}{{}_{2n}C_n}\right)^{\frac{1}{n}}=\dfrac{27}{16}
\end{align}
