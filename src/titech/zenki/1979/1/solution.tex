{\bf[解]} 

$\vec{a}=(1,1,1)$，$\vec{b}=(2,3,-1)$，$\vec{c}=(0,1,0)$とすると，$\vec{a}$，$\vec{b}$はそれぞれ，
$l$，$m$の方向ベクトルである．$P_n$，$Q_n$が各直線上にあるので，実数$t_n$，$s_n$を用いて
\begin{align}
 &\vec{OP_n}=t_n\vec{a}&\vec{OQ_n}=\vec{c}+s_n\vec{b} \label{eq:1}
\end{align}
と表される．題意の条件に\cref{eq:1}を代入して
\begin{align}
 &\begin{cases}\displaystyle
  \vec{b}\cdot\vec{P_nQ_n}=0 \\
  \vec{a}\cdot\vec{Q_nP_{n+1}}=0
  \end{cases}\nonumber \\
 \Longleftrightarrow     
 &\begin{cases}\displaystyle
  \vec{b}\cdot(\vec{c}+s_n\vec{b}-t_n\vec{a})=0 \\
  \vec{a}\cdot(\vec{c}+s_n\vec{b}-t_{n+1}\vec{a})=0
  \end{cases}\nonumber \\
\Longleftrightarrow
 &\begin{cases}\displaystyle
  \vec{b}\cdot\vec{c}+s_n|\vec{b}|^2-t_n\vec{a}\cdot\vec{b}=0 \\
  \vec{a}\cdot\vec{c}+s_n\vec{a}\cdot\vec{b}-t_{n+1}|\vec{a}|^2=0 
 \end{cases}\label{eq:2}
\end{align}
ここで，\cref{eq:1}より
\begin{align}
 &|\vec{a}|^2=3&|\vec{b}|^2=14 \\
 &\vec{a}\cdot\vec{c}=1&\vec{b}\cdot\vec{c}=3 \\
 &\vec{a}\cdot\vec{b}=4
\end{align}
だから，これを\cref{eq:2}に代入して
\begin{align} 
 \begin{cases}\displaystyle
  3+14s_n-4t_n=0 \\
  1+4s_n-3t_{n+1}=0
 \end{cases}\label{eq:3}
\end{align}
を得る．\cref{eq:3}から$s_n$を消去して$t_n$のみの漸化式を作ると
\begin{align}
 t_{n+1}=\dfrac{1}{21}(8t_n+1) \\
 t_{n+1}-\dfrac{1}{13}=\dfrac{8}{21}\left(t_n-\dfrac{1}{13}\right) 
\end{align}
だから，これを繰り返し用いて$t_n$の一般項は
\begin{align}
  t_n=\dfrac{1}{13}+\left(\dfrac{8}{21}\right)^{n-1}\left(t_1-\dfrac{1}{13}\right)
\end{align}
となり，極限値は
\begin{align}
  \lim_{n\to\infty}t_n = \dfrac{1}{13}\label{eq:4}
\end{align}
である．
$s_n$についてはこの結果を\cref{eq:3}に代入して
\begin{align}
 \lim_{n\to\infty}s_n
 &=\lim_{n\to\infty}\left(\dfrac{4t_n-3}{14}\right)\nonumber \\
 &=\dfrac{1}{14}\left(4\lim_{n\to\infty}t_n-3\right)\nonumber\\
 &=\dfrac{1}{14}\left(4\dfrac{1}{13}-3\right)&(\because\text{\cref{eq:3}})\nonumber \\
 &=\dfrac{-5}{26}\label{eq:5}
\end{align}          
である．
以上\cref{eq:4,eq:5}を\cref{eq:1}に代入して
\begin{align}
 \lim_{n\to\infty}\vec{OP_n}
  &=\lim_{n\to\infty} t_n\vec{a}\\
  &=\dfrac{1}{13}\vec{a} \\
  &=\left(\dfrac{1}{13},\dfrac{1}{13},\dfrac{1}{13}\right) \\ 
 \lim_{n\to\infty}\vec{OQ_n}
  &=\vec{c}+\lim_{n\to\infty}s_n\vec{b}\\
  &=\vec{c}-\dfrac{5}{26}\vec{b} \\
  &=\left(\dfrac{-5}{13},\dfrac{11}{26},\dfrac{5}{26}\right)
\end{align}
となって，求める点は
\begin{align}
 &P_n\to\left(\dfrac{1}{13},\dfrac{1}{13},\dfrac{1}{13}\right) \\
 &Q_n\to\left(\dfrac{-5}{13},\dfrac{11}{26},\dfrac{5}{26}\right)
\end{align}
である．$\cdots$(答)
