{\bf [解]}
与えられた数列
\begin{align}
I_n=(-1)^n\int_0^{\frac{\pi}{4}}\dfrac{x\cos(2n-1)x}{\cos x}dx \label{eq:1}
\end{align}
について考える．

\textbf{(1)}
$n \ge 2$について$I_n-I_{n-1}$に\cref{eq:1}を代入して具体的に計算すると
\begin{align}
\begin{split}
I_n-I_{n-1}
 &=(-1)^n\left[\int_0^{\frac{\pi}{4}}\dfrac{x\cos(2n-1)x}{\cos x}dx+\int_0^{\frac{\pi}{4}}\dfrac{x\cos(2n-3)x}{\cos x}dx\right] \\
 &=(-1)^n\int_0^{\frac{\pi}{4}}\dfrac{x\{\cos(2n-3)x+\cos(2n-1)x\}}{\cos x}dx \\
 &=(-1)^n\int_0^{\frac{\pi}{4}}\dfrac{x}{\cos x}\cdot2\cos(2n-2)x\cos x\,dx \\
 &=(-1)^n\int_0^{\frac{\pi}{4}}2x\cos(2n-2)x\,dx \quad\cdots\text{①}
\end{split} \label{eq:2}
\end{align}
である．
$n\ge2$に注意して部分積分すると
\begin{align}
\begin{split}
\int_0^{\frac{\pi}{4}}2x\cos(2n-2)x\,dx
  &=2\left[\dfrac{x}{2n-2}\sin(2n-2)x+\dfrac{1}{(2n-2)^2}\cos(2n-2)x\right]_0^{\frac{\pi}{4}} \\
  &=2\left\{\dfrac{\pi}{4}\cdot\dfrac{1}{2n-2}\sin\dfrac{n-1}{2}\pi+\dfrac{1}{(2n-2)^2}\left(\cos\dfrac{n-1}{2}\pi-1\right)\right\}
\end{split}
\end{align}
だから，\cref{eq:2}に代入して
\begin{align}
I_n-I_{n-1}=2(-1)^n\left\{\dfrac{\pi}{4}\cdot\dfrac{1}{2n-2}\sin\dfrac{n-1}{2}\pi+\dfrac{1}{(2n-2)^2}\left(\cos\dfrac{n-1}{2}\pi-1\right)\right\}
\end{align}
である．

\textbf{(2)} 
まず$I_1$を計算すると\cref{eq:1}より
\begin{align}
  I_1
   &=-\int_0^{\frac{\pi}{4}}x\,dx \\
   &=-\dfrac{1}{2}\left(\dfrac{\pi}{4}\right)^2
\end{align}
だから，(1)で得た漸化式で$n=2,3$として
\begin{align}
\begin{split}
I_2&=I_1+2\left(\dfrac{\pi}{4}\cdot\dfrac{1}{2}-\dfrac{1}{4}\right)=I_1+\dfrac{\pi}{4}-\dfrac{1}{2} \\
I_3&=I_2-2\left(\dfrac{\pi}{4}\cdot\dfrac{1}{4}\cdot0-\dfrac{1}{8}\right)=I_2+\dfrac{1}{4} \\
&\therefore\ I_3=I_1+\dfrac{\pi}{4}-\dfrac{1}{2}+\dfrac{1}{4}=-\dfrac{\pi^2}{32}+\dfrac{\pi}{4}-\dfrac{1}{4}
\end{split}
\end{align}
である．
