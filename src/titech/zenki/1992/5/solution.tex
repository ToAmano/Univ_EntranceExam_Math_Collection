{\bf [解]}
\begin{align}
I_n=(-1)^n\int_0^{\dfrac{\pi}{4}}\dfrac{x\cos(2n-1)x}{\cos x}dx
\end{align}
である．

\textbf{(1)}
\begin{align}
I_n-I_{n-1}=(-1)^n\left[\int_0^{\dfrac{\pi}{4}}\dfrac{x\cos(2n-1)x}{\cos x}dx+\int_0^{\dfrac{\pi}{4}}\dfrac{x\cos(2n-3)x}{\cos x}dx\right]
\end{align}
\begin{align}
=(-1)^n\int_0^{\dfrac{\pi}{4}}\dfrac{x\{\cos(2n-3)x+\cos(2n-1)x\}}{\cos x}dx
\end{align}
\begin{align}
=(-1)^n\int_0^{\dfrac{\pi}{4}}\dfrac{x}{\cos x}\cdot2\cos(2n-2)x\cos x\,dx
\end{align}
\begin{align}
=(-1)^n\int_0^{\dfrac{\pi}{4}}2x\cos(2n-2)x\,dx \quad\cdots\text{①}
\end{align}
これは$n\ge2$であることから
\begin{align}
\int_0^{\dfrac{\pi}{4}}2x\cos(2n-2)x\,dx=2\left[\dfrac{x}{2n-2}\sin(2n-2)x+\dfrac{1}{(2n-2)^2}\cos(2n-2)x\right]_0^{\dfrac{\pi}{4}}
\end{align}
\begin{align}
=2\left\{\dfrac{\pi}{4}\cdot\dfrac{1}{2n-2}\sin\dfrac{n-1}{2}\pi+\dfrac{1}{(2n-2)^2}\left(\cos\dfrac{n-1}{2}\pi-1\right)\right\}
\end{align}
だから，①に代入して
\begin{align}
I_n-I_{n-1}=2(-1)^n\left\{\dfrac{\pi}{4}\cdot\dfrac{1}{2n-2}\sin\dfrac{n-1}{2}\pi+\dfrac{1}{(2n-2)^2}\left(\cos\dfrac{n-1}{2}\pi-1\right)\right\}
\end{align}

\textbf{(2)} $I_1=-\displaystyle\int_0^{\dfrac{\pi}{4}}x\,dx=-\dfrac{1}{2}\left(\dfrac{\pi}{4}\right)^2$だから，(1)で$n=2,3$として
\begin{align}
I_2=I_1+2\left(\dfrac{\pi}{4}\cdot\dfrac{1}{2}-\dfrac{1}{4}\right)=I_1+\dfrac{\pi}{4}-\dfrac{1}{2}
\end{align}
\begin{align}
I_3=I_2-2\left(\dfrac{\pi}{4}\cdot\dfrac{1}{4}\cdot0-\dfrac{1}{8}\right)=I_2+\dfrac{1}{4}
\end{align}
\begin{align}
\therefore\ I_3=I_1+\dfrac{\pi}{4}-\dfrac{1}{2}+\dfrac{1}{4}=-\dfrac{\pi^2}{32}+\dfrac{\pi}{4}-\dfrac{1}{4}
\end{align}
である．
