{\bf [解]}
与えられた関数
\begin{align}
f(x)=
\begin{cases}\displaystyle
2x & (0\le x<1/2) \\
2x-1 & (1/2\le x\le1)
\end{cases} \label{eq:1}
\end{align}
について考える．このグラフは以下のようになる．境界部分は白丸は含まず，黒丸のみ含む．
また，関数列$\{f_n(x)\}$は$f_1(x)=f(x)$を満たす．

\paragraph{(1)}

題意の漸化式および\cref{eq:1}より
\begin{align}
f_{n+1}(x)=
\begin{cases}\displaystyle
2f_n(x) & (0\le f_n(x)<1/2) \\
2f_n(x)-1 & (1/2\le f_n(x)\le1)
\end{cases} \label{eq:2}
\quad\cdots\text{①}
\end{align}
である．

具体的に$f_{2}(x)$を計算すると\cref{eq:1,eq:2}より
\begin{align}
 f_{2}(x) &=
  \begin{cases}
    2f_1(x) & (0\le f_1(x)<1/2) \\
    2f_1(x)-1 & (1/2\le f_1(x)\le1)
  \end{cases} \\
&=
  \begin{cases}
    4x   & \left(0 \le x < 1/4 \right) \\
    4x-1 & \left(1/4 \le x < 1/2 \right) \\
    4x-2 & \left(1/2 \le x < 3/4 \right) \\
    4x-3 & \left(1/4 \le x < 1 \right) \\
    1    & \left(x = 1 \right) \\
  \end{cases}
\end{align}
である．全く同様に$f_{3}(x)$を計算すると
\begin{align}
 f_3(x)=
  \begin{cases}
   8x-k & \quad\left(\dfrac{k}{8}\le x<\dfrac{k+1}{8}\right) \quad (k=0,1,\cdots,7) \\
   1    & (x=1)
  \end{cases}
\end{align}
であり，グラフを図示すると下図のようになる．

\paragraph{(2)}

まず$f_n(x)$の一般形を考えよう．(1)以下同様に考えると$t=0,1,\cdots,2^n-1$に対して
\begin{align}
 \begin{cases}
  f_n(x)&=2^nx-t\ \left(\dfrac{t}{2^n}\le x<\dfrac{t+1}{2^n}\right) \\
  f_n(1)&=1  
 \end{cases} \label{eq:3}
\end{align}
であることを数学的帰納法に示す．$n=1$では成立するから，以下$n=\ell\ (\ell\in\mathbb{N})$での成立を仮定して$n=\ell+1$での成立を示す．
仮定および\cref{eq:2}より$f_{\ell+1}(x)$は
\begin{align}
f_{\ell+1}(x)
&=
\begin{cases}\displaystyle
2(2^\ell x-t) & \left(\dfrac{t}{2^\ell}\le x<\dfrac{t+1/2}{2^\ell}\right) \\
2(2^\ell x-t)-1 & \left(\dfrac{t+1/2}{2^\ell}\le x<\dfrac{t+1}{2^\ell}\right)
\end{cases} \\
&= \begin{cases}\displaystyle
2^{\ell+1}x-2t & \left(\dfrac{2t}{2^{\ell+1}}\le x<\dfrac{2t+1}{2^{\ell+1}}\right) \\
2^{\ell+1}x-(2t+1) & \left(\dfrac{2t+1}{2^{\ell+1}}\le x<\dfrac{2t+2}{2^{\ell+1}}\right)
\end{cases}
\end{align}
だから，$f_{\ell+1}(1)=1$とあわせて，
\begin{align}
 f_{\ell+1}(x)=2^{\ell+1}x-t'\ \left(\dfrac{t'}{2^{\ell+1}}\le x<\dfrac{t'+1}{2^{\ell+1}}\right)
\end{align}
となって$n=\ell+1$でも\cref{eq:3}は成立する．
よって数学的帰納法により任意の$n\in\mathbb{N}$で\cref{eq:3}が成立する．


% ，$f_n(x)=2^nx-t\ \left(\dfrac{t}{2^n}\le x<\dfrac{t+1}{2^n}\right)$，$f_n(1)=1$（$t=0,1,\cdots,2^n-1$）であること$\cdots\diamondsuit$を帰納的に示す．$n=\ell\ (\ell\in\mathbb{N})$での$\diamondsuit$の成立を仮定すると，①から，

さて，以下題意の極限値を計算しよう．自然数$k,m$が
\begin{align}
 1 \le k \le 2^{m}-1 \label{eq:4}
\end{align}
を満たすとして$p=\dfrac{k}{2^m}$とおく．新しく
\begin{align}
 A_n = \dfrac{1}{n}\sum_{l=1}^{n}f_{l}(p) \label{eq:5}
\end{align}
とおくと求めるべき極限は
\begin{align}
 \lim_{n\to\infty}A_n \label{eq:6}
\end{align}
である．$n\to\infty$を考えるので$n$は$m$より大きいとして良く，$m\le n$で考えれば良い．

さて $1\le\ell\le n$ なる自然数$\ell$に対し整数$t_p$が存在して，
\begin{align}
f_\ell(p)=2^\ell p-t_p \quad(\text{ただし}\ t_p\text{は}\ t_p/2^\ell\le p<(t_p+1)/2^\ell\text{をみたす0以上の整数}) \quad\cdots\text{②} \label{eq:7}
\end{align}
と書ける．$t_p$についての不等式を整理すると
\begin{align}
\dfrac{t_p}{2^\ell}\le\dfrac{k}{2^m}<\dfrac{t_p+1}{2^\ell} \iff k\cdot2^{\ell-m}-1<t_p\le k\cdot2^{\ell-m} \quad\cdots\text{③} \label{eq:8}
\end{align}
である．$\ell\ge m$の時$2^{\ell-m}$が整数だから\cref{eq:8}の両辺は整数で1つ差である．$t_p\in\mathbb{Z}$だから右側の等号が成立するしかなく，この時\cref{eq:7}より$f_\ell(p)=0$である．したがって，\cref{eq:5}において$\ell=1,2,\cdots,m$で和をとれば良い．\cref{eq:7}より$f_{\ell}(p) \le 1$だから
\begin{align}
 A_n = \dfrac{1}{n}\sum_{\ell=1}^{m}f_{\ell}(p) \le \dfrac{1}{n}\sum_{\ell=1}^{m} 1 = \dfrac{m}{n}
\end{align}
である．また$f_{\ell}(p) \ge 0$より$0 \le A_n$でもある．以上まとめて
\begin{align}
 0 \le A_n \le \dfrac{m}{n}
\end{align}
である．両辺は$n\to\infty$で$0$に収束するからはさみうちの定理より求める極限値は
\begin{align}
 \lim_{n\to\infty} A_n = 0
\end{align}
である．
