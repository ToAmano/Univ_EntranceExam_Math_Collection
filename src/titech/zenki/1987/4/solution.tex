{\bf [解]}
$x(t)=t+e^{at},\ y(t)=-t+e^{at}$とおく．又，$p=e^t$とする．
\begin{align}
x'(t)=1+ap^a, \qquad y'(t)=-1+ap^a \quad\cdots\text{①}
\end{align}
$a>0$から，$x'(t)>0$だから
\begin{align}
\dfrac{dy}{dx}=\dfrac{y'(t)}{x'(t)}=\dfrac{-1+ap^a}{1+ap^a} \quad\cdots\text{②}
\end{align}
題意から，$y(t)=0$の時，$dy/dx=0$なる$t$がある．$dy/dx=0\iff a\cdot p^a=1$（$\because②$）で，これをといて，（$a>0$）
\begin{align}
t=-\dfrac{1}{a}\log a
\end{align}
この時$y(t)=0$となるから，
\begin{align}
\dfrac{1}{a}\log a+\dfrac{1}{a}=0 \quad\therefore\ a=\dfrac{1}{e}\ (>0)
\end{align}

\textbf{(2)} ②及び(1)の結果から下表をうる．

$t\to+\infty$の時，$x(t),y(t)\to+\infty$．$t\to-\infty$の時，$x(t)\to-\infty,\ y(t)\to+\infty$．
\begin{align}
x(t)\le y(t)\iff t+e^{t/e}\le-t+e^{t/e}\iff t\le0 \quad(x(0)=1,\ y(0)=1)
\end{align}

\begin{align}
S=\triangle+\int_1^{2e}y(t)dx=\dfrac{1}{2}+\int_1^{2e}y(t)dx=\dfrac{1}{2}+\int_0^ey(t)\dfrac{dx}{dt}dt \quad\cdots\text{③}
\end{align}
ここで，
\begin{align}
y(t)\dfrac{dx}{dt}=\left(-t+e^{\dfrac{t}{e}}\right)\left(1+\dfrac{1}{e}e^{\dfrac{t}{e}}\right)=\dfrac{1}{e}e^{\dfrac{2}{e}t}+\left(1-\dfrac{t}{e}\right)e^{\dfrac{t}{e}}-t \quad\cdots\text{④}
\end{align}
であり，各項計算して，
\begin{align}
\begin{split}
\int_0^e\dfrac{1}{e}e^{\dfrac{2}{e}t}dt&=\dfrac{1}{2}\left[e^{\dfrac{2}{e}t}\right]_0^e=\dfrac{1}{2}(e^2-1) \\
\int_0^e\left(1-\dfrac{t}{e}\right)e^{\dfrac{t}{e}}dt&=e\int_0^1(1-x)e^xdx=e^2-2e \\
\int_0^et\,dt&=\dfrac{1}{2}e^2
\end{split}
\end{align}
だから④に代入して，
\begin{align}
④=\dfrac{1}{2}(e^2-1)+e^2-2e-\dfrac{1}{2}e^2=e^2-2e-\dfrac{1}{2}
\end{align}
だから，③から
\begin{align}
S=e(e-2)
\end{align}
