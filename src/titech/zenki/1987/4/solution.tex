{\bf [解]}

\paragraph{(1)}
わかりやすいように
\begin{align}
 \begin{cases}
  x(t)=t+e^{at} \\
  y(t)=-t+e^{at}
 \end{cases} \label{eq:1}
\end{align}
とおく．一階微分を計算すると
\begin{align}
 \begin{cases}
   x'(t)=1+ae^{at} \\
   y'(t)=-1+ae^{at} \quad\cdots\text{①}  
 \end{cases} \label{eq:2}
\end{align}
である．
$a>0$より$x'(t)>0$だから$\dfrac{dy}{dx}$は
\begin{align}
\dfrac{dy}{dx}=\dfrac{y'(t)}{x'(t)}=\dfrac{-1+ae^{at}}{1+ae^{at}} \quad\cdots\text{②}
\end{align}
となる．

題意より$y(t)=0$かつ$dy/dx=0$なる$t$があるから，これを$t_0$と置く．
まず\cref{eq:2}より
\begin{align}
  &\dfrac{dy}{dx}=0 \\
  &-1+ae^{at_0} = 0 \\
\therefore 
 &t_0=-\dfrac{1}{a}\log a
\end{align}
である．ついで$y(t_0)=0$に\cref{eq:1}に代入して
\begin{align}
\dfrac{1}{a}\log a+\dfrac{1}{a}=0 \quad\therefore\ a=\dfrac{1}{e}\ (>0)
\end{align}
が求める$a$である．

\paragraph{(2)} 
(1)の結果を\cref{eq:1,eq:2}に代入すると曲線$C$の増減表は以下のようになる．
\begin{table}[htb]
  \centering
\begin{tabular}{c|ccccc}
$t$ & & & $e$ & & \\
\hline
$x'$ & $+$ & $+$ & $+$ & & \\
\hline
$y'$ & & $-$ & $0$ & $+$ & \\
\hline
$(x,y)$ & $\searrow$ & & $(2e,0)$ & & $\nearrow$
\end{tabular}
\end{table}
また極限値は
\begin{align}
 \lim_{t\to+\infty}x(t) = +\infty, \quad \lim_{t\to+\infty}y(t) = +\infty \\
 \lim_{t\to-\infty}x(t) = -\infty, \quad \lim_{t\to-\infty}y(t) = +\infty 
\end{align}
である．

また，$y=x$との上下関係を調べると
\begin{align}
x(t)\le y(t)\iff t+e^{t/e}\le-t+e^{t/e}\iff t\le0 \quad(x(0)=1,\ y(0)=1)
\end{align}
である．

以上から$C$の概形は下図で，もとめる面積$S$は下図斜線部である．

図のように$O(0,0), A(1,1), B(1,0), D(2e,0)$と置くと面積$S$は$triangle OAB$と，曲線$D$と$x=1, y=0$の囲む部分の二つの面積の合計であるから
\begin{align}
S=\triangle OAB + ABC = \dfrac{1}{2}+\int_1^{2e}y(t)dx=\dfrac{1}{2}+\int_0^ey(t)\dfrac{dx}{dt}dt  \label{eq:3}
\end{align}
である．

ここで\cref{eq:1,eq:2}から被積分関数は
\begin{align}
 y(t)\dfrac{dx}{dt}
   &=\left(-t+e^{\frac{t}{e}}\right)\left(1+\dfrac{1}{e}e^{\frac{t}{e}}\right) \\
   &=\dfrac{1}{e}e^{\frac{2}{e}t}+\left(1-\dfrac{t}{e}\right)e^{\frac{t}{e}}-t \label{eq:4}
\end{align}
であり，各項個別に積分を計算すると
\begin{align}
\begin{split}
\int_0^e\dfrac{1}{e}e^{\frac{2}{e}t}dt&=\dfrac{1}{2}\left[e^{\frac{2}{e}t}\right]_0^e=\dfrac{1}{2}(e^2-1) \\
\int_0^e\left(1-\dfrac{t}{e}\right)e^{\frac{t}{e}}dt&=e\int_0^1(1-x)e^xdx=e^2-2e \\
\int_0^et\,dt&=\dfrac{1}{2}e^2
\end{split}
\end{align}
だから\cref{eq:3}に代入して，
\begin{align}
S = \dfrac{1}{2} + \dfrac{1}{2}(e^2-1)+e^2-2e-\dfrac{1}{2}e^2=e^2-2e
\end{align}
が求める面積である．
