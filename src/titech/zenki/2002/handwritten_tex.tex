\section*{第1問}

\begin{proof}[解]
$S=\sin x,\ C=\cos x$と書き，$g(x)=S-aC$（$=\sin x-a\cos x$）とおく。$g'(x)=C+aS$。

\textbf{$1^\circ$ $a\le0$の時：} $[0,\pi/4]$で$S\ge0,\ C\ge0$だから$g(x)=S-aC\ge0$。

\textbf{$2^\circ$ $a\ge0$の時：} $g'(x)=C+aS\ge0$（$C,S\ge0,\ a\ge0$）だから$g$は単調増加。$g(0)=-a,\ g(\pi/4)=\dfrac{\sqrt2}2(1-a)$。

したがって，
\begin{itemize}
\item $a\le0$の時：$g(x)\ge0$（$[0,\pi/4]$で）だから
\[
f(a)=\int_0^{\pi/4}(S-aC)\,dx=a\int_0^{\pi/4}(-C)\,dx+\int_0^{\pi/4}S\,dx=-\frac{\sqrt2}2a+\left(1-\frac{\sqrt2}2\right)
\]
これは$a$の単調減少関数。
\item $a\ge1$の時：$g(\pi/4)\le0$，$g$単調増加で$g(0)=-a\le0$だから$g(x)\le0$（$[0,\pi/4]$で）。
\[
f(a)=-\int_0^{\pi/4}(S-aC)\,dx=\frac{\sqrt2}2a-\left(1-\frac{\sqrt2}2\right)
\]
これは$a$の単調増加関数。
\item $0\le a\le1$の時：$g(0)\le0\le g(\pi/4)$，$g$単調増加だから，$g(\alpha)=0$なる$\alpha\in[0,\pi/4]$がただ1つ存在し
\[
f(a)=-\int_0^\alpha g(x)\,dx+\int_\alpha^{\pi/4}g(x)\,dx=G(\pi/4)+G(0)-2G(\alpha) \quad\cdots\text{①}
\]
（$G(x)=-\cos x-a\sin x$は$g$の原始関数の一つ）。
\end{itemize}

$G(0)=-1,\ G(\pi/4)=-\dfrac{\sqrt2}2(a+1),\ G(\alpha)=-(\cos\alpha+a\sin\alpha)$を①に代入して
\[
f(a)=-1-\frac{\sqrt2}2(a+1)+2\cos\alpha+2a\sin\alpha \quad\cdots\text{③}
\]
$g(\alpha)=0$から$a=\tan\alpha$（$0\le\alpha<\pi/4$）。これを③に代入して$\alpha$の関数とみると
\[
f=-1-\frac{\sqrt2}2(\tan\alpha+1)+2\cos\alpha+2\tan\alpha\sin\alpha
\]
\[
\frac{df}{d\alpha}=-\frac{\sqrt2}2\cdot\frac1{\cos^2\alpha}-2\sin\alpha+2\left(\frac{\sin\alpha}{\cos^2\alpha}+\sin\alpha\right)=\frac{4\sin\alpha-\sqrt2}{2\cos^2\alpha}
\]
これは$\sin\alpha=\sqrt2/4$で符号が$-$から$+$に変わるので，下表のように$f$はこの点で最小。
\begin{center}
\begin{tabular}{c|ccccc}
$\alpha$ & $0$ & & (\ $\sin\alpha=\sqrt2/4$\ ) & & $\pi/4$ \\
\hline
$f'$ & & $-$ & $0$ & $+$ & \\
\hline
$f$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
この時，$\sin\alpha=\sqrt2/4$，$\cos\alpha=\sqrt{1-2/16}=\sqrt{14}/4$，$\tan\alpha=\dfrac{\sqrt2/4}{\sqrt{14}/4}=\dfrac1{\sqrt7}=\dfrac{\sqrt7}7$だから，③に代入して
\[
\min f=-1-\frac{\sqrt2}2\left(\frac{\sqrt7}7+1\right)+2\cdot\frac{\sqrt{14}}4+2\cdot\frac{\sqrt7}7\cdot\frac{\sqrt2}4
\]
$-\dfrac{\sqrt2}2\cdot\dfrac{\sqrt7}7=-\dfrac{\sqrt{14}}{14}$と$2\cdot\dfrac{\sqrt7}7\cdot\dfrac{\sqrt2}4=\dfrac{\sqrt{14}}{14}$が打ち消し合うので
\[
\min f=-1-\frac{\sqrt2}2+\frac{\sqrt{14}}2=\frac{\sqrt{14}}2-\frac{\sqrt2}2-1
\]
\end{proof}

\section*{第2問}

\begin{proof}[解]
楕円$\dfrac{x^2}{17}+\dfrac{y^2}8=1$の外部の点$P(a,b)$を考える。

\textbf{$a=\pm\sqrt{17}$の時：} 一方の接線が$x=\pm\sqrt{17}$（垂直）だから，直交するにはもう一方が$y=\pm\sqrt8$（水平）である必要があり，このとき$P=(\pm\sqrt{17},\pm\sqrt8)$。

\textbf{$a\ne\pm\sqrt{17}$の時：} $P$から引いた接線は$y$軸に平行にならず，傾き$m$を持つので$\ell:y=m(x-a)+b$とおける。$P$は楕円の外部だから
\[
\frac{a^2}{17}+\frac{b^2}8>1 \quad\cdots\text{①}
\]
$X=x/\sqrt{17},\ Y=y/\sqrt8$なる変換を行うと，楕円は$X^2+Y^2=1$（単位円）に，接線$\ell$は
\[
\ell':\ \sqrt8\,Y=m(\sqrt{17}X-a)+b
\]
に移り，これも単位円に接する（接する条件は座標変換で保たれる）。よって，原点から$\ell'$までの距離が$1$：
\[
\frac{|b-ma|}{\sqrt{17m^2+8}}=1
\]
両辺2乗して
\[
(b-ma)^2=17m^2+8 \iff (a^2-17)m^2-2abm+(b^2-8)=0 \quad\cdots\text{②}
\]
$a\ne\pm\sqrt{17}$だから②は$m$の2次方程式で，2つの解が2本の接線の傾き$m_1,m_2$を与える。判別式（の$1/4$）は
\[
D/4=(ab)^2-(a^2-17)(b^2-8)=8a^2+17b^2-8\cdot17
\]
①（両辺を$136=8\cdot17$倍）から$8a^2+17b^2>136$，よって$D/4>0$。したがって②は常に2つの異なる実解$m_1,m_2$を持つ（$P$から引いた2接線は常に存在する）。2接線が直交する条件は$m_1m_2=-1$で，解と係数の関係$m_1m_2=(b^2-8)/(a^2-17)$から
\[
\frac{b^2-8}{a^2-17}=-1 \iff a^2+b^2=25 \quad\cdots\text{③}
\]
（$a=\pm\sqrt{17}$の場合の点$(\pm\sqrt{17},\pm\sqrt8)$も$17+8=25$をみたし，③に含まれる。）さらに，$a^2+b^2=25$をみたす点は自動的に①をみたす：$8<17$より$b^2/8\ge b^2/17$だから，$a^2/17+b^2/8\ge(a^2+b^2)/17=25/17>1$。

以上から，もとめる軌跡は円
\[
a^2+b^2=25
\]
\end{proof}

\section*{第3問}

\begin{proof}[解]
$A=(0,0,1)$，$L=(0,0,1+\sqrt2)$とする。$z$軸に関する対称性から，$B,C$は$xz$平面に関して対称（$y$座標の符号のみ逆）としてよい。$\triangle ABC$は1辺1の正三角形だから，$BC$の中点を$M$とすると$AM=\sqrt3/2$，$BM=CM=1/2$，$AM\perp BC$。$\overrightarrow{AM}$の方向を角$\theta$（$0<\theta<\pi$）で
\[
\overrightarrow{AM}=\frac{\sqrt3}2(\sin\theta,\,0,\,\cos\theta)
\]
とおくと（$BC$は$AM$に垂直な$y$方向），
\[
B=\left(\frac{\sqrt3}2\sin\theta,\ \frac12,\ 1+\frac{\sqrt3}2\cos\theta\right),\qquad
C=\left(\frac{\sqrt3}2\sin\theta,\ -\frac12,\ 1+\frac{\sqrt3}2\cos\theta\right)
\]
（実際$AB^2=\frac34\sin^2\theta+\frac14+\frac34\cos^2\theta=1$，$BC=1$となり正しく正三角形になる。）

光源$L$から点$P=(p_x,p_y,p_z)$を通る直線が$xy$平面（$z=0$）と交わる点$P'$は，$P'=t\,(p_x,p_y,0)$，$t=\dfrac{1+\sqrt2}{1+\sqrt2-p_z}$で与えられる。

$A$は$x,y$座標が$0$だから，$A$の影$A'$は原点$O$である。$B,C$は$x,z$座標が等しく$y$座標のみ符号が逆だから，$t_B=t_C=\dfrac{1+\sqrt2}{1+\sqrt2-\left(1+\frac{\sqrt3}2\cos\theta\right)}=\dfrac{1+\sqrt2}{\sqrt2-\frac{\sqrt3}2\cos\theta}$，
\[
B'=\left(t_B\cdot\frac{\sqrt3}2\sin\theta,\ \frac{t_B}2\right),\qquad C'=\left(t_B\cdot\frac{\sqrt3}2\sin\theta,\ -\frac{t_B}2\right)
\]
影の三角形$A'B'C'$（$A'=O$）は，底辺$B'C'$（長さ$t_B$，$x=t_B\frac{\sqrt3}2\sin\theta$の位置）を底辺とする二等辺三角形で，面積は
\[
T=\frac12\cdot t_B\cdot t_B\frac{\sqrt3}2\sin\theta=\frac{\sqrt3}4t_B^2\sin\theta
\]
$c=\cos\theta,\ s=\sin\theta$とし，$D=\sqrt2-\frac{\sqrt3}2c$（$D>0$は常に成立）とおくと$t_B=\dfrac{1+\sqrt2}D$だから
\[
T=\frac{\sqrt3}4(1+\sqrt2)^2\cdot\frac s{D^2}
\]
$\theta\in(0,\pi)$で$T$を最大化する。$D'=dD/d\theta=\frac{\sqrt3}2s$に注意して，$T\propto s/D^2$を$\theta$で微分すると，その符号は
\[
cD^2-s\cdot2DD'=D(cD-\sqrt3s^2)
\]
の符号（$D>0$だから）と一致し，$D>0$より$cD-\sqrt3s^2$の符号を調べれば良い。
\[
cD-\sqrt3s^2=c\left(\sqrt2-\frac{\sqrt3}2c\right)-\sqrt3(1-c^2)=\sqrt2c+\frac{\sqrt3}2c^2-\sqrt3
\]
これが$0$になるのは，$\sqrt3c^2+2\sqrt2c-2\sqrt3=0$，すなわち
\[
c=\frac{-2\sqrt2\pm\sqrt{8+24}}{2\sqrt3}=\frac{-2\sqrt2\pm4\sqrt2}{2\sqrt3}
\]
$|c|\le1$をみたすのは$c=\dfrac{2\sqrt2}{2\sqrt3}=\dfrac{\sqrt2}{\sqrt3}=\dfrac{\sqrt6}3$。$c<$この値で$cD-\sqrt3s^2<0$（減少），$c>$この値で$>0$（増加）と確かめられるので（$\theta$の向きに注意して），$T$はこの$c=\sqrt6/3$で最大となる。

この時$s=\sqrt{1-c^2}=\sqrt{1-2/3}=\sqrt{1/3}=1/\sqrt3$，$D=\sqrt2-\dfrac{\sqrt3}2\cdot\dfrac{\sqrt6}3=\sqrt2-\dfrac{\sqrt{18}}6=\sqrt2-\dfrac{3\sqrt2}6=\sqrt2-\dfrac{\sqrt2}2=\dfrac{\sqrt2}2$だから
\[
\max T=\frac{\sqrt3}4(1+\sqrt2)^2\cdot\frac{1/\sqrt3}{(√2/2)^2}=\frac{\sqrt3}4(1+\sqrt2)^2\cdot\frac1{\sqrt3}\cdot2=\frac12(\sqrt2+1)^2
\]
\end{proof}

\section*{第4問}

\begin{proof}[解]
\textbf{(1)} $y=1/x$は単調減少だから，区間$[k,k+1]$での面積比較により
\[
\sum_{k=1}^n\frac1k<1+\int_1^n\frac1x\,dx \quad\cdots\text{①}, \qquad
\int_1^n\frac1x\,dx+\frac1{n+1}<\sum_{k=1}^n\frac1k \quad\cdots\text{②}
\]
①②から（$\int_1^n(1/x)dx=\log n$）
\[
\log n+\frac1{n+1}<\sum_{k=1}^n\frac1k<\log n+1
\]
両辺$\log n$（$>0$）で割って
\[
1+\frac1{(n+1)\log n}<\frac1{\log n}\sum_{k=1}^n\frac1k<1+\frac1{\log n} \quad\cdots\text{③}
\]
③の両辺は$n\to\infty$で$1$に収束するから，はさみうちの定理から
\[
\lim_{n\to\infty}\frac1{\log n}\sum_{k=1}^n\frac1k=1
\]

\textbf{(2)} $f_n(x)=\displaystyle\prod_{k=0}^n(x-k)$とおく。積の微分法から
\[
f_n'(x)=(x-1)\cdots(x-n)+x(x-2)\cdots(x-n)+\cdots+x(x-1)\cdots(x-(n-1))
\]
（各項は$k$番目の因子$(x-k)$を除いた積）。$0\le k\le n$なる整数$k$に対し$f_n(k)\ne0$（$x_n$は整数でない）だから，極値を与える$x=x_n$では$f_n(x_n)\ne0$であり，$f_n'(x_n)=0$の両辺を$f_n(x_n)$で割って
\[
0=\frac1{x_n}+\frac1{x_n-1}+\cdots+\frac1{x_n-n}
\]
すなわち
\[
\frac1{x_n}=\frac1{1-x_n}+\frac1{2-x_n}+\cdots+\frac1{n-x_n}
\]

次に$0<x_n\le1/2$を示す。$f_n(0)=f_n(1)=0$（$x=0,1$がともに$f_n$の根）だから，平均値の定理（ロルの定理）より，$f_n'(c)=\dfrac{f_n(1)-f_n(0)}{1-0}=0$をみたす$c\in(0,1)$が存在する。$x_n$は極値を与える最小の$x$であり，この$c$は$(0,1)$内にあるので，$0<x_n<1$である。

次に$1/2<x_n<1$と仮定すると，$1<1/x_n<2$であり，$0<1-x_n<1/2$より$1/(1-x_n)>2$，また$k\ge2$では$k-x_n>k-1\ge1>0$より$1/(k-x_n)>0$。したがって定義式の右辺は
\[
\frac1{1-x_n}+\frac1{2-x_n}+\cdots+\frac1{n-x_n}>\frac1{1-x_n}>2
\]
となり，左辺$1/x_n<2$と矛盾する。よって$1/2<x_n<1$ではあり得ず，$0<x_n\le1/2$である。
\end{proof}

\noindent\textbf{[解2]}（(2)の別証明の一部：$f_n'(0)$，$f_n'(1/2)$を直接評価する方法）

\begin{proof}[解2]
$f_n(0)=(-1)(-2)\cdots(-n)=(-1)^n\,n!$

$n=1$の時，$f_1(x)=x(x-1)$は放物線で，頂点（極値点）はちょうど$x=1/2$にあり，$x_1=1/2$。

$n\ge2$の時，$f_n'(0)$と$f_n'(1/2)$を直接計算すると符号が異なることが確かめられ（$f_n'(0)f_n'(1/2)<0$），$f_n'$は連続だから中間値の定理により$f_n'(x)=0$となる$x$が$(0,1/2)$内に存在する。以上いずれの場合も，$0<x_n\le1/2$に極値を与える$x$がある。
\end{proof}

\begin{proof}[解]
\textbf{(3)} $0<x_n\le1/2$より，$n$が十分大きいとして，各$k\ge1$で$k-1<k-x_n\le k-\dfrac12$（$k\ge1$では$k-x_n<k$，$k\ge1$では$x_n\le1/2$から$k-x_n\ge k-1/2$）に注意すると，
\[
\sum_{k=1}^n\frac1k<\frac1{x_n}=\sum_{k=1}^n\frac1{k-x_n}<\frac1{1-x_n}+\sum_{k=2}^n\frac1{k-1}\le\frac1{1-1/2}+\sum_{k=2}^n\frac1{k-1}=2+\sum_{j=1}^{n-1}\frac1j
\]
（左の不等式は$k-x_n<k$から$1/(k-x_n)>1/k$を各項について足し合わせたもの，右は$k\ge2$で$k-x_n>k-1$から$1/(k-x_n)<1/(k-1)$，$k=1$の項は$1/(1-x_n)\le1/(1-1/2)=2$による。）
$\displaystyle\sum_{j=1}^{n-1}\frac1j=\sum_{k=1}^n\frac1k-\frac1n$だから
\[
\sum_{k=1}^n\frac1k<\frac1{x_n}<\sum_{k=1}^n\frac1k+2-\frac1n
\]
両辺を$\log n$（$>0$）で割って
\[
\frac1{\log n}\sum_{k=1}^n\frac1k<\frac1{x_n\log n}<\frac1{\log n}\sum_{k=1}^n\frac1k+\frac1{\log n}\left(2-\frac1n\right) \quad\cdots\text{⑦}
\]
⑦の両辺は(1)から$n\to\infty$で$1$に収束するので，はさみうちの定理から
\[
\frac1{x_n\log n}\longrightarrow1 \quad(n\to\infty)
\]
$1/x$は$x>0$で連続だから
\[
x_n\log n\longrightarrow1 \quad(n\to\infty)
\]
\end{proof}
