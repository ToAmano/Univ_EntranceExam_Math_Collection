$n$を自然数とする．
\begin{enumerate}
  \item 次の極限を求めよ．
\begin{align*}
  \lim_{n \to \infty}\dfrac{1}{\log n} \left( 1+\dfrac{1}{2}+\dfrac{1}{3}+\cdots+\dfrac{1}{n} \right)
\end{align*}
  \item 関数$y=x(x-1)(x-2)\cdots(x-n)$の極値を与える$x$の最小値を$x_n$とする．このとき
\begin{align*}
  \dfrac{1}{x_n}=\dfrac{1}{1-x_n}+\dfrac{1}{2-x_n}+\cdots+\dfrac{1}{n-x_n}
\end{align*}
および$\displaystyle 0 < x_n \leqq \dfrac{1}{2}$を示せ．
  \item (2)の$x_n$に対して，極限$\displaystyle\lim_{n \to \infty}x_n\log n$を求めよ．
\end{enumerate}
