{\bf [解]}
実数$x,y$が
\begin{align}
 x^2+y^2\le1 \label{eq:1}
\end{align}
をみたす場合を考える．

\paragraph{(1)}

$s=x+y,\ t=xy$とおく．$x,y$は二次方程式$z^2-sz+t=0$の2実解だから，判別式$D$が$0$以上であることが必要で
\begin{align}
s^2-4t\ge0 \label{eq:2}
\end{align}
である．このもとで$x^2+y^2=s^2-2t$を\cref{eq:1}に代入して
\begin{align}
 s^2-2t\le1 \label{eq:3}
\end{align}
である．
以上\cref{eq:2,eq:3}が求めるべき$(s,t)$の存在範囲で，まとめると
\begin{align}
 \dfrac{s^2-1}{2}\le t\le\dfrac{s^2}{4} \label{eq:4}
\end{align}
である．これを図示すると下図斜線部（境界含む）である．

\paragraph{(2)}

新しく
\begin{align}
 f(s,t)=t+ms
\end{align}
とおいて，この最大最小値を求めれば良い．
$s$を固定すると$f$は$t$の増加関数だから\cref{eq:4}から
\begin{align}
 \dfrac{s^2-1}{2}+ms\le f(s,t)\le\dfrac{s^2}{4}+ms \quad (\sqrt{2} \le s \le \sqrt{2})
\end{align}
である．
この左辺を$g(s)$，右辺を$h(s)$とおく: 
\begin{align}
\begin{cases}
g(s)&=\dfrac{1}{2}(s^2-1)+ms=\dfrac{1}{2}(s+m)^2-\dfrac{1}{2}m^2-\dfrac{1}{2} \\
h(s)&=\dfrac{1}{4}s^2+ms=\dfrac{1}{4}(s+2m)^2-m^2
\end{cases}
\end{align}
以下$m \ge 0$に注意して$g(s)$の最小値および$h(s)$の最大値を求めれば良い．


\subparagraph{$g(s)$の最小値について：} 

$g$は$s=-m$で最小の下に凸な放物線だから，$\sqrt{2} \le s \le \sqrt{2}$と$m$の大小関係に注意して
\begin{itemize}
\item $-\sqrt{2}\le-m$（$m\le\sqrt{2}$）の時：頂点が区間内にあり，$\min g=g(-m)=-\dfrac{1}{2}m^2-\dfrac{1}{2}$
\item $-m<-\sqrt{2}$（$m>\sqrt{2}$）の時：$g$は区間内で単調増加，$\min g=g(-\sqrt{2})=\dfrac{1}{2}(m-\sqrt{2})^2-\dfrac{1}{2}m^2-\dfrac{1}{2}=-\sqrt{2}m+\dfrac{1}{2}$
\end{itemize}
となる．

\subparagraph{$h(s)$の最大値について：} 

$h$は$s=-2m\le0$で最小をとる下に凸な放物線だから，区間$[-\sqrt{2},\sqrt{2}]$上の最大値は頂点$-2m$から遠い方の端点$s=\sqrt{2}$でとる．
よって
\begin{align}
\max h=h(\sqrt{2})=\dfrac{1}{4}(\sqrt{2}+2m)^2-m^2=\sqrt{2}m+\dfrac{1}{2}
\end{align}
である．

\casesend

以上をまとめて，もとめる最大値・最小値は
\begin{align}
\text{最大値}=\sqrt{2}\,m+\dfrac{1}{2},\qquad
\text{最小値}=
\begin{cases}\displaystyle
-\dfrac{1}{2}m^2-\dfrac{1}{2} & (0\le m\le\sqrt{2})\\[2mm]
-\sqrt{2}\,m+\dfrac{1}{2} & (m\ge\sqrt{2})
\end{cases}
\end{align}
である．
