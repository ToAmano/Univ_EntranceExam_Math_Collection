{\bf [解]}
与えられた不等式
\begin{align}
 \log_{a}\dfrac{x-2}{x-1} > \log_{a}\dfrac{3-x}{x} \label{eq:1}
\end{align}
について考える．
まず真数条件から 
\begin{align}
 \begin{cases}
  \dfrac{x-2}{x-1}>0 \\
  \dfrac{3-x}{x} > 0 
 \end{cases} \\
\therefore
 \begin{cases}
  0<x<1 \\
  2<x<3
 \end{cases} \label{eq:2}
\end{align}
である．
以下底の$a$が$1$より大きいか小さいかで場合分けして考える．

\subparagraph{$0<a<1$の時}

\cref{eq:1}の$\log$を外して
\begin{align}
 \dfrac{x-2}{x-1}<\dfrac{3-x}{x}
\end{align}
両辺に$(x-1)^2x^2>0$をかけて整理して
 \begin{align}
  (x-2)(x-1)x^2<(3-x)x(x-1)^2 \\
  x(x-1)(2x^2-6x+3)<0 \\ 
  0<x<\dfrac{3-\sqrt{3}}{2},\ 1<x<\dfrac{3+\sqrt{3}}{2}
\end{align}
だから，\cref{eq:2}とあわせて，
\begin{align}
0<x<\dfrac{3-\sqrt{3}}{2},\ 2<x<\dfrac{3+\sqrt{3}}{2}
\end{align}
である．

\subparagraph{$1<a$の時}

\cref{eq:1}の$\log$を外して
\begin{align}
 \dfrac{x-2}{x-1}>\dfrac{3-x}{x}
\end{align}
である．両辺に$(x-1)^2x^2>0$をかけて整理して
\begin{align}
 (x-2)(x-1)x^2 > (3-x)(x-1)^2x \\
 (2x^2-6x+3)x(x-1) > 0 \\
 x<0,\ \dfrac{3-\sqrt{3}}{2}<x<1,\ \dfrac{3+\sqrt{3}}{2}<x
\end{align}
である．\cref{eq:2}と合わせて
\begin{align}
\dfrac{3-\sqrt{3}}{2}<x<1,\ \dfrac{3+\sqrt{3}}{2}<x<3
\end{align}
である．
\casesend

以上二つのケースをまとめて
\begin{align}
 \begin{cases}
  0<a<1\ \text{の時} & 0<x<\dfrac{3-\sqrt{3}}{2},\ 2<x<\dfrac{3+\sqrt{3}}{2} \\
  1<a\ \text{の時}   & \dfrac{3-\sqrt{3}}{2}<x<1,\ \dfrac{3+\sqrt{3}}{2}<x<3
 \end{cases}
\end{align}
