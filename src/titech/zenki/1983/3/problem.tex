$A=
\begin{pmatrix} \displaystyle\cos\dfrac{2\pi}{3} & \displaystyle-\sin\dfrac{2\pi}{3}
  \\ \displaystyle\sin\dfrac{2\pi}{3} & \displaystyle\cos\dfrac{2\pi}{3}
\end{pmatrix}$とし，
ベクトル$\vec{u}_n=
\begin{pmatrix} x_n \\ y_n
\end{pmatrix}$ $(n=1, \, 2, \, \cdots\cdots)$を次のように定義する．
\begin{align}
  \vec{u}_1=
  \begin{pmatrix} 0 \\ 0
  \end{pmatrix}, \hspace{1\zw}
  \vec{u}_2=
  \begin{pmatrix} 1 \\ 0
  \end{pmatrix}, \hspace{1\zw}
  \vec{u}_n=\vec{u}_{n-1}+
  \dfrac{1}{2}A(\vec{u}_{n-1}-\vec{u}_{n-2}) \hspace{1\zw} (n\geqq3)
\end{align}
\begin{enumerate}
  \item $x_n$，$y_n$を求めよ．
  \item $\displaystyle\lim_{n\to\infty}x_n$，$\displaystyle\lim_{n\to\infty}y_n$を求めよ．
\end{enumerate}
