{\bf [解]}
\begin{align}
\ell_1: y=-x-2, \qquad \ell_2: y=-\dfrac{1}{t^2}x+\dfrac{2}{t}
\end{align}
$\ell_1$と$y=-\dfrac{3}{x}$の交点のうち$A(1,-3)$，$\ell_2$と$y=-\dfrac{3}{x}$の交点のうち$B\left(3t,-\dfrac{1}{t}\right)$，$\ell_1$と$\ell_2$の交点$C\left(-\dfrac{2t}{t-1},\dfrac{2}{t-1}\right)$とおく．$S(t)$は，直線$AC$，$BC$と曲線$AB$（$y=-3/x$）とで囲まれた図形の面積であり，右のように分割できる．

補助点$D\left(1,-\dfrac{1}{t}\right)$をとると，
\begin{align}
\begin{split}
\triangle ADB&=\dfrac{1}{2}(3t-1)\left(3-\dfrac{1}{t}\right)=\dfrac{(3t-1)^2}{2t} \\
\text{（曲線と弦}DB\text{の間の面積）}&=\int_1^{3t}\dfrac{3}{x}\,dx-(3t-1)\cdot\dfrac{1}{t}=3\log3t-\dfrac{3t-1}{t}
\end{split}
\end{align}
であり，$\triangle ABC$の面積は
\begin{align}
\triangle ABC=\dfrac{1}{2}\left|(3t-1)\left(\dfrac{2}{t-1}+3\right)-\left(-\dfrac{1}{t}+3\right)\left(-\dfrac{2t}{t-1}-1\right)\right|=\dfrac{1}{2}\cdot\dfrac{(3t-1)^2}{t-1}\cdot\dfrac{t+1}{t}
\end{align}
これらを合わせて計算すると，
\begin{align}
S(t)=\dfrac{2(3t-1)}{t-1}+3\log3t
\end{align}
を得る．よって
\begin{align}
S'(t)=\dfrac{3}{t}-\dfrac{4}{(t-1)^2}=\dfrac{(3t-1)(t-3)}{t(t-1)^2}
\end{align}
$t>1$において下表を得る．
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $1$ & & $3$ & & \\
\hline
$S'$ & & $-$ & $0$ & $+$ & \\
\hline
$S$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
したがって，
\begin{align}
\min S(t)=S(3)=8+6\log3
\end{align}
