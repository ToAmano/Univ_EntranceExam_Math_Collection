{\bf [解]}
\textbf{(1)} $a,b>0$とし$g(t)=\dfrac{1}{b}t^a-\log t$（$t>0$）とおく．
\begin{align}
g'(t)=\dfrac{a}{b}t^{a-1}-\dfrac{1}{t}=\dfrac{1}{t}\Bigl(\dfrac{a}{b}t^a-1\Bigr)
\end{align}
より，$g'(t)=0\iff t=\bigl(\dfrac{b}{a}\bigr)^{\frac{1}{a}}$．この点の前後で$g'$は$-$から$+$に変わるので，$g$は$t=\bigl(\dfrac{b}{a}\bigr)^{1/a}$で極小値
\begin{align}
g\Bigl(\Bigl(\dfrac{b}{a}\Bigr)^{\frac{1}{a}}\Bigr)=\dfrac{1}{a}-\dfrac{1}{a}\log\dfrac{b}{a}
\end{align}
をとる（$t\to0^+,\,t\to\infty$でともに$g(t)\to+\infty$なのでこれは最小値でもある）．

\textbf{(2)} $m>0$とする．条件(b)は$t>0$すべてで成り立つ不等式なので，(1)の結果を$a=x,\ b=y$として用いて言い換える．$g(t)=\dfrac{1}{y}t^x-\log t$の最小値は$t=(y/x)^{1/x}$で$\dfrac{1}{x}\bigl(1-\log\dfrac{y}{x}\bigr)$だから，条件(b)は
\begin{align}
\dfrac{1}{x}\Bigl(1-\log\dfrac{y}{x}\Bigr)\ge m
\end{align}
と同値．$x>0$に注意して整理すると$\log\dfrac{y}{x}\le1-mx$，すなわち
\begin{align}
y\le x\cdot e^{1-mx}\equiv f(x)
\end{align}
となる．よって$D$は
\begin{align}
0<x<y\le f(x)=x\,e^{1-mx}
\end{align}
で定まる領域である．
\begin{align}
f'(x)=e^{1-mx}(1-mx)
\end{align}
より$f$は$x=\dfrac{1}{m}$で最大値$f(\dfrac{1}{m})=\dfrac{1}{m}$をとり，$x\to0^+$と$x\to\infty$でともに$f(x)\to0$．また
\begin{align}
f(x)\ge x\iff e^{1-mx}\ge1\iff x\le\dfrac{1}{m}
\end{align}
（$x>0$より）．したがって$0<x\le\dfrac{1}{m}$では直線$y=x$と曲線$y=f(x)$の間（$x<y\le f(x)$）が存在し，$x>\dfrac{1}{m}$では$f(x)<x$となるので$D$は空である．よって$D$は図の斜線部分（境界は太線部を含み，点線部および$x=0$は含まない）．

\textbf{(3)} 面積$S$は
\begin{align}
S=\int_0^{\frac{1}{m}}\bigl(x\,e^{1-mx}-x\bigr)\,dx=e\int_0^{\frac{1}{m}}x\,e^{-mx}\,dx-\dfrac{1}{2}\Bigl(\dfrac{1}{m}\Bigr)^2
\end{align}
部分積分により$\displaystyle\int xe^{-mx}\,dx=-e^{-mx}\Bigl(\dfrac{x}{m}+\dfrac{1}{m^2}\Bigr)+C$なので
\begin{align}
\int_0^{\frac{1}{m}}xe^{-mx}\,dx=\Bigl[-e^{-mx}\Bigl(\dfrac{x}{m}+\dfrac{1}{m^2}\Bigr)\Bigr]_0^{\frac{1}{m}}=-e^{-1}\cdot\dfrac{2}{m^2}+\dfrac{1}{m^2}=\dfrac{1}{m^2}\Bigl(1-\dfrac{2}{e}\Bigr)
\end{align}
ゆえに
\begin{align}
S=e\cdot\dfrac{1}{m^2}\Bigl(1-\dfrac{2}{e}\Bigr)-\dfrac{1}{2m^2}=\dfrac{e-2}{m^2}-\dfrac{1}{2m^2}=\Bigl(e-\dfrac{5}{2}\Bigr)\dfrac{1}{m^2}
\end{align}
