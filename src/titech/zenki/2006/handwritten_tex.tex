\section*{第1問}

\begin{proof}[解]
\textbf{(1)} $A_k=\displaystyle\int_{(k-1)\pi}^{k\pi}|\sin x|\,dx$とおく。$|\sin x|$は周期$\pi$を持ち，$[0,\pi]$上$\sin x\ge0$だから，置換$x=t+(k-1)\pi$により
\begin{align*}
A_k=\int_0^\pi\sin t\,dt=2\qquad(k=1,2,\dots)
\end{align*}
が成り立つ。$n$が偶数$n=2m$（$m\in\mathbb N$）のとき
\begin{align*}
I(n)=\int_0^{m\pi}|\sin x|\,dx=\sum_{k=1}^mA_k=2m=n.
\end{align*}
$n$が奇数$n=2m+1$（$m=0,1,2,\dots$）のとき
\begin{align*}
I(n)=\sum_{k=1}^mA_k+\int_{m\pi}^{m\pi+\frac\pi2}|\sin x|\,dx=2m+\int_0^{\frac\pi2}\sin t\,dt=2m+1=n
\end{align*}
（最後の積分は$x=t+m\pi$と置換し，$|\sin x|$の周期性より$[m\pi,m\pi+\frac\pi2]$上の値は$[0,\frac\pi2]$上の$\sin t$の値に一致することを用いた）。以上より，すべての自然数$n$に対し
\begin{align*}
I(n)=n
\end{align*}
である。

\textbf{(2)} $\displaystyle\int_0^{\frac{s\pi}2}\cos x\,dx=\sin\frac{s\pi}2$であるから，示すべき不等式は
\begin{align*}
s\le\sin\frac{s\pi}2\le\frac\pi2s\qquad(0\le s\le1)\tag{$\ast$}
\end{align*}
と同値である。

（左側）$f(s)=\sin\dfrac{s\pi}2-s$とおくと，$f(0)=0$，$f(1)=\sin\dfrac\pi2-1=0$，
\begin{align*}
f'(s)=\frac\pi2\cos\frac{s\pi}2-1
\end{align*}
は単調減少であり，$\cos\dfrac{\alpha\pi}2=\dfrac2\pi$をみたす$\alpha\in(0,1)$を境に符号が$+$から$-$に変わる。よって$f$は$[0,\alpha]$で増加，$[\alpha,1]$で減少し，両端で$f=0$だから区間$[0,1]$全体で$f(s)\ge0$，すなわち$s\le\sin\dfrac{s\pi}2$。

（右側）$h(s)=\dfrac\pi2s-\sin\dfrac{s\pi}2$とおくと$h(0)=0$，
\begin{align*}
h'(s)=\frac\pi2\Bigl(1-\cos\frac{s\pi}2\Bigr)\ge0
\end{align*}
より$h$は単調増加，よって$h(s)\ge h(0)=0$，すなわち$\sin\dfrac{s\pi}2\le\dfrac\pi2s$。

以上で$(\ast)$が示され，もとの不等式
\begin{align*}
0\le\int_0^{\frac{s\pi}2}\cos x\,dx-s\le\Bigl(\frac\pi2-1\Bigr)s\qquad(0\le s\le1)
\end{align*}
が成り立つ。

\textbf{(3)} $J(a)=\displaystyle\int_0^{\frac{a\pi}2}|\sin t|\,dt$とおく（$a>0$）。$[a]$が奇数なので$[a]=2m-1$（$m\in\mathbb N$）と表せ，$\beta=a-[a]\in[0,1)$とおくと，(1)の$A_k=2$と$I(2m-1)=2m-1$を用いて
\begin{align*}
J(a)&=\int_0^{\frac{(2m-1)\pi}2}|\sin t|\,dt+\int_{\frac{(2m-1)\pi}2}^{\frac{a\pi}2}|\sin t|\,dt\\
&=(2m-1)+\int_0^{\frac{\beta\pi}2}\Bigl|\sin\Bigl(\frac{(2m-1)\pi}2+u\Bigr)\Bigr|\,du
\end{align*}
（$t=\dfrac{(2m-1)\pi}2+u$と置換）。$2m-1$は奇数だから$\sin\bigl(\frac{(2m-1)\pi}2+u\bigr)=\pm\cos u$となり，$0\le u\le\frac{\beta\pi}2<\frac\pi2$では$\cos u\ge0$なので絶対値は$\cos u$。よって
\begin{align*}
J(a)=(2m-1)+\int_0^{\frac{\beta\pi}2}\cos u\,du=[a]+\sin\frac{\beta\pi}2.
\end{align*}
(2)の$(\ast)$を$s=\beta\in[0,1)$に適用すると$\beta\le\sin\dfrac{\beta\pi}2\le\dfrac\pi2\beta$なので
\begin{align*}
[a]+\beta\ \le\ J(a)\ \le\ [a]+\frac\pi2\beta,
\quad\text{すなわち}\quad
a\ \le\ J(a)\ \le\ [a]+\frac\pi2(a-[a]).
\end{align*}
左側の不等式から$J(a)/a\ge1$。右側の不等式から
\begin{align*}
J(a)-a\le[a]+\frac\pi2(a-[a])-a=\Bigl(\frac\pi2-1\Bigr)(a-[a])
\end{align*}
となり，両辺を$a(>0)$で割ると
\begin{align*}
\frac{J(a)}a-1\le\Bigl(\frac\pi2-1\Bigr)\Bigl(1-\frac{[a]}a\Bigr).
\end{align*}
$t=au$と置換すれば$\displaystyle\int_0^{\frac\pi2}|\sin at|\,dt=\frac1aJ(a)$だから，以上をまとめて
\begin{align*}
0\le\int_0^{\frac\pi2}|\sin at|\,dt-1\le\Bigl(\frac\pi2-1\Bigr)\Bigl(1-\frac{[a]}a\Bigr)
\end{align*}
が示された。
\end{proof}

\section*{第2問}

\begin{proof}[解]
\textbf{(1)} $a,b>0$とし$g(t)=\dfrac1bt^a-\log t$（$t>0$）とおく。
\begin{align*}
g'(t)=\frac abt^{a-1}-\frac1t=\frac1t\Bigl(\frac abt^a-1\Bigr)
\end{align*}
より，$g'(t)=0\iff t=\bigl(\frac ba\bigr)^{\frac1a}$。この点の前後で$g'$は$-$から$+$に変わるので，$g$は$t=\bigl(\frac ba\bigr)^{1/a}$で極小値
\begin{align*}
g\Bigl(\Bigl(\frac ba\Bigr)^{\frac1a}\Bigr)=\frac1a-\frac1a\log\frac ba
\end{align*}
をとる（$t\to0^+,\,t\to\infty$でともに$g(t)\to+\infty$なのでこれは最小値でもある）。

\textbf{(2)} $m>0$とする。条件(b)は$t>0$すべてで成り立つ不等式なので，(1)の結果を$a=x,\ b=y$として用いて言い換える。$g(t)=\frac1yt^x-\log t$の最小値は$t=(y/x)^{1/x}$で$\frac1x\bigl(1-\log\frac yx\bigr)$だから，条件(b)は
\begin{align*}
\frac1x\Bigl(1-\log\frac yx\Bigr)\ge m
\end{align*}
と同値。$x>0$に注意して整理すると$\log\dfrac yx\le1-mx$，すなわち
\begin{align*}
y\le x\cdot e^{1-mx}\equiv f(x)
\end{align*}
となる。よって$D$は
\begin{align*}
0<x<y\le f(x)=x\,e^{1-mx}
\end{align*}
で定まる領域である。
\begin{align*}
f'(x)=e^{1-mx}(1-mx)
\end{align*}
より$f$は$x=\frac1m$で最大値$f(\frac1m)=\frac1m$をとり，$x\to0^+$と$x\to\infty$でともに$f(x)\to0$。また
\begin{align*}
f(x)\ge x\iff e^{1-mx}\ge1\iff x\le\frac1m
\end{align*}
（$x>0$より）。したがって$0<x\le\frac1m$では直線$y=x$と曲線$y=f(x)$の間（$x<y\le f(x)$）が存在し，$x>\frac1m$では$f(x)<x$となるので$D$は空である。よって$D$は図の斜線部分（境界は太線部を含み，点線部および$x=0$は含まない）。

\begin{figure}[htb]
\centering
\begin{tikzpicture}[scale=2.6]
  \draw[->] (-0.1,0) -- (1.6,0) node[right]{$x$};
  \draw[->] (0,-0.1) -- (0,1.3) node[above]{$y$};
  \fill[gray!25] plot[domain=0:1,samples=40] (\x,{\x*exp(1-\x)}) -- (0,0) -- cycle;
  \draw[dashed] (0,0) -- (1.35,1.35) node[right]{\scriptsize$y=x$};
  \draw[thick,domain=0:1.5,samples=60] plot (\x,{\x*exp(1-\x)}) node[right]{\scriptsize$y=xe^{1-mx}$};
  \draw[dotted] (1,0) -- (1,1) -- (0,1);
  \node[below] at (1,0) {$\frac1m$};
  \node[left] at (0,1) {$\frac1m$};
  \filldraw (1,1) circle(0.015);
\end{tikzpicture}
\caption{領域$D$（斜線部，$m=1$の場合の概形）}
\end{figure}

\textbf{(3)} 面積$S$は
\begin{align*}
S=\int_0^{\frac1m}\bigl(x\,e^{1-mx}-x\bigr)\,dx=e\int_0^{\frac1m}x\,e^{-mx}\,dx-\frac12\Bigl(\frac1m\Bigr)^2
\end{align*}
部分積分により$\displaystyle\int xe^{-mx}\,dx=-e^{-mx}\Bigl(\frac xm+\frac1{m^2}\Bigr)+C$なので
\begin{align*}
\int_0^{\frac1m}xe^{-mx}\,dx=\Bigl[-e^{-mx}\Bigl(\frac xm+\frac1{m^2}\Bigr)\Bigr]_0^{\frac1m}=-e^{-1}\cdot\frac2{m^2}+\frac1{m^2}=\frac1{m^2}\Bigl(1-\frac2e\Bigr)
\end{align*}
ゆえに
\begin{align*}
S=e\cdot\frac1{m^2}\Bigl(1-\frac2e\Bigr)-\frac1{2m^2}=\frac{e-2}{m^2}-\frac1{2m^2}=\Bigl(e-\frac52\Bigr)\frac1{m^2}
\end{align*}
\end{proof}

\section*{第3問}

\begin{proof}[解]
点$P$を原点とする。3個の円板を$A,B,C$とし，回転して$A$の中心$O_A=(1,0)$に固定してよい。$B,C$の中心$O_B,O_C$は単位円上にあるので，偏角を$\theta,\alpha$として
\begin{align*}
O_B=(\cos\theta,\sin\theta)\ (0\le\theta\le\pi),\qquad O_C=(\cos\alpha,\sin\alpha)\ (0\le\alpha\le2\pi)
\end{align*}
とおく。条件(b)（3円板すべてが共有する点は$P$のみ）をみたす$O_C$の条件は
\begin{align*}
\pi\le\alpha\le\theta+\pi\tag{①}
\end{align*}
である。

中心間の偏角の差が$\theta$（$0\le\theta\le\pi$）である2つの単位円板の共通部分の面積を$S(\theta)$とする。

\begin{figure}[htb]
\centering
\begin{tikzpicture}[scale=1.6]
  \coordinate (P) at (0,0);
  \coordinate (OA) at (1,0);
  \coordinate (OB) at ({cos(100)},{sin(100)});
  \fill[gray!25] (P) arc (180:100:1) arc (0:-80:1);
  \draw[thick] (OA) circle (1);
  \draw[thick] (OB) circle (1);
  \draw[dashed] (P) -- (OA);
  \draw[dashed] (P) -- (OB);
  \draw (0.35,0) arc (0:100:0.35) node[midway,above right]{\scriptsize$\theta$};
  \filldraw (P) circle(0.02) node[below left]{$P$};
  \filldraw (OA) circle(0.02) node[below right]{$O_A$};
  \filldraw (OB) circle(0.02) node[above]{$O_B$};
\end{tikzpicture}
\caption{中心角$\theta$の2円板の共通部分（斜線部）の面積が$S(\theta)$}
\end{figure}

2円の中心間距離は$2\sin\frac\theta2$なので，$x=\sin t$と置換して
\begin{align*}
S(\theta)=4\int_{\sin\frac\theta2}^1\sqrt{1-x^2}\,dx=4\int_{\frac\theta2}^{\frac\pi2}\cos^2t\,dt=4\Bigl[\frac t2+\frac{\sin2t}4\Bigr]_{\frac\theta2}^{\frac\pi2}=\pi-\theta-\sin\theta.
\end{align*}

$O_A,O_B,O_C$の偏角はそれぞれ$0,\theta,\alpha$（$0\le\theta\le\pi\le\alpha\le\theta+\pi$）なので，これらを結ぶ3つの弧の中心角は
\begin{align*}
\overset{\frown}{O_AO_B}=\theta,\qquad\overset{\frown}{O_BO_C}=\alpha-\theta,\qquad\overset{\frown}{O_CO_A}=2\pi-\alpha
\end{align*}
（和は$2\pi$）。3円板の和集合の面積$S$は，3円の面積の和から2円ずつの重なりを引けば得られる（3円共通部分は$P$のみで面積0だから，包除原理はここで止まる）：
\begin{align*}
S=3\pi-\bigl\{S(\theta)+S(\alpha-\theta)+S(2\pi-\alpha)\bigr\}.
\end{align*}
$S(\theta)+S(\alpha-\theta)+S(2\pi-\alpha)=3\pi-2\pi-\{\sin\theta+\sin(\alpha-\theta)+\sin(2\pi-\alpha)\}$であり，$\sin(2\pi-\alpha)=-\sin\alpha$だから
\begin{align*}
S=2\pi+\sin\theta+\sin(\alpha-\theta)-\sin\alpha.
\end{align*}
和積公式$\sin(\alpha-\theta)-\sin\alpha=-2\sin\dfrac\theta2\cos\Bigl(\alpha-\dfrac\theta2\Bigr)$を用いると
\begin{align*}
S-2\pi=-2\sin\frac\theta2\cos\Bigl(\alpha-\frac\theta2\Bigr)+\sin\theta\equiv f(\alpha)\tag{②}
\end{align*}

$\theta$を固定して$\alpha$を動かす。①より$\alpha-\dfrac\theta2\in\bigl[\pi-\frac\theta2,\ \pi+\frac\theta2\bigr]$であり，$0\le\theta\le\pi$より$\sin\frac\theta2\ge0$だから，$f(\alpha)$は$\cos\bigl(\alpha-\frac\theta2\bigr)$が最小となるとき，すなわち$\alpha-\dfrac\theta2=\pi$（この値は許容範囲内）のとき最大となり，
\begin{align*}
\max_\alpha f(\alpha)=2\sin\frac\theta2+\sin\theta\equiv g(\theta)\qquad(0\le\theta\le\pi).
\end{align*}
次に$\theta$を動かす。$t=\cos\dfrac\theta2$とおくと
\begin{align*}
g'(\theta)=\cos\frac\theta2+2\cos^2\frac\theta2-1=(2t-1)(t+1)
\end{align*}
$0\le\theta\le\pi$では$t\in[0,1]$なので$t+1>0$，符号は$2t-1$で決まり，$t=\frac12$すなわち$\theta=\dfrac{2\pi}3$を境に$g'$は$+$から$-$に変わる。よって$g$は$\theta=\dfrac{2\pi}3$で最大値
\begin{align*}
g\Bigl(\frac{2\pi}3\Bigr)=2\sin\frac\pi3+\sin\frac{2\pi}3=\sqrt3+\frac{\sqrt3}2=\frac{3\sqrt3}2
\end{align*}
をとる。したがって
\begin{align*}
\max S=2\pi+\frac{3\sqrt3}2
\end{align*}
である。このとき$\theta=\dfrac{2\pi}3$，$\alpha=\pi+\dfrac\theta2=\dfrac{4\pi}3$となるから，3つの弧の中心角は
\begin{align*}
\overset{\frown}{O_AO_B}=\overset{\frown}{O_BO_C}=\overset{\frown}{O_CO_A}=\frac{2\pi}3
\end{align*}
となり，3つの中心が$P$のまわりに正三角形状に等間隔配置されるときに最大値が実現される。

\centering

    \coordinate (O\a) at ({cos(\a)},{sin(\a)});
    \draw[thick] (O\a) circle (1);
    \filldraw (O\a) circle(0.02);
    \draw[dashed] (0,0) -- (O\a);
  }
  \filldraw (0,0) circle(0.025) node[below]{$P$};
  \node[right] at (O0) {$O_A$};
  \node[above left] at (O120) {$O_B$};
  \node[below left] at (O240) {$O_C$};
\end{tikzpicture}
\caption{$S$が最大となる配置（3つの中心が$P$のまわりに等間隔）}
\end{figure}
\end{proof}

\section*{第4問}

\begin{proof}[解]
各点の位置ベクトルを対応する小文字で表す：$A,B,C,D$の位置ベクトルを$\vec a,\vec b,\vec c,\vec d$とすると
\begin{align*}
\vec k=\frac{\vec a+\vec b}2,\quad \vec l=\frac{\vec b+\vec c}2,\quad \vec m=\frac{\vec c+\vec d}2,\quad \vec n=\frac{\vec d+\vec a}2.
\end{align*}

\textbf{(1)}
\begin{align*}
\overrightarrow{MK}\cdot\overrightarrow{LN}&=(\vec k-\vec m)\cdot(\vec n-\vec l)
=\frac14(\vec a+\vec b-\vec c-\vec d)\cdot(\vec a+\vec d-\vec b-\vec c)\\
&=\frac14\bigl[(\vec a-\vec c)+(\vec b-\vec d)\bigr]\cdot\bigl[(\vec a-\vec c)-(\vec b-\vec d)\bigr]\\
&=\frac14\bigl(|\vec a-\vec c|^2-|\vec b-\vec d|^2\bigr)=\frac14\bigl(|\overrightarrow{AC}|^2-|\overrightarrow{BD}|^2\bigr).
\end{align*}
よって$4\overrightarrow{MK}\cdot\overrightarrow{LN}=|\overrightarrow{AC}|^2-|\overrightarrow{BD}|^2$が示された。

\textbf{(2)} 6つの辺の長さを$AB=c_1,\,CD=c_2,\,AC=b_1,\,BD=b_2,\,AD=a_1,\,BC=a_2$とおく。4つの面はそれぞれ3辺の組
\begin{align*}
\triangle ABC:(a_2,b_1,c_1),\quad\triangle ABD:(a_1,b_2,c_1),\quad\triangle ACD:(a_1,b_1,c_2),\quad\triangle BCD:(a_2,b_2,c_2)
\end{align*}
を持つ。全ての面が互いに合同なので，これらの3辺の組はすべて等しい：
\begin{align*}
\{a_2,b_1,c_1\}=\{a_1,b_2,c_1\}=\{a_1,b_1,c_2\}=\{a_2,b_2,c_2\}.
\end{align*}
$\triangle ABC$と$\triangle ABD$は辺$AB$（長さ$c_1$）を共有しており，四面体が退化しない一般の場合にはこの共有辺どうしが対応するから，残りの2辺が一致する：
\begin{align*}
\{a_2,b_1\}=\{a_1,b_2\}.
\end{align*}
同様に，$\triangle ABC,\triangle ACD$は辺$AC$を共有するので$\{a_2,c_1\}=\{a_1,c_2\}$，$\triangle ABD,\triangle ACD$は辺$AD$を共有するので$\{b_2,c_1\}=\{b_1,c_2\}$。これら3つの関係を同時にみたす組合せは
\begin{align*}
a_1=a_2,\qquad b_1=b_2,\qquad c_1=c_2
\end{align*}
のみであるから，
\begin{align*}
|\overrightarrow{AC}|=|\overrightarrow{BD}|,\qquad |\overrightarrow{BC}|=|\overrightarrow{AD}|,\qquad|\overrightarrow{AB}|=|\overrightarrow{CD}|
\end{align*}
が示された。

\textbf{(3)} (2)より対辺が等しい四面体なので，$AB=CD=\sqrt3$，$BC=AD=\sqrt5$，$CA=BD=\sqrt6$である。このような（対辺が等しい）四面体は直方体の4頂点として実現できる。直方体の辺の長さを$x,y,z$とし
\begin{align*}
A=(0,0,0),\ B=(x,y,0),\ C=(x,0,z),\ D=(0,y,z)
\end{align*}
とおくと
\begin{align*}
AB=CD=\sqrt{x^2+y^2},\quad CA=BD=\sqrt{x^2+z^2},\quad BC=AD=\sqrt{y^2+z^2}
\end{align*}
となり，これは対辺が等しい構造と一致する。与えられた長さから
\begin{align*}
x^2+y^2=3,\qquad y^2+z^2=5,\qquad x^2+z^2=6.
\end{align*}
辺々を加えると$2(x^2+y^2+z^2)=14$，すなわち$x^2+y^2+z^2=7$。よって
\begin{align*}
z^2=7-3=4,\qquad x^2=7-5=2,\qquad y^2=7-6=1
\end{align*}
より（すべて正だから）$x=\sqrt2,\ y=1,\ z=2$。

このとき（それぞれ$AB,BC,DA,AC$の中点として）
\begin{align*}
K=\Bigl(\frac{\sqrt2}2,\frac12,0\Bigr),\quad L=\Bigl(\sqrt2,\frac12,1\Bigr),\quad N=\Bigl(0,\frac12,1\Bigr),\quad P=\Bigl(\frac{\sqrt2}2,0,1\Bigr).
\end{align*}
$P$を基準にすると
\begin{align*}
\overrightarrow{PK}=\Bigl(0,\frac12,-1\Bigr),\quad \overrightarrow{PL}=\Bigl(\frac{\sqrt2}2,\frac12,0\Bigr),\quad\overrightarrow{PN}=\Bigl(-\frac{\sqrt2}2,\frac12,0\Bigr)
\end{align*}
であるから，四面体$PKLN$の体積は
\begin{align*}
V=\frac16\bigl|\det(\overrightarrow{PK},\overrightarrow{PL},\overrightarrow{PN})\bigr|.
\end{align*}
ここで
\begin{align*}
\det(\overrightarrow{PK},\overrightarrow{PL},\overrightarrow{PN})
=\begin{vmatrix}0&\frac12&-1\\[2pt]\frac{\sqrt2}2&\frac12&0\\[2pt]-\frac{\sqrt2}2&\frac12&0\end{vmatrix}
=(-1)\times\Bigl(\frac{\sqrt2}2\cdot\frac12-\frac12\cdot\Bigl(-\frac{\sqrt2}2\Bigr)\Bigr)=(-1)\times\frac{\sqrt2}2=-\frac{\sqrt2}2
\end{align*}
（第3列で展開した）。よって
\begin{align*}
V=\frac16\cdot\frac{\sqrt2}2=\frac{\sqrt2}{12}.
\end{align*}
\end{proof}
