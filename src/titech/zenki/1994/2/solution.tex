{\bf [解]}
$C: xy=-2$，$C=\cos\theta,\ S=\sin\theta$とする．

\textbf{(1)} $Q(X,Y)$とおくと，
\begin{align*}
X+Yi=\left(t-\frac2ti\right)(C+iS)=\left(tC+\frac2tS\right)+i\left(tS-\frac2tC\right)
\end{align*}
だから，$t,S,C$は実数から，実・虚部比較して，
\begin{align*}
Q\left(tC+\frac2tS,\ tS-\frac2tC\right)
\end{align*}

\textbf{(2)} $Q(X,Y)$を時計回りに$\theta$だけ戻すと$P$になるので，$P(x,y)$として
\begin{align*}
x+yi=(X+Yi)(C-iS)=(XC+YS)+i(-XS+YC)
\end{align*}
だから，(1)と同様に
\begin{align*}
x=XC+YS, \qquad y=-XS+YC
\end{align*}
だから$C$に代入して
\begin{align*}
(XC+YS)(-XS+YC)=-2
\end{align*}
よって$Q$は
\begin{align*}
(x\cos\theta+y\sin\theta)(-x\sin\theta+y\cos\theta)=-2
\end{align*}
を通る．

\textbf{(3)} $(x,y)=(\sqrt3+1,\sqrt3-1)$を代入して，
\begin{align*}
\left\{(\sqrt3+1)C+(\sqrt3-1)S\right\}\left\{-(\sqrt3+1)S+(\sqrt3-1)C\right\}=-2
\end{align*}
\begin{align*}
-(\sqrt3+1)(\sqrt3-1)S^2+(\sqrt3+1)(\sqrt3-1)C^2+\left\{(\sqrt3-1)^2-(\sqrt3+1)^2\right\}SC=-2
\end{align*}
\begin{align*}
2(C^2-S^2)-4\sqrt3CS=-2
\end{align*}
\begin{align*}
-\cos2\theta+\sqrt3\sin2\theta=+1
\end{align*}
\begin{align*}

\end{align*}
