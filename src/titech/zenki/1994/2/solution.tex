{\bf [解]}
$C: xy=-2$，$C=\cos\theta,\ S=\sin\theta$とする．

\textbf{(1)} $Q(X,Y)$とおくと，
\begin{align*}
X+Yi=\left(t-\dfrac{2}{t}i\right)(C+iS)=\left(tC+\dfrac{2}{t}S\right)+i\left(tS-\dfrac{2}{t}C\right)
\end{align*}
だから，$t,S,C$は実数から，実・虚部比較して，
\begin{align*}
Q\left(tC+\dfrac{2}{t}S,\ tS-\dfrac{2}{t}C\right)
\end{align*}

\textbf{(2)} $Q(X,Y)$を時計回りに$\theta$だけ戻すと$P$になるので，$P(x,y)$として
\begin{align*}
x+yi=(X+Yi)(C-iS)=(XC+YS)+i(-XS+YC)
\end{align*}
だから，(1)と同様に
\begin{align*}
x=XC+YS, \qquad y=-XS+YC
\end{align*}
だから$C$に代入して
\begin{align*}
(XC+YS)(-XS+YC)=-2
\end{align*}
よって$Q$は
\begin{align*}
(x\cos\theta+y\sin\theta)(-x\sin\theta+y\cos\theta)=-2
\end{align*}
を通る．

\textbf{(3)} $(x,y)=(\sqrt{3}+1,\sqrt{3}-1)$を代入して，
\begin{align*}
\left\{(\sqrt{3}+1)C+(\sqrt{3}-1)S\right\}\left\{-(\sqrt{3}+1)S+(\sqrt{3}-1)C\right\}=-2
\end{align*}
\begin{align*}
-(\sqrt{3}+1)(\sqrt{3}-1)S^2+(\sqrt{3}+1)(\sqrt{3}-1)C^2+\left\{(\sqrt{3}-1)^2-(\sqrt{3}+1)^2\right\}SC=-2
\end{align*}
\begin{align*}
2(C^2-S^2)-4\sqrt{3}CS=-2
\end{align*}
\begin{align*}
-\cos2\theta+\sqrt{3}\sin2\theta=+1
\end{align*}
\begin{align*}
2\sin\left(2\theta-\dfrac{\pi}{6}\right)=1 \quad\cdots\text{①}
\end{align*}
$0<\theta<2\pi$から$-\dfrac{\pi}{6}<2\theta-\dfrac{\pi}{6}<\dfrac{23}{6}\pi$だから，①をみたすのは，
\begin{align*}

\end{align*}
\begin{align*}
