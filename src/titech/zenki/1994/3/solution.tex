{\bf [解]}
\paragraph{(1)}

\begin{align}
\begin{split}
\int_0^\pi e^{-x}\sin x\,dx&=\dfrac{1}{1+1}\left[-e^{-x}\sin x-e^{-x}\cos x\right]_0^\pi \\
&=\dfrac{1}{2}\left(1+e^{-\pi}\right)
\end{split}
\end{align}

\paragraph{(2)}
与えられた定積分を
\begin{align}
 A_n=\displaystyle\int_0^{n\pi}e^{-x}|\sin x|dx \label{eq:1}
\end{align}
とおく．さらに$a_k=\displaystyle\int_{(k-1)\pi}^{k\pi}e^{-x}|\sin x|dx$とおくと，$t=x-(k-1)\pi$と置換して
\begin{align}
\begin{split}
a_k
 &=\int_0^\pi e^{-t-(k-1)\pi}|\sin t|\,dt \\
 &=e^{-(k-1)\pi}\int_0^\pi e^{-t}\sin t\,dt \quad(\because0\le t\le\pi\text{で}\sin t\ge0) \\
 &=e^{-(k-1)\pi}a_1
\end{split}
\end{align}
だから，\cref{eq:1}に代入して
\begin{align}
A_n=\sum_{k=1}^na_k=a_1\dfrac{1-e^{-n\pi}}{1-e^{-\pi}}\xrightarrow{n\to\infty}\dfrac{a_1}{1-e^{-\pi}}=\dfrac{1}{2}\cdot\dfrac{1+e^{-\pi}}{1-e^{-\pi}} \quad(\because(1))
\end{align}
が求める極限値である．
