\section*{第1問}

\begin{proof}[解]
$C: y=x^2$，$P(\alpha,\alpha^2)$，$Q(\beta,\beta^2)$（$\alpha<\beta$）とおく。題意の面積に関する規定から
\[
\frac16=\frac16(\beta-\alpha)^3 \quad\therefore\ \beta-\alpha=1 \quad\cdots\text{①}
\]
である。又，$P,Q$における$C$の接線は
\[
\ell_P: y=2\alpha x-\alpha^2
\]
\[
\ell_Q: y=2\beta x-\beta^2
\]
だから，$R\left(\dfrac{\alpha+\beta}{2},\alpha\beta\right)$である。$X=\dfrac{\alpha+\beta}{2},\ Y=\alpha\beta$とおく。①から$\beta$を消して，$X,Y$に代入
\[
\begin{cases}
\alpha<\alpha+1 \\
X=\dfrac{2\alpha+1}{2} \\
Y=\alpha(\alpha+1)
\end{cases}
\]
$\alpha=X-\dfrac12$だから，$Y=\left(X-\dfrac12\right)\left(X+\dfrac12\right)=X^2-\dfrac14$となり，もとめるのは
\[
y=x^2-\frac14
\]
\end{proof}

\section*{第2問}

\begin{proof}[解]
$C: xy=-2$，$C=\cos\theta,\ S=\sin\theta$とする。

\textbf{(1)} $Q(X,Y)$とおくと，
\[
X+Yi=\left(t-\frac2ti\right)(C+iS)=\left(tC+\frac2tS\right)+i\left(tS-\frac2tC\right)
\]
だから，$t,S,C$は実数から，実・虚部比較して，
\[
Q\left(tC+\frac2tS,\ tS-\frac2tC\right)
\]

\textbf{(2)} $Q(X,Y)$を時計回りに$\theta$だけ戻すと$P$になるので，$P(x,y)$として
\[
x+yi=(X+Yi)(C-iS)=(XC+YS)+i(-XS+YC)
\]
だから，(1)と同様に
\[
x=XC+YS, \qquad y=-XS+YC
\]
だから$C$に代入して
\[
(XC+YS)(-XS+YC)=-2
\]
よって$Q$は
\[
(x\cos\theta+y\sin\theta)(-x\sin\theta+y\cos\theta)=-2
\]
を通る。

\textbf{(3)} $(x,y)=(\sqrt3+1,\sqrt3-1)$を代入して，
\[
\left\{(\sqrt3+1)C+(\sqrt3-1)S\right\}\left\{-(\sqrt3+1)S+(\sqrt3-1)C\right\}=-2
\]
\[
-(\sqrt3+1)(\sqrt3-1)S^2+(\sqrt3+1)(\sqrt3-1)C^2+\left\{(\sqrt3-1)^2-(\sqrt3+1)^2\right\}SC=-2
\]
\[
2(C^2-S^2)-4\sqrt3CS=-2
\]
\[
-\cos2\theta+\sqrt3\sin2\theta=+1
\]
\[
2\sin\left(2\theta-\frac{\pi}{6}\right)=1 \quad\cdots\text{①}
\]
$0<\theta<2\pi$から$-\dfrac{\pi}{6}<2\theta-\dfrac{\pi}{6}<\dfrac{23}{6}\pi$だから，①をみたすのは，
\[
2\theta-\frac{\pi}{6}=\frac{\pi}{6},\frac{5}{6}\pi,\frac{13}{6}\pi,\frac{17}{6}\pi
\]
\[
\theta=\frac16\pi,\frac36\pi,\frac76\pi,\frac96\pi
\]
\[
=\frac16\pi,\frac12\pi,\frac76\pi,\frac32\pi
\]
\end{proof}

\section*{第3問}

\begin{proof}[解]
\textbf{(1)}
\[
\int_0^\pi e^{-x}\sin x\,dx=\frac{1}{1+1}\left[-e^{-x}\sin x-e^{-x}\cos x\right]_0^\pi
\]
\[
=\frac12\left(1+e^{-\pi}\right)
\]

\textbf{(2)} $A_n=\displaystyle\int_0^{n\pi}e^{-x}|\sin x|dx$とおく。$a_k=\displaystyle\int_{(k-1)\pi}^{k\pi}e^{-x}|\sin x|dx$とおくと，$t=x-(k-1)\pi$として，
\[
a_k=\int_0^\pi e^{-t-(k-1)\pi}|\sin t|\,dt
\]
\[
=e^{-(k-1)\pi}\int_0^\pi e^{-t}\sin t\,dt \quad(\because0\le t\le\pi\text{で}\sin t\ge0)
\]
\[
=e^{-(k-1)\pi}a_1
\]
だから，
\[
A_n=\sum_{k=1}^na_k=a_1\frac{1-e^{-n\pi}}{1-e^{-\pi}}\xrightarrow{n\to\infty}\frac{a_1}{1-e^{-\pi}}=\frac12\cdot\frac{1+e^{-\pi}}{1-e^{-\pi}} \quad(\because(1))
\]
\end{proof}

\section*{第4問}

\begin{proof}[解]
\[
f(m,n)=\frac12(m+n)(m+n+1)+n \quad(m,n\in\mathbb{Z}_{\ge0})
\]
である。

\textbf{(1)} $(m,n)=(0,0)$は題意をみたす。以下他の場合をかんがえる。
\[
\frac12(m+n)(m+n+1)+n\le5
\]
\[
(m+n)(m+n+1)+2n\le10 \quad\cdots\text{①}
\]
$t=m+n$とすると，$n\ge0$から$t(t+1)\le10$となり，$1\le t$，$t\in\mathbb{N}$とあわせて，$t=1,2$となる。

\textbf{$1^\circ$ $t=1$}

$(m,n)=(0,1),(1,0)$で共に①をみたし十分。

\textbf{$2^\circ$ $t=2$}

$(m,n)=(0,2),(1,1),(2,0)$で共に①をみたし十分。

以上から，もとめる$(m,n)$は
\[
(m,n)=(0,0),(0,1),(0,2),(1,0),(1,1),(2,0)
\]
で図示して右図黒丸。

\begin{center}
\begin{tikzpicture}[scale=0.9, >=stealth]
  \draw[->] (-0.3,0) -- (3,0) node[right] {$m$};
  \draw[->] (0,-0.3) -- (0,3) node[above] {$n$};
  \foreach \p in {(0,0),(0,1),(0,2),(1,0),(1,1),(2,0)}{
    \fill \p circle (2pt);
  }
  \node[below] at (1,0) {$1$};
  \node[below] at (2,0) {$2$};
  \node[left] at (0,1) {$1$};
  \node[left] at (0,2) {$2$};
\end{tikzpicture}
\end{center}

\textbf{(2)} $f(m,n)=f(m',n')$とする。$(m,n)=(0,0)$の時は明らかで，他の場合をかんがえる。
\[
\frac12(m+n)(m+n+1)+n=\frac12(m'+n')(m'+n'+1)+n'
\]
\[
t(t+1)+2n=t'(t'+1)+2n' \quad(t'=m'+n') \quad\cdots\text{②}
\]
ここで，$f(x,y)=x^2+x+2y\ (0\le y\le x)$とおくと，これは$y$の単調増加関数で，
\[
x^2+x\le f(x,y)\le x^2+3x \quad\cdots\text{③}
\]
となる。この両辺は$x$については単調増加で，$x=t$として，
\[
t^2+t\le f(t,n)\le t^2+3t
\]
であり，$\{(t+1)^2+(t+1)\}-(t^2+3t)=2>0$から，異なる2自然数$t,t'$に対し，$f(t,n)=f(t',n')$となることはない。すなわち②が成立するには$t=t'$が必要で，この時，$n=n'$とすれば十分，この時$m=m'$も成立する。以上から，$f(m,n)=f(m',n')$ならば，$(m,n)=(m',n')$となる。
\end{proof}

\section*{第5問}

\begin{proof}[解]
$f(x)=x^2$，$g(x)=-x^2-16x-65=-(x+8)^2-1$とおく。$P,Q$の$x$座標を$\alpha$，$\beta-8$とする。
\[
\overrightarrow{PQ}=\begin{pmatrix}\beta-8\\g(\beta-8)\end{pmatrix}-\begin{pmatrix}\alpha\\\alpha^2\end{pmatrix}=\begin{pmatrix}\beta-\alpha-8\\-\alpha^2-\beta^2-1\end{pmatrix}
\]
だから，$a=\beta-\alpha,\ b=\beta+\alpha$とすると
\[
\alpha^2+\beta^2=\frac{a^2+b^2}{2}
\]
に注意して
\[
|\overrightarrow{PQ}|^2=(a-8)^2+\left(\frac{a^2+b^2}{2}+1\right)^2
\]
\[
=\frac14\left\{b^2+(a^2+2)\right\}^2+(a-8)^2 \quad\cdots\text{①}
\]
$\alpha,\beta$は$x$の2次式$x^2-bx+\dfrac{b^2-a^2}{4}=0$の2実解だから，判別式$D$として，
\[
D=b^2-(b^2-a^2)\ge0 \quad\therefore\ a^2\ge0
\]
これは$a\in\mathbb{R}$から常に成立。よって①の$\min$をかんがえれば良い。$a$を固定すると$b=0$で$\min$で，
\[
\min|\overrightarrow{PQ}|^2=\frac14(a^2+2)^2+(a-8)^2\ (\equiv f(a))
\]
とすると
\[
f'(a)=\frac12(a^2+2)\cdot2a+2(a-8)=a^3+4a-16=(a-2)(a^2+2a+8)
\]
から下表をうる。
\begin{center}
\begin{tabular}{c|ccccc}
$a$ & & & $2$ & & \\
\hline
$f'$ & & $-$ & $0$ & $+$ & \\
\hline
$f$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
よって$a=2$で$|\overrightarrow{PQ}|^2$は$\min$をとる。$|\overrightarrow{PQ}|\ge0$から，この時$|\overrightarrow{PQ}|$も最小で，
\[
\min|\overrightarrow{PQ}|=\sqrt{45}=3\sqrt5
\]
\end{proof}
