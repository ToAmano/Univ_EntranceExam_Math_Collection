$f(x)=x^4-2ax^2$とおくと一階微分は$f'(x)=4x^3-4ax$だから$x=t$での接線は
\begin{align}
 \ell_t: y=(4t^3-4at)x-3t^4+2at^2 \label{eq:1}
\end{align}
となる．$C$と$\ell_t$の交点の$x$座標は$f(x)$と$\ell_t$を連立した四次方程式
\begin{align}
\begin{split}
x^4-2ax^2&=(4t^3-4at)x-3t^4+2at^2 \\
x^4-2ax^2-4t(t^2-a)x+3t^4-2at^2&=0 \\
(x-t)^2\{x^2+2tx+(3t^2-2a)\}&=0
\end{split}
\end{align}
の解である．このうち$x=t$は点$P$に対応し，
\begin{align}
 x^2+2tx+(3t^2-2a)=0 \label{eq:2}
\end{align}
の解が$Q,R$に対応する．以下この左辺を$g(x)$とおく．
\cref{eq:2}の判別式$D$として，$-\sqrt{a}\le t\le\sqrt{a}$から
\begin{align}
D/4=t^2-(3t^2-2a)=2a-t^2\ge0
\end{align}
となり，\cref{eq:2}は2実解（重解含む）を持つ．したがって\cref{eq:2}の2解が$\alpha,\beta$で，
解と係数の関係から
\begin{align}
\alpha+\beta=-2t, \qquad \alpha\beta=3t^2-2a \label{eq:3}
\end{align}
である．

\paragraph{(2)}
$\alpha\le\beta$だから，題意の条件は（重なる場合をのぞいて）
\begin{align}
\alpha<t<\beta \quad\cdots\text{②} \label{eq:4}
\end{align}
である．すなわち\cref{eq:2}の解が$x<t$に1つ，$t<x$に1つある時である．
この条件は
\begin{align}
  g(t)<0 \iff 6t^2-2a<0 \iff -\sqrt{\dfrac{a}{3}}<t<\sqrt{\dfrac{a}{3}} \label{eq:5}
\end{align}
である．

したがってピタゴラスの定理から
\begin{align}
\begin{split}
L^2
  &=(\beta-\alpha)^2+(\beta-\alpha)^2(4t^3-4at)^2 \\
  &=\{1+16t^2(t^2-a)^2\}\{(\alpha+\beta)^2-4\alpha\beta\} \\
  &=\{1+16t^2(t^2-a)^2\}\{(-2t)^2-4(3t^2-2a)\} \quad(\because(1)) \\
  &=\{1+16t^2(t^2-a)^2\}(8a-8t^2)
\end{split} \label{eq:6}
\end{align}

\paragraph{(4)}
(3)より$L^2$は$p=t^2$のみに依存するのでこれを$h(p)$とおく．
$-\sqrt{a}\le t\le\sqrt{a}$から
\begin{align}
 0\le p\le a \label{eq:7}
\end{align}
である．\cref{eq:6}より
\begin{align}
 h(p)=-8(p-a)\{1+16p(p-a)^2\}
\end{align}
だから，
\begin{align}
\begin{split}
-\dfrac{1}{8}h'(p)
&=1+16p(p-a)^2+(p-a)\cdot16(3p^2-4ap+a^2) \\
&=16(p-a)\{p(p-a)+3p^2-4ap+a^2\}+1 \\
&=16(p-a)^2(4p-a)+1 \\
&=64p^3-9\cdot 16 ap^2 + 6\cdot 16a^2p -16a^3 + 1 
\end{split}
\end{align}
と展開できる．$a=\dfrac{7}{12}$を代入してさらに整理して
\begin{align}
 -\dfrac{1}{8}h'(p) 
  &=2^6p^3 - 2^2\cdot 3\cdot 7 p^2 + \dfrac{2\cdot 7^2}{3}p - \dfrac{7^3}{2^{2}3^3} + 1
\end{align}
両辺に$2^{2}3^{3}$をかけると
\begin{align}
 -\dfrac{27}{2}h'(p) 
  &=2^{8}3^{3}p^3 - 2^{4}\cdot 3^{4}\cdot 7 p^2 + 2^{3}7^{2}3^{2}p - 7^{3} + 2^{2}3^{3} 
\end{align}
簡単のため
\begin{align}
 q = 12 p
\end{align}
とおいて代入すると
\begin{align}
 -\dfrac{27}{2}h'(p) 
  &=4q^3 - 63q^2 + 2\cdot 3\cdot 7q - 3^{3}\cdot 5 \\
  &=\left(q-1\right)\left(4q^2-59q+135\right) \\
  &=4\left(q-1\right)\left[\left(q-\dfrac{59}{8}\right)^2+\dfrac{2101}{16}\right] 
\end{align}
だから，$h(p)$の増減表は以下のようになる．

\begin{table}[htb]
  \centering
\caption{$h(p)$の増減表}
\begin{tabular}{c|ccccc}
$q$ & $0$ & & $1$ & & $7$ \\
\hline
$p$ & $0$ & & $1/12$ & & $7/12(=a)$ \\
\hline
$h'$ & & $+$ & $0$ & $-$ & \\
\hline
$h$ & & $\nearrow$ & & $\searrow$ &
\end{tabular}
\end{table}

したがって，$h(p)$は$p=\dfrac{1}{12}$で最大値をとる．その最大値は
\begin{align}
\begin{split}
\max h(p) 
  &= \max L^2 \\
  &= h\left(\dfrac{1}{12}\right) \\
  &=-8\left(\dfrac{1}{12}-\dfrac{7}{12}\right)\left\{1+16\cdot\dfrac{1}{12}\left(\dfrac{1}{12}-\dfrac{7}{12}\right)^2\right\} \\
  &=4\left(1+\dfrac{4}{3}\cdot\dfrac{1}{4}\right)=\dfrac{16}{3}
\end{split}
\end{align}
である．従って$L\ge 0$より
\begin{align}
 \max L=\sqrt{\dfrac{16}{3}}=\dfrac{4\sqrt{3}}{3}
\end{align}
が求める$L$の最大値である．
