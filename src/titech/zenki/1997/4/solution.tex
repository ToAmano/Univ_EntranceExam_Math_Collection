\textbf{(1)} 右のように各点をおく．正弦定理から
\begin{align}
\begin{split}
\dfrac{AB}{\sin\beta}&=\dfrac{AC}{\sin\alpha}=\dfrac{\ell}{\sin(\alpha+\beta)} \quad\cdots\text{①} \\
S&=\dfrac{1}{2}AB\cdot AC\cdot\sin(\alpha+\beta) \\
&=\dfrac{1}{2}\cdot\dfrac{\ell\sin\beta}{\sin(\alpha+\beta)}\cdot\dfrac{\ell\sin\alpha}{\sin(\alpha+\beta)}\cdot\sin(\alpha+\beta) \quad(\because①) \\
&=\dfrac{\ell^2}{2}\cdot\dfrac{\sin\alpha\sin\beta}{\sin(\alpha+\beta)} \\
&=\dfrac{\ell^2}{4}\cdot\dfrac{\cos(\alpha-\beta)-\cos(\alpha+\beta)}{\sin(\alpha+\beta)}
\end{split}
\end{align}

\begin{align}
\begin{split}

\end{split}
\end{align}
$BD+CD=1$から
\begin{align}
\left(\dfrac{2}{3}\sqrt{3}\sin\theta+\cos\theta\right)x=1 \quad\cdots\text{①}
\end{align}
