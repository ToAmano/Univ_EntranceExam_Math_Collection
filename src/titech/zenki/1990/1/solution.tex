\begin{flushright}
\footnotesize\textit{【自動文字起こし・要確認】}
\end{flushright}

{\bf [解]}
以下
\begin{align}
 x,y,z,w>0 \label{eq:1}
\end{align}
で考える．小文字$k$に対してその$\dfrac{1}{m}$乗を大文字$K$とおく．すなわち
\begin{align}
 K = k^{\frac{1}{m}} \label{eq:2}
\end{align}
とする．題意の方程式をこの大文字を使って書き直すと
\begin{align}
(X+Y)^n+(Z+W)^n=\left\{(X^n+Z^n)^{\frac{1}{n}}+(Y^n+W^n)^{\frac{1}{n}}\right\}^n \quad\cdots\text{②} \label{eq:3}
\end{align}
である．これが任意の自然数の組$(m,n)$で成立するので$n=2$での成立が必要である．実際に\cref{eq:3}で$n=2$とすると
\begin{align}
2XY+2ZW=2\sqrt{(X^2+Z^2)(Y^2+W^2)} \quad\cdots\text{③} \label{eq:4}
\end{align}
\cref{eq:1}から大文字$X,Y,Z,W$も正であって，\cref{eq:4}の両辺正で2乗しても良く，
\begin{align}
(XY+ZW)^2&=(X^2+Z^2)(Y^2+W^2) \\
2XYZW&=X^2W^2+Y^2Z^2 \\
(XW-YZ)^2&=0 \\
\therefore
XW&=YZ \quad\cdots\text{④} \label{eq:5}
\end{align}
が必要である．


逆に\cref{eq:5}が成立つ時，正の実数$k$を用いて
\begin{align}
 \left(\dfrac{Z}{W}\right)=k\left(\dfrac{X}{Y}\right)
\end{align}
とおける．これを\cref{eq:3}に代入して成立するかどうか確かめると
\begin{align}
\begin{split}
(\text{②の左辺})&=(X+Y)^n+k^n(X+Y)^n=(1+k^n)(X+Y)^n \\
(\text{②の右辺})&=\left[(1+k^n)^{\frac{1}{n}}X^{\frac{1}{n}\cdot n}+(1+k^n)^{\frac{1}{n}}Y^{\frac{1}{n}\cdot n}\right]^n=(1+k^n)(X+Y)^n
\end{split}
\end{align}
となり，$k$の値にかかわらず\cref{eq:3}は成立していて十分である．

以上から\cref{eq:3}が成立する条件は
\begin{align}
XW=YZ \\
\left(xw\right)^{\frac{1}{m}} =  \left(yz\right)^{\frac{1}{m}}
\end{align}
である．両辺$m$乗して求める条件は
\begin{align}
xw=yz
\end{align}
である．
