\begin{flushright}
\footnotesize\textit{【自動文字起こし・要確認】}
\end{flushright}

{\bf [解]}
$x,y,z,w>0\ \cdots$①．文字$k$に対し，$k^{\frac{1}{m}}=K$とおく．
\begin{align}
(X+Y)^n+(Z+W)^n=\left\{(X^n+Z^n)^{\frac{1}{n}}+(Y^n+W^n)^{\frac{1}{n}}\right\}^n \quad\cdots\text{②}
\end{align}
②がsuffix$(m,n)$で成立するので（$m,n\in\mathbb{N}$），$n=2$での成立が必要．
\begin{align}
2XY+2ZW=2\sqrt{(X^2+Z^2)(Y^2+W^2)} \quad\cdots\text{③}
\end{align}
①から，③の両辺正で2乗しても良く，
\begin{align}
\begin{split}
(XY+ZW)^2&=(X^2+Z^2)(Y^2+W^2) \\
2XYZW&=X^2W^2+Y^2Z^2 \\
(XW-YZ)^2&=0 \\
XW&=YZ \quad\cdots\text{④}
\end{split}
\end{align}
が必要．逆に④が成立つ時，$\left(\dfrac{Z}{W}\right)=k\left(\dfrac{X}{Y}\right)$（$k\in\mathbb{R}_{>0}$）とおける．
\begin{align}
\begin{split}
(\text{②の左辺})&=(X+Y)^n+k^n(X+Y)^n=(1+k^n)(X+Y)^n \\
(\text{②の右辺})&=\left[(1+k^n)^{\frac{1}{n}}X^{\frac{1}{n}\cdot n}+(1+k^n)^{\frac{1}{n}}Y^{\frac{1}{n}\cdot n}\right]^n=(1+k^n)(X+Y)^n
\end{split}
\end{align}
とかり，たしかに②は成立して十分．

以上から
\begin{align}
XW=YZ
\end{align}
つまり
\begin{align}
xw=yz
\end{align}
