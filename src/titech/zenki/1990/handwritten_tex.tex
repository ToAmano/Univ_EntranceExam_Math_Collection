\section*{第1問}

\begin{proof}[解]
$x,y,z,w>0\ \cdots$①。文字$k$に対し，$k^{\frac1m}=K$とおく。
\[
(X+Y)^n+(Z+W)^n=\left\{(X^n+Z^n)^{\frac1n}+(Y^n+W^n)^{\frac1n}\right\}^n \quad\cdots\text{②}
\]
②がsuffix$(m,n)$で成立するので（$m,n\in\mathbb{N}$），$n=2$での成立が必要。
\[
2XY+2ZW=2\sqrt{(X^2+Z^2)(Y^2+W^2)} \quad\cdots\text{③}
\]
①から，③の両辺正で2乗しても良く，
\[
(XY+ZW)^2=(X^2+Z^2)(Y^2+W^2)
\]
\[
2XYZW=X^2W^2+Y^2Z^2
\]
\[
(XW-YZ)^2=0
\]
\[
XW=YZ \quad\cdots\text{④}
\]
が必要。逆に④が成立つ時，$\left(\dfrac{Z}{W}\right)=k\left(\dfrac{X}{Y}\right)$（$k\in\mathbb{R}_{>0}$）とおける。
\[
(\text{②の左辺})=(X+Y)^n+k^n(X+Y)^n=(1+k^n)(X+Y)^n
\]
\[
(\text{②の右辺})=\left[(1+k^n)^{\frac1n}X^{\frac1n\cdot n}+(1+k^n)^{\frac1n}Y^{\frac1n\cdot n}\right]^n=(1+k^n)(X+Y)^n
\]
とかり，たしかに②は成立して十分。

以上から
\[
XW=YZ
\]
つまり
\[
xw=yz
\]
\end{proof}

\section*{第2問}

\begin{proof}[解]
\[
x_i>0, \qquad \sum_{i=1}^nx_i=k \quad\cdots\text{①}
\]
$A_n=\displaystyle\sum_{i=1}^nx_i\log x_i$，$B_n=k\log\dfrac{k}{n}$とおく。「$A_n\ge B_n\ \cdots$②が任意の$n\in\mathbb{N}$で成立すること」…③を帰納法で示す。$n=1$の時の成立は明らかなので，$n=p$での③の成立を仮定し，$n=p+1$でも成立することを示す。

$p+1$個の正数$x_i\ (i=1,2,\cdots,p+1)$に対し，$\displaystyle\sum_{i=1}^{p+1}x_i=k$とし，$\displaystyle\sum_{i=1}^px_i=\ell\ (0<\ell<k)$とおく。まず$\ell$を固定する。
\[
A_{p+1}=A_p+x_{p+1}\log x_{p+1}
\]
\[
\ge\ell\log\frac{\ell}{p}+(k-\ell)\log(k-\ell)\equiv f(\ell) \quad(\because\text{仮定}) \quad\cdots\text{②}
\]
次に$\ell$をうごかす。
\[
f'(\ell)=1+\log\ell-\log p-\log(k-\ell)-1=\log\frac{\ell}{k-\ell}-\log p
\]
$\log x$が$x$の単調増加関数であるから下表をうる（$\ell,k,p>0,\ k-\ell>0$）。
\begin{center}
\begin{tabular}{c|ccccc}
$\ell$ & $0$ & & $\dfrac{pk}{p+1}$ & & $k$ \\
\hline
$f'$ & & $-$ & $0$ & $+$ & \\
\hline
$f$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
よって
\[
f(\ell)\ge\frac{pk}{p+1}\log\frac{k}{p+1}+\frac{k}{p+1}\log\frac{k}{p+1}
\]
\[
=k\log\frac{k}{p+1}=B_{p+1} \quad\cdots\text{③}
\]
②，③から
\[
A_{p+1}\ge B_{p+1}
\]
となり，$n=p+1$でも②は成立。以上から③は示された。$\blacksquare$
\end{proof}

\section*{第3問}

\begin{proof}[解]
$C: \dfrac{x^2}{4}+\dfrac{y^2}{3}=1$，$C_1: (x-1)^2+y^2=1$，$C_2: (x+1)^2+y^2=1$とおく。対称性から，$P$が$x\ge0$にあるとして良い。$P$をこの範囲内で$P(X,Y)$と固定し，$R,Q$をうごかす。

まず，$PQ$について考える。以下$S=\sin\theta,\ C=\cos\theta$と書くことにすると，$Q(1+C,S)\ (\pi\le\theta\le2\pi)$とおけて，
\[
PQ^2=(1+C-X)^2+(S-Y)^2
\]
\[
=(1+C)^2+S^2+X^2+Y^2-2X(1+C)-2YS
\]
\[
=X^2+Y^2-2(X-1)(C+1)-2YS
\]
ここで，$(X-1)(C+1)+YS=\begin{pmatrix}X-1\\Y\end{pmatrix}\cdot\begin{pmatrix}C+1\\S\end{pmatrix}$から，$PQ^2$は，$PQ$が$(1,0)$を通る時に最大で，$\overline{PQ}\ge0$から，この時$\overline{PQ}$も最大。

同様のことが$R$についても言える。そこで，$F_1(1,0)$，$F_2(-1,0)$とおくと，
\[
\max(\overline{PQ}+\overline{PR})=\overline{PF_1}+\overline{PF_2}+2
\]
\[
=6
\]
（$C$は長軸$2a=4$の楕円で$\overline{PF_1}+\overline{PF_2}=4$（$F_1,F_2$は$C$の焦点，$\because a^2=4,b^2=3,c^2=a^2-b^2=1$）だから）
\end{proof}

\section*{第4問}

\begin{proof}[解]
$0<x<\dfrac{\pi}{2}\ \cdots$①
\[
f(x)=\int_0^x\frac{d\theta}{\cos\theta}+\int_x^{\frac{\pi}{2}}\frac{d\theta}{\sin\theta}
\]
\[
f'(x)=\frac{1}{\cos x}-\frac{1}{\sin x}=\frac{S-C}{CS} \quad(S=\sin x,\ C=\cos x\text{から，下表を得る})
\]
\begin{center}
\begin{tabular}{c|ccccc}
$x$ & $0$ & & $\dfrac{\pi}{4}$ & & $\dfrac{\pi}{2}$ \\
\hline
$f'$ & & $-$ & $0$ & $+$ & \\
\hline
$f$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
したがって，$f(x)$は$x=\dfrac{\pi}{4}$で最小。
\[
\int_{\frac{\pi}{4}}^{\frac{\pi}{2}}\frac{d\theta}{\sin\theta}=\int_{\frac{\pi}{4}}^0\frac{-dt}{\sin\left(\frac{\pi}{2}-t\right)} \quad\left(t=\frac{\pi}{2}-\theta\right)
\]
\[
=\int_0^{\frac{\pi}{4}}\frac{d\theta}{\cos\theta}
\]
からこの時
\[
f\left(\frac{\pi}{4}\right)=2\int_0^{\frac{\pi}{4}}\frac{d\theta}{\cos\theta}
\]
\[
=2\int_0^{\frac{\sqrt2}{2}}\frac{1}{1-x^2}dx
\]
\[
=\left[\log(x+1)-\log(1-x)\right]_0^{\frac{\sqrt2}{2}}
\]
\[
=\log\frac{1+\frac{\sqrt2}{2}}{1-\frac{\sqrt2}{2}}=\log\frac{2+\sqrt2}{2-\sqrt2}
\]
\end{proof}

\section*{第5問}

\begin{proof}[解]
$P$から引いた接線が座標軸に平行になる時，(i)から$P$の$x$座標が$\pm\dfrac{1}{\sqrt3}$である。このうち(i)を満たすのは$P$の$y$座標が$|y|\le1$をみたす時。以下，他の場合を考える。

この時$P$から引いた接線$\ell$として
\[
\ell: y=m(x-a)+b
\]
とおける。$m\in\mathbb{R}$，$P(a,b)$とした。$\cdots\bigstar$

ここで，$X=\sqrt3x,\ Y=y$なる変換を考える。この変換によって図形$A$が$A'$にうつると表す。楕円は$X^2+Y^2=1$に，$\ell$は$\ell'$にうつり，$\ell'$と円$X^2+Y^2=1$は接している。
\[
\ell': Y=m\left(\frac{1}{\sqrt3}X-a\right)+b
\]
で接する条件から
\[
\frac{|-am+b|}{\sqrt{\frac13m^2+1}}=1
\]
両辺0以上から2乗しても良く，
\[
\left(a^2-\frac13\right)m^2-2abm+b^2-1=0 \quad\cdots\text{①}
\]
$m$の2次方程式①の2実解$m_1,m_2\ (m_1\le m_2)$とする。
\[
\text{①の判別式}D\text{として，}\quad\frac{D}{4}=a^2b^2-\left(a^2-\frac13\right)(b^2-1)=a^2+\frac13b^2-\frac13\ge0
\]
又，$a\ne\dfrac{1}{\sqrt3}$から①は2次式。

まず，条件(i)を考える。(i)がみたされる時，$A\left(\dfrac{1}{\sqrt3},0\right)$として，線分$AP$上の点$B(t(a-\tfrac{1}{\sqrt3})+\tfrac{1}{\sqrt3},\ tb)\ (0\le t\le1)$と楕円の共有点が$A$のみであるから，
\[
3\left\{t\left(a-\frac{1}{\sqrt3}\right)+\frac{1}{\sqrt3}\right\}^2+t^2b^2=1
\]
\[
\left\{3\left(a-\frac{1}{\sqrt3}\right)^2+b^2\right\}t^2+2\sqrt3\left(a-\frac{1}{\sqrt3}\right)t=0
\]
の$t\ne0$の解が負ならば良く，
\[
a-\frac{1}{\sqrt3}\ge0 \iff a\ge\frac{1}{\sqrt3} \quad\cdots\text{②}
\]

次に，条件(ii)をかんがえる。$\angle QPR\ge\angle R$の時，$\overrightarrow{PQ}\cdot\overrightarrow{PR}\le0$である。
\[
\begin{pmatrix}-1\\m_1\end{pmatrix}\cdot\begin{pmatrix}-1\\m_2\end{pmatrix}\le0 \quad\therefore\ 1+m_1m_2\le0
\]
①の$m_1m_2=(b^2-1)/(a^2-\tfrac13)$だから，代入して
\[
1\le\frac{1-b^2}{a^2-\frac13}
\]
②から，$a^2-\dfrac13\ge0$だから
\[
a^2-\frac13\le1-b^2 \quad\therefore\ a^2+b^2\le\frac43 \quad\cdots\text{③}
\]
②③と前半の議論により図示するのは下図斜線部（境界含む）で，面積$S$として
\[
S=\frac12\cdot\frac{2\pi}{3}\cdot\frac43-\frac{\sqrt3}{3}
\]
\[
=\frac13\left(\frac{4}{3}\pi-\sqrt3\right)
\]

  \node[above right] at (0.9,0.9) {$x^2+y^2=\frac43$};
\end{tikzpicture}
\end{center}
\end{proof}
