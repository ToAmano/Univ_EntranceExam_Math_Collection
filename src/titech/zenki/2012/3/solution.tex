{\bf [解]}
与えられた関数を
\begin{align}
 f(x)=x^3-3x^2+2x \label{eq:1}
\end{align}  
とおく．

\paragraph{(1)}

方程式$f(x)=ax$の$x\neq 0$の解を求めれば良い．\cref{eq:1}を代入して
\begin{align}
 x^3-3x^2+2x = ax 
\end{align}
$x\neq 0$より両辺$x$で割って
\begin{align}
 x^2-3x+2=a \label{eq:2}
\end{align}
である．これが$x \neq 0$に実数解を持てば良い．
左辺を平方完成すると
\begin{align}
 \text{左辺} = \left(x-\dfrac{3}{2}\right)^2-\dfrac{1}{4} \ge \dfrac{-1}{4}
\end{align}
だから，このような条件は
\begin{align}
a\ge-\dfrac{1}{4} \label{eq:3}
\end{align}
である．

\cref{eq:3}の範囲で考えよう．\cref{eq:2}の重解を含む二つの実数解を$\alpha,\beta$とする．ただし$\alpha\le\beta$とする．
$a=2$の時\cref{eq:2}より$\alpha=0,\beta=3$となり$C$と$\ell$は$x=0$で接し，$a$がそれ以上になれば図より$S(a)$は単調増加するから，
最小値は
\begin{align}
 -\dfrac{1}{4} \le a \le 2 \label{eq:4}
\end{align}
でとる．従って以下この範囲で考える．この時解と係数の関係から
\begin{align}
 \begin{cases}
  \alpha+\beta=3 \\
  \alpha\beta=2-a
 \end{cases} \label{eq:5}
\end{align}
だから，$\alpha,\beta$は常に$0$以上である．

$C$と$\ell$の差を
\begin{align}
 g(x) = f(x) - ax \label{eq:5}
\end{align}
とおくと，$g(x)=0$の解が$0,\alpha,\beta$だから因数定理より
\begin{align}
 g(x) = x(x-\alpha)(x-\beta) \label{eq:6}
\end{align}
と書ける．これは$0<x<\alpha$で正，$\alpha<x<\beta$で負だから，$C$と$l$で囲まれる面積は
\begin{align}
  S(a)
   &=\int_0^\alpha g(x)dx-\int_\alpha^\beta g(x)dx \\
   &=\int_0^\alpha f(x)dx-\int_0^\alpha ax\,dx-\int_\alpha^\beta f(x)dx+\int_\alpha^\beta ax\,dx \label{eq:7}
\end{align}
と書ける．
$\alpha,\beta$は$a$の関数であることに注意して両辺を$a$で微分する．
まず\cref{eq:5,eq:6}より
\begin{align}
 f(x) = x(x-\alpha)(x-\beta) + ax
\end{align}
だから，
\begin{align}
 f(\alpha)=a\alpha, \quad f(\beta)=a\beta \label{eq:8}
\end{align}
なので，右辺の各項を微分すると
\begin{align}
\dfrac{d}{da}\int_0^\alpha f(x)dx&=f(\alpha)\alpha'=a\alpha\alpha',\\
\dfrac{d}{da}\int_\alpha^\beta f(x)dx&=f(\beta)\beta'-f(\alpha)\alpha'=a(\beta\beta'-\alpha\alpha'),\\
\dfrac{d}{da}\int_0^\alpha ax\,dx&=\int_0^\alpha x\,dx+a\alpha\alpha'=\dfrac{1}{2}\alpha^2+a\alpha\alpha',\\
\dfrac{d}{da}\int_\alpha^\beta ax\,dx&=\int_\alpha^\beta x\,dx+a(\beta\beta'-\alpha\alpha') = \dfrac{\beta^2-\alpha^2}{2} +a(\beta\beta'-\alpha\alpha')
\end{align}
となる．
よって\cref{eq:7}の両辺を$a$で微分してこれらを代入すると
\begin{align}
S'(a)
  &=a\alpha\alpha'-\Bigl(\dfrac{1}{2}\alpha^2+a\alpha\alpha'\Bigr)-a(\beta\beta'-\alpha\alpha')+\dfrac{\beta^2-\alpha^2}{2}+a(\beta\beta'-\alpha\alpha')\\
  &=-\dfrac{1}{2}\alpha^2+\dfrac{1}{2}(\beta^2-\alpha^2) \\
  &=\dfrac{1}{2}(\beta^2-2\alpha^2) \\
  &=\dfrac{1}{2}(\beta-\sqrt{2}\alpha)(\beta+\sqrt{2}\alpha) \label{eq:8}
\end{align}
となる．
さて，$0\le\alpha\le\beta$より$\beta+\sqrt{2}\alpha\ge0$（等号は$\alpha=\beta=0$のときのみ）だから，$S'(a)$の符号は$\beta-\sqrt{2}\alpha$の符号で決まる．
簡単のため
\begin{align}
 A=\sqrt{4a+1} \label{eq:9}
\end{align}
とおくと
\begin{align}
 \begin{cases}
   \alpha&=\dfrac{3-A}{2} \\
   \beta&=\dfrac{3+A}{2}
  \end{cases}
\end{align}
だから，
\begin{align}
\beta-\sqrt{2}\alpha
  &=\dfrac{1}{2}\bigl\{(3+A)-\sqrt{2}(3-A)\bigr\} \\
  &=\dfrac{1+\sqrt{2}}{2}\bigl\{A-3(3-2\sqrt{2})\bigr\}
\end{align}
よって$S'(a)$は$A=3(3-2\sqrt{2})=9-6\sqrt{2}$を境に符号が$-$から$+$に変わる．
このとき
\begin{align}
4a+1=A^2=(9-6\sqrt{2})^2=81-108\sqrt{2}+72=153-108\sqrt{2}
\end{align}
より
\begin{align}
a=\dfrac{152-108\sqrt{2}}{4}=38-27\sqrt{2}
\end{align}
であり，これは\cref{eq:4}の範囲内である．
以上から$S(a)$の増減表は以下のようになる．

\begin{table}[htb]
\centering
\caption{$S(a)$ の増減表 $\left(-\dfrac{1}{4} \le a \le 2\right)$}
\begin{tabular}{c|ccccc}
$a$ & $-\dfrac{1}{4}$ & $\cdots$ & $38-27\sqrt{2}$ & $\cdots$ & $2$ \\
\hline
$A$ & &  & $3(3-2\sqrt{2})$ &  & \\
\hline
$S'(a)$ & & $-$ & $0$ & $+$ & \\
\hline
$S(a)$ & & $\searrow$ & 極小 & $\nearrow$ & 
\end{tabular}
\end{table}

従って$S(a)$は
\begin{align}
a=38-27\sqrt{2}
\end{align}
で最小となる．
