\section*{第1問}

\begin{proof}[解]
接点を$A\bigl(\alpha,\frac12\alpha^2\bigr)$，$B\bigl(\beta,\frac12\beta^2\bigr)$（$\alpha<\beta$）とおく。$y=\frac12x^2$より$y'=x$だから，2接線は
\begin{align*}
l_A:\ y=\alpha x-\frac12\alpha^2,\qquad l_B:\ y=\beta x-\frac12\beta^2.
\end{align*}
$PA\perp PB$（すなわち$l_A\perp l_B$）は$\alpha\beta=-1$（①）と同値。$l_A,l_B$の交点$P$の$x$座標は，$\alpha x-\frac12\alpha^2=\beta x-\frac12\beta^2$を解いて$x=\dfrac{\alpha+\beta}2$。

放物線は自身の接線より常に上にあり，その差は$\frac12x^2-\bigl(\alpha x-\frac12\alpha^2\bigr)=\frac12(x-\alpha)^2$（$l_B$についても同様）だから
\begin{align*}
S=\int_\alpha^{\frac{\alpha+\beta}2}\frac12(x-\alpha)^2dx+\int_{\frac{\alpha+\beta}2}^\beta\frac12(x-\beta)^2dx
=\frac1{12}\Bigl(\frac{\beta-\alpha}2\Bigr)^3\cdot2=\frac1{24}(\beta-\alpha)^3.\tag{②}
\end{align*}
①より$\beta=-\dfrac1\alpha$（$\alpha\neq0$）なので，②に代入して
\begin{align*}
S=\frac1{24}\Bigl(-\frac1\alpha-\alpha\Bigr)^3=-\frac1{24}\Bigl(\alpha+\frac1\alpha\Bigr)^3.\tag{②$'$}
\end{align*}
また$\alpha<\beta=-\dfrac1\alpha$より，$\alpha>0$なら$-\frac1\alpha<0<\alpha$となり矛盾するので$\alpha<0$（③）。$t=-\alpha>0$とおくと，②$'$は
\begin{align*}
S=-\frac1{24}\Bigl(-t-\frac1t\Bigr)^3=\frac1{24}\Bigl(t+\frac1t\Bigr)^3.
\end{align*}
$t>0$より相加相乗平均から$t+\dfrac1t\ge2$（等号は$t=1$）だから
\begin{align*}
S\ge\frac1{24}\cdot2^3=\frac13.
\end{align*}
よって$S$の最小値は$\dfrac13$（$t=1$，すなわち$\alpha=-1,\beta=1$のとき）。
\end{proof}

\section*{第2問}

\begin{proof}[解]
$f(x,y)=(ax+(a-2)y,\ (a-2)x+ay)$とする。条件をみたす直線$L$（$(0,1)$を通り，$f$で不変）が存在する$a$を求める。

\textbf{(i) $L$が$y$軸（$x=0$）でない場合．} $L:y=kx+1$とおける。$(x,kx+1)$の像は
\begin{align*}
X=ax+(a-2)(kx+1)=\bigl[a+(a-2)k\bigr]x+(a-2),\qquad
Y=(a-2)x+a(kx+1)=\bigl[(a-2)+ak\bigr]x+a.
\end{align*}
これがすべての$x$について$Y=kX+1$をみたすための条件を求める。$Y-kX-1$を$x$の1次式として整理すると，$x$の係数は
\begin{align*}
\bigl[(a-2)+ak\bigr]-k\bigl[a+(a-2)k\bigr]=(a-2)-(a-2)k^2=(a-2)(1-k^2),
\end{align*}
定数項は
\begin{align*}
a-k(a-2)-1.
\end{align*}
これらがともに$0$であることが必要十分：
\begin{align*}
(a-2)(1-k^2)=0,\qquad a-k(a-2)-1=0.\tag{$\ast$}
\end{align*}

$a=2$のとき，第2式は$2-1=1\neq0$となり不可能。よって$a\neq2$，$(\ast)$の第1式から$k^2=1$，$k=\pm1$。

$k=1$のとき，第2式は$a-(a-2)-1=1\neq0$となり不可能。

$k=-1$のとき，第2式は$a+(a-2)-1=2a-3=0$より$a=\dfrac32$。

よって非鉛直な直線の場合，$a=\dfrac32$（このとき$L:y=-x+1$）。

\textbf{(ii) $L$が$y$軸（$x=0$）の場合．} $(0,y)$の像は$f(0,y)=((a-2)y,\ ay)$。これが常に$x=0$上にあるためには$(a-2)y=0$がすべての$y$で成り立つ必要があり，$a=2$。実際$a=2$のとき$f(0,y)=(0,2y)$は$x=0$上にある。

以上より，条件をみたす$a$は
\begin{align*}
a=2,\ \frac32.
\end{align*}
\end{proof}

\section*{第3問}

\begin{proof}[解]
$m,n\in\mathbb N$，$1\le m,n\le2N$（①）とする。$f(x)=x^2-nx+m$とおく。$f(x)=0$が$N$以上の実数解を持つ条件を考える。$f$の軸は$x=\dfrac n2$であり，①より$\dfrac n2\le N$。放物線$y=f(x)$は下に凸で軸が$N$以下にあるから，「$N$以上の実数解を持つ」ことと「$f(N)\le0$」は同値である（軸が$N$以下にある以上，2解がともに$N$以上になることはなく，$f(N)\le0$なら大きい方の解が$N$以上，$f(N)>0$なら実数解があっても両方$N$未満）。よって条件は
\begin{align*}
f(N)=N^2-nN+m\le0\quad\therefore\ m\le Nn-N^2.\tag{②}
\end{align*}

①，②をみたす領域を$nm$平面上に図示すると，直線$m=Nn-N^2$は$n=N$で$m=0$，$n=N+2$で$m=2N$を通る（境界は$m\ge1$の制約により$m=0$は含まれない）。

\begin{figure}[htb]
\centering
\begin{tikzpicture}[scale=0.42]
  \def\N{5}
  \draw[->] (0,0) -- (12,0) node[right]{$n$};
  \draw[->] (0,0) -- (0,11) node[above]{$m$};
  \draw[dashed] (\N,0) -- (\N,10.5);
  \draw[dashed] ({\N+2},0) -- ({\N+2},10.5);
  \draw[dashed] (2*\N,0) -- (2*\N,10.5);

  \node[below] at (\N,-0.2) {$N$};
  \node[below] at ({\N+2},-0.2) {$N+2$};
  \node[below] at (2*\N,-0.2) {$2N$};
  \node[left] at (0,2*\N) {$2N$};
  \fill[gray!25] plot[domain=\N:{\N+2},samples=2] (\x,{\N*\x-\N*\N}) -- ({\N+2},0) -- (\N,0) -- cycle;
  \fill[gray!25] ({\N+2},0) rectangle (2*\N,2*\N);
  \draw[thick,domain=\N:{\N+2},samples=2] plot (\x,{\N*\x-\N*\N});
  \node at (8,4) {\scriptsize$m=Nn-N^2$};
\end{tikzpicture}
\caption{条件①，②をみたす領域（斜線部，$N\ge3$の場合の概形）}
\end{figure}

$N\ge2$のとき，$n=N+1,\dots,2N$の各列について，
\begin{itemize}
\item $n=N+1$：$Nn-N^2=N$より，$m=1,\dots,N$の$N$通り。
\item $n=N+2,\dots,2N$（$N-1$列）：$Nn-N^2\ge2N$となり$m$の上限$2N$が効くので，各列$m=1,\dots,2N$の$2N$通り。
\end{itemize}
（$n=1,\dots,N$では$Nn-N^2\le0$となり該当する$m\ge1$は存在しない。）よって，もとめる組の総数を$F(N)$とすると
\begin{align*}
F(N)=N+(N-1)\cdot2N=N+2N^2-2N=N(2N-1).
\end{align*}

$N=1$の場合はこの一般論の前提（$N+2\le2N$すなわち$N\ge2$）が成り立たないので直接確認する。$N=1$のとき$2N=2$で，$n=1$では該当する$m$はなく，$n=2$では$Nn-N^2=1$より$m=1$の$1$通り。よって$F(1)=1=1\cdot(2\cdot1-1)$となり，上と同じ式が成り立つ。

以上より，すべての正の整数$N$に対して，求める組$(m,n)$の個数は
\begin{align*}
N(2N-1).
\end{align*}
\end{proof}

\section*{第4問}

\begin{proof}[解]
$l$の方向ベクトルは$\vec a=(1,1,1)$である。

\textbf{(1)} 点$\bigl(\frac t3,\frac t3,\frac t3\bigr)$を通り$l$と垂直な平面$\pi$は，法線ベクトル$(1,1,1)$を持つから
\begin{align*}
\pi:\ x+y+z=\frac t3+\frac t3+\frac t3=t.
\end{align*}
これと$xy$平面（$z=0$）との交線は
\begin{align*}
x+y=t,\qquad z=0.
\end{align*}

\textbf{(2)} $l$上に$\overrightarrow{OP}=\dfrac t3(1,1,1)$なる点$P$（$t\in\mathbb R$）をとる。$P$を通り$l$と垂直な平面による回転体の切り口を考える。この平面と$xy$平面の交線$x+y=t,\ z=0$は，(1)で見た通り$\pi\cap\{z=0\}$そのものである。

直線$x+y=t$と放物線$y=x(1-x)$の交点$Q$は，$x+x(1-x)=t$すなわち$x^2-2x+t=0$を解いて$x=1\pm\sqrt{1-t}$のうち$[0,1]$内の解
\begin{align*}
x=1-\sqrt{1-t}\qquad(0\le t\le1)
\end{align*}
に対応する点で，$Q=\bigl(1-\sqrt{1-t},\,(1-\sqrt{1-t})\sqrt{1-t},\,0\bigr)$。また直線$x+y=t$と$x$軸（$y=0$）の交点を$R=(t,0,0)$とする。

\begin{align*}
\overrightarrow{PR}=\Bigl(\frac23t,-\frac13t,-\frac13t\Bigr),\qquad|\overrightarrow{PR}|^2=\frac{4}9t^2+\frac19t^2+\frac19t^2=\frac23t^2.
\end{align*}
また
\begin{align*}
\overrightarrow{PQ}=\Bigl(1-\sqrt{1-t}-\frac t3,\ (1-\sqrt{1-t})\sqrt{1-t}-\frac t3,\ -\frac t3\Bigr)
\end{align*}
より，$|\overrightarrow{PQ}|^2$を計算すると
\begin{align*}
|\overrightarrow{PQ}|^2=\frac23t^2-4t+4-2(2-t)\sqrt{1-t}.
\end{align*}
（$t=0,1$で検算すると，$t=0$で$0$，$t=1$で$\frac23$となり，直接計算とも一致する。）

領域$D$の平面$\pi$による切り口は，線分$QR$（$P$を中心とする半径$|\overrightarrow{PQ}|$から$|\overrightarrow{PR}|$までの範囲；$|\overrightarrow{PQ}|\le|\overrightarrow{PR}|$）に対応し，これを$l$のまわりに回転させると，半径$|\overrightarrow{PQ}|$から$|\overrightarrow{PR}|$までの円環（アニュラス）になる。よってこの平面での立体の断面積$S(t)$は
\begin{align*}
\frac{S(t)}\pi=|\overrightarrow{PR}|^2-|\overrightarrow{PQ}|^2=\frac23t^2-\Bigl(\frac23t^2-4t+4-2(2-t)\sqrt{1-t}\Bigr)=4t-4+2(2-t)\sqrt{1-t}.
\end{align*}

$t$は平面$\pi$を指定するパラメータであり，$l$の単位方向ベクトルは$\frac1{\sqrt3}(1,1,1)$，$\pi$上の点$\bigl(\frac t3,\frac t3,\frac t3\bigr)$の$l$に沿った弧長は$\dfrac t{\sqrt3}=\dfrac{\sqrt3}3t$だから，もとめる体積$V$は
\begin{align*}
V=\int_0^1S(t)\cdot\frac{\sqrt3}3\,dt=\frac{\sqrt3}3\pi\int_0^1\bigl\{4(t-1)+2(2-t)\sqrt{1-t}\bigr\}dt.
\end{align*}
$2(2-t)\sqrt{1-t}=2(1-t)\sqrt{1-t}+2\sqrt{1-t}$と分けて，
\begin{align*}
\int_0^1(t-1)dt=\Bigl[\frac12t^2-t\Bigr]_0^1=-\frac12,\quad
\int_0^1\sqrt{1-t}\,dt=\Bigl[-\frac23(1-t)^{3/2}\Bigr]_0^1=\frac23,\quad
\int_0^1(1-t)\sqrt{1-t}\,dt=\Bigl[-\frac25(1-t)^{5/2}\Bigr]_0^1=\frac25
\end{align*}
を用いると
\begin{align*}
V=\frac{\sqrt3}3\pi\Bigl[4\Bigl(-\frac12\Bigr)+2\cdot\frac25+2\cdot\frac23\Bigr]=\frac{\sqrt3}3\pi\Bigl[-2+\frac45+\frac43\Bigr]=\frac{\sqrt3}3\pi\cdot\frac2{15}=\frac{2\sqrt3}{45}\pi.
\end{align*}
\end{proof}
