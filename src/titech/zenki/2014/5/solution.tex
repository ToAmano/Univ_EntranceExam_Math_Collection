{\bf [解]}
$f(x)=x^3+x^2+1$とおく．$f'(x)=3x^2+2x$より，増減表から$f$は$x=-\dfrac{2}{3}$で極大値$\dfrac{31}{27}$，$x=0$で極小値$1$をとる．

点$(x_k,f(x_k))$における$C$の接線$l_k$は
\begin{align}
l_k:\ y=g_k(x)=(3x_k^2+2x_k)x-2x_k^3-x_k^2+1
\end{align}
である．$C$と$l_k$の共有点の$x$座標のうち$x_k$でないものを求める．$f(x)=g_k(x)$より
\begin{align}
x^3+x^2-(3x_k^2+2x_k)x+2x_k^3+x_k^2=0
\end{align}
は$x=x_k$を重解に持つから，$(x-x_k)^2(x-r)=0$の形に因数分解でき，係数比較（$x^2$の係数）より$-(2x_k+r)=1$，すなわち$r=-(2x_k+1)$．よって
\begin{align}
x_{k+1}=-(2x_k+1).\tag{①}
\end{align}
また
\begin{align}
S_k=\Bigl|\int_{x_{k-1}}^{x_k}\{f(x)-g_{k-1}(x)\}dx\Bigr|=\Bigl|\int_{x_{k-1}}^{x_k}(x-x_{k-1})^2(x-x_k)\,dx\Bigr|=\dfrac{1}{12}(x_k-x_{k-1})^4\tag{②}
\end{align}
（標準公式$\int_p^q(x-p)^2(x-q)dx=-\dfrac{(q-p)^4}{12}$を用いた）．

\textbf{(1)} $x_0=1$だから，①より$x_1=-(2\cdot1+1)=-3$．②に代入して
\begin{align}
S_1=\dfrac{(-3-1)^4}{12}=\dfrac{4^4}{12}=\dfrac{4^3}{3}=\dfrac{64}{3}.
\end{align}

\textbf{(2)} ①より$x_{k+1}+\dfrac{1}{3}=-2\Bigl(x_k+\dfrac{1}{3}\Bigr)$だから，$x_k+\dfrac{1}{3}$は公比$-2$の等比数列．$x_0+\dfrac{1}{3}=\dfrac{4}{3}$だから
\begin{align}
x_k+\dfrac{1}{3}=\dfrac{4}{3}(-2)^k\quad\therefore\ x_k=\dfrac{4}{3}(-2)^k-\dfrac{1}{3}=\dfrac{1}{3}\bigl\{(-2)^{k+2}-1\bigr\}.
\end{align}

\textbf{(3)} (2)より
\begin{align}
x_k-x_{k-1}=\dfrac{1}{3}\bigl\{(-2)^{k+2}-(-2)^{k+1}\bigr\}=\dfrac{1}{3}(-2)^{k+1}(-3)=-(-2)^{k+1}
\end{align}
だから，②より
\begin{align}
S_k=\dfrac{1}{12}(x_k-x_{k-1})^4=\dfrac{1}{12}\{-(-2)^{k+1}\}^4=\dfrac{1}{12}(-2)^{4(k+1)}=\dfrac{1}{12}\cdot16\cdot2^{4k}=\dfrac{4}{3}\cdot2^{4k}.
\end{align}
よって
\begin{align}
\dfrac{1}{S_k}=\dfrac{3}{4}\Bigl(\dfrac{1}{2}\Bigr)^{4k}=\dfrac{3}{4}\Bigl(\dfrac{1}{16}\Bigr)^k
\end{align}
であり，等比級数の和の公式から
\begin{align}
\sum_{k=1}^\infty\dfrac{1}{S_k}=\dfrac{3}{4}\cdot\dfrac{1/16}{1-1/16}=\dfrac{3}{4}\cdot\dfrac{1}{15}=\dfrac{1}{20}.
\end{align}
