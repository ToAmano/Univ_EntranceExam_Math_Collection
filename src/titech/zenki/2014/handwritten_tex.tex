\section*{第1問}

\begin{proof}[解]
題意から$n=2m+1$（$m\in\mathbb N$）とおける。
\begin{align*}
a_n=\frac16\sum_{k=1}^{n-1}(k-1)k(k+1)=\frac1{24}(n+1)n(n-1)(n-2),\qquad
b_n=\frac{(n+1)(n-1)}8=\frac12m(m+1)\tag{①}
\end{align*}
とかける（$a_n$は$(k-1)k(k+1)(k+2)-(k-2)(k-1)k(k+1)=4(k-1)k(k+1)$による望遠鏡和で計算した）。

\textbf{(1)} ①で$(n+1)n(n-1)(n-2)$は連続4整数の積だから$4!=24$で割り切れるので，$a_n\in\mathbb Z$。また$m(m+1)$も連続2整数の積だから$2!=2$で割り切れるので，$b_n\in\mathbb Z$。

\textbf{(2)} $n=2m+1$を用いると
\begin{align*}
a_n=\frac1{24}(2m+2)(2m+1)\cdot2m(2m-1)=\frac{(2m+1)(2m-1)(m+1)m}6
\end{align*}
だから
\begin{align*}
a_n-b_n&=m(m+1)\Bigl[\frac{(2m+1)(2m-1)}6-\frac12\Bigr]=m(m+1)\cdot\frac{4m^2-4}6=\frac{m(m+1)}6\cdot4(m+1)(m-1)\\
&=\frac23(m+1)(m+1)m(m-1).
\end{align*}
$(m+1)m(m-1)$は連続3整数の積だから$3!=6$で割り切れ，$A:=(m+1)m(m-1)/6\in\mathbb Z$とおけて
\begin{align*}
a_n-b_n=4(m+1)A\in4\mathbb Z.
\end{align*}
よって$a_n-b_n$は$4$の倍数である。
\end{proof}

\section*{第2問}

\begin{proof}[解]
\textbf{(1)} $a=2$のとき，$t>0$から
\begin{align*}
(\ast)\iff e^t-e^{t/2}t-1\ge0.\tag{①}
\end{align*}
この左辺を$f(t)$とおく。
\begin{align*}
f'(t)=e^t-e^{t/2}\Bigl(1+\frac t2\Bigr)=e^{t/2}\Bigl\{e^{t/2}-\Bigl(\frac t2+1\Bigr)\Bigr\}>0\qquad(\because t>0,\ e^{t/2}>1+\frac t2)
\end{align*}
だから，$f(t)$は単調増加で，$0<t$のとき$f(t)>f(0)=0$。よって①が成り立ち，(*)が示された。

\textbf{(2)} $g(t)=e^t-e^{t/a}t-1$とおく。$t>0$で$g(t)\ge0$であることを示せばよい。$x=t/a$とおくと
\begin{align*}
g'(t)=e^t-e^{t/a}\Bigl(1+\frac ta\Bigr)=e^{ax}-e^x(1+x)=e^x\bigl[e^{(a-1)x}-(x+1)\bigr].\tag{②}
\end{align*}
$a\ge2$のとき，$a-1\ge1$より$e^{(a-1)x}\ge1+(a-1)x\ge1+x$（$x>0$，$e^u\ge1+u$と$a-1\ge1$より）だから②は$\ge0$，すなわち$g'(t)\ge0$。$g$は単調増加だから$t>0$に対して$g(t)>g(0)=0$となる。（③）

一方，$1<a<2$のとき，$0<x<\alpha$なる十分小さい$\alpha$で$e^{(a-1)x}-(x+1)<0$となる（$x\to0^+$で$e^{(a-1)x}-(x+1)\to0$かつその微分係数は$x=0$で$(a-1)-1=a-2<0$だから，小さい$x>0$で負）。したがって②より$g'(t)<0$となる$t$が存在し，$g(0)=0$から出発して$g$は減少するので$g(t)<0$となる$t$が存在し，(*)は成り立たない。（④）

以上③④から，求める条件は
\begin{align*}
a\ge2
\end{align*}
である。（②の直感：$y=e^x$の$x=0$での接線は$y=x+1$であり，$(e^{(a-1)x})'|_{x=0}=a-1$だから，$a-1$が$1$以上か未満かでグラフの交わり方が変わり，これが上記の場合分けに対応する。）
\end{proof}

\section*{第4問}

\begin{proof}[解]
\textbf{(1)} 題意から，複素数平面上で$45^\circ$回転を考えると
\begin{align*}
x+yi&=\Bigl(\cos\frac\pi4+i\sin\frac\pi4\Bigr)(t+si)=\frac{\sqrt2}2(1+i)(t+si)=\frac{\sqrt2}2\bigl\{(t-s)+i(t+s)\bigr\}
\end{align*}
であり，$x,y,s,t$は実数だから
\begin{align*}
x=\frac{\sqrt2}2(t-s)=\frac{\sqrt2}2(-\sqrt2t^2+3t)\equiv f(t),\qquad
y=\frac{\sqrt2}2(s+t)=\frac{\sqrt2}2(\sqrt2t^2-t)\equiv g(t).
\end{align*}

\textbf{(2)} $g(t)=\dfrac{\sqrt2}2\Bigl[\sqrt2\Bigl(t-\dfrac1{2\sqrt2}\Bigr)^2-\dfrac{\sqrt2}8\Bigr]$だから，$g$は$t=\dfrac1{2\sqrt2}=\dfrac{\sqrt2}4$で最小値
\begin{align*}
\frac{\sqrt2}2\cdot\Bigl(-\frac{\sqrt2}8\Bigr)=-\frac18
\end{align*}
をとる。$g(t)$は$t$の2次関数（上に凸ではなく下に凸）で，直線$y=a$が曲線$C$とただ1点で交わるのは，この最小値の位置，すなわち
\begin{align*}
a=-\frac18
\end{align*}
のときである。

\textbf{(3)} $g(t)=0\iff t=0,\ \dfrac{\sqrt2}2$（$\sqrt2t^2-t=t(\sqrt2t-1)=0$より）。よって$0\le t\le\dfrac{\sqrt2}2$の部分がグラフと$x$軸で囲む領域$D$の境界を与え，$t\in\bigl(0,\frac{\sqrt2}4\bigr)$で$x_-(y)$（左側の枝），$t\in\bigl(\frac{\sqrt2}4,\frac{\sqrt2}2\bigr)$で$x_+(y)$（右側の枝）を与える。$D$を$y$軸のまわりに回転した体積$V$は
\begin{align*}
\frac V\pi=\int_a^0\bigl(x_+^2-x_-^2\bigr)dy=\int_0^{\sqrt2/2}x(t)^2\frac{dy}{dt}dt\tag{①}
\end{align*}
（両方の枝の寄与が$t=0$から$\frac{\sqrt2}2$までの一つの積分にまとまる）。$g'(t)=\dfrac{\sqrt2}2(2\sqrt2t-1)$であり，$x(t)=\dfrac{\sqrt2}2t(3-\sqrt2t)$だから
\begin{align*}
x(t)^2g'(t)&=\frac12t^2(3-\sqrt2t)^2\cdot\frac{\sqrt2}2(2\sqrt2t-1)=\frac{\sqrt2}4t^2(3-\sqrt2t)^2(2\sqrt2t-1)\\
&=\frac{\sqrt2}4t^2(4\sqrt2t^3-26t^2+24\sqrt2t-9)
\end{align*}
（$(3-\sqrt2t)^2(2\sqrt2t-1)=4\sqrt2t^3-26t^2+24\sqrt2t-9$を展開して得た）。①に代入して
\begin{align*}
\frac V\pi&=\frac{\sqrt2}4\Bigl[\frac{4\sqrt2}6t^6-\frac{26}5t^5+\frac{24\sqrt2}4t^4-\frac93t^3\Bigr]_0^{\sqrt2/2}\\
&=\frac{\sqrt2}4\Bigl[\frac{2\sqrt2}3t^6-\frac{26}5t^5+6\sqrt2t^4-3t^3\Bigr]_0^{\sqrt2/2}
\end{align*}
$t=\dfrac{\sqrt2}2$のとき$t^2=\dfrac12,t^3=\dfrac{\sqrt2}4,t^4=\dfrac14,t^5=\dfrac{\sqrt2}8,t^6=\dfrac18$を代入すると
\begin{align*}
\frac{2\sqrt2}3\cdot\frac18-\frac{26}5\cdot\frac{\sqrt2}8+6\sqrt2\cdot\frac14-3\cdot\frac{\sqrt2}4=\frac{\sqrt2}{12}-\frac{13\sqrt2}{20}+\frac{3\sqrt2}2-\frac{3\sqrt2}4=\frac{11\sqrt2}{60}
\end{align*}
（通分すると$\frac{5-39+90-45}{60}\sqrt2=\frac{11}{60}\sqrt2$）。よって
\begin{align*}
\frac V\pi=\frac{\sqrt2}4\cdot\frac{11\sqrt2}{60}=\frac{11\cdot2}{240}=\frac{11}{120}
\end{align*}
より
\begin{align*}
V=\frac{11}{120}\pi.
\end{align*}
\end{proof}

\section*{第5問}

\begin{proof}[解]
$f(x)=x^3+x^2+1$とおく。$f'(x)=3x^2+2x$より，増減表から$f$は$x=-\dfrac23$で極大値$\dfrac{31}{27}$，$x=0$で極小値$1$をとる。

点$(x_k,f(x_k))$における$C$の接線$l_k$は
\begin{align*}
l_k:\ y=g_k(x)=(3x_k^2+2x_k)x-2x_k^3-x_k^2+1
\end{align*}
である。$C$と$l_k$の共有点の$x$座標のうち$x_k$でないものを求める。$f(x)=g_k(x)$より
\begin{align*}
x^3+x^2-(3x_k^2+2x_k)x+2x_k^3+x_k^2=0
\end{align*}
は$x=x_k$を重解に持つから，$(x-x_k)^2(x-r)=0$の形に因数分解でき，係数比較（$x^2$の係数）より$-(2x_k+r)=1$，すなわち$r=-(2x_k+1)$。よって
\begin{align*}
x_{k+1}=-(2x_k+1).\tag{①}
\end{align*}
また
\begin{align*}
S_k=\Bigl|\int_{x_{k-1}}^{x_k}\{f(x)-g_{k-1}(x)\}dx\Bigr|=\Bigl|\int_{x_{k-1}}^{x_k}(x-x_{k-1})^2(x-x_k)\,dx\Bigr|=\frac1{12}(x_k-x_{k-1})^4\tag{②}
\end{align*}
（標準公式$\int_p^q(x-p)^2(x-q)dx=-\dfrac{(q-p)^4}{12}$を用いた）。

\textbf{(1)} $x_0=1$だから，①より$x_1=-(2\cdot1+1)=-3$。②に代入して
\begin{align*}
S_1=\frac{(-3-1)^4}{12}=\frac{4^4}{12}=\frac{4^3}3=\frac{64}3.
\end{align*}

\textbf{(2)} ①より$x_{k+1}+\dfrac13=-2\Bigl(x_k+\dfrac13\Bigr)$だから，$x_k+\dfrac13$は公比$-2$の等比数列。$x_0+\dfrac13=\dfrac43$だから
\begin{align*}
x_k+\frac13=\frac43(-2)^k\quad\therefore\ x_k=\frac43(-2)^k-\frac13=\frac13\bigl\{(-2)^{k+2}-1\bigr\}.
\end{align*}

\textbf{(3)} (2)より
\begin{align*}
x_k-x_{k-1}=\frac13\bigl\{(-2)^{k+2}-(-2)^{k+1}\bigr\}=\frac13(-2)^{k+1}(-3)=-(-2)^{k+1}
\end{align*}
だから，②より
\begin{align*}
S_k=\frac1{12}(x_k-x_{k-1})^4=\frac1{12}\{-(-2)^{k+1}\}^4=\frac1{12}(-2)^{4(k+1)}=\frac1{12}\cdot16\cdot2^{4k}=\frac43\cdot2^{4k}.
\end{align*}
よって
\begin{align*}
\frac1{S_k}=\frac34\Bigl(\frac12\Bigr)^{4k}=\frac34\Bigl(\frac1{16}\Bigr)^k
\end{align*}
であり，等比級数の和の公式から
\begin{align*}
\sum_{k=1}^\infty\frac1{S_k}=\frac34\cdot\frac{1/16}{1-1/16}=\frac34\cdot\frac1{15}=\frac1{20}.
\end{align*}
\end{proof}
