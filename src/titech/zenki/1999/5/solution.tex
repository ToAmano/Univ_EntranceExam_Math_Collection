{\bf [解]}
$e(\theta)=\cos\theta+i\sin\theta$とおく．

\textbf{(1)} $P_0=0,\ P_1=1+i$．$\overrightarrow{P_{n-1}P_n}$を表す複素数を$d_n$とすると，題意から
\begin{align}
d_{n+1}=\dfrac{2}{3}e\left(\dfrac{\pi}{3}\right)d_n \quad\cdots\text{②}
\end{align}
$d_1=1+i$（①）とあわせて
\begin{align}
d_n=\left\{\dfrac{2}{3}e\left(\dfrac{\pi}{3}\right)\right\}^{n-1}(1+i)
\end{align}
だから
\begin{align}
\overrightarrow{OP_n}=\overrightarrow{OP_0}+d_1+\cdots+d_n=\sum_{k=1}^n\left\{\dfrac{2}{3}e\left(\dfrac{\pi}{3}\right)\right\}^{k-1}(1+i)=(1+i)\cdot\dfrac{1-\left\{\dfrac{2}{3}e(\pi/3)\right\}^n}{1-\dfrac{2}{3}e(\pi/3)}
\end{align}
$\left|\dfrac{2}{3}e(\pi/3)\right|=\dfrac{2}{3}<1$だから$n\to\infty$で
\begin{align}
P_\infty=\dfrac{1+i}{1-\dfrac{2}{3}e(\pi/3)}
\end{align}
$e(\pi/3)=\dfrac{1}{2}+\dfrac{\sqrt{3}}{2}i$より$\dfrac{2}{3}e(\pi/3)=\dfrac{1}{3}+\dfrac{\sqrt{3}}{3}i$，$1-\dfrac{2}{3}e(\pi/3)=\dfrac{2}{3}-\dfrac{\sqrt{3}}{3}i=\dfrac{2-\sqrt{3}i}{3}$だから
\begin{align}
P_\infty=\dfrac{3(1+i)}{2-\sqrt{3}i}=\dfrac{3(1+i)(2+\sqrt{3}i)}{(2-\sqrt{3}i)(2+\sqrt{3}i)}=\dfrac{3(1+i)(2+\sqrt{3}i)}{7}
\end{align}
$(1+i)(2+\sqrt{3}i)=(2-\sqrt{3})+(2+\sqrt{3})i$だから
\begin{align}
P_\infty=\dfrac{3}{7}\left\{(2-\sqrt{3})+(2+\sqrt{3})i\right\}
\end{align}

\textbf{(2)} $Q_0=0,\ Q_1=z$．(1)と同様に，$\overrightarrow{Q_{n-1}Q_n}$を表す複素数を$\beta_n$とすると$\beta_1=z$，
\begin{align}
\begin{split}
\beta_{n+1}&=\dfrac{1}{2}e\left(\dfrac{\pi}{6}\right)\beta_n \quad\therefore\ \beta_n=\left\{\dfrac{1}{2}e\left(\dfrac{\pi}{6}\right)\right\}^{n-1}z \\
\overrightarrow{OQ_n}&=\sum_{k=1}^n\beta_k=z\cdot\dfrac{1-\left\{\dfrac{1}{2}e(\pi/6)\right\}^n}{1-\dfrac{1}{2}e(\pi/6)} \ \xrightarrow{n\to\infty}\ Q_\infty=\dfrac{z}{1-\dfrac{1}{2}e(\pi/6)}
\end{split}
\end{align}
$e(\pi/6)=\dfrac{\sqrt{3}}{2}+\dfrac{1}{2}i$より$\dfrac{1}{2}e(\pi/6)=\dfrac{\sqrt{3}}{4}+\dfrac{1}{4}i$，$1-\dfrac{1}{2}e(\pi/6)=\dfrac{4-\sqrt{3}}{4}-\dfrac{1}{4}i=\dfrac{(4-\sqrt{3})-i}{4}$だから
\begin{align}
Q_\infty=\dfrac{4z}{(4-\sqrt{3})-i}
\end{align}

$Q_\infty=P_\infty$より
\begin{align}
\begin{split}
\dfrac{4z}{(4-\sqrt{3})-i}&=\dfrac{3}{7}\left\{(2-\sqrt{3})+(2+\sqrt{3})i\right\} \\
z&=\dfrac{3}{28}\left\{(2-\sqrt{3})+(2+\sqrt{3})i\right\}\left\{(4-\sqrt{3})-i\right\}
\end{split}
\end{align}
右辺の$\{\ \}\{\ \}$を展開する（実部$=pr-qs$，虚部$=ps+qr$，ただし$p=2-\sqrt{3},q=2+\sqrt{3},r=4-\sqrt{3},s=-1$として）：
\begin{align}
\begin{split}
pr&=(2-\sqrt{3})(4-\sqrt{3})=11-6\sqrt{3},\qquad qs=-(2+\sqrt{3}) \\
pr-qs&=11-6\sqrt{3}+2+\sqrt{3}=13-5\sqrt{3} \\
ps&=-(2-\sqrt{3}),\qquad qr=(2+\sqrt{3})(4-\sqrt{3})=5+2\sqrt{3} \\
ps+qr&=-(2-\sqrt{3})+5+2\sqrt{3}=3+3\sqrt{3}=3(1+\sqrt{3})
\end{split}
\end{align}
よって
\begin{align}
z=\dfrac{3}{28}\left\{(13-5\sqrt{3})+3(1+\sqrt{3})i\right\}
\end{align}
