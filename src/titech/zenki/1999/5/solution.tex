{\bf [解]}
実数$\theta$に対して$e(\theta)=\cos\theta+i\sin\theta$とおく．

\paragraph{(1)}

初期条件は
\begin{align}
 P_0=0,\ P_1=1+i \label{eq:1}
\end{align}
である．$\overrightarrow{P_{n-1}P_n}$を表す複素数を$d_n$とすると\cref{eq:1}から
\begin{align}
 d_1 = 1+i \label{eq:2}
\end{align}
および題意から
\begin{align}
d_{n+1}=\dfrac{2}{3}e\left(\dfrac{\pi}{3}\right)d_n \quad\cdots\text{②} \label{eq:3}
\end{align}
である．
漸化式を繰り返し用いて
\begin{align}
d_n=\left\{\dfrac{2}{3}e\left(\dfrac{\pi}{3}\right)\right\}^{n-1}(1+i) \label{eq:4}
\end{align}
である．

従って$P_n$へのベクトルは
\begin{align}
\overrightarrow{OP_n}
 &=\overrightarrow{OP_0}+d_1+\cdots+d_n \\
 &=\sum_{k=1}^n\left\{\dfrac{2}{3}e\left(\dfrac{\pi}{3}\right)\right\}^{k-1}(1+i) \\
 &=(1+i)\cdot\dfrac{1-\left\{\dfrac{2}{3}e(\pi/3)\right\}^n}{1-\dfrac{2}{3}e(\pi/3)} \label{eq:5}
\end{align}
と書ける．

ここで$\left|\dfrac{2}{3}e(\pi/3)\right|=\dfrac{2}{3}<1$だから$n\to\infty$では$\left|\dfrac{2}{3}e(\pi/3)\right|^{n}\to 0$であって，
\begin{align}
 \overrightarrow{OP_\infty} = \lim_{n\to\infty}\overrightarrow{OP_n}=\dfrac{1+i}{1-\dfrac{2}{3}e(\pi/3)}
\end{align}
となる．ここで$e(\pi/3)=\dfrac{1}{2}+\dfrac{\sqrt{3}}{2}i$より$1-\dfrac{2}{3}e(\pi/3)=\dfrac{2}{3}-\dfrac{\sqrt{3}}{3}i=\dfrac{2-\sqrt{3}i}{3}$だから
\begin{align}
P_\infty
 &=\dfrac{3(1+i)}{2-\sqrt{3}i} \\
 &=\dfrac{3(1+i)(2+\sqrt{3}i)}{(2-\sqrt{3}i)(2+\sqrt{3}i)}\\
 &=\dfrac{3(1+i)(2+\sqrt{3}i)}{7} \\
 &= \dfrac{3}{7}\left\{(2-\sqrt{3})+(2+\sqrt{3})i\right\} \label{eq:6}
\end{align}
となる．

\paragraph{(2)}

初期条件は
\begin{align}
 Q_0=0,\ Q_1=z \label{eq:7}
\end{align}
である．
(1)と同様に$\overrightarrow{Q_{n-1}Q_n}$を表す複素数を$\beta_n$とすると\cref{eq:7}および題意の条件から
\begin{align}
 \begin{cases}
  \beta_1=z \\
  \beta_{n+1}=\dfrac{1}{2}e\left(\dfrac{\pi}{6}\right)\beta_n 
 \end{cases}
\end{align}
である．従って漸化式を繰り返し用いて一般項は
\begin{align}
 \beta_n=\left\{\dfrac{1}{2}e\left(\dfrac{\pi}{6}\right)\right\}^{n-1}z
\end{align}
である．従って$Q_$へのベクトルは
\begin{align}
\overrightarrow{OQ_n}
  &=\sum_{k=1}^n\beta_k \\
  &=z\cdot\dfrac{1-\left\{\dfrac{1}{2}e(\pi/6)\right\}^n}{1-\dfrac{1}{2}e(\pi/6)} 
\end{align}
だから$n\to\infty$への極限では
\begin{align}
 Q_{\infty} = \lim_{n\to\infty}Q_{n} = \dfrac{z}{1-\dfrac{1}{2}e(\pi/6)}
\end{align}
となる．

ここで$e(\pi/6)=\dfrac{\sqrt{3}}{2}+\dfrac{1}{2}i$より$1-\dfrac{1}{2}e(\pi/6)=\dfrac{4-\sqrt{3}}{4}-\dfrac{1}{4}i=\dfrac{(4-\sqrt{3})-i}{4}$だから
\begin{align}
Q_\infty=\dfrac{4z}{(4-\sqrt{3})-i} \label{eq:8}
\end{align}
である．

$Q_\infty=P_\infty$に\cref{eq:6,eq:8}を代入して
\begin{align}
\begin{split}
\dfrac{4z}{(4-\sqrt{3})-i}&=\dfrac{3}{7}\left\{(2-\sqrt{3})+(2+\sqrt{3})i\right\} \\
\therefore
z&=\dfrac{3}{28}\left\{(2-\sqrt{3})+(2+\sqrt{3})i\right\}\left\{(4-\sqrt{3})-i\right\}
\end{split}
\end{align}
である．以下右辺を綺麗化すると
\begin{align}
 z 
  &= \dfrac{3}{28} \left[(2-\sqrt{3})(4-\sqrt{3})+(2+\sqrt{3}) + (2+\sqrt{3})(4-\sqrt{3})i -(2-\sqrt{3})i\right] \\
  &= \dfrac{3}{28} \left[(11-6\sqrt{3})+(2+\sqrt{3}) + (5+2\sqrt{3})i -(2-\sqrt{3})i\right] \\
  &= \dfrac{3}{28} \left[(13-5\sqrt{3}) + (3+3\sqrt{3})i \right] 
\end{align}
である．
