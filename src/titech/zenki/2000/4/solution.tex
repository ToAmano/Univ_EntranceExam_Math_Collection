{\bf [解]}

二つの関数の差分を
\begin{align}
 f_n(x)=e^{nx}-1-e^x \label{eq:1}
\end{align}
とおく．

\paragraph{(1)}

(ア)と(イ)の交点の$x$座標は方程式$f_n(x)=0$の解で与えられるから，$f_n(x)$の挙動について調べる．
一階微分は
\begin{align}
 f_n'(x)=ne^{nx}-e^x=ne^{nx}-e^{x}=e^{x}(ne^{(n-1)x}-1)
\end{align}
である．$x \ge 0$では$e^x\ge1$だから，$n\ge2$より$ne^{(n-1)x}\ge n\ge2>1$である．
従って$f_n'(x)>0$であり，増減表は以下のようになる．

\begin{table}[htb]
  \centering
\caption{$f(x)$の増減表}
\begin{tabular}{c|ccc}
$x$ & $0$ & & $\infty$\\
\hline
$f'$ & & $+$ & \\
\hline
$f$ & $-1$ & $\nearrow$ &$\infty$
\end{tabular}
\end{table}

これと\cref{eq:1}より$f_n(0)=1-1-1=-1<0$および$f_n(x)\to\infty\ (x\to\infty)$から$f_n(x)=0$は$x>0$にただ1つの実解を持つ．
したがって(ア)，(イ)は第1象限にただ1つの交点を持つ．

\paragraph{(2)}
(1)の交点を$P_n(a_n,b_n)$とおくと(1)より
\begin{align}
 \begin{cases}
  f_n(a_n) = 0 \\
  0 < a_n
 \end{cases} \label{eq:2}
\end{align}
である．

まず$a_n$の極限値から考えよう．$x=\dfrac{1}{n}$の時の$f_n(x)$の値は
\begin{align}
f\left(\dfrac{1}{n}\right)=e-1-e^{1/n}
\end{align}
であり，$n\ge 2$より$\dfrac{1}{n}\le \dfrac{1}{2}$だから$e^{1/n}\le e^{1/2}$であることを用いて
\begin{align}
f\left(\dfrac{1}{n}\right)=e-1-e^{1/n}\ge e-1-e^{1/2}
\end{align}
である．
ここで$(1.7)^2=2.89>e$より$e^{1/2}<1.7$，また$e>2.7$だから右辺は下から評価できて
\begin{align}
 e-1-e^{1/2}>2.7-1-1.7=0 \\
\therefore
f\left(\dfrac{1}{n}\right) > 0 
\end{align}
である．(1)より$0<x$では$f_n(x)$は単調増加だから$a_n<\dfrac{1}{n}$で，\cref{eq:2}と合わせて
\begin{align}
0<a_n<\dfrac{1}{n}
\end{align}
である．右辺は$n\to\infty$で$0$に収束するからはさみうちの定理から
\begin{align}
 \lim_{n\to\infty}a_n = 0 \label{eq:3}
\end{align}
である．

次に$na_n$の極限値を考える．$f(a_n)=0$に\cref{eq:1}を代入して
\begin{align}
 e^{na_n}=e^{a_n}+1
\end{align}
である．両辺正だから自然対数をとって
\begin{align}
 na_n=\log(e^{a_n}+1) 
\end{align}
である．両辺の$n\to\infty$の極限をとって右辺に\cref{eq:3}を用いると
\begin{align}
 \lim_{n\to\infty}na_n = \lim_{n\to\infty} \log(e^{a_n}+1) = \log(e^{\lim_{n\to\infty}a_n}+1) = \log 2 \label{eq:4}
\end{align}
である．

\cref{eq:3,eq:4}をまとめて
\begin{align}
 \begin{cases}
  \lim_{n\to\infty}a_n = 0 \\
  \lim_{n\to\infty}na_n = \log 2
 \end{cases} \label{eq:5}
\end{align}
である．

$0<x<a_n$では$f_n(x)<0$つまり$e^x>e^{nx}-1$である．よってこの区間では(ア)が(イ)より上にある．
従って求める面積$S_n$は
\begin{align}
\begin{split}
S_n
 &=\int_0^{a_n}\left\{-f_n(x)\right\}dx\\
 &=\int_0^{a_n}\left\{e^x-(e^{nx}-1)\right\}dx\\
 &=\left[e^x-\dfrac{1}{n}e^{nx}+x\right]_0^{a_n} \\
 &=e^{a_n}-\dfrac{1}{n}e^{na_n}+a_n-1+\dfrac{1}{n} 
\end{split}
\end{align}
と書けるから，両辺に$n$をかけて
\begin{align}
nS_n
 &=n(e^{a_n}-1)+na_n+1-e^{na_n} \\
 &=na_n\cdot\dfrac{e^{a_n}-1}{a_n}+na_n+1-e^{na_n} \label{eq:6}
\end{align}
である．
$n\to\infty$の時の各項の振る舞いを考えると\cref{eq:5}より
\begin{align}
 \lim_{n\to\infty} \dfrac{e^{a_n}-1}{a_n} = 1 \\
 \lim_{n\to\infty} e^{na_n} = e^{\log 2} = 2
\end{align}
だから\cref{eq:6}に代入して
\begin{align}
 \lim_{n\to\infty} nS_n = \log 2 + \log 2 +1 -2 = 2\log 2-1
\end{align}
である．
