{\bf [解]}
題意より
\begin{align}
 0<\alpha,\beta<1 \quad\cdots\text{①}
\end{align}
である．

\paragraph{(1)}
$n$回目に$E$が起こって$n+1$回目にも$E$が起こる確率は$\alpha p_n$，
$n$回目に$E$が起こらず$n+1$回目に$E$が起こる確率は$(1-\beta)(1-p_n)$だから
漸化式は
\begin{align}
 p_{n+1} 
  &= \alpha p_n + (1-\beta)(1-p_n) \\
  &= (\alpha+\beta-1)p_n + (1-\beta) \quad\cdots\text{②}
\end{align}
である．

\paragraph{(2)}
まず漸化式②を解く．
新しく
\begin{align}
  t = \dfrac{1-\beta}{2-(\alpha+\beta)}
\end{align}
とおいて②に代入して
\begin{align}
  p_{n+1}-t = \left(\alpha+\beta-1\right)\left(p_n-t\right)
\end{align}
である．
等比数列の公式から
\begin{align}
  p_n = \left(\alpha+\beta-1\right)^{n-1}(p_1-t)+t
\end{align}
である．①より$-1<\alpha+\beta-1<1$だから，
\begin{align}
  \lim_{n\to\infty} p_n 
    &= t \\
    &= \dfrac{1-\beta}{2-(\alpha+\beta)}
\end{align}
が求める極限値である．
