{\bf [解]}
題意より
\begin{align}
 m,n\in\mathbb{Z}_{>0}, m<n, 0<x<1 \label{eq:1}
\end{align}
である．新しく
\begin{align}
 f(t)=\left(1+\dfrac{x}{t^2}\right)^t \label{eq:2}
\end{align}
とおくと，\cref{eq:1}の元で$f(m)$と$f(n)$の大小を比較すれば良い．
そこで$y=f(t)$の一階微分から挙動を調べる．

$f(t)>0$だから\cref{eq:2}の両辺自然対数をとって
\begin{align}
 \log f(t) = t \log \left(1+\dfrac{x}{t^2}\right)
\end{align}
となり，この両辺を$t$で微分すると
\begin{align}
\dfrac{f'(t)}{f(t)}
  &=\log\left(1+\dfrac{x}{t^2}\right)+\dfrac{t}{1+x/t^2}(-2)\dfrac{1}{t^3} \\
  &= \log\left(1+\dfrac{x}{t^2}\right)-\dfrac{2}{x+t^2} \quad \cdots \text{②} \label{eq:3}
\end{align}
となる．

この右辺を $g(t)$ とする：
\begin{align}
 \begin{cases}
 g(t) = \log\left(1+\dfrac{x}{t^2}\right)-\dfrac{2}{x+t^2} \\
 \dfrac{f'(t)}{f(t)} = g(t)
 \end{cases} \label{eq:4}
\end{align}
一階微分を計算すると
\begin{align}
g'(t)
  &=\dfrac{1}{1+x/t^2}(-2)\dfrac{1}{t^3}+2\left(\dfrac{1}{t^2+x}\right)^2\cdot2t \\
  &= -\dfrac{2}{t}\cdot\dfrac{1}{x+t^2}+4t\left(\dfrac{1}{x+t^2}\right)^2 \\
  &= \dfrac{2(t^2-x)}{t(x+t^2)^2}
\end{align}
である．\cref{eq:1}より$1\le t$で考えれば良いが，この時\cref{eq:1}より$x<t^2$だから$g'(t)\ge0$ で $g(t)$ は単調増加となる．
これと$g(t)$の極限値が
\begin{align}
g(t)\xrightarrow{t\to+\infty}0
\end{align}
であることから$g(t)$の増減表は以下のようになる．

% --- g(t) の増減表 ---
\begin{table}[htb]
\centering
\caption{$g(t)$ の増減表 ($1 \le t$)}
\begin{tabular}{c|cccc}
$x$ & $1$ & $\dots$ & $(+\infty)$ \\ \hline
$g'(t)$ &  & $+$ &  \\ \hline
$g(t)$ & $\log(1+x) - \dfrac{2}{x+1}$ & $\nearrow$ & $0$
\end{tabular}
\end{table}

従って $1\le t$ の時 $g(t)<0$ となる．
これを\cref{eq:4}に代入して$f(t)>0$と合わせて $f'(t)<0$ つまり $f(t)$ は単調減少である．
よって\cref{eq:1}にも注意して増減表は以下のようになる．

\begin{table}[htb]
\centering
\caption{$f(t)$ の増減表 ($1 \le t$)}
\begin{tabular}{c|cccccc}
$t$ & $1$ & $\dots$ & $m$ & $\dots$ & $n$ & $\dots$ \\ \hline
$f'(t)$ & & $-$ & $-$ & $-$ & $-$ & \\ \hline
$f(t)$ & $1+x$ & $\searrow$ & $f(m)$ & $\searrow$ & $f(n)$ & $\searrow$
\end{tabular}
\end{table}

以上から $f(n)<f(m)$ つまり，
\begin{align}
\left(1+\dfrac{x}{m^2}\right)^m>\left(1+\dfrac{x}{n^2}\right)^n
\end{align}
である．
