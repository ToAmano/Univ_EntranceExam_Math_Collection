{\bf [解]}
$n\in\mathbb{Z}_{\ge3}\ \cdots$①，$x,y,z\in\mathbb{N}\ \cdots$②
\begin{align}
\begin{cases}\displaystyle
x+y+z=n \\
x\le y+z \\
y\le z+x \\
z\le x+y
\end{cases}
\quad\cdots *_1
\end{align}
まず，$z=k$と固定する．$*_1$に代入して
\begin{align}
\begin{cases}\displaystyle
x+y=n-k\ge k \\
y\ge x-k \\
y\le x+k
\end{cases}
\quad\cdots *_2
\end{align}
$y=n-k-x$を代入，$y>0$から$x<n-k\ \cdots$③
\begin{align}
\begin{split}
x-k&\le n-k-x\le x+k \\
\dfrac{n}{2}-k&\le x\le\dfrac{n}{2}\quad\cdots\text{④}
\end{split}
\end{align}
又，$1\le k\le\dfrac{n}{2}$である．同区間では$\dfrac{n}{2}\le n-k$である $\cdots$⑤

\subparagraph{⑦ $n=2p\ (p\in\mathbb{N})$の時（$p=2,3,\cdots$）}

$z=k$と固定（$k=1,2,\cdots,p$）した時の$(x,y)$の数は，④⑤より
\begin{align}
\begin{cases}\displaystyle
k=1,2,\cdots,p-1\text{の時} & k+1\text{個} \\
k=p\text{の時} & p-1\text{個}
\end{cases}
\end{align}
より，あわせて
\begin{align}
\sum_{k=1}^{p-1}(k+1)+(p-1)=\left(\dfrac{p(p+1)}{2}-1\right)+(p-1)=\dfrac{1}{2}p^2+\dfrac{3}{2}p-2=\dfrac{1}{8}n^2+\dfrac{3}{4}n-2\ \text{個}
\end{align}

\subparagraph{⑦ $n=2p+1$の時}

$z=k$と固定（$k=1,2,\cdots,p$）した時の$(x,y)$の数は，③④より$k$個で，
\begin{align}
\sum_{k=1}^{p}k=\dfrac{1}{2}p(p+1)=\dfrac{1}{2}\cdot\dfrac{n-1}{2}\cdot\dfrac{n+1}{2}\ \text{個}
\end{align}

したがって，
\begin{align}
\begin{cases}\displaystyle
\dfrac{1}{8}n^2+\dfrac{3}{4}n-2\text{個} & (n:\text{even}) \\[2mm]
\dfrac{1}{2}\cdot\dfrac{n-1}{2}\cdot\dfrac{n+1}{2}\text{個} & (n:\text{odd})
\end{cases}
\end{align}
