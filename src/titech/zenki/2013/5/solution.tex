\paragraph{(1)}

$C_1$が$C_2$に内接するためには
\begin{align}
\min_P|AP|=(C_1\text{の半径})=a \label{eq:1}
\end{align}
であることが必要である．実際に$\overrightarrow{AP}$を計算すると
\begin{align}
|AP|^2
 &=(\cos\theta-a)^2+b^2\sin^2\theta \\
 &=(\cos\theta-a)^2+b^2(1-\cos^2\theta) \\
 &=(1-b^2)\cos^2\theta-2a\cos\theta+(a^2+b^2) \label{eq:2}
\end{align}
である．これを$f(\cos\theta)$とおく．
$a>0$より接点は$x \ge 0$側にあるから，
\begin{align}
 0 \le \cos\theta \le 1 \label{eq:3}
\end{align}
としてよく，このもとで$x=\cos\theta$として
\begin{align}
 f(x) = (1-b^2)x^2 -2ax + (a^2+b^2) \label{eq:4}
\end{align}
の最小値を考えよう．

$f(x)$は高々$2$次式であり，二次係数の大きさで場合分けする．

\subparagraph{$1-b^2=0$すなわち$b=1$の時（$\because b>0$）}

この時は\cref{eq:4}は
\begin{align}
 f(x) = -2ax +a^2
\end{align}
となり，$a>0$よりこれは$[0,1]$で単調減少だから
\begin{align}
 \min f(x) = f(1) = -2a + a^2 + 1 = (a-1)^2 \label{eq:5}
\end{align}
である．

\subparagraph{$1-b^2 > 0$すなわち$0 < b < 1$の時（$\because b>0$）}

この時は$f(x)$は下に凸な二次式であり，\cref{eq:4}を平方完成して
\begin{align}
 f(x) = \left(1-b^2\right)\left(x-\dfrac{a}{1-b^2}\right)^2 -\dfrac{a^2}{1-b^2} +a^2+b^2
\end{align}
である．$a>0$および$1-b^2>0$より軸$\dfrac{a}{1-b^2}$は正だから，この最小値は軸が$[0,1]$に含まれるかどうかを考えて
\begin{align}
 \min f(x) 
 &=
 \begin{cases}
  f\left(\dfrac{a}{1-b^2}\right)  & 0<\dfrac{a}{1-b^2}<1 \\
  f\left(1\right) & 1 \le \dfrac{a}{1-b^2}  
 \end{cases}  \\
 &=
 \begin{cases}
  a^2+b^2 -\dfrac{a^2}{1-b^2} & 0< a <1-b^2 \\
  (a-1)^2 & 1-b^2 \le a 
 \end{cases} \label{eq:6}
\end{align}
である．

\subparagraph{$1-b^2 < 0$すなわち$1 < b$の時（$\because b>0$）}

この時は$f(x)$は上に凸な二次式であり，\cref{eq:4}を平方完成して
\begin{align}
 f(x) = \left(1-b^2\right)\left(x-\dfrac{a}{1-b^2}\right)^2 -\dfrac{a^2}{1-b^2} +a^2+b^2
\end{align}
である．$a>0$および$1-b^2<0$より軸$\dfrac{a}{1-b^2}$は負だから，この$[0,1]$での最小値は常に$f(1)$である．よって
\begin{align}
 \min f(x) = f(1) = (a-1)^2 \label{eq:7}
\end{align}
である．
\casesend

以上\cref{eq:5,eq:6,eq:7}をまとめて
\begin{align}
 \min f(x)
 &=
 \begin{cases}
  a^2+b^2 -\dfrac{a^2}{1-b^2} & 0< a <1-b^2, 0<b<1  \\
  (a-1)^2 & 1-b^2 \le a
 \end{cases} \label{eq:8}
\end{align}
である．よって\cref{eq:1}に代入して求める条件は
\begin{align}
 &\begin{cases}
  a^2+b^2 -\dfrac{a^2}{1-b^2} = a^2 & 0< a <1-b^2, 0<b<1 \\
  (a-1)^2 = a^2 & 1-b^2 \le a
 \end{cases} \\
\therefore
 &\begin{cases}
  a^2 = b^2(1-b^2) & b^2 < \dfrac{1}{2} \\
  a = \dfrac{1}{2} & \dfrac{1}{2} \le b^2
 \end{cases} \label{eq:9}
\end{align}
である．前者の時は接点は$\cos\theta=\dfrac{a}{1-b^2}$となる点，後者は接点$(1,0)$である．
この時，いずれも$C_1$が$C_2$の内側にあり，$C_1$が$C_2$に内接しており十分である．


\paragraph{(2)}

$b=\dfrac{1}{\sqrt{3}}$のとき$b^2=\dfrac{1}{3}\le\dfrac{1}{2}$だから\cref{eq:9}の前者の場合に該当する．この時$a$は\cref{eq:9}より
\begin{align}
 a^2
  &=b^2(1-b^2) \\
  &=\dfrac{1}{3}\cdot\dfrac{2}{3} \\
  &=\dfrac{2}{9} \\
\therefore\ 
 a&=\dfrac{\sqrt{2}}{3} \label{eq:10}
\end{align}
となるから，接点の座標$P(p,q)=(\cos\theta,b\sin\theta)$は
接点の$x$座標は
\begin{align}
 p &=\cos\theta \\
   &= \dfrac{a}{1-b^2} \\
   &=\dfrac{\sqrt{2}/3}{2/3} \\
   &=\dfrac{\sqrt{2}}{2} 
\end{align}
であり，$P$は第一象限にあるので$q>0$だから
\begin{align}
 q &= b\sin\theta \\
   &= b\sqrt{1-\cos^2\theta} \\
   &= b\sqrt{1-p^2}  \\
   &=\dfrac{1}{\sqrt{3}}\sqrt{1-\dfrac{1}{2}} \\
   &=\dfrac{1}{\sqrt{6}} \\
   &=\dfrac{\sqrt{6}}{6}
\end{align}
である．これらをまとめて求める座標は
\begin{align}
 P\left(\dfrac{\sqrt{2}}{2},\dfrac{\sqrt{6}}{6}\right) \label{eq:11}
\end{align}
である．

\paragraph{(3)}

\begin{align}
 T = 2 T' \label{eq:12}
\end{align}

{\bf [別解]}

(3)は極力通常の積分を避けて単純な扇形と三角形の面積のみで計算を実行した．
比較のため通常の積分を用いるとどうなるかを示す．

$C_2$の上半分は$y=b\sqrt{1-x^2}$（$p\le x\le1$），$C_1$の上半分は$y=\sqrt{a^2-(x-a)^2}$（$p\le x\le2a$，$C_1$は$x\le2a$までしか存在しない）．よって$T'$は
\begin{align}
T'=\int_p^1b\sqrt{1-x^2}\,dx-\int_p^{2a}\sqrt{a^2-(x-a)^2}\,dx
\end{align}
である．
第二項は$u=\dfrac{x-a}{a}$と置換すると$a du = dx$より
\begin{align}
T'=b \int_p^1\sqrt{1-x^2}\,dx-a^2 \int_{p/a-1}^{1}\sqrt{1-u^2}\,du
\end{align}
である．ここで$\sqrt{1-x^2}$の原始関数は
\begin{align}
 \int\sqrt{1-x^2}dx=\dfrac{1}{2}\bigl[x\sqrt{1-x^2}+\arcsin x\bigr] \equiv f(x)
\end{align}
だから代入して
\begin{align}
 T'
  &= b\left(f(1) - f(p)\right) - a^2 \left(f(1)-f\left(\dfrac{p}{a}-1\right)\right) \\
  &= (b-a^2)f(1) -bf(p) + a^2f\left(\dfrac{p}{a}-1\right)
\end{align}
である．各項個別に計算すると，$p=\dfrac{\sqrt{2}}{2}$，$\dfrac{p}{a}-1 = \dfrac{3}{2}-1 = \dfrac{1}{2}$より
\begin{align}
 f(1) &= \dfrac{1}{2}\arcsin 1 = \dfrac{\pi}{4} \\
 f(p) &= \dfrac{1}{2}\bigl[\dfrac{\sqrt{2}}{2}\cdot\dfrac{\sqrt{2}}{2}+\dfrac{\pi}{4}\bigr]  \\
      &=\dfrac{1}{4}+\dfrac{\pi}{8} \\
 f\left(\dfrac{p}{a}-1\right) 
      &= \dfrac{1}{2}\bigl[\dfrac{1}{2}\cdot\dfrac{\sqrt{3}}{2}+\dfrac{\pi}{6}\bigr] \\
      &= \dfrac{\sqrt{3}}{8} + \dfrac{\pi}{12}
\end{align}
だから代入して
\begin{align}
 T' 
  &= \left(\dfrac{\sqrt{3}}{3} - \dfrac{2}{9}\right) \dfrac{\pi}{4} - \dfrac{\sqrt{3}}{3}\left[\dfrac{1}{4}+\dfrac{\pi}{8}\right] + \dfrac{2}{9}\left[\dfrac{\sqrt{3}}{8} + \dfrac{\pi}{12}\right] \\
  &= \sqrt{3}\left(\dfrac{\pi}{12}-\dfrac{1}{12} - \dfrac{\pi}{24} + \dfrac{1}{36}\right) + \pi\left(\dfrac{-1}{18}+\dfrac{1}{54}\right) \\
  &= \sqrt{3}\left(\dfrac{\pi}{24} - \dfrac{1}{18}\right) - \dfrac{\pi}{27}
\end{align}
となって本解答と一致する．
