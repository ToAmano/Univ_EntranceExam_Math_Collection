{\bf [解]}
$C=\cos\chi$，$S=\sin\chi$とし，$C_2$上の点$P(C,bS)$（$0\le C\le1$）とおく．また$A(a,0)$（$C_1$の中心）とする．

\textbf{(1)} $C_1$が$C_2$に内接する条件は
\begin{align}
\min_P|AP|=(C_1\text{の半径})=a\tag{①}
\end{align}
である．
\begin{align}
|AP|^2=(C-a)^2+b^2S^2=(C-a)^2+b^2(1-C^2)=(1-b^2)C^2-2aC+(a^2+b^2)\equiv f(C)
\end{align}
とおく（$0\le C\le1$で考えればよい：$a>0$より対称性から接点は$x\ge0$側）．$f'(C)=2(1-b^2)C-2a$であり，$f'(0)=-2a<0$．$f'(1)=2(1-a-b^2)$の正負で場合分けする．

$1^\circ$ $f'(1)\le0\iff1\le a+b^2$のとき：$f'(C)$は$C$の1次式で両端が非正だから区間内で$f'(C)\le0$，すなわち$f$は単調減少で
\begin{align}
\min f(C)=f(1)=(1-b^2)-2a+(a^2+b^2)=(1-a)^2.
\end{align}

$2^\circ$ $f'(1)\ge0\iff1\ge a+b^2$のとき：このとき$1-b^2\ge a>0$だから，$f(C)$は下に凸な$C$の2次関数で，$C=\dfrac{a}{1-b^2}\in(0,1]$で最小となり
\begin{align}
\min f(C)=f\Bigl(\dfrac{a}{1-b^2}\Bigr)=\dfrac{a^2}{1-b^2}-\dfrac{2a^2}{1-b^2}+(a^2+b^2)=(a^2+b^2)-\dfrac{a^2}{1-b^2}.
\end{align}

①（$\min f(C)=a^2$）より，$1^\circ$では$(1-a)^2=a^2$，すなわち$a=\dfrac{1}{2}$，$2^\circ$では$(a^2+b^2)-\dfrac{a^2}{1-b^2}=a^2$，整理して$b^2(1-b^2)=a^2$．これらはいずれも$C_1$が$C_2$の内側にあり接することに対応し，まとめると条件は
\begin{align}
a=\dfrac{1}{2}\ \text{かつ}\ \dfrac{1}{2}\le b^2\quad(\text{接点}(1,0))\qquad\text{または}\qquad b^2\le\dfrac{1}{2}\ \text{かつ}\ a^2=b^2(1-b^2)\quad\Bigl(\text{接点は}C=\dfrac{a}{1-b^2}\text{となる点}\Bigr).
\end{align}

\textbf{(2)} $b=\dfrac{1}{\sqrt{3}}$のとき$b^2=\dfrac{1}{3}\le\dfrac{1}{2}$だから後者の場合に該当し，
\begin{align}
a^2=b^2(1-b^2)=\dfrac{1}{3}\cdot\dfrac{2}{3}=\dfrac{2}{9}\quad\therefore\ a=\dfrac{\sqrt{2}}{3}.
\end{align}
接点の$x$座標は
\begin{align}
p=\dfrac{a}{1-b^2}=\dfrac{\sqrt{2}/3}{2/3}=\dfrac{\sqrt{2}}{2}
\end{align}
であり，$q=b\sqrt{1-p^2}=\dfrac{1}{\sqrt{3}}\sqrt{1-\dfrac{1}{2}}=\dfrac{1}{\sqrt{3}}\cdot\dfrac{1}{\sqrt{2}}=\dfrac{1}{\sqrt{6}}=\dfrac{\sqrt{6}}{6}$（第1象限より$q>0$）．よって
\begin{align}
(p,q)=\Bigl(\dfrac{\sqrt{2}}{2},\ \dfrac{\sqrt{6}}{6}\Bigr).
\end{align}

\textbf{(3)} $x\ge p$の範囲で$C_1$と$C_2$に囲まれた部分の面積を$T$とする．$y$軸に関する対称性（$x\ge p>0$の範囲は上下対称）から，上半分（$y\ge0$）の面積を$T'$として$T=2T'$．

$C_2$の上半分は$y=b\sqrt{1-x^2}$（$p\le x\le1$），$C_1$の上半分は$y=\sqrt{a^2-(x-a)^2}$（$p\le x\le2a$，$C_1$は$x\le2a$までしか存在しない）．よって
\begin{align}
T'=\int_p^1b\sqrt{1-x^2}\,dx-\int_p^{2a}\sqrt{a^2-(x-a)^2}\,dx.
\end{align}
第1項：$\displaystyle\int\sqrt{1-x^2}dx=\dfrac{1}{2}\bigl[x\sqrt{1-x^2}+\arcsin x\bigr]$を用いて，$x=1$で$\dfrac{1}{2}\cdot\dfrac{\pi}{2}=\dfrac{\pi}{4}$，$x=p=\dfrac{\sqrt{2}}{2}$で$\sqrt{1-p^2}=\dfrac{\sqrt{2}}{2}$だから$\dfrac{1}{2}\bigl[\dfrac{\sqrt{2}}{2}\cdot\dfrac{\sqrt{2}}{2}+\dfrac{\pi}{4}\bigr]=\dfrac{1}{2}\bigl[\dfrac{1}{2}+\dfrac{\pi}{4}\bigr]=\dfrac{1}{4}+\dfrac{\pi}{8}$．よって
\begin{align}
\int_p^1\sqrt{1-x^2}dx=\dfrac{\pi}{4}-\Bigl(\dfrac{1}{4}+\dfrac{\pi}{8}\Bigr)=\dfrac{\pi}{8}-\dfrac{1}{4},\qquad
b\int_p^1\sqrt{1-x^2}dx=\dfrac{1}{\sqrt{3}}\Bigl(\dfrac{\pi}{8}-\dfrac{1}{4}\Bigr)=\dfrac{\sqrt{3}}{24}\pi-\dfrac{\sqrt{3}}{12}.
\end{align}
第2項：$u=x-a$と置換すると，$x=p$で$u=p-a=\dfrac{\sqrt{2}}{2}-\dfrac{\sqrt{2}}{3}=\dfrac{\sqrt{2}}{6}$，$x=2a$で$u=a$．$\displaystyle\int\sqrt{a^2-u^2}du=\dfrac{1}{2}\bigl[u\sqrt{a^2-u^2}+a^2\arcsin\dfrac{u}{a}\bigr]$を用いる．$u=a$で$\dfrac{1}{2}\cdot a^2\cdot\dfrac{\pi}{2}=\dfrac{a^2\pi}{4}$．$u=\dfrac{\sqrt{2}}{6}$のとき，$\sqrt{a^2-u^2}=q=\dfrac{\sqrt{6}}{6}$（接点の$y$座標）であり，$\dfrac{u}{a}=\dfrac{\sqrt{2}/6}{\sqrt{2}/3}=\dfrac{1}{2}$より$\arcsin\dfrac{1}{2}=\dfrac{\pi}{6}$だから
\begin{align}
\dfrac{1}{2}\Bigl[\dfrac{\sqrt{2}}{6}\cdot\dfrac{\sqrt{6}}{6}+a^2\cdot\dfrac{\pi}{6}\Bigr]=\dfrac{1}{2}\Bigl[\dfrac{\sqrt{12}}{36}+\dfrac{2}{9}\cdot\dfrac{\pi}{6}\Bigr]=\dfrac{1}{2}\Bigl[\dfrac{\sqrt{3}}{18}+\dfrac{\pi}{27}\Bigr]=\dfrac{\sqrt{3}}{36}+\dfrac{\pi}{54}.
\end{align}
よって
\begin{align}
\int_p^{2a}\sqrt{a^2-(x-a)^2}dx=\dfrac{a^2\pi}{4}-\Bigl(\dfrac{\sqrt{3}}{36}+\dfrac{\pi}{54}\Bigr)=\dfrac{2}{9}\cdot\dfrac{\pi}{4}-\dfrac{\sqrt{3}}{36}-\dfrac{\pi}{54}=\dfrac{\pi}{18}-\dfrac{\sqrt{3}}{36}-\dfrac{\pi}{54}=\dfrac{\pi}{27}-\dfrac{\sqrt{3}}{36}
\end{align}
（$\pi/18-\pi/54=3\pi/54-\pi/54=2\pi/54=\pi/27$を用いた）．以上より
\begin{align}
T'=\Bigl(\dfrac{\sqrt{3}}{24}\pi-\dfrac{\sqrt{3}}{12}\Bigr)-\Bigl(\dfrac{\pi}{27}-\dfrac{\sqrt{3}}{36}\Bigr)=\dfrac{\sqrt{3}}{24}\pi-\dfrac{\sqrt{3}}{12}+\dfrac{\sqrt{3}}{36}-\dfrac{\pi}{27}=\Bigl(\dfrac{\pi}{24}-\dfrac{1}{18}\Bigr)\sqrt{3}-\dfrac{\pi}{27}
\end{align}
（$-\sqrt{3}/12+\sqrt{3}/36=-3\sqrt{3}/36+\sqrt{3}/36=-2\sqrt{3}/36=-\sqrt{3}/18$を用いた）．よって
\begin{align}
T=2T'=\Bigl(\dfrac{\pi}{12}-\dfrac{1}{9}\Bigr)\sqrt{3}-\dfrac{2\pi}{27}.
\end{align}
