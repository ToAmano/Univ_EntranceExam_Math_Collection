{\bf [解]}
\paragraph{(1)}

全ての組み合わせ${}_nP_2=n(n-1)$通りについて，引いた2枚の積の和$S_2$は
\begin{align}
S_2 
 &= \Pi_{i\neq j} ij \\
 &= \Pi_{ij} ij - \Pi_{i=j} ij \\
 &= (1+2+\cdots + n)^2 - \left(1^2+2^2+\cdots+n^2\right) \\
 &=\dfrac{1}{4}n^2(n+1)^2-\dfrac{1}{6}n(n+1)(2n+1)
\end{align}
である．従って求める期待値は
\begin{align}
\begin{split}
E
&=\dfrac{S_2}{{}_nP_2} \\
&=\dfrac{1}{{}_nP_2}\left[\dfrac{1}{4}n^2(n+1)^2-\dfrac{1}{6}n(n+1)(2n+1)\right] \\
&=\dfrac{1}{n-1}\left[\dfrac{1}{4}n(n+1)^2-\dfrac{1}{6}(n+1)(2n+1)\right] \\
&=\dfrac{(n+1)(3n^2-n-2)}{12(n-1)} \\
&=\dfrac{(n+1)(3n+2)}{12}
\end{split}
\end{align}
である．

\paragraph{(2)}
全ての組み合わせ${}_nP_3=n(n-1)(n-2)$通りについての引いた3枚の積$S_3$は
\begin{align}
S_3 
&=\Pi_{i\neq j, i\neq k, j\neq k} ijk \\
&=\Pi_{ijk} ijk - \Pi_{i=j} ijk - \Pi_{i=k}ijk - \Pi_{j=k}ijk + 2\Pi_{i=j=k}ijk \label{eq:1}
\end{align}
である．各項計算すると
\begin{align}
 \Pi_{ijk} ijk  &= (1+2+\cdots+n)^3 = \left\{\dfrac{1}{2}n(n+1)\right\}^3 \\
 \Pi_{i=j} ijk  &= (1^2+2^2+\cdots+n^2)(1+2+\cdots+n) = \dfrac{1}{2}n(n+1)\cdot\dfrac{1}{6}n(n+1)(2n+1) \\
 \Pi_{i=j=k}ijk &= \left\{\dfrac{1}{2}n(n+1)\right\}^2
\end{align}
だから\cref{eq:1}に代入して
\begin{align}
 S_3 
 &=\left\{\dfrac{1}{2}n(n+1)\right\}^3-3\cdot\dfrac{1}{2}n(n+1)\cdot\dfrac{1}{6}n(n+1)(2n+1)+2\left\{\dfrac{1}{2}n(n+1)\right\}^2 \\
 &=\left\{\dfrac{1}{2}n(n+1)\right\}^2\left[\dfrac{1}{2}n(n+1)-(2n+1)+2\right] \\
 &=\left\{\dfrac{1}{2}n(n+1)\right\}^2\left[\dfrac{1}{2}n^2-\dfrac{3}{2}n+1\right] \\
 &=\dfrac{1}{2}\left\{\dfrac{1}{2}n(n+1)\right\}^2\left(n-1\right)\left(n-2\right)
\end{align}
だから，求める期待値は
\begin{align}
 E(n) 
 &= \dfrac{S_3}{{}_nP_3} \\
 &= \dfrac{1}{2}\dfrac{\left\{\dfrac{1}{2}n(n+1)\right\}^2\left(n-1\right)\left(n-2\right)}{n(n-1)(n-2)} \\
 &= \dfrac{1}{8n}\left\{n(n+1)\right\}^2 
\end{align}
である．従って求める極限値は
\begin{align}
 \dfrac{E(n)}{n^3} 
  &= \dfrac{1}{8n^4}\left\{n(n+1)\right\}^2 \\
  &= \dfrac{1}{8}\left(1+\dfrac{1}{n}\right)^2 \\
  &\to \dfrac{1}{8} \quad(n\to\infty)
\end{align}
である．
