{\bf [解]}
$n\ge3\ \cdots$①

\textbf{(1)} 全ての場合$nP_2$通りについての引いた2枚の積の和は（1枚目，2枚目で区別）
\begin{align}
\begin{split}
1\cdot(2+\cdots+n)+2(1+3+\cdots+n)+\cdots+n(1+2+\cdots+n-1) \\
&=(1+2+\cdots+n)^2-(1^2+2^2+\cdots+n^2) \\
&=\dfrac{1}{4}n^2(n+1)^2-\dfrac{1}{6}n(n+1)(2n+1)
\end{split}
\end{align}
だから，
\begin{align}
\begin{split}
E&=\dfrac{1}{{}_nP_2}\left[\dfrac{1}{4}n^2(n+1)^2-\dfrac{1}{6}n(n+1)(2n+1)\right] \\
&=\dfrac{1}{n-1}\left[\dfrac{1}{4}n(n+1)^2-\dfrac{1}{6}(n+1)(2n+1)\right] \\
&=\dfrac{(n+1)(3n^2-n-2)}{12(n-1)}=\dfrac{(n+1)(3n+2)}{12}
\end{split}
\end{align}

\textbf{(2)} 全ての場合$nP_3$通りについての引いた3枚の積は
\begin{align}
\begin{split}
(1+2+\cdots+n)^3-{}_3C_2(1^2+2^2+\cdots+n^2)(1+2+\cdots+n)+2(1^3+2^3+\cdots+n^3) \\
&=\left\{\dfrac{1}{2}n(n+1)\right\}^3-3\cdot\dfrac{1}{2}n(n+1)\cdot\dfrac{1}{6}n(n+1)(2n+1)+2\left\{\dfrac{1}{2}n(n+1)\right\}^2 \\
&=\left\{\dfrac{1}{2}n(n+1)\right\}^2\left[\dfrac{1}{2}n(n+1)-(2n+1)+2\right]
\end{split}
\end{align}
だから，
\begin{align}
\begin{split}
\dfrac{E(n)}{n^3}&=\dfrac{\dfrac{1}{4}n^2(n+1)^2\left(\dfrac{1}{2}n^2-\dfrac{3}{2}n+1\right)}{n^4(n-1)(n-2)} \\
&=\dfrac{1}{4}\left(1+\dfrac{1}{n}\right)^2\cdot\dfrac{(n-1)(n-2)}{2(n-1)(n-2)}\longrightarrow\dfrac{1}{8} \quad(n\to\infty)
\end{split}
\end{align}
