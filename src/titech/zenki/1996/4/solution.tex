{\bf [解]}
$x\ge0$で$f'(x)>0\ \cdots\bigstar$，$f(0)=a\ (a>1)$

$P$での接線$\ell: y=f'(t)(x-t)+f(t)$，法線$m: y=-\dfrac{1}{f'(t)}(x-t)+f(t)$だから，
\begin{align}
Q\left(-\dfrac{f(t)}{f'(t)}+t,0\right), \qquad R(f(t)f'(t)+t,0)
\end{align}
とおける．$f(x)$は単調増加なことから，
\begin{align}
F(t)=f(t)f'(t)+\dfrac{f(t)}{f'(t)}
\end{align}
したがって，(ハ)から
\begin{align}
f'(t)+\dfrac{1}{f'(t)}=\dfrac{f(t)}{f'(t)} \quad\cdots\text{①}
\end{align}

\textbf{(1)} ①から
\begin{align}
\{f'(t)\}^2=f(t)-1 \quad\cdots\text{☆}
\end{align}
両辺$t$で微分して，
\begin{align}
2f'(t)f''(t)=f'(t)
\end{align}
\begin{align}
f''(t)=\dfrac{1}{2} \quad(\because f'(t)>0) \quad\cdots\text{②}
\end{align}
したがって，$f'(t)$は単調増加で，②の両辺積分して
\begin{align}
f'(t)=\dfrac{1}{2}t+C \quad\cdots\text{③}
\end{align}
とおける．$\bigstar$から，$f'(0)=C>0\ \cdots$④である．③の両辺積分して
\begin{align}
f(t)=\dfrac{1}{4}t^2+Ct+a \quad\cdots\text{⑤}
\end{align}
とおける（$\bigstar$，$f(0)=a$）．③，⑤を☆に代入
\begin{align}
\left(\dfrac{1}{2}t+C\right)^2=\dfrac{1}{4}t^2+Ct+a-1
\end{align}
\begin{align}
C^2=a-1
\end{align}
④から，$C=\sqrt{a-1}$である．$f(x)$の$[x,x+h]$に平均値の定理を適用して（微分可能かつ連続），
\begin{align}
f(x+h)-f(x)=f'(p)\cdot h
\end{align}
となる$p\ (x<p<x+h)$がある．②とあわせて
\begin{align}
f(x+h)-f(x)\ge\sqrt{a-1}\cdot h
\end{align}

\textbf{(2)} (1)から，$f(t)=\dfrac{1}{4}t^2+\sqrt{a-1}\,t+a$である．$F(t)=\dfrac{\{f(t)\}^2}{f'(t)}$から
\begin{align}
F'(t)=\dfrac{2f(t)\{f'(t)\}^2-f''(t)\{f(t)\}^2}{\{f'(t)\}^2}=\dfrac{f(t)}{\{f'(t)\}^2}\left\{2\{f'(t)\}^2-f(t)f''(t)\right\}
\end{align}
の符号は$t\ge0$の時，$\{\ \}$部の符号に等しい．
\begin{align}
\{\ \}=2\left\{\dfrac{1}{2}t+\sqrt{a-1}\right\}^2-\left(\dfrac{1}{4}t^2+\sqrt{a-1}\,t+a\right)\cdot\dfrac{1}{2}
\end{align}
\begin{align}
=2\left[\dfrac{1}{4}t^2+\sqrt{a-1}\cdot t+(a-1)\right]-\dfrac{1}{8}t^2-\dfrac{1}{2}\sqrt{a-1}\cdot t-\dfrac{1}{2}a
\end{align}
\begin{align}
=\dfrac{3}{8}t^2+\dfrac{3}{2}\sqrt{a-1}\cdot t+\dfrac{3}{2}a-2\equiv g(t)
\end{align}
$g(t)=0$の2解を$\alpha,\beta$として，下表を得る（$\alpha<\beta$）．

\textbf{$1^\circ$ $\alpha<\beta\le0$つまり$\dfrac{4}{3}\le a$の時}

$t\ge0$で$F(t)$は単調増加で，$\min F(t)=F(0)=\dfrac{a^2}{\sqrt{a-1}}$

\textbf{$2^\circ$ $\alpha<0\le\beta$つまり$1<a\le\dfrac{4}{3}$の時}
\begin{center}
\begin{tabular}{c|ccccc}
$t$ & $0$ & & $\beta$ & & \\
\hline
$F'$ & & $-$ & $0$ & $+$ & \\
\hline
$F$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
$g(\beta)=0$と☆から，$f'(\beta)^2=1/3$，$f(\beta)=4f'(\beta)^2=4/3$となり，
\begin{align}
\min F(t)=F(\beta)=\dfrac{16}{9}\sqrt{3}
\end{align}
以上まとめて，
\begin{align}
\min F(t)=
\begin{cases}
\dfrac{16}{9}\sqrt{3} & \left(1<a\le\dfrac{4}{3}\right) \\
\dfrac{a^2}{\sqrt{a-1}} & \left(\dfrac{4}{3}\le a\right)
\end{cases}
\end{align}
