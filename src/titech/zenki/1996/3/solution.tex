\textbf{(1)} $0<\alpha<\beta$の時，グラフの概形は右図（$p\ne0\ \cdots$①）．又，
\begin{align*}
f'(x)=7px^6(x-\alpha)(x-\beta)+px^7\{(x-\beta)+(x-\alpha)\}
\end{align*}
\begin{align*}
=px^6\left\{9x^2-8(\alpha+\beta)x+7\alpha\beta\right\} \quad\cdots\text{②}
\end{align*}
である．$f'(1)=0$から②に代入して
\begin{align*}
p\left\{9-8(\alpha+\beta)+7\alpha\beta\right\}=0
\end{align*}
$p\ne0$から
\begin{align*}
9-8(\alpha+\beta)+7\alpha\beta=0 \quad\cdots\text{③}
\end{align*}
$f(1)=1$から，
\begin{align*}
p(1-\alpha)(1-\beta)=1 \quad\cdots\text{④}
\end{align*}
題意から$\displaystyle\int_0^\beta f(x)dx=0$で，
\begin{align*}
\int_0^\beta f(x)dx=p\int_0^\beta\left\{x^9-(\alpha+\beta)x^8+\alpha\beta x^7\right\}dx=p\left[\frac1{10}x^{10}-\frac19(\alpha+\beta)x^9+\frac{\alpha\beta}{8}x^8\right]_0^\beta
\end{align*}
\begin{align*}
=p\left(\frac1{10}\beta^{10}-\frac19(\alpha+\beta)\beta^9+\frac{\alpha\beta}{8}\beta^8\right)=0
\end{align*}
$p\beta\ne0$から
\begin{align*}
\frac1{10}\beta-\frac19(\alpha+\beta)+\frac18\alpha=0 \quad\therefore\ \beta=\frac{90}{72}\alpha=\frac54\alpha \quad\cdots\text{⑤}
\end{align*}
⑤を③に代入して，
\begin{align*}
(5\alpha-6)(7\alpha-6)=0 \quad\therefore\ (\alpha,\beta)=\left(\frac65,\frac32\right),\left(\frac67,\frac{15}{14}\right)
\end{align*}
④から，順に$p=10,-98$である．以上から
\begin{align*}
(\alpha,\beta,p)=\left(\frac65,\frac32,10\right),\left(\frac67,\frac{15}{14},-98\right)
\end{align*}
が必要．逆にこの時，$x=1$で極大値1をとることを確認する．

\textbf{(2)} $\alpha<0<\beta$の時，グラフは右図（$p\ne0$）で，$x=1$で極大なことから，
\begin{align*}
p<0, \qquad 0<\beta<1 \quad\cdots\text{⑧}
\end{align*}
が必要である．題意から③④は成立し，さらに
\begin{align*}
\int_\alpha^\beta f(x)dx=0
\end{align*}
\begin{align*}
\left[\frac1{10}x^{10}-\frac19(\alpha+\beta)x^9+\frac{\alpha\beta}8x^8\right]_\alpha^\beta=0 \quad(\because②)
\end{align*}
\begin{align*}
\frac1{10}(\beta^{10}-\alpha^{10})-\frac19(\alpha+\beta)(\beta^9-\alpha^9)+\frac{\alpha\beta}8(\beta^8-\alpha^8)=0
\end{align*}
$t=\dfrac\alpha\beta$とおく（$t<0$）と，
\begin{align*}
-4t^{10}+4+5t^9-5t=0
\end{align*}
\begin{align*}
4(1-t^{10})+5t(t^8-1)=0
\end{align*}
$-1<t<0$の時，左辺は正，$t<-1$の時，左辺は負でいずれも不適で，$t=-1$のみ解．この時$\beta=-\alpha$で，③④に代入して，
\begin{align*}
9-7\alpha^2=0, \qquad p(1-\alpha^2)=1
\end{align*}
\begin{align*}
\therefore\ (\alpha,\beta,p)=\left(-\frac{3\sqrt7}7,\frac{3\sqrt7}7,-\frac72\right) \quad(\alpha<0<\beta\text{を確認})
\end{align*}
逆にこの時，$f'(x)=px^6\{9x^2-9\}$となり，下表をうる．
\begin{center}
\begin{tabular}{c|ccccccc}
$x$ & & $-1$ & & $0$ & & $1$ & \\
\hline
$f'$ & $-$ & $0$ & $+$ & $0$ & $+$ & $0$ & $-$ \\
\hline
$f$ & $\searrow$ & & $\nearrow$ & & $\nearrow$ & & $\searrow$
\end{tabular}
\end{center}
よって$x=1$で極大となり十分．以上から
\begin{align*}
f(x)=-\frac72x^7\left(x^2-\frac97\right)
\end{align*}
