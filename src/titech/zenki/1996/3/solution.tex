{\bf [解]}
\textbf{(1)} $0<\alpha<\beta$の時，グラフの概形は右図（$p\ne0\ \cdots$①）．又，
\begin{align}
\begin{split}
f'(x)&=7px^6(x-\alpha)(x-\beta)+px^7\{(x-\beta)+(x-\alpha)\} \\
&=px^6\left\{9x^2-8(\alpha+\beta)x+7\alpha\beta\right\} \quad\cdots\text{②}
\end{split}
\end{align}
である．$f'(1)=0$から②に代入して
\begin{align}
p\left\{9-8(\alpha+\beta)+7\alpha\beta\right\}=0
\end{align}
$p\ne0$から
\begin{align}
9-8(\alpha+\beta)+7\alpha\beta=0 \quad\cdots\text{③}
\end{align}
$f(1)=1$から，
\begin{align}
p(1-\alpha)(1-\beta)=1 \quad\cdots\text{④}
\end{align}
題意から$\displaystyle\int_0^\beta f(x)dx=0$で，
\begin{align}
\begin{split}
\int_0^\beta f(x)dx&=p\int_0^\beta\left\{x^9-(\alpha+\beta)x^8+\alpha\beta x^7\right\}dx=p\left[\dfrac{1}{10}x^{10}-\dfrac{1}{9}(\alpha+\beta)x^9+\dfrac{\alpha\beta}{8}x^8\right]_0^\beta \\
&=p\left(\dfrac{1}{10}\beta^{10}-\dfrac{1}{9}(\alpha+\beta)\beta^9+\dfrac{\alpha\beta}{8}\beta^8\right)=0
\end{split}
\end{align}
$p\beta\ne0$から
\begin{align}
\dfrac{1}{10}\beta-\dfrac{1}{9}(\alpha+\beta)+\dfrac{1}{8}\alpha=0 \quad\therefore\ \beta=\dfrac{90}{72}\alpha=\dfrac{5}{4}\alpha \quad\cdots\text{⑤}
\end{align}
⑤を③に代入して，
\begin{align}
(5\alpha-6)(7\alpha-6)=0 \quad\therefore\ (\alpha,\beta)=\left(\dfrac{6}{5},\dfrac{3}{2}\right),\left(\dfrac{6}{7},\dfrac{15}{14}\right)
\end{align}
④から，順に$p=10,-98$である．以上から
\begin{align}
(\alpha,\beta,p)=\left(\dfrac{6}{5},\dfrac{3}{2},10\right),\left(\dfrac{6}{7},\dfrac{15}{14},-98\right)
\end{align}
が必要．逆にこの時，$x=1$で極大値1をとることを確認する．

\textbf{(2)} $\alpha<0<\beta$の時，グラフは右図（$p\ne0$）で，$x=1$で極大なことから，
\begin{align}
p<0, \qquad 0<\beta<1 \quad\cdots\text{⑧}
\end{align}
が必要である．題意から③④は成立し，さらに
\begin{align}
\begin{split}
\int_\alpha^\beta f(x)dx&=0 \\
\left[\dfrac{1}{10}x^{10}-\dfrac{1}{9}(\alpha+\beta)x^9+\dfrac{\alpha\beta}{8}x^8\right]_\alpha^\beta&=0 \quad(\because②) \\
\dfrac{1}{10}(\beta^{10}-\alpha^{10})-\dfrac{1}{9}(\alpha+\beta)(\beta^9-\alpha^9)+\dfrac{\alpha\beta}{8}(\beta^8-\alpha^8)&=0
\end{split}
\end{align}
$t=\dfrac{\alpha}{\beta}$とおく（$t<0$）と，
\begin{align}
\begin{split}
-4t^{10}+4+5t^9-5t&=0 \\
4(1-t^{10})+5t(t^8-1)&=0
\end{split}
\end{align}
$-1<t<0$の時，左辺は正，$t<-1$の時，左辺は負でいずれも不適で，$t=-1$のみ解．この時$\beta=-\alpha$で，③④に代入して，
\begin{align}
\begin{split}
9-7\alpha^2&=0, \qquad p(1-\alpha^2)=1 \\
&\therefore\ (\alpha,\beta,p)=\left(-\dfrac{3\sqrt{7}}{7},\dfrac{3\sqrt{7}}{7},-\dfrac{7}{2}\right) \quad(\alpha<0<\beta\text{を確認})
\end{split}
\end{align}
逆にこの時，$f'(x)=px^6\{9x^2-9\}$となり，下表をうる．
\begin{table}[htb]
  \centering
\begin{tabular}{c|ccccccc}
$x$ & & $-1$ & & $0$ & & $1$ & \\
\hline
$f'$ & $-$ & $0$ & $+$ & $0$ & $+$ & $0$ & $-$ \\
\hline
$f$ & $\searrow$ & & $\nearrow$ & & $\nearrow$ & & $\searrow$
\end{tabular}
\end{table}
よって$x=1$で極大となり十分．以上から
\begin{align}
f(x)=-\dfrac{7}{2}x^7\left(x^2-\dfrac{9}{7}\right)
\end{align}
