{\bf [解]}
\textbf{(1)} $Q(\alpha,\alpha^2)$，$R(\beta,\beta^2)$（$\alpha<\beta$）とおいて良い．この時$Q,R$での接線は各々
\begin{align}
y=2\alpha x-\alpha^2, \qquad y=2\beta x-\beta^2
\end{align}
だから，$P$はこの交点で，$\left(\dfrac{\alpha+\beta}{2},\alpha\beta\right)$である．$P$は$y=x^2$の下側だから，
\begin{align}
\alpha\beta<\dfrac{1}{4}(\alpha+\beta)^2 \quad\therefore\ (\alpha-\beta)^2>0
\end{align}
これは$\alpha<\beta$から常にみたされる．そこで，$\alpha,\beta$が，$A=\alpha+\beta,\ b=\alpha\beta$とした時，実数解条件及び題意から
\begin{align}
\begin{split}
b&\le\dfrac{1}{2}A-1,\ -\dfrac{1}{2}A+1 \quad\cdots\text{①} \\
-1&\le b,\quad b\le\dfrac{1}{4}A^2
\end{split}
\end{align}
をみたしながら動く時のQRの中点$M(X,Y)$の軌跡を求めれば良い．
\begin{align}
\begin{split}
X&=\dfrac{1}{2}(\alpha+\beta)=\dfrac{1}{2}A \quad\cdots\text{②} \\
Y&=\dfrac{1}{2}(\alpha^2+\beta^2)=\dfrac{1}{2}(A^2-2b)
\end{split}
\end{align}
であり，①を図示すると右図斜線部（境界含む）だから，
\begin{align}
\begin{split}
-1&\le b\le\dfrac{1}{2}A-1 \quad(0\le A\le2) \quad\cdots\text{③} \\
-1&\le b\le-\dfrac{1}{2}A+1 \quad(2\le A\le4)
\end{split}
\end{align}
②③から
\begin{align}
\begin{split}
\dfrac{1}{2}A^2-\dfrac{1}{2}A+1&\le Y\le\dfrac{1}{2}A^2+1 \quad(0\le A\le2) \quad\cdots\text{④} \\
\dfrac{1}{2}A^2+\dfrac{1}{2}A-1&\le Y\le\dfrac{1}{2}A^2+1 \quad(2\le A\le4)
\end{split}
\end{align}
④に②を代入して
\begin{align}
\begin{split}
2X^2-X+1&\le Y\le2X^2+1 \quad(0\le X\le1) \\
2X^2+X-1&\le Y\le2X^2+1 \quad(1\le X\le2)
\end{split}
\end{align}
これがもとめる領域で，図示して下図斜線部．

\textbf{(2)}
\begin{align}
\begin{split}
\overrightarrow{PQ}&=\left(\alpha-\dfrac{\alpha+\beta}{2},\ \alpha^2-\alpha\beta\right)=\left(\dfrac{\alpha-\beta}{2},\ \alpha(\alpha-\beta)\right) \\
\overrightarrow{PR}&=\left(\beta-\dfrac{\alpha+\beta}{2},\ \beta^2-\alpha\beta\right)=\left(\dfrac{\beta-\alpha}{2},\ \beta(\beta-\alpha)\right)
\end{split}
\end{align}
だから，サラスの公式から，
\begin{align}
\triangle PQR=\dfrac{1}{2}\left|\dfrac{1}{2}(\alpha-\beta)\beta(\beta-\alpha)-\dfrac{1}{2}(\beta-\alpha)\alpha(\alpha-\beta)\right|=\dfrac{1}{4}(\beta-\alpha)^3 \quad(\because\beta>\alpha) \quad\cdots\text{⑤}
\end{align}
ここで，$\beta,\alpha$は$t$の2次式$t^2-at+b=0$の2実解で$t=\dfrac{a\pm\sqrt{a^2-4b}}{2}$でかつ$\alpha<\beta$から
\begin{align}
\beta-\alpha=\sqrt{a^2-4b} \quad\cdots\text{⑥}
\end{align}
なので，⑥を⑤に代入して，
\begin{align}
\triangle PQR=\dfrac{1}{4}(a^2-4b)^{\frac{3}{2}}
\end{align}
$\triangle PQR=2$の時，
\begin{align}
8=(a^2-4b)^{\frac{3}{2}} \iff 4=a^2-4b \quad(\because a^2-4b>0)
\end{align}
だから，$P(X,Y)$について$X=\dfrac{1}{2}a,\ Y=b$だから，
\begin{align}
Y=X^2-1
\end{align}
である．
