{\bf [解]}
\begin{align}
\begin{array}{ccc}
k-1 & \xrightarrow{1-\dfrac{k-1}{n}} & k \\
r-1 & & r \\
r & \xrightarrow{\dfrac{r}{n}} & r
\end{array}
\end{align}

\paragraph{(1)}
$S_k=r$となるのは，$S_{k-1}=r$でかつ$k$回目に今までと同じカードを引くか，$S_{k-1}=r-1$でかつ$k$回目に新しいカードを引くかのどちらかだから，
\begin{align}
P(S_k=r)=\dfrac{n+1-r}{n}P(S_{k-1}=r-1)+\dfrac{r}{n}P(S_{k-1}=r) \quad(r=1,2,\cdots,k\text{の時}, P(S_0=r)=0\text{と扱う})
\end{align}
である．


\paragraph{(2)}

確率変数$X_t$を，
\begin{align}
X_t=
\begin{cases}\displaystyle
0 & (t\text{のカードを}k\text{回目まで引かなかった}) \\
1 & (t\text{のカードを}k\text{回目までに引いた})
\end{cases}
\end{align}
と定めると$S_k=\displaystyle\sum_{t=1}^nX_t$だから，期待値の加法性および$X_t$に関する対称性から
\begin{align}
E(S_k)=E\left(\sum_{t=1}^nX_t\right)=\sum_{t=1}^nE(X_t)=nE(X_1) \quad\cdots\text{①} \label{eq:1}
\end{align}
である．$k$回目までに1のカードを引く確率$P(k, X_1=1)$は排反事象をかんがえて，
\begin{align}
P(k, X_1=1)=1-\left(\dfrac{n-1}{n}\right)^k
\end{align}
であり，また期待値の定義から$k$回目までの$X_1$の期待値は
\begin{align}
 E(k, X_1)=\sum_{X_1}X_1 P(k,X_1) = 0\cdot P(k,X_1=0)+1\cdot P(k,X_1=1) = 1-\left(\dfrac{n-1}{n}\right)^k
\end{align}
だから，\cref{eq:1}に代入して求める$S_k$の期待値は
\begin{align}
  E_k=n\left[1-\left(\dfrac{n-1}{n}\right)^k\right]
\end{align}
である．

{\bf [別解]}

\paragraph{(2)}
解答では誘導を使わずに直接求めたが，(1)の誘導に乗ると以下のようになる．

(1)の漸化式の両辺$r$をかけて$r$について和をとると
\begin{align}
\sum_{r=1}^krP(S_k=r)=\sum_{r=1}^k\dfrac{r^2}{n}P(S_{k-1}=r)+\sum_{r=1}^k\dfrac{nr+r(r-1)}{n}P(S_{k-1}=r-1)
\end{align}
を得る．$P(S_{k-1}=k)=0$だから右辺第一項の和の上端は$k-1$で書き換えられて
\begin{align}
\begin{split}
E_k
 &=\sum_{r=1}^{k-1}\dfrac{r^2}{n}P(S_{k-1}=r)+\sum_{r=1}^k\dfrac{(n-1)(r-1)+n-(r-1)^2}{n}P(S_{k-1}=r-1) \\
 &=\sum_{r=1}^{k-1}\dfrac{r^2}{n}P(S_{k-1}=r)+\dfrac{n-1}{n}E_{k-1}+\sum_{r=1}^kP(S_{k-1}=r-1)-\sum_{r=1}^k\dfrac{(r-1)^2}{n}P(S_{k-1}=r-1)
\end{split}
\end{align}
最初の和と最後の和は（添字を$j=r-1$とおきかえると）互いに打ち消しあう．また確率の定義より第3項は$\displaystyle\sum_{r=1}^kP(S_{k-1}=r-1)=1$であるから，結局
\begin{align}
E_k=\dfrac{n-1}{n}E_{k-1}+1
\end{align}
とかける．$E_1=1$と合わせてこの漸化式をとくと
\begin{align}
E_k=n\left[1-\left(\dfrac{n-1}{n}\right)^k\right]
\end{align}
となる．
