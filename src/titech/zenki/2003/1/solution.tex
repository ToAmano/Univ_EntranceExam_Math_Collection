題意の三次関数を
\begin{align}
 C:y=f(x)=-x^3+ax^2+bx \quad (a>0) \label{eq:1}
\end{align}
とする．原点を通る直線を
\begin{align}
 y=mx \quad (m\in\mathbb R) \label{eq:2}
\end{align}
とおいて，これが$C$と2点を共有するとする．
すると\cref{eq:1,eq:2}を連立した$x$の方程式がちょうど二つの異なる実数解を持つ：
\begin{align}
 & -x^3+ax^2+bx=mx \\
 & x^3-ax^2+(m-b)x=0 \\
 & x\left\{x^2-ax+(m-b)\right\}=0 \quad\cdots\text{①} \label{eq:3}
\end{align}
これは$x=0$を必ず解を持つので，残りの二次方程式
\begin{align}
 x^2-ax+(m-b) = 0 \label{eq:4}
\end{align}
が$x=0$と$x\neq 0$に解を持つか，あるいは$x\neq 0$に重解を持つかのいずれかである．

\subparagraph{$1^\circ$ $x^2-ax+m-b=0$が$x=0$を解に持つ時：}

\cref{eq:4}に$x=0$を代入して
\begin{align}
 m = b
\end{align}
である．これを\cref{eq:4}に代入して
\begin{align}
 &x^2-ax=0 \\
 &x(x-a)=0 \\
 & x = 0, a
\end{align}
である．ここで\cref{eq:1}より$a>0$だから\cref{eq:3}の全体の解は$\{0,a\}$（$x=0$は重解）となりちょうど2点を共有する．

\subparagraph{$2^\circ$ $x^2-ax+m-b=0$が$x\ne0$に重解を持つ時：} 

まず\cref{eq:4}の判別式$D$が$0$であるから
\begin{align}
 a^2-4(m-b)=0 \quad \therefore m=\dfrac{1}{4}a^2+b
\end{align}
である．この時の重解は$x=\dfrac{a}{2}\ne0$（$a>0$）である．
よって\cref{eq:3}の解は$\{0,a/2\}$（$a/2$が重解）となりちょうど2点を共有する．
\casesend

以上から，もとめる2本の直線は
\begin{align}
y=bx,\qquad y=\left(\dfrac{1}{4}a^2+b\right)x
\end{align}
である．

\paragraph{(2)}

$\dfrac{1}{4}a^2>0$（$a>0$）だから，傾きの大きい方は
\begin{align}
 \ell_1: y=\ell_1(x) = \left(\dfrac{1}{4}a^2+b\right)x
\end{align}
であり，小さい方は
\begin{align}
 \ell_2: y=\ell_2(x) = bx
\end{align}
である．$S_1,S_2$を図示すると下図の斜線部となる．


$\ell_2$と$C$の交点は$x=0$（重解，接点）と$x=a$だから二つの関数の差は
\begin{align}
 f(x) - \ell_2(x) = x^2(a-x)
\end{align}
であり，$[0,a]$で符号は正で一定だから$S_2$は
\begin{align}
S_2
  &= \int_0^a \left(f(x) - \ell_2(x)\right)\,dx \\
  &= \int_0^a x^2(a-x)\,dx \\
  &= \dfrac{1}{12}a^4 \label{eq:5}
\end{align}
である．

同様に$\ell_1$と$C$の交点は$x=0$と$x=a/2$（重解，接点）だから二つの関数の差は
\begin{align}
 \ell_1(x) - f(x) = x\left(x-\dfrac{a}{2}\right)^2
\end{align}
であり，$[0,a/2]$で符号は正で一定だから$S_1$は
\begin{align}
S_1
 &=\int_0^{a/2}\left(\ell_1(x) - f(x)\right) dx \\
 &=\int_0^{a/2}x\left(x-\dfrac{a}{2}\right)^2dx \\
 &=\dfrac{1}{12}\left(\dfrac{a}{2}\right)^4 \\
 &=\dfrac{1}{192}a^4 \label{eq:6}
\end{align}
である．

したがって\cref{eq:5,eq:6}から求める面積の比は
\begin{align}
S_1:S_2=\dfrac{1}{192}:\dfrac{1}{12}=1:16
\end{align}
である．
