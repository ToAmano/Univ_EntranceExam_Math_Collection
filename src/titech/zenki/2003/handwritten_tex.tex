\section*{第1問}

\begin{proof}[解]
$C:y=f(x)=-x^3+ax^2+bx$（$a>0$）とし，$y=mx$（$m\in\mathbb R$）が$C$と2点を共有するとする。この時
\[
-x^3+ax^2+bx=mx \iff x^3-ax^2+(m-b)x=0 \iff x\left\{x^2-ax+(m-b)\right\}=0 \quad\cdots\text{①}
\]

\textbf{(1)} \textbf{$1^\circ$ $x^2-ax+m-b=0$が$x=0$を解に持つ時：} 代入して$m=b$。この時方程式は$x^2-ax=0$すなわち$x(x-a)=0$となり，残りの解は$x=a\ne0$（$a>0$）。①全体の解は$\{0,a\}$（$x=0$は重解），ちょうど2点を共有する。

\textbf{$2^\circ$ $x^2-ax+m-b=0$が$x\ne0$に重解を持つ時：} 判別式$=0$：$a^2-4(m-b)=0$，すなわち$m=\dfrac14a^2+b$。この時の重解は$x=\dfrac a2\ne0$（$a>0$）。①全体の解は$\{0,a/2\}$（$a/2$が重解），ちょうど2点を共有する。

以上から，もとめる2本の直線は
\[
y=bx,\qquad y=\left(\frac14a^2+b\right)x
\]

\textbf{(2)} $\dfrac14a^2>0$（$a>0$）だから，傾きの大きい方は$\ell_1:y=\left(\dfrac14a^2+b\right)x$，小さい方は$\ell_2:y=bx$。

$\ell_2$と$C$の交点は$x=0$（重解，接点）と$x=a$。$C$と$\ell_2$の差
\[
f(x)-bx=-x^3+ax^2=x^2(a-x)
\]
は$[0,a]$で符号一定だから
\[
S_2=\left|\int_0^a x^2(a-x)\,dx\right|=\left|a\cdot\frac{a^3}3-\frac{a^4}4\right|=\frac1{12}a^4
\]

$\ell_1$と$C$の交点は$x=0$と$x=a/2$（重解，接点）。$\ell_1$と$C$の差
\[
\left(\frac14a^2+b\right)x-f(x)=x^3-ax^2+\frac14a^2x=x\left(x-\frac a2\right)^2
\]
は$[0,a/2]$で符号一定だから
\[
S_1=\left|\int_0^{a/2}x\left(x-\frac a2\right)^2dx\right|=\frac1{12}\left(\frac a2\right)^4=\frac1{192}a^4
\]

したがって
\[
S_1:S_2=\frac1{192}:\frac1{12}=1:16
\]
\end{proof}

\section*{第2問}

\begin{proof}[解]
縦横比$1:a$（$a>1$）の長方形から正方形を切り取る操作は，2辺の長さ$(p,q)$（$p<q$）に対し，$q$から$p$を繰り返し引き（$p$の倍数を引ききって）残りが$p$未満になったところで止める，という減法の互除法（連分数展開）そのものである。実際，$a=3$の時：$(1,3)\to(1,2)\to(1,1)$（2回），$a=3/2$の時：$(1,3/2)\to(1,1/2)\to(1/2,1/2)$（2回）となり，題意と一致する。

\textbf{(1)} $n=3$回で終わる$a$を，直接この操作をシミュレートして求める。

$a=4$：$(1,4)\to(1,3)\to(1,2)\to(1,1)$。3回で終了。

$a=5/2$：$(1,5/2)\to(1,3/2)\to(1,1/2)\to(1/2,1/2)$。3回で終了。

$a=5/3$：$(1,5/3)\to(1,2/3)\to(2/3,1/3)\to(1/3,1/3)$。3回で終了。

$a=4/3$：$(1,4/3)\to(1,1/3)\to(1/3,2/3)\to(1/3,1/3)$。3回で終了。

これら以外の$a$では，同様のシミュレートにより2回以下または4回以上で終わることが確かめられる。以上から，もとめる$a$の値は
\[
a=\frac43,\ \frac53,\ \frac52,\ 4
\]

\textbf{(2)} \textbf{$a$が最大になる場合：} 常に長さ$1$の辺に沿って正方形を取り除き続ける場合（$a$が整数の場合）で，$(1,a)\to(1,a-1)\to\cdots\to(1,1)$と$a-1$回で終わる。これが$n$回になるのは$a-1=n$，すなわち
\[
a=n+1
\]

\textbf{$a$が最小になる場合：} 逆に，操作のたびに辺の長さの差が小さくなる度合いを最小限に抑える，すなわち最初の長さ$a$の辺（長い方）から，$n$回目の直前まで一貫して正方形を取り除き続ける（長さ$1$の辺は最初の1回でほぼ使い切り，以後は毎回「現在の短い方の辺」を基準に僅かずつ削っていく最も遅い進み方）とき，$a$は最小となる。この時，$n$回目の操作後の2辺の比は
\[
(a-1):\{1-(n-1)(a-1)\}
\]
となり，これが$1:1$（正方形＝終了）に等しくなるとき
\[
1-(n-1)(a-1)=a-1 \iff 1=n(a-1) \iff a=1+\frac1n
\]

したがって，$n$回の操作で終わる$a$の最大値は$n+1$，最小値は$1+\dfrac1n$である。

（検算：$n=3$の場合，最大値$n+1=4$，最小値$1+1/3=4/3$は，(1)で求めた4つの値の最大・最小と一致する。）
\end{proof}

\section*{第3問}

\begin{proof}[解]
$A,B,C$の位置ベクトルを$\vec a,\vec b,\vec c$とする。$P$は$AB$を$x:(1-x)$に内分するから$\vec p=(1-x)\vec a+x\vec b$，$Q$は$AC$を$y:(1-y)$に内分するから$\vec q=(1-y)\vec a+y\vec c$（$0\le x,y<1$）。

$R$は直線$BQ$と直線$CP$の交点。$\vec r=t(1-x)\vec a+tx\vec b+(1-t)\vec c$（直線$CP$上，パラメータ$t$）とおき，直線$BQ$上の点としての表示 $\vec r=s(1-y)\vec a+(1-s)\vec b+sy\vec c$ と係数比較すると
\[
tx=1-s,\qquad 1-t=sy
\]
から$s=1-tx$を代入して$1-t=(1-tx)y$，整理して$t=\dfrac{1-y}{1-xy}$。このとき$R$の重心座標（$A,B,C$に関する係数）は
\[
\left(\frac{(1-x)(1-y)}{1-xy},\ \frac{x(1-y)}{1-xy},\ \frac{y(1-x)}{1-xy}\right)
\]
（和が$1$であることは容易に確かめられる）。

$M,N$はそれぞれ$AB,AC$の中点だから，重心座標で$M=(1/2,1/2,0)$，$N=(1/2,0,1/2)$，$A=(1,0,0)$。$R=\lambda A+\mu M+\nu N$（$\lambda+\mu+\nu=1$）とおくと，$R$の$B,C$成分から$\mu=2\cdot\dfrac{x(1-y)}{1-xy}$，$\nu=2\cdot\dfrac{y(1-x)}{1-xy}$であり，$R\in\triangle AMN$（境界を含む）となる条件は$\lambda,\mu,\nu\ge0$。$\mu,\nu\ge0$は$0\le x,y<1$から自動的にみたされるので，本質的な条件は$\lambda\ge0$，すなわち$R$の$A$成分が$1/2$以上であること：
\[
\frac{(1-x)(1-y)}{1-xy}\ge\frac12
\]
（$1-xy>0$に注意して）これは
\[
2(1-x)(1-y)\ge1-xy \iff 1-2x-2y+3xy\ge0 \iff y(2-3x)\le1-2x \quad\cdots\text{②}
\]
と同値。

$0\le x<1,\ 0\le y<1$，②をみたす$(x,y)$の領域を考える。$x=1/2$で右辺$=0$，$2-3x=1/2>0$だから$y\le0$，すなわち$y=0$のみ。$1/2<x<2/3$では右辺$1-2x<0$かつ$2-3x>0$だから$y\le(\text{負})$となり不適（$y\ge0$と両立しない）。$x\ge2/3$も同様に不適。よって，領域は$0\le x\le1/2$，$0\le y\le\dfrac{1-2x}{2-3x}$（境界を含む）である。

この面積$S$は
\[
S=\int_0^{1/2}\frac{1-2x}{2-3x}\,dx
\]
$\dfrac{1-2x}{2-3x}=\dfrac23+\dfrac1{3(3x-2)}$（実際，$\frac23+\frac1{3(3x-2)}=\frac{2(3x-2)+1}{3(3x-2)}=\frac{6x-3}{3(3x-2)}=\frac{2x-1}{3x-2}=\frac{1-2x}{2-3x}$）だから
\[
S=\int_0^{1/2}\left(\frac23+\frac1{3(3x-2)}\right)dx=\left[\frac23x+\frac19\log|3x-2|\right]_0^{1/2}
\]
\[
=\left(\frac13+\frac19\log\frac12\right)-\frac19\log2=\frac13+\frac19\left(\log\frac12-\log2\right)=\frac13+\frac19\log\frac14=\frac13-\frac29\log2
\]
\end{proof}

\section*{第4問}

\begin{proof}[解]
$f_1(x)=x^2$，$f_{n+1}(x)=f_n(x)+x^3f_n^{(2)}(x)$ ($f_n^{(k)}$は$k$次導関数) $\cdots$①

\textbf{(1)} $f_n(x)$が$(n+1)$次多項式であり，その$x^{n+1}$の係数を$a_n$とかけることを帰納的に示す。$n=1$の時は$a_1=1$として成立するので，以下$n=k$での成立を仮定する。①から，$f_k(x)=a_kx^{k+1}+\cdots$（$k+1$次）なら，$f_k^{(2)}(x)$は$(k-1)$次で最高次の項は$k(k+1)a_kx^{k-1}$，したがって$x^3f_k^{(2)}(x)$の最高次は$k(k+1)a_kx^{k+2}$。これは$f_k(x)$（$k+1$次）より高次だから，$f_{k+1}(x)=f_k(x)+x^3f_k^{(2)}(x)$は$k+2=(k+1)+1$次であり，最高次係数は
\[
a_{k+1}=k(k+1)a_k
\]
よって$n=k+1$でも成立する。以上から，$f_n(x)$は$(n+1)$次多項式である。

漸化式$a_{k+1}=k(k+1)a_k$（$a_1=1$）をくり返し用いて
\[
a_n=\prod_{k=1}^{n-1}k(k+1)=\left(\prod_{k=1}^{n-1}k\right)\left(\prod_{k=1}^{n-1}(k+1)\right)=(n-1)!\cdot n!=\frac{(n!)^2}n
\]

\textbf{(2)} $f_n(x)$の$x,x^2,x^3,x^4$の係数をそれぞれ$b_n,c_n,d_n,e_n$とおくと，$f_n^{(k)}(0)=k!\times(x^k\text{の係数})$である。$f_1(x)=x^2$から
\[
b_1=d_1=e_1=0,\qquad c_1=1 \quad\cdots\text{②}
\]

①の右辺$x^3f_n^{(2)}(x)$において，$f_n(x)=\sum_jp_jx^j$なら$f_n^{(2)}(x)=\sum_jp_jj(j-1)x^{j-2}$，$x^3f_n^{(2)}(x)=\sum_jp_jj(j-1)x^{j+1}$だから，$x^m$の係数は$p_{m-1}(m-1)(m-2)$（$f_n(x)$の$x^{m-1}$の係数の$(m-1)(m-2)$倍）。$m=1,2,3,4$に適用して
\[
b_{n+1}=b_n,\qquad c_{n+1}=c_n,\qquad d_{n+1}=d_n+2c_n,\qquad e_{n+1}=e_n+6d_n \quad\cdots\text{③}
\]
（$m=1$：$(0)(-1)=0$，$m=2$：$(1)(0)=0$，$m=3$：$(2)(1)=2$，$m=4$：$(3)(2)=6$。）

②③から
\[
b_n=0,\qquad c_n=1,\qquad d_n=2(n-1),\qquad e_n=6(n-1)(n-2) \quad\cdots\text{④}
\]
（$d_n$は初項$0$，公差$2c_n=2$の等差数列。$e_n$は$e_{n+1}-e_n=6d_n=12(n-1)$の累積和：$e_n=\sum_{j=1}^{n-1}12(j-1)=12\cdot\dfrac{(n-2)(n-1)}2=6(n-1)(n-2)$。）

$f_n(x)=\cdots+e_nx^4+d_nx^3+c_nx^2+b_nx+A$だから
\[
f_n^{(1)}(x)=\cdots+4e_nx^3+3d_nx^2+2c_nx+b_n,\quad f_n^{(2)}(x)=\cdots+12e_nx^2+6d_nx+2c_n,
\]
\[
f_n^{(3)}(x)=\cdots+24e_nx+6d_n,\qquad f_n^{(4)}(x)=\cdots+24e_n
\]
④から
\[
f_n^{(1)}(0)=b_n=0,\qquad f_n^{(2)}(0)=2c_n=2,
\]
\[
f_n^{(3)}(0)=6d_n=12(n-1),\qquad f_n^{(4)}(0)=24e_n=144(n-1)(n-2)
\]
\end{proof}
