円$O_k\ (k=1,2,\cdots,5)$の半径を$r_k$とすると題意から
\begin{align}
r_1=1, \qquad r_2=a, \qquad r_3=r_4=\dfrac{1-a}{2} \label{eq:1}
\end{align}
である．
また$\angle DAB=\theta$とおく（$0<\theta<\pi/2\ \cdots$①）．
$O_3$と$O_4$の接点を$E$とすると$E$は$AD, BC$の交点であることに注意する．


\paragraph{(1)}
四角形ABDCの面積は
\begin{align}
 S(a) = |AD||BE| \label{eq:2}
\end{align}
であるから，以下$|AD|, |BE|$を$a$を用いて表す．
そのために$r_5$が必要なのでまずはこれを求めよう．
$\triangle ABE, \triangle BDE$にピタゴラスの定理を用いて
\begin{align}
AE&=\sqrt{|AB|^2-|BE|^2}=\sqrt{(r_2+r_3)^2-r_3^2}=\sqrt{\left(a+\dfrac{1-a}{2}\right)^2-\left(\dfrac{1-a}{2}\right)^2}=\sqrt{a} \label{eq:3} \\
DE&=\sqrt{|BD|^2-|BE|^2}=\sqrt{(r_3+r_4)^2-r_3^2}=\sqrt{\left(r_5+\dfrac{1-a}{2}\right)^2-\left(\dfrac{1-a}{2}\right)^2}=\sqrt{r_5^2+(1-a)r_5} \label{eq:4}
\end{align}
である．$r_1=1$を$r_k$を用いて表すと
\begin{align}
 r_1 = 1 = |AE|+|DE|+r_5
\end{align}
だからここに\cref{eq:3,eq:4}を代入して
\begin{align}
1&=\sqrt{a}+\sqrt{r_5^2+(1-a)r_5}+r_5 \\
(1-\sqrt{a})-r_5&=\sqrt{r_5^2+(1-a)r_5} 
\end{align}
だから，
\begin{align}
 (1-\sqrt{a})-r_5\ge 0 \label{eq:5}
\end{align}
の元で両辺二乗して
\begin{align}
  & r_5^2-2(1-\sqrt{a})r_5+(1-\sqrt{a})^2=r_5^2+(1-a)r_5  \\
\end{align}
だから整理して
 \begin{align}
  r_5 
    & =\dfrac{(1-\sqrt{a})^2}{2(1-\sqrt{a})+(1-a)} \\
    & =\dfrac{(1-\sqrt{a})^2}{3-a-2\sqrt{a}} \\
    & =\dfrac{(1-\sqrt{a})^2}{(3+\sqrt{a})(1-\sqrt{a})} \\
    & =\dfrac{1-\sqrt{a}}{3+\sqrt{a}} \label{eq:6}
\end{align}
となる．これを\cref{eq:5}に代入すると$0<a<1$より
\begin{align}
 (1-\sqrt{a})-r_5= (1-\sqrt{a})+\dfrac{1-\sqrt{a}}{3+\sqrt{a}}=\dfrac{(1-\sqrt{a})(\sqrt{a}+2)}{\sqrt{a}+3} \ge 0
\end{align}
であり，よって\cref{eq:6}の$r_5$は\cref{eq:5}を満たして十分である．

このもとで\cref{eq:4}に代入して，\cref{eq:1}に注意すると
\begin{align}
 DE 
   &=\sqrt{\dfrac{1-\sqrt{a}}{\sqrt{a}+3}\left(\dfrac{1-\sqrt{a}}{\sqrt{a}+3}+1-\sqrt{a}\right)} \\
   &=\sqrt{\dfrac{(1-\sqrt{a})^2}{(\sqrt{a}+3)^2}(\sqrt{a}+2)^2} \\
   &=\dfrac{(1-\sqrt{a})(\sqrt{a}+2)}{\sqrt{a}+3} 
\end{align}
だから\cref{eq:2}に代入すると
\begin{align}
\begin{split}
S(a)
  &=AD\cdot BE \\
  &=(AE+ED)\cdot BE \\
  &=\left(AE+ED\right)\cdot r_3 \\
  &=\left(\sqrt{a}+\dfrac{(1-\sqrt{a})(\sqrt{a}+2)}{\sqrt{a}+3}\right)\cdot\dfrac{1-a}{2} \\
  &=\dfrac{(\sqrt{a}+1)(1-a)}{\sqrt{a}+3} 
\end{split}\label{eq:7}
\end{align}
が求める面積である．


\paragraph{(2)}
簡単のため
\begin{align}
 p = \sqrt{a}+3
\end{align}
とおくと，\cref{eq:1}の元で
\begin{align}
 3 < p < 4 \label{eq:8}
\end{align}
である．\cref{eq:7}の$S$の表示と合わせて，以下\cref{eq:8}の元で
\begin{align}
 S(a) = f(p) = \dfrac{(p-2)^2(4-p)}{p}
\end{align}
の最大値を求めれば良い．
$f(p)$の一階微分を計算すると
\begin{align}
  \dfrac{df}{dp}&=\dfrac{p[2(p-2)(4-p)-(p-2)^2]-(p-2)^2(4-p)}{p^2}=\dfrac{-2(p-2)(p^2-2p-4)}{p^2}
\end{align}
だから増減表は以下のようになる．

\begin{table}[htb]
  \centering
\begin{tabular}{c|ccccc}
$p$ & $3$ & & $1+\sqrt{5}$ & & $4$ \\
\hline
$f'$ & & $+$ & $0$ & $-$ & \\
\hline
$f$ & & $\nearrow$ & & $\searrow$ &
\end{tabular}
\end{table}
したがって$f$すなわち$S$は$p=1+\sqrt{5}$で最大値をとり，その時の値は
\begin{align}
 \max S = f(p=1+\sqrt{5}) = 10\sqrt{5}-22
\end{align}
である．
