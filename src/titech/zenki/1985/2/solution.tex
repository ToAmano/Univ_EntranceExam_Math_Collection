{\bf [解]}
\textbf{(1)} $t$にまって，切断面は変化する直方体の頂点を通る．$\pi(t): x+\dfrac{1}{2}y+\dfrac{1}{3}z=t$とおく．$V$の体積は$T=6$である $\cdots$①

\textbf{$1^\circ$ $0<t\le1$の時}

切断立体のうち，三角錐のものの体積は
\begin{align}
\dfrac{1}{3}\cdot t\cdot\left(\dfrac{1}{2}\cdot t\cdot 2t\cdot\dfrac{3}{2}\right)=t^3
\end{align}
であり，これは$t\le3/2$より小さいから，$f(t)=t^3$

\textbf{$2^\circ$ $1\le t\le2$の時}

切断立体のうち，原点$O$を含む方の体積は，コーナーの三角錐を3つ分あわせて切り取ることになり，
\begin{align}
t^3-3(t-1)^3=-2t^3+9t^2-9t+3\equiv g(t)
\end{align}
である．$g'(t)=-6(t^2-3t+2)=-6(t-2)(t-1)\ge0\ (1\le t\le2)$だから$g(t)$は非減少で，$g(3/2)=3$だから，
\begin{align}
f(t)=
\begin{cases}
g(t) & \left(1\le t\le\dfrac{3}{2}\right) \\
6-g(t) & \left(\dfrac{3}{2}\le t\le2\right)
\end{cases}
\end{align}

\textbf{$3^\circ$ $2\le t<3$の時}

対称性から，$f(t)=(3-t)^3$

以上から
\begin{align}
f(t)=
\begin{cases}
t^3 & (0\le t\le1) \\
-2t^3+9t^2-9t+3 & \left(1\le t\le\dfrac{3}{2}\right) \\
2t^3-9t^2+9t+3 & \left(\dfrac{3}{2}\le t\le2\right) \\
(3-t)^3 & (2\le t\le3)
\end{cases}
\end{align}

\textbf{(2)} $h(t)=t^3,\ P(t)=6-g(t)$とおく．以下$h\to0$と扱う．$f(t)$は各区間で連続である．

\textbf{⑦ $t=1$の時}
\begin{align}
\dfrac{f(1+h)-f(1)}{h}\xrightarrow{h\to0}g'(1)=3, \qquad \dfrac{f(1-h)-f(1)}{-h}\xrightarrow{h\to0}f'(1)=3
\end{align}
より左右極限が一致し，微分可能．

\textbf{⑦ $t=3/2$の時}
\begin{align}
\dfrac{f(3/2+h)-f(3/2)}{h}\xrightarrow{h\to0}P'(3/2)=-\dfrac{9}{2}, \qquad \dfrac{f(3/2-h)-f(3/2)}{-h}\xrightarrow{h\to0}g'(3/2)=+\dfrac{9}{2}
\end{align}
より左右極限が一致せず，微分不可．

以上から
\begin{align}
t=1:\ \text{possible}, \qquad t=\dfrac{3}{2}:\ \text{impossible}
\end{align}
