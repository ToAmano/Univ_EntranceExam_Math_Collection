{\bf [解]}

\paragraph{(1)}

 $w\in\mathbb{R} \iff w=\bar w$ だから $w=az^2+bz$ を代入して
\begin{align}
\begin{split}
 & az^2+bz=a\bar z^2+b\bar z \\
 & a(z^2-\bar z^2)+b(z-\bar z)=0 \\
 & \{a(z+\bar z)+b\}(z-\bar z)=0 \\
&\iff z+\bar z=-\dfrac{b}{a}\ \text{or}\ z\in\mathbb{R} \\
&\iff x=-\dfrac{b}{2a}\ \text{or}\ y=0
\end{split}
\end{align}
が求める集合である．

\paragraph{(2)}

$y=0$ の時，$w$ は
\begin{align}
 w = f(x)=ax^2+bx \label{eq:1}
\end{align}
で表される．

次に $x=-\dfrac{b}{2a}$ の時，
\begin{align}
\begin{split}
 w &= az^2+bz \\
   &= a\left(-\dfrac{b}{2a}+iy\right)^2+b\left(-\dfrac{b}{2a}+iy\right) \\
   &= a\left(\dfrac{b^2}{4a^2}-\dfrac{by}{a}i-y^2\right)+byi-\dfrac{b^2}{2a} \\
   &= -ay^2-\dfrac{b^2}{4a} \equiv g(y) \label{eq:2}
\end{split}
\end{align}
である．

従って\cref{eq:1,eq:2}をまとめて
\begin{align}
 w =
 \begin{cases}
  f(x) = ax^2+bx = a\left(x+\dfrac{b}{2a}\right)^2 - \dfrac{b^2}{4a}\\
  g(y) = -ay^2-\dfrac{b^2}{4a} 
 \end{cases}
\end{align}
である．

\subparagraph{$1^\circ\ a>0$ の時}

$f(x)\ge-\dfrac{b^2}{4a}$，$g(y)\le-\dfrac{b^2}{4a}$ でこの間の値はくまなくとるから，$w$ は任意の実数値をとる．

\subparagraph{$2^\circ\ a<0$ の時}

$f(x)\le-\dfrac{b^2}{4a}$，$g(y)\ge-\dfrac{b^2}{4a}$ で $1^\circ$ と同様に $w$ は任意の実数値をとる．
\casesend

以上から題意は示された．
