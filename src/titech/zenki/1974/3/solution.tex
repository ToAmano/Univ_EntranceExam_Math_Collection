{\bf [解]}
\begin{enumerate}
  \item[(1)] $w\in\mathbb{R} \iff w=\bar w$ だから $w=az^2+bz$ を代入して
  \begin{align}
\begin{split}
az^2+bz&=a\bar z^2+b\bar z
  \iff a(z^2-\bar z^2)+b(z-\bar z)=0
  \iff \{a(z+\bar z)+b\}(z-\bar z)=0 \\
&\iff z+\bar z=-\dfrac{b}{a}\ \text{or}\ z\in\mathbb{R}
  \iff x=-\dfrac{b}{2a}\ \text{or}\ y=0
\end{split}
\end{align}

  \item[(2)] $y=0$ の時，$w$ は $f(x)=ax^2+bx$ で表される． $\cdots$①

  次に $x=-\dfrac{b}{2a}$ の時，
  \begin{align}
\begin{split}
az^2+bz &= a\left(-\dfrac{b}{2a}+iy\right)^2+b\left(-\dfrac{b}{2a}+iy\right)
  = a\left(\dfrac{b^2}{4a^2}-\dfrac{by}{a}i-y^2\right)+byi-\dfrac{b^2}{2a} \\
&= -ay^2-\dfrac{b^2}{4a} \equiv g(y) \quad \cdots \text{②}
\end{split}
\end{align}
  である．

  \subparagraph{$1^\circ\ a>0$ の時}

  $f(x)\ge-\dfrac{b^2}{4a}$，$g(y)\le-\dfrac{b^2}{4a}$ でこの間の値はくまなくとるから，$w$ は任意の実数値をとる．

  \subparagraph{$2^\circ\ a<0$ の時}

  $f(x)\le-\dfrac{b^2}{4a}$，$g(y)\ge-\dfrac{b^2}{4a}$ で $1^\circ$ と同様に $w$ は任意の実数値をとる．

  
\casesend以上から示された．
\end{enumerate}
