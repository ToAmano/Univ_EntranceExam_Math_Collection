\begin{proof}[解]
\textbf{(1)} $F(x)=\displaystyle\int_0^{mx}f(\sin x)g(\cos x)dx$（左辺），$G(x)=m\displaystyle\int_0^\pi f(\sin x)g(\cos x)dx$（右辺）とおく。$I_k=\displaystyle\int_{(k-1)\pi}^{k\pi}f(\sin x)g(\cos x)dx$（$k\in\mathbb N$）とおき，$t=x-(k-1)\pi$とおくと
\[
I_k=\int_0^\pi f\bigl(\sin(t+(k-1)\pi)\bigr)g\bigl(\cos(t+(k-1)\pi)\bigr)dt
\]
$\sin(t+(k-1)\pi)=(-1)^{k-1}\sin t$，$\cos(t+(k-1)\pi)=(-1)^{k-1}\cos t$であり，$f,g$は偶関数だから
\[
I_k=\int_0^\pi f(\sin t)g(\cos t)\,dt
\]
（$k$によらず一定）。$k=1,\dots,m$について足して
\[
F(\pi)=\sum_{k=1}^mI_k=m\int_0^\pi f(\sin t)g(\cos t)dt=G(\pi)
\]
（一般に$mx$を$m\pi$と読み替えて同じ議論を用いれば良い）。

\textbf{(2)} $D=\displaystyle\int_0^\pi\frac{|\sin x|}{(1+\cos^2x)^2}dx$とおく。$[0,\pi]$で$\sin x\ge0$だから$D=\displaystyle\int_0^\pi\frac{\sin x}{(1+\cos^2x)^2}dx$。$f(x)=|x|$，$g(x)=\dfrac1{(1+x^2)^2}$（ともに偶関数）として，(1)から
\[
\int_0^{m\pi}f(\sin x)g(\cos x)\,dx=mD \quad\cdots\text{②}
\]
$P_t=\displaystyle\int_0^tf(\sin x)g(\cos x)dx$とおくと，$f,g\ge0$だから$P_t\ge0$で$t$について単調増加。②から$P_{m\pi}=mD$，$P_{(m+1)\pi}=(m+1)D$である。

$B=\displaystyle\int_0^1\frac{|\sin nx|}{(1+\cos^2nx)^2}dx$について，$t=nx$とおくと
\[
B=\frac1n\int_0^nf(\sin t)g(\cos t)dt=\frac1nP_n
\]
$m\pi\le n<(m+1)\pi$と$P_t$の単調性から$P_{m\pi}\le P_n\le P_{(m+1)\pi}$，すなわち$mD\le P_n\le(m+1)D$。したがって
\[
\frac{mD}{(m+1)\pi}\le\frac{P_n}n=B\le\frac{(m+1)D}{m\pi}
\]
（左は$P_n\ge mD$と$n<(m+1)\pi$から，右は$P_n\le(m+1)D$と$n\ge m\pi$から）。これがもとめる不等式である。

\textbf{(3)} $n\to\infty$の時，$m\pi\le n$から$m\to\infty$でもあり，(2)の両辺は$m\to\infty$でともに$D/\pi$に収束する（$m/(m+1)\to1$）。はさみうちの定理から
\[
\lim_{n\to\infty}\int_0^1\frac{|\sin nx|}{(1+\cos^2nx)^2}dx=\frac{D}\pi
\]
$D$を求める。$t=\cos x$とおくと（$dt=-\sin x\,dx$，$x:0\to\pi$で$t:1\to-1$）
\[
D=\int_1^{-1}\frac1{(1+t^2)^2}(-1)\,dt=\int_{-1}^1\frac1{(1+t^2)^2}dt=2\int_0^1\frac1{(1+t^2)^2}dt
\]
$t=\tan\theta$とおくと$dt=d\theta/\cos^2\theta$，$1+t^2=1/\cos^2\theta$，$\theta:0\to\pi/4$で
\[
D=2\int_0^{\pi/4}\cos^4\theta\cdot\frac1{\cos^2\theta}d\theta=2\int_0^{\pi/4}\cos^2\theta\,d\theta=2\left[\frac\theta2+\frac{\sin2\theta}4\right]_0^{\pi/4}=2\left(\frac\pi8+\frac14\right)=\frac\pi4+\frac12
\]
したがって
\[
\lim_{n\to\infty}\int_0^1\frac{|\sin nx|}{(1+\cos^2nx)^2}dx=\frac D\pi=\frac{\pi/4+1/2}\pi=\frac14+\frac1{2\pi}
\]
\end{proof}
