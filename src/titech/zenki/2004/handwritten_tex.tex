\section*{第1問}

\begin{proof}[解]
$a,b>0$とする。

\textbf{(1)} $f(x)=\dfrac{x^4}{(x-a)^3}$は$x>a$で正である。$\log f(x)=4\log x-3\log(x-a)$を$x$で微分して
\[
\frac{f'(x)}{f(x)}=\frac4x-\frac3{x-a}=\frac{x-4a}{x(x-a)}
\]
\[
\therefore\ f'(x)=\frac{x^3(x-4a)}{(x-a)^4}
\]
$x>a$では$x^3>0,\ (x-a)^4>0$だから，$f'(x)$の符号は$(x-4a)$の符号と一致する。
\begin{center}
\begin{tabular}{c|ccccc}
$x$ & $a$ & & $4a$ & & \\
\hline
$f'$ & & $-$ & $0$ & $+$ & \\
\hline
$f$ & & $\searrow$ & & $\nearrow$ &
\end{tabular}
\end{center}
すなわち，$f$は$a<x<4a$で単調減少，$4a<x$で単調増加し，$x=4a$で最小値$f(4a)=\dfrac{(4a)^4}{(3a)^3}=\dfrac{4^4}{3^3}a$をとる。

\textbf{(2)} $g(x)=\dfrac1{(x-a)^2}-\dfrac b{x^3}$，$g'(x)=\dfrac{-2}{(x-a)^3}+\dfrac{3b}{x^4}$。$x$軸に平行な直線と3点で交わるためには，$g'(x)=0$，$a<x$の前後で$g'$が符号を変える点が少なくとも2つあることが必要十分（$*$）。

$x>a$で
\[
g'(x)\ge0 \iff \frac{3b}{x^4}\ge\frac2{(x-a)^3} \iff \frac32b\ge\frac{x^4}{(x-a)^3}=f(x)
\]
(1)から$f$は$x=4a$で最小値$\dfrac{4^4}{3^3}a$をとり，$x\to a^+,\ x\to\infty$で$f(x)\to\infty$。したがって，$y=\dfrac32b$のグラフが$y=f(x)$のグラフと2点で交わり，$(*)$をみたすための条件は
\[
\frac{4^4}{3^3}a<\frac32b \quad\cdots\text{②}
\]
このとき$g'(x)=0$は2解$\alpha,\beta$（$\alpha<\beta$）を持ち，下表を得る。
\begin{center}
\begin{tabular}{c|ccccccc}
$x$ & $a$ & & $\alpha$ & & $\beta$ & & \\
\hline
$g'$ & & $-$ & $0$ & $+$ & $0$ & $-$ & \\
\hline
$g$ & & $\searrow$ & & $\nearrow$ & & $\searrow$ &
\end{tabular}
\end{center}
$g(x)\to+\infty\ (x\to a^+)$，$g(x)\to0\ (x\to+\infty)$だから，$g$のグラフは，$+\infty$から$g(\alpha)$（極小）まで減少，$g(\beta)$（極大）まで増加，その後$0$に向かって減少する形になる。この時，$g(\alpha)<k<g(\beta)$をみたす$k$に対し，直線$y=k$は$g$のグラフとちょうど3点で交わる。$g(\alpha)<g(\beta)$は，まさに$g$が極大・極小を両方持つこと，すなわち②と同値である。

以上から，もとめる必要十分条件は②を整理して
\[
b>\frac{2\cdot4^4}{3^4}a
\]
\end{proof}

\section*{第2問}

\begin{proof}[解]
\textbf{(1)} $F(x)=\displaystyle\int_0^{mx}f(\sin x)g(\cos x)dx$（左辺），$G(x)=m\displaystyle\int_0^\pi f(\sin x)g(\cos x)dx$（右辺）とおく。$I_k=\displaystyle\int_{(k-1)\pi}^{k\pi}f(\sin x)g(\cos x)dx$（$k\in\mathbb N$）とおき，$t=x-(k-1)\pi$とおくと
\[
I_k=\int_0^\pi f\bigl(\sin(t+(k-1)\pi)\bigr)g\bigl(\cos(t+(k-1)\pi)\bigr)dt
\]
$\sin(t+(k-1)\pi)=(-1)^{k-1}\sin t$，$\cos(t+(k-1)\pi)=(-1)^{k-1}\cos t$であり，$f,g$は偶関数だから
\[
I_k=\int_0^\pi f(\sin t)g(\cos t)\,dt
\]
（$k$によらず一定）。$k=1,\dots,m$について足して
\[
F(\pi)=\sum_{k=1}^mI_k=m\int_0^\pi f(\sin t)g(\cos t)dt=G(\pi)
\]
（一般に$mx$を$m\pi$と読み替えて同じ議論を用いれば良い）。

\textbf{(2)} $D=\displaystyle\int_0^\pi\frac{|\sin x|}{(1+\cos^2x)^2}dx$とおく。$[0,\pi]$で$\sin x\ge0$だから$D=\displaystyle\int_0^\pi\frac{\sin x}{(1+\cos^2x)^2}dx$。$f(x)=|x|$，$g(x)=\dfrac1{(1+x^2)^2}$（ともに偶関数）として，(1)から
\[
\int_0^{m\pi}f(\sin x)g(\cos x)\,dx=mD \quad\cdots\text{②}
\]
$P_t=\displaystyle\int_0^tf(\sin x)g(\cos x)dx$とおくと，$f,g\ge0$だから$P_t\ge0$で$t$について単調増加。②から$P_{m\pi}=mD$，$P_{(m+1)\pi}=(m+1)D$である。

$B=\displaystyle\int_0^1\frac{|\sin nx|}{(1+\cos^2nx)^2}dx$について，$t=nx$とおくと
\[
B=\frac1n\int_0^nf(\sin t)g(\cos t)dt=\frac1nP_n
\]
$m\pi\le n<(m+1)\pi$と$P_t$の単調性から$P_{m\pi}\le P_n\le P_{(m+1)\pi}$，すなわち$mD\le P_n\le(m+1)D$。したがって
\[
\frac{mD}{(m+1)\pi}\le\frac{P_n}n=B\le\frac{(m+1)D}{m\pi}
\]
（左は$P_n\ge mD$と$n<(m+1)\pi$から，右は$P_n\le(m+1)D$と$n\ge m\pi$から）。これがもとめる不等式である。

\textbf{(3)} $n\to\infty$の時，$m\pi\le n$から$m\to\infty$でもあり，(2)の両辺は$m\to\infty$でともに$D/\pi$に収束する（$m/(m+1)\to1$）。はさみうちの定理から
\[
\lim_{n\to\infty}\int_0^1\frac{|\sin nx|}{(1+\cos^2nx)^2}dx=\frac{D}\pi
\]
$D$を求める。$t=\cos x$とおくと（$dt=-\sin x\,dx$，$x:0\to\pi$で$t:1\to-1$）
\[
D=\int_1^{-1}\frac1{(1+t^2)^2}(-1)\,dt=\int_{-1}^1\frac1{(1+t^2)^2}dt=2\int_0^1\frac1{(1+t^2)^2}dt
\]
$t=\tan\theta$とおくと$dt=d\theta/\cos^2\theta$，$1+t^2=1/\cos^2\theta$，$\theta:0\to\pi/4$で
\[
D=2\int_0^{\pi/4}\cos^4\theta\cdot\frac1{\cos^2\theta}d\theta=2\int_0^{\pi/4}\cos^2\theta\,d\theta=2\left[\frac\theta2+\frac{\sin2\theta}4\right]_0^{\pi/4}=2\left(\frac\pi8+\frac14\right)=\frac\pi4+\frac12
\]
したがって
\[
\lim_{n\to\infty}\int_0^1\frac{|\sin nx|}{(1+\cos^2nx)^2}dx=\frac D\pi=\frac{\pi/4+1/2}\pi=\frac14+\frac1{2\pi}
\]
\end{proof}

\section*{第3問}

\begin{proof}[解]
コイン$P,Q,R$の表の出る確率を$p,q,r$とする。各回で，まず$P$を投げ，表（確率$p$）なら$Q$を，裏（確率$1-p$）なら$R$を選び，選んだコインの表（$Q$なら確率$q$，$R$なら確率$r$）で赤玉，裏で白玉を入れる。したがって1回の試行で
\[
P(Q\text{選択・赤})=pq,\quad P(Q\text{選択・白})=p(1-q),\quad P(R\text{選択・赤})=(1-p)r,\quad P(R\text{選択・白})=(1-p)(1-r)
\]

\textbf{(1)} $n$回とも$Q$を選び，そのうち$k$回赤（$n-k$回白）となる確率は，どの$k$回が赤かを選ぶ場合の数をかけて
\[
{}_nC_k(pq)^k\{p(1-q)\}^{n-k}={}_nC_k\,p^n\,q^k(1-q)^{n-k}
\]

\textbf{(2)} $n$回とも赤玉が出る（コインの選択によらない）確率を求める。$k$回が「$Q$選択・赤」，残り$n-k$回が「$R$選択・赤」となる確率は${}_nC_k(pq)^k\{(1-p)r\}^{n-k}$であり，これを$k=0,\dots,n$について足すと，二項定理から
\[
\sum_{k=0}^n{}_nC_k(pq)^k\{(1-p)r\}^{n-k}=\{pq+(1-p)r\}^n
\]
がもとめる確率である。

\textbf{(3)} $n=2004$，$p=q=\dfrac12$，$r=\dfrac15$とする。$b_k$を，$n$回の試行の結果，赤玉がちょうど$k$個になる確率とする。赤玉$k$個のうち，どの回が「$Q$選択・赤」でどの回が「$R$選択・赤」かを$\ell$で，白玉$n-k$個のうち「$Q$選択・白」「$R$選択・白」の内訳を同様に足し合わせると，(2)と同じ二項定理の要領で
\[
b_k={}_nC_k\{pq+(1-p)r\}^k\{p(1-q)+(1-p)(1-r)\}^{n-k}
\]
$pq=\dfrac14$，$(1-p)r=\dfrac1{10}$から$pq+(1-p)r=\dfrac14+\dfrac1{10}=\dfrac7{20}$。$p(1-q)=\dfrac14$，$(1-p)(1-r)=\dfrac12\cdot\dfrac45=\dfrac25$から$p(1-q)+(1-p)(1-r)=\dfrac14+\dfrac25=\dfrac{13}{20}$。したがって
\[
b_k={}_nC_k\left(\frac7{20}\right)^k\left(\frac{13}{20}\right)^{n-k}={}_nC_k\left(\frac{13}{20}\right)^n\left(\frac7{13}\right)^k
\]

$b_k$を最大にする$k$をもとめる。
\[
\frac{b_k}{b_{k+1}}=\frac{{}_nC_k(7/13)^k}{{}_nC_{k+1}(7/13)^{k+1}}=\frac{13}7\cdot\frac{{}_nC_k}{{}_nC_{k+1}}=\frac{13}7\cdot\frac{k+1}{n-k}=:f(k)
\]
$f(k)\ge1\iff13(k+1)\ge7(n-k)$（$k\le n-1=2003$，$n-k>0$）。$n=2004$を代入して
\[
13(k+1)\ge7(2004-k)\iff13k+13\ge14028-7k\iff20k\ge14015\iff k\ge700.75
\]
したがって，整数$k$について，$k\ge701$で$f(k)\ge1$（$b_k\ge b_{k+1}$，減少），$k\le700$で$f(k)<1$（$b_k<b_{k+1}$，増加）。よって
\[
b_1<b_2<\cdots<b_{700}<b_{701}>b_{702}>\cdots>b_{2004}
\]
となり，$b_k$は$k=701$で最大となる。すなわち，もっとも起こりやすいのは赤玉が\textbf{701個}の時である。
\end{proof}

\section*{第4問}

\begin{proof}[解]
2球は$x^2+y^2+z^2=r^2$，$(x-1)^2+y^2+z^2=1-r^2$。辺々引くと，交面は$x=r^2$である（$0<r<1$より$0<r^2<r<1$）。

\textbf{(1)} 交わりの立体は，球1（中心$0$，半径$r$）の$x\ge r^2$の部分（球欠）と，球2（中心$1$，半径$\sqrt{1-r^2}$）の$x\le r^2$の部分（球欠）の和である。球$i$を単位球に正規化した時の対応する球欠の体積（$\pi$倍する前）を$V_i$とすると，スケーリングにより
\[
V(r)=r^3V_1+(1-r^2)^{3/2}V_2 \quad\cdots\text{①}
\]
球1の交面$x=r^2$は，半径$r$で正規化すると$x'=r^2/r=r$（単位球上）に対応するので
\[
V_1=\int_r^1(1-x^2)\,dx=\left[x-\frac{x^3}3\right]_r^1=\frac23-r+\frac13r^3
\]
球2の交面$x=r^2$は，中心$1$，半径$d:=\sqrt{1-r^2}$で正規化すると，中心からの距離が$1-r^2=d^2$，正規化座標で$-d$に対応するので，対称性から同じ形で
\[
V_2=\frac23-d+\frac13d^3 \quad(d=\sqrt{1-r^2})
\]
①に代入して（$d^2=1-r^2,\ d^4=(1-r^2)^2,\ d^6=(1-r^2)^3$に注意）
\[
V(r)=r^3\left(\frac23-r+\frac13r^3\right)+d^3\left(\frac23-d+\frac13d^3\right)
\]
\[
=\frac23r^3-r^4+\frac13r^6+\frac23d^3-(1-r^2)^2+\frac13(1-r^2)^3
\]
整理して（$\pi$倍を忘れずに）
\[
V(r)=\left(-r^4+\frac23r^3+r^2-\frac23+\frac23(1-r^2)^{3/2}\right)\pi
\]

\textbf{(2)}
\[
\frac{V'(r)}\pi=-4r^3+2r^2+2r+\frac23\cdot\frac32(1-r^2)^{1/2}\cdot(-2r)=-4r^3+2r^2+2r-2r(1-r^2)^{1/2}
\]
\[
=-2r\left[2r^2-r-1+(1-r^2)^{1/2}\right]
\]
$0<r<1$では$-2r<0$だから，$f(r):=2r^2-r-1+\sqrt{1-r^2}$とおくと，$V'(r)$の符号は$-f(r)$の符号に等しい。

$f(r)\ge0\iff\sqrt{1-r^2}\ge1+r-2r^2$。$0<r<1$で右辺が正であることを確かめた上で両辺を2乗すると
\[
1-r^2\ge(1+r-2r^2)^2=4r^4-4r^2(r+1)+(r+1)^2=4r^4-4r^3-3r^2+2r+1
\]
\[
\iff4r^4-4r^3-2r^2+2r\le0\iff2r(2r^3-2r^2-r+1)\le0\iff2r(r-1)(2r^2-1)\le0
\]
（$2r^3-2r^2-r+1=(r-1)(2r^2-1)$は$r=1$が根であることを確認して因数分解）。$0<r<1$では$2r>0,\ (r-1)<0$だから$2r(r-1)<0$，したがって
\[
2r(r-1)(2r^2-1)\le0\iff2r^2-1\ge0\iff r\ge\frac1{\sqrt2}
\]
すなわち，$f(r)\ge0\iff r\ge1/\sqrt2$。下表を得る。
\begin{center}
\begin{tabular}{c|ccccc}
$r$ & $0$ & & $1/\sqrt2$ & & $1$ \\
\hline
$f$ & & $-$ & $0$ & $+$ & \\
\hline
$V'$ & & $+$ & $0$ & $-$ & \\
\hline
$V$ & & $\nearrow$ & & $\searrow$ &
\end{tabular}
\end{center}
よって$V(r)$は$r=1/\sqrt2=\sqrt2/2$で最大。この時$r^2=1/2,\ r^3=\sqrt2/4,\ r^4=1/4,\ 1-r^2=1/2,\ (1-r^2)^{3/2}=\sqrt2/4$だから
\[
\frac{V(1/\sqrt2)}\pi=-\frac14+\frac23\cdot\frac{\sqrt2}4+\frac12-\frac23+\frac23\cdot\frac{\sqrt2}4=\left(-\frac14+\frac12-\frac23\right)+\frac{4}{3}\cdot\frac{\sqrt2}4=-\frac5{12}+\frac{\sqrt2}3
\]
したがって，$r=\dfrac{\sqrt2}2$で最大値
\[
\max V=\left(\frac{\sqrt2}3-\frac5{12}\right)\pi
\]
をとる。
\end{proof}
